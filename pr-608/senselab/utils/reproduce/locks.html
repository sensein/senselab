senselab.utils.reproduce.locks

Lock headers, the pre-install integrity check and freeze parsing for reproduce envs.

A requirements-derived body lock opens with a header::

# reproduce lock
# requirements-sha256: <requirements_sha256(...) of the compile inputs>
# python: 3.11

followed by uv's pinned, hashed lines. Paths in an EnvSpec are relative to the manifest repository, passed as base_dir.

 62def requirements_sha256(
 63    requirements: Iterable[str],
 64    python: str,
 65    max_cuda: str | None,
 66    exclude_newer: datetime.date,
 67    config_sha: str | None,
 68    overrides_sha: str | None,
 69    extra_no_emit: Iterable[str],
 70) -> str:
 71    """Digest of everything a requirements-derived body lock is compiled from.
 72
 73    Args:
 74        requirements: The requirement lines of ``requirements.in``; order does not matter.
 75        python: The env's ``major.minor`` Python version.
 76        max_cuda: The env's CUDA ceiling, if any.
 77        exclude_newer: The compile's ``--exclude-newer`` date.
 78        config_sha: sha256 of the uv ``--config-file``, if any.
 79        overrides_sha: sha256 of the ``--overrides`` file, if any.
 80        extra_no_emit: Names left out beyond the torch family and the sdists; order does not matter.
 81
 82    Returns:
 83        A hex sha256.
 84    """
 85    fields = {
 86        "python": python,
 87        "max_cuda": max_cuda,
 88        "exclude_newer": exclude_newer.isoformat(),
 89        "config_file_sha256": config_sha,
 90        "overrides_sha256": overrides_sha,
 91        "extra_no_emit": sorted(extra_no_emit),
 92    }
 93    return uv_tools.requirements_sha256(requirements, header_fields=fields)

 96def write_header(body: str, fields: Mapping[str, str]) -> str:
 97    """Prefix a lock body with the reproduce header.
 98
 99    Args:
100        body: uv's pinned, hashed lines.
101        fields: Header keys (lowercase, digits and ``-``) and single-line values, in order.
102
103    Returns:
104        The header followed by ``body``.
105
106    Raises:
107        ValueError: If a key is malformed or a value spans lines.
108    """
109    lines = [HEADER_MARKER]
110    for key, value in fields.items():
111        if not _KEY_RE.match(key) or "\n" in value or "\r" in value:
112            raise ValueError(f"bad header field {key!r}: {value!r}")
113        lines.append(f"# {key}: {value}")
114    return "\n".join(lines) + "\n" + body

117def parse_header(text: str) -> dict[str, str]:
118    """Read the reproduce header of a lock.
119
120    Args:
121        text: The lock file's contents.
122
123    Returns:
124        The header fields, keyed as written.
125
126    Raises:
127        ReproduceLockError: If the first line is not the reproduce marker.
128    """
129    lines = text.splitlines()
130    if not lines or lines[0].strip() != HEADER_MARKER:
131        raise ReproduceLockError(f"lock does not start with {HEADER_MARKER!r}")
132    header: dict[str, str] = {}
133    for line in lines[1:]:
134        if not line.startswith("#"):
135            break
136        match = _HEADER_RE.match(line)
137        if match:
138            header[match.group(1)] = match.group(2)
139    return header

142def _logical_lines(text: str) -> list[str]:
143    joined = re.sub(r"\\\r?\n\s*", " ", text)
144    lines = (_COMMENT_RE.sub("", line).strip() for line in joined.splitlines())
145    return [line for line in lines if line]

148def _is_index_option(line: str) -> bool:
149    """Whether ``line`` is an index or find-links option; any other option line is refused."""
150    if not line.startswith("-"):
151        return False
152    if re.split(r"[\s=]", line, maxsplit=1)[0] in INDEX_OPTIONS:
153        return True
154    raise ReproduceLockError(f"option line refused: {line!r}")

157def lock_entries(text: str) -> list[LockEntry]:
158    """Parse a pinned requirements file: a uv lock, an export, a resolve or a freeze.
159
160    Index and find-links option lines are skipped; every other option line is refused.
161
162    Args:
163        text: The file's contents; comments and backslash continuations are allowed.
164
165    Returns:
166        One entry per requirement line, in file order.
167
168    Raises:
169        ReproduceLockError: If a line is an ``-e``, ``-r``, ``-c`` or other option, or not an ``==``/``@`` pin.
170    """
171    entries = []
172    for line in _logical_lines(text):
173        if _is_index_option(line):
174            continue
175        match = _LINE_RE.match(_HASH_RE.sub("", line).strip())
176        if not match:
177            raise ReproduceLockError(f"not a pinned requirement: {line!r}")
178        name, op, pin, marker = match.groups()
179        hashes = frozenset(_HASH_RE.findall(line))
180        entries.append(LockEntry(normalize_name(name), pin, op == "@", " ".join((marker or "").split()), hashes))
181    return entries

184def entry_version(entry: LockEntry) -> str:
185    """Return an entry's full version, local tag included.
186
187    Args:
188        entry: A parsed requirement.
189
190    Returns:
191        The ``==`` version, or the version in a ``.whl`` URL's file name.
192
193    Raises:
194        ReproduceLockError: If a URL pin is not a wheel over ``file://``, ``http://`` or ``https://``, or the
195            wheel's project name is not the entry's name.
196    """
197    if not entry.is_url:
198        return entry.pin
199    url = urllib.parse.urlsplit(entry.pin)
200    filename = urllib.parse.unquote(url.path.rsplit("/", 1)[-1])
201    parts = filename.split("-")
202    if url.scheme not in ("file", "http", "https") or not filename.endswith(".whl") or len(parts) < 5:
203        raise ReproduceLockError(f"{entry.name} is not pinned to a wheel: {entry.pin}")
204    if normalize_name(parts[0]) != entry.name:
205        raise ReproduceLockError(f"{entry.name} is pinned to a wheel of {parts[0]}: {entry.pin}")
206    return parts[1]

210    """Parse ``uv pip freeze`` output into ``{name: version}``.

224def versions_agree(a: str, b: str) -> bool:
225    """Compare versions, ignoring a ``+local`` tag only when exactly one side has one.
226
227    Args:
228        a: A version, e.g. ``2.8.0+cu128``.
229        b: Another version.
230
231    Returns:
232        Whether they name the same release.
233    """
234    if ("+" in a) != ("+" in b):
235        return a.split("+", 1)[0] == b.split("+", 1)[0]
236    return a == b

240    path = base_dir / rel

242        raise ReproduceLockError(f"{rel} resolves outside {base_dir}")

246def _verify(
247    base_dir: Path, rel: str, expected: str, what: str, verified: dict[str, str], contents: dict[str, bytes]
248) -> None:
249    if rel not in contents:
250        contents[rel] = _path(base_dir, rel).read_bytes()
251    actual = hashlib.sha256(contents[rel]).hexdigest()
252    if actual != expected:
253        raise ReproduceLockError(f"{what} {rel}: sha256 {actual} != manifest {expected}")
254    verified[rel] = actual

277def requirements_digest_for(env: EnvSpec, base_dir: Path) -> str:
278    """Return the ``requirements-sha256`` header value for a requirements-derived body.
279
280    ``requirements.in`` lines are normalised first: continuations joined, comments stripped, inner
281    whitespace collapsed. Index options are digested with the requirements; ``-r``, ``-c``, ``-e`` and
282    every other option are refused, because the files they name are not hashed.
283
284    Args:
285        env: The step's env; its body source must be ``requirements.in``.
286        base_dir: The manifest repository.
287
288    Returns:
289        :func:`requirements_sha256` over the requirement lines and the env's compile inputs.
290
291    Raises:
292        ReproduceLockError: If the body is not requirements-derived, a line is a non-index option, or a
293            requirement line repeats.
294    """
295    body = env.body
296    if body.source != "requirements.in" or body.requirements_in is None:
297        raise ReproduceLockError("only a requirements.in body has a requirements digest")
298    return _requirements_digest(env, _path(base_dir, body.requirements_in).read_text())

301def _requirements_digest(env: EnvSpec, text: str) -> str:
302    body = env.body
303    lines = [" ".join(line.split()) for line in _logical_lines(text)]
304    for line in lines:
305        _is_index_option(line)
306    dupes = sorted({line for line in lines if lines.count(line) > 1})
307    if dupes:
308        raise ReproduceLockError(f"{body.requirements_in} repeats requirement lines: {dupes}")
309    return requirements_sha256(
310        lines,
311        env.python,
312        env.max_cuda,
313        env.exclude_newer,
314        body.config_file_sha256,
315        body.overrides_sha256,
316        env.extra_no_emit,
317    )

327def _overlay_problems(
328    env: EnvSpec,
329    overlay: list[LockEntry],
330    base: list[LockEntry],
331    torch_names: set[str],
332    in_family: Callable[[str], bool],
333) -> list[str]:
334    pinned = _group(base)
335    held_out = {normalize_name(s.name) for s in env.sdist_allow} | {normalize_name(n) for n in env.extra_no_emit}
336    problems = []
337    for name, entries in sorted(_group(overlay).items()):
338        if name in held_out:
339            problems.append(f"{name}: held out of the base locks (sdist_allow or extra_no_emit)")
340        elif in_family(name) and name not in torch_names:
341            problems.append(f"{name}: torch-family name absent from torch.lock")
342        elif name not in pinned:
343            continue
344        elif len(pinned[name]) > 1 or len(entries) > 1:
345            problems.append(f"{name}: pinned in more than one entry; markers are not evaluated")
346        else:
347            mine, theirs = entries[0], pinned[name][0]
348            if (mine.pin, mine.marker, mine.hashes) != (theirs.pin, theirs.marker, theirs.hashes):
349                problems.append(f"{name}: {mine.pin} differs from base {theirs.pin} or its marker or hashes differ")
350    return problems

353def _torch_problems(torch: list[LockEntry], reference: dict[str, list[tuple[str, set[str] | None]]]) -> list[str]:
354    names = {e.name for e in torch}
355    problems = [f"{e.name}: no --hash in torch.lock" for e in torch if not e.hashes]
356    for entry in torch:
357        version = entry_version(entry)
358        candidates = reference.get(entry.name, [])
359        if not any(versions_agree(version, v) and (h is None or entry.hashes <= h) for v, h in candidates):
360            problems.append(f"{entry.name}: torch.lock {version}, reference {sorted(v for v, _ in candidates)}")
361    problems += [f"{n}: in the reference, absent from torch.lock" for n in sorted(reference) if n not in names]
362    return problems

373def uv_lock_torch_family(packages: Iterable[Mapping[str, Any]], roots: Iterable[str]) -> set[str]:
374    """Return the torch family a repository ``uv.lock`` implies.
375
376    The family is the roots plus the torch-family-named packages (``nvidia-*``, ``triton``, …) in the
377    transitive closure of the roots' ``dependencies``, over every marker. A CUDA-named package that only
378    another project dependency pulls in is not part of it.
379
380    Args:
381        packages: The ``[[package]]`` tables of a parsed ``uv.lock``.
382        roots: ``torch``, ``torchvision`` and ``torchaudio`` names present in ``torch.lock``.
383
384    Returns:
385        Normalised names.
386    """
387    deps: dict[str, set[str]] = {}
388    for package in packages:
389        names = {normalize_name(d["name"]) for d in package.get("dependencies", [])}
390        deps.setdefault(normalize_name(package["name"]), set()).update(names)
391    seen = {normalize_name(r) for r in roots}
392    stack = list(seen)
393    while stack:
394        for dep in deps.get(stack.pop(), set()) - seen:
395            seen.add(dep)
396            stack.append(dep)
397    return {n for n in seen if is_torch_family(n)}

400def _uv_lock_reference(
401    packages: list[dict[str, Any]], family: set[str]
402) -> dict[str, list[tuple[str, set[str] | None]]]:
403    out: dict[str, list[tuple[str, set[str] | None]]] = {}
404    for package in packages:
405        name = normalize_name(package["name"])
406        if name in family:
407            hashes = {w["hash"].removeprefix("sha256:") for w in package.get("wheels", []) if "hash" in w}
408            out.setdefault(name, []).append((package["version"], hashes))
409    for name in family - set(out):
410        out[name] = []
411    return out

414def check_integrity(env: EnvSpec, base_dir: Path, uv_lock: Path | None = None) -> dict[str, str]:
415    """Check an env's hashed files and pin agreement before anything is installed.
416
417    The checks run in order: lock and sdist-patch sha256s (torch, body, wheels, overlays, build envs);
418    the ``requirements.in``, config and overrides sha256s and the recomputed header digest; the
419    ``first_resolve`` sha256; with a torch lock, no torch-lock or torch-family name in the body; overlay
420    pins against the base locks; torch-family pins against ``first_resolve`` (``requirements.in``
421    bodies) or the repository ``uv.lock`` (uv-export bodies). Against ``uv.lock`` the torch family is
422    :func:`uv_lock_torch_family`, which also decides which names the body and overlays may not pin;
423    otherwise it is the name rule of :func:`is_torch_family`.
424
425    Each file is read once; the header, pin and torch checks run on the bytes that were hashed. The
426    returned sha256s describe those bytes. The installer copies each file into ``{tmp}``,
427    checks the copy against this mapping and installs from the copy, never from ``base_dir``.
428
429    Args:
430        env: The step's env.
431        base_dir: The manifest repository; every path in ``env`` is relative to it and must resolve inside it.
432        uv_lock: The checkout's ``uv.lock``; needed for a uv-export body with a torch lock.
433
434    Returns:
435        ``{relative path: sha256}`` of every file verified.
436
437    Raises:
438        ReproduceLockError: On the first check that fails.
439    """
440    verified: dict[str, str] = {}
441    contents: dict[str, bytes] = {}
442
443    def text(rel: str) -> str:
444        return contents[rel].decode()
445
446    locks = [("torch lock", env.torch), ("body lock", env.body), ("wheels lock", env.wheels)]
447    locks += [("test lock", env.test), ("notebook_tools lock", env.notebook_tools)]
448    locks += [(f"build_env lock of {s.name}", s.build_env) for s in env.sdist_allow]
449    for what, ref in locks:
450        if ref is not None:
451            _verify(base_dir, ref.lock, ref.lock_sha256, what, verified, contents)
452    for sdist in env.sdist_allow:
453        for patch in sdist.patches:
454            _verify(base_dir, patch.file, patch.sha256, f"patch of {sdist.name}", verified, contents)
455
456    body = env.body
457    if body.source == "requirements.in":
458        if body.requirements_in is None or body.requirements_in_sha256 is None:
459            raise ReproduceLockError("a requirements.in body needs requirements_in and its sha256")
460        _verify(base_dir, body.requirements_in, body.requirements_in_sha256, "requirements.in", verified, contents)
461        for what, rel, sha in (
462            ("config file", body.config_file, body.config_file_sha256),
463            ("overrides", body.overrides, body.overrides_sha256),
464        ):
465            if rel is not None and sha is not None:
466                _verify(base_dir, rel, sha, what, verified, contents)
467        recorded = parse_header(text(body.lock)).get("requirements-sha256")
468        expected = _requirements_digest(env, text(body.requirements_in))
469        if recorded != expected:
470            raise ReproduceLockError(
471                f"body lock {body.lock}: header requirements-sha256 {recorded} != recomputed {expected}"
472            )
473
474    if env.first_resolve is not None:
475        _verify(base_dir, env.first_resolve.file, env.first_resolve.sha256, "first_resolve", verified, contents)
476
477    torch = lock_entries(text(env.torch.lock)) if env.torch is not None else []
478    torch_names = {e.name for e in torch}
479    body_entries = lock_entries(text(body.lock))
480    base = body_entries + torch
481    in_family: Callable[[str], bool] = is_torch_family
482    uv_packages: list[dict[str, Any]] = []
483    family: set[str] = set()
484    if env.torch is not None and body.source == "uv-export":
485        if uv_lock is None:
486            raise ReproduceLockError("a uv-export body with a torch lock needs the repository uv.lock to check against")
487        uv_packages = tomllib.loads(uv_lock.read_text()).get("package", [])
488        family = uv_lock_torch_family(uv_packages, torch_names & TORCH_NAMES)
489        in_family = family.__contains__
490    if env.torch is not None:
491        stray = sorted({e.name for e in body_entries if e.name in torch_names or in_family(e.name)})
492        if stray:
493            raise ReproduceLockError(f"body lock {body.lock} pins torch.lock or torch-family names: {stray}")
494    for what, ref in (("test", env.test), ("notebook_tools", env.notebook_tools)):
495        if ref is not None:
496            problems = _overlay_problems(env, lock_entries(text(ref.lock)), base, torch_names, in_family)
497            if problems:
498                raise ReproduceLockError(f"{what} overlay refused: {problems}")
499
500    if env.torch is None:
501        return verified
502    if body.source == "requirements.in":
503        if env.first_resolve is None:
504            raise ReproduceLockError("a torch lock with a requirements.in body needs first_resolve")
505        reference = _resolve_reference(lock_entries(text(env.first_resolve.file)))
506    else:
507        reference = _uv_lock_reference(uv_packages, family)
508    problems = _torch_problems(torch, reference)
509    if problems:
510        raise ReproduceLockError(f"torch lock {env.torch.lock} disagrees: {problems}")
511    return verified

374def uv_lock_torch_family(packages: Iterable[Mapping[str, Any]], roots: Iterable[str]) -> set[str]:
375    """Return the torch family a repository ``uv.lock`` implies.
376
377    The family is the roots plus the torch-family-named packages (``nvidia-*``, ``triton``, …) in the
378    transitive closure of the roots' ``dependencies``, over every marker. A CUDA-named package that only
379    another project dependency pulls in is not part of it.
380
381    Args:
382        packages: The ``[[package]]`` tables of a parsed ``uv.lock``.
383        roots: ``torch``, ``torchvision`` and ``torchaudio`` names present in ``torch.lock``.
384
385    Returns:
386        Normalised names.
387    """
388    deps: dict[str, set[str]] = {}
389    for package in packages:
390        names = {normalize_name(d["name"]) for d in package.get("dependencies", [])}
391        deps.setdefault(normalize_name(package["name"]), set()).update(names)
392    seen = {normalize_name(r) for r in roots}
393    stack = list(seen)
394    while stack:
395        for dep in deps.get(stack.pop(), set()) - seen:
396            seen.add(dep)
397            stack.append(dep)
398    return {n for n in seen if is_torch_family(n)}

417

419    the ``requirements.in``, config and overrides sha256s and the recomputed header digest; the

443

445        return contents[rel].decode()
446

463            ("config file", body.config_file, body.config_file_sha256),
464            ("overrides", body.overrides, body.overrides_sha256),
465        ):
