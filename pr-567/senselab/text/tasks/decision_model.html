senselab.text.tasks.decision_model

Decision models: small local models that answer typed questions about a text with probabilities.

Triage uses Cloudflare's Clef, served by Ollama, as a second opinion beside its instruction-tuned reviewer. .ollama runs a pinned model; .second_opinion holds the questions triage asks and reads the answers back.

218    def __enter__(self) -> "OllamaServer":
219        """Verify the pin, start the server, wait until it answers and, with ``require_gpu``, load the model."""
220        verify_pin(self.models_dir, self.pin, verified_dir=self.verified_dir)
221        self.host = f"127.0.0.1:{_free_port()}"
222        env = {
223            "OLLAMA_VULKAN": "0",
224            **os.environ,
225            "OLLAMA_MODELS": str(self.models_dir),
226            "OLLAMA_HOST": self.host,
227            "OLLAMA_KEEP_ALIVE": self.keep_alive,
228            "OLLAMA_NUM_PARALLEL": str(self.num_parallel),
229            "OLLAMA_NOPRUNE": "1",
230        }
231        self._log = self.log_path.open("ab") if self.log_path is not None else subprocess.DEVNULL
232        self._process = subprocess.Popen(  # noqa: S603 — a fixed binary with fixed arguments
233            [str(self.binary), "serve"], env=env, stdout=self._log, stderr=subprocess.STDOUT
234        )
235        deadline = time.monotonic() + self.startup_timeout_s
236        while time.monotonic() < deadline:
237            if self._process.poll() is not None:
238                raise RuntimeError(f"ollama serve exited with {self._process.returncode} before answering")
239            try:
240                with urllib.request.urlopen(f"http://{self.host}/api/version", timeout=5):  # noqa: S310
241                    break
242            except (urllib.error.URLError, OSError):
243                time.sleep(1.0)
244        else:
245            self.__exit__(None, None, None)
246            raise RuntimeError(f"ollama serve did not answer within {self.startup_timeout_s} s")

251                self.__exit__(None, None, None)
252                raise

255    def log_tail(self) -> str:
256        """The last non-empty line of the server's log, or an empty string."""
257        if self.log_path is None or not self.log_path.exists():
258            return ""
259        if self._log not in (None, subprocess.DEVNULL):
260            self._log.flush()
261        with self.log_path.open("rb") as handle:
262            handle.seek(0, os.SEEK_END)
263            handle.seek(max(0, handle.tell() - 8192))
264            lines = [line.strip() for line in handle.read().decode("utf-8", "replace").splitlines() if line.strip()]
265        return lines[-1] if lines else ""

267    def _unusable(self, reason: str) -> ServerUnusableError:
268        tail = self.log_tail()
269        return ServerUnusableError(f"{reason}; server log: {tail}" if tail else reason)

275            ServerUnusableError: If the load fails or any of the model is held outside GPU memory.
276        """
277        body = json.dumps({"model": self.model, **WARM_UP, "keep_alive": self.keep_alive}).encode("utf-8")
278        request = urllib.request.Request(  # noqa: S310 — loopback only
279            f"http://{self.host}/v1/systemone", data=body, headers={"Content-Type": "application/json"}
280        )
281        try:
282            with urllib.request.urlopen(request, timeout=self.load_timeout_s):  # noqa: S310
283                pass
284        except (urllib.error.URLError, OSError) as error:
285            raise self._unusable(f"{self.model} did not load: {type(error).__name__}: {error}") from error

288    def require_resident(self) -> None:
289        """Require the pinned model loaded and wholly in GPU memory, by the server's ``/api/ps``.
290
291        Raises:
292            ServerUnusableError: If it is not loaded, or ``size_vram`` is below ``size``.
293        """
294        try:
295            with urllib.request.urlopen(f"http://{self.host}/api/ps", timeout=30) as response:  # noqa: S310
296                loaded = json.loads(response.read()).get("models") or []
297        except (urllib.error.URLError, OSError, ValueError) as error:
298            raise self._unusable(f"/api/ps did not answer: {type(error).__name__}: {error}") from error
299        held = [entry for entry in loaded if self.model in (entry.get("name"), entry.get("model"))]
300        if not held:
301            raise self._unusable(f"{self.model} is not loaded")
302        size, vram = int(held[0].get("size") or 0), int(held[0].get("size_vram") or 0)
303        if size <= 0 or vram < size:
304            raise self._unusable(f"{self.model} is not wholly on the GPU: size_vram {vram} of size {size}")

def read_answers( answers: Mapping[str, Any]) -> SecondOpinion: 
 86def read_answers(answers: Mapping[str, Any]) -> SecondOpinion:
 87    """Read a decision model's answers to :data:`QUESTIONS`.
 88
 89    Args:
 90        answers: The ``answers`` mapping, question name to typed answer.
 91
 92    Returns:
 93        The probabilities and the ``other_voice`` choice.
 94
 95    Raises:
 96        ValueError: If a question is unanswered or an answer has no probability.
 97    """
 98    probabilities: dict[str, float] = {}
 99    choice = ""
100    for name in QUESTION_NAMES:
101        answer = answers.get(name)
102        if not isinstance(answer, Mapping):
103            raise ValueError(f"no answer to {name!r}")
104        if QUESTIONS[name]["type"] == "choice":
105            classes = answer.get("probabilities")
106            if not isinstance(classes, Mapping) or MORE_THAN_ONE not in classes:
107                raise ValueError(f"{name!r} answered without class probabilities")
108            probabilities[name] = float(classes[MORE_THAN_ONE])
109            choice = str(answer.get("choice") or "")
110        else:
111            if answer.get("noul") is None:
112                raise ValueError(f"{name!r} answered without a probability")
113            probabilities[name] = float(answer["noul"])
114    return SecondOpinion(probabilities=probabilities, other_voice_choice=choice, raw=dict(answers))

 95
 96    Raises:
 97        PinMismatchError: If the manifest is missing, names other digests, or the blob's size or
 98            content does not match.
 99    """
100    manifest_path = _manifest_path(models_dir, pin)
101    try:
102        manifest_bytes = manifest_path.read_bytes()
103        manifest = json.loads(manifest_bytes)
104    except (OSError, json.JSONDecodeError) as error:
105        raise PinMismatchError(f"no readable manifest for {pin.model_id} at {manifest_path}: {error}") from error
106    manifest_digest = f"sha256:{hashlib.sha256(manifest_bytes).hexdigest()}"
107    if manifest_digest != pin.manifest_digest:
108        raise PinMismatchError(f"{pin.model_id}: manifest is {manifest_digest}, pinned {pin.manifest_digest}")
109    config_digest = str((manifest.get("config") or {}).get("digest") or "")
110    if config_digest != pin.config_digest:
111        raise PinMismatchError(f"{pin.model_id}: manifest config is {config_digest}, pinned {pin.config_digest}")
112    layers = [layer for layer in manifest.get("layers") or () if layer.get("mediaType") == _MODEL_LAYER]
113    if len(layers) != 1 or layers[0].get("digest") != pin.blob_digest:
114        named = [layer.get("digest") for layer in layers]
115        raise PinMismatchError(f"{pin.model_id}: model layer is {named}, pinned {pin.blob_digest}")
116    for layer in manifest.get("layers") or ():
117        if layer.get("mediaType") == _MODEL_LAYER:
118            continue
119        small = _blob_path(models_dir, str(layer.get("digest")))
120        try:
121            held = _sha256(small)
122        except OSError as error:
123            raise PinMismatchError(f"{pin.model_id}: layer {small} is missing: {error}") from error
124        if held != layer.get("digest"):
125            raise PinMismatchError(f"{pin.model_id}: layer {small.name} hashes to {held}")
126    blob = _blob_path(models_dir, pin.blob_digest)
127    try:
128        stat = blob.stat()
129    except OSError as error:
130        raise PinMismatchError(f"{pin.model_id}: blob {blob} is missing: {error}") from error
131    if stat.st_size != int(layers[0].get("size") or -1):
132        raise PinMismatchError(f"{pin.model_id}: blob is {stat.st_size} bytes, manifest says {layers[0].get('size')}")
133    marker = None
134    if verified_dir is not None:
135        marker = verified_dir / f"{pin.blob_digest.replace(':', '-')}.{stat.st_size}.{int(stat.st_mtime)}"
136        if marker.exists():
137            return blob
138    actual = _sha256(blob)
139    if actual != pin.blob_digest:
140        raise PinMismatchError(f"{pin.model_id}: blob hashes to {actual}, pinned {pin.blob_digest}")
141    if marker is not None:
142        marker.parent.mkdir(parents=True, exist_ok=True)
143        marker.write_text(actual + "\n", encoding="utf-8")
144    return blob

Check that the store holds exactly the pinned model, and return its blob.

Arguments:
  • models_dir: The Ollama model store (OLLAMA_MODELS).
  • pin: The model and its digests.
  • verified_dir: Where a completed full-blob hash is remembered, keyed by digest, size and modification time; None hashes the blob every call.
Returns:

The path of the verified weights blob.

Raises:
  • PinMismatchError: If the manifest is missing, names other digests, or the blob's size or content does not match.
