125def f0_track(
126    audio: Audio,
127    *,
128    f0_min_hz: float,
129    f0_max_hz: float,
130    hop_s: float,
131) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
132    """F0 and its strength per frame via Praat's cc pitch: ``(times_s, f0_hz, strength)``.
133
134    Unvoiced frames carry NaN in ``f0_hz`` with their ``strength`` retained, so F0 always travels
135    with the periodicity that placed it and a reader cannot separate them.
136
137    Args:
138        audio: The recording. ``get_sound`` handles channel merging and resampling.
139        f0_min_hz: Lowest F0 to search. Read it from ``voice.f0_range_hz``.
140        f0_max_hz: Highest F0 to search. Read it from ``voice.f0_range_hz``.
141        hop_s: Praat's ``time_step``. Read it from ``phonation.hop_s``.
142
143    Returns:
144        ``(times_s, f0_hz, strength)``, one value per frame, all three the same length.
145
146    Raises:
147        ModuleNotFoundError: If parselmouth is not installed.
148    """
149    _require_parselmouth()
150    snd = get_sound(audio)
151    pitch = snd.to_pitch_cc(time_step=hop_s, pitch_floor=f0_min_hz, pitch_ceiling=f0_max_hz)
152    times = np.asarray(pitch.xs(), dtype=np.float64)
153    f0 = np.asarray(pitch.selected_array["frequency"], dtype=np.float64)
154    strength = np.asarray(pitch.selected_array["strength"], dtype=np.float64)
155    f0[f0 == 0.0] = np.nan
156    return times, f0, strength

175    audio: Audio,
176    *,
177    hop_s: float,
178    max_formants: int,
179    formant_max_hz: float,
180    window_s: float,
181    preemphasis_hz: float,
182) -> FormantTrack:
183    """F1-F4 and their bandwidths over the whole stream, by Praat's Burg method.
184
185    Computed once over the stream so a consumer slices the track rather than re-tracking a fragment.
186
187    Args:
188        audio: The recording. ``get_sound`` handles channel merging and resampling.
189        hop_s: Praat's ``time_step``. Read it from ``phonation_spans.hop_s``.
190        max_formants: Praat's ``max_number_of_formants``. Read it from ``phonation_spans.max_formants``.
191        formant_max_hz: Praat's ``maximum_formant``. Read it from ``phonation_spans.formant_max_hz``.
192        window_s: Praat's ``window_length``. Read it from ``phonation_spans.formant_window_s``.
193        preemphasis_hz: Praat's ``pre_emphasis_from``. Read it from
194            ``phonation_spans.formant_preemphasis_hz``.
195
196    Returns:
197        The track. A frame where Praat placed no formant carries NaN in both arrays, so a missing
198        formant is absent rather than zero.
199
200    Raises:
201        ModuleNotFoundError: If parselmouth is not installed.
202    """
203    _require_parselmouth()
204    snd = get_sound(audio)
205    formants = snd.to_formant_burg(
206        time_step=hop_s,
207        max_number_of_formants=max_formants,
208        maximum_formant=formant_max_hz,
209        window_length=window_s,
210        pre_emphasis_from=preemphasis_hz,
211    )
212    times = np.asarray(formants.xs(), dtype=np.float64)
213    values: list[np.ndarray] = []
214    bandwidths: list[np.ndarray] = []
215    for order in (1, 2, 3, 4):
216        values.append(
217            np.asarray(
218                [formants.get_value_at_time(order, t, unit=parselmouth.FormantUnit.HERTZ) for t in times],
219                dtype=np.float64,
220            )
221        )
222        bandwidths.append(
223            np.asarray(
224                [formants.get_bandwidth_at_time(order, t, unit=parselmouth.FormantUnit.HERTZ) for t in times],
225                dtype=np.float64,
226            )
227        )
228    return FormantTrack(
229        times_s=times,
230        f_hz=(values[0], values[1], values[2], values[3]),
231        bandwidth_hz=(bandwidths[0], bandwidths[1], bandwidths[2], bandwidths[3]),
232    )

def period_marks( audio: senselab.audio.data_structures.audio.Audio, start_s: float, end_s: float, *, f0_min_hz: float, f0_max_hz: float) -> list[PeriodMark]: 
 79def period_marks(
 80    audio: Audio,
 81    start_s: float,
 82    end_s: float,
 83    *,
 84    f0_min_hz: float,
 85    f0_max_hz: float,
 86) -> list[PeriodMark]:
 87    """Glottal period marks inside one span, from Praat's point process.
 88
 89    Args:
 90        audio: The recording.
 91        start_s: Span onset.
 92        end_s: Span offset.
 93        f0_min_hz: Lowest F0 to search. Read it from ``voice.f0_range_hz``.
 94        f0_max_hz: Highest F0 to search. Read it from ``voice.f0_range_hz``.
 95
 96    Returns:
 97        One mark per pair of consecutive pulses whose gap is a plausible period — within
 98        ``[1/f0_max_hz, 1/f0_min_hz]``; a longer gap is a voicing break, not a period. Empty when
 99        Praat places no pulses — absent, not zero.
100
101    Raises:
102        ModuleNotFoundError: If parselmouth is not installed.
103    """
104    _require_parselmouth()
105    snd = get_sound(audio)
106    part = snd.extract_part(from_time=start_s, to_time=end_s, preserve_times=True)
107    point_process = parselmouth.praat.call(part, "To PointProcess (periodic, cc)", f0_min_hz, f0_max_hz)
108    n_points = int(parselmouth.praat.call(point_process, "Get number of points"))
109    pulses = [float(parselmouth.praat.call(point_process, "Get time from index", i)) for i in range(1, n_points + 1)]
110    x = np.asarray(part.values, dtype=np.float64).squeeze(0)
111    t0 = float(part.xmin)
112    sr = 1.0 / float(part.dx)
113    marks: list[PeriodMark] = []
114    for opening, closing in zip(pulses, pulses[1:]):
115        period = closing - opening
116        if not (1.0 / f0_max_hz <= period <= 1.0 / f0_min_hz):
117            continue
118        i0 = max(0, int((opening - t0) * sr))
119        i1 = min(len(x), int((closing - t0) * sr))
120        amplitude = float(np.abs(x[i0:i1]).max()) if i1 > i0 else 0.0
121        marks.append(PeriodMark(time_s=opening, period_s=period, amplitude=amplitude))
122    return marks

Glottal period marks inside one span, from Praat's point process.

Arguments:
  • audio: The recording.
  • start_s: Span onset.
  • end_s: Span offset.
  • f0_min_hz: Lowest F0 to search. Read it from voice.f0_range_hz.
  • f0_max_hz: Highest F0 to search. Read it from voice.f0_range_hz.
Returns:
