60        ``(times_s, hnr_db)``, one value per frame. Frames Praat judges silent carry its floor
61        value rather than being dropped, so the track and its times stay aligned.
62
63    Raises:
64        ModuleNotFoundError: If parselmouth is not installed.
65    """
66    _require_parselmouth()
67    snd = get_sound(audio)
68    harmonicity = snd.to_harmonicity_cc(
69        time_step=hop_s,
70        minimum_pitch=f0_min_hz,
71        silence_threshold=silence_threshold,
72        periods_per_window=periods_per_window,
73    )
74    times = np.asarray(harmonicity.xs(), dtype=np.float64)
75    values = np.asarray(harmonicity.values, dtype=np.float64).squeeze(0)

 87    """Glottal period marks inside one span, from Praat's point process.
 88
 89    Args:
 90        audio: The recording.
 91        start_s: Span onset.
 92        end_s: Span offset.
 93        f0_min_hz: Lowest F0 to search. Read it from ``phonation.f0_min_hz``.
 94        f0_max_hz: Highest F0 to search. Read it from ``phonation.f0_max_hz``.
 95
 96    Returns:
 97        One mark per pair of consecutive pulses whose gap is a plausible period — within
 98        ``[1/f0_max_hz, 1/f0_min_hz]``; a longer gap is a voicing break, not a period. Empty when
 99        Praat places no pulses — absent, not zero.
100
101    Raises:
102        ModuleNotFoundError: If parselmouth is not installed.
103    """
104    _require_parselmouth()
105    snd = get_sound(audio)
106    part = snd.extract_part(from_time=start_s, to_time=end_s, preserve_times=True)
107    point_process = parselmouth.praat.call(part, "To PointProcess (periodic, cc)", f0_min_hz, f0_max_hz)
108    n_points = int(parselmouth.praat.call(point_process, "Get number of points"))
109    pulses = [float(parselmouth.praat.call(point_process, "Get time from index", i)) for i in range(1, n_points + 1)]
110    x = np.asarray(part.values, dtype=np.float64).squeeze(0)
111    t0 = float(part.xmin)
112    sr = 1.0 / float(part.dx)
113    marks: list[PeriodMark] = []
114    for opening, closing in zip(pulses, pulses[1:]):
115        period = closing - opening
116        if not (1.0 / f0_max_hz <= period <= 1.0 / f0_min_hz):
117            continue
118        i0 = max(0, int((opening - t0) * sr))
119        i1 = min(len(x), int((closing - t0) * sr))
120        amplitude = float(np.abs(x[i0:i1]).max()) if i1 > i0 else 0.0
121        marks.append(PeriodMark(time_s=opening, period_s=period, amplitude=amplitude))
122    return marks
