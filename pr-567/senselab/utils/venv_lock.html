senselab.utils.venv_lock

Committed, fully pinned locks for the subprocess venvs that ensure_venv() builds.

Every venv name has one lock under data/venv_locks/<name>.txt, compiled by scripts/lock_subprocess_venvs.py from the backend's requirement list with uv pip compile --universal --generate-hashes. A lock opens with a header::

# senselab subprocess-venv lock: ppgs
# requirements-sha256: <sha256 of the backend's requirements, Python version and CUDA cap>
# python: 3.11
# torch: torch==2.8.0 torchaudio==2.8.0
# exclude-newer: 2026-10-02T00:00:00Z
# uv: uv 0.11.3

followed by uv's pinned, hashed requirements. torch, torchaudio and the CUDA-variant packages only they pull in (nvidia-*, triton) are left out of the pinned body: ensure_venv installs the exact torch: pins from the host's CUDA-matched PyTorch index, then the body with --no-deps.

The design is in specs/20261002-subprocess-venv-locks/design.md.

 89def requirements_sha256(
 90    requirements: list[str], python_version: str, max_cuda_version: Optional[tuple[int, int]] = None
 91) -> str:
 92    """Digest of everything a lock is compiled from.
 93
 94    Args:
 95        requirements: The backend's requirement specs.
 96        python_version: The venv's ``major.minor`` Python version.
 97        max_cuda_version: The backend's CUDA index ceiling, if any.
 98
 99    Returns:
100        A hex SHA-256.
101    """
102    payload = {
103        "requirements": sorted(requirements),
104        "ipc": list(IPC_REQUIREMENTS),
105        "python": python_version,
106        "max_cuda": list(max_cuda_version) if max_cuda_version else None,
107    }
108    return hashlib.sha256(json.dumps(payload, sort_keys=True).encode()).hexdigest()

116def parse_lock(path: Path, name: Optional[str] = None) -> VenvLock:
117    """Read a lock file's header.
118
119    Args:
120        path: The lock file.
121        name: The venv name; defaults to the file stem.
122
123    Returns:
124        The parsed lock.
125
126    Raises:
127        VenvLockError: When a required header field is missing.
128    """
129    text = path.read_text()
130    header: dict[str, str] = {}
131    for line in text.splitlines():
132        if not line.startswith("#"):
133            break
134        match = _HEADER_RE.match(line)
135        if match:
136            header[match.group(1)] = match.group(2)
137    for key in ("requirements-sha256", "python", "torch"):
138        if key not in header:
139            raise VenvLockError(f"lock {path} has no '{key}:' header line")
140    torch = header["torch"]
141    pins = tuple(torch.split()) if torch and torch != "none" else ()
142    return VenvLock(
143        name=name or path.stem,
144        path=path,
145        requirements_sha256=header["requirements-sha256"],
146        python=header["python"],
147        torch_pins=pins,
148        sha256=hashlib.sha256(text.encode()).hexdigest(),
149        header=header,
150    )

153def load_lock(
154    name: str,
155    requirements: list[str],
156    python_version: Optional[str] = None,
157    max_cuda_version: Optional[tuple[int, int]] = None,
158    lock_dir: Path = LOCK_DIR,
159) -> VenvLock:
160    """Load the committed lock for a venv and check it was compiled from these requirements.
161
162    Args:
163        name: The venv name.
164        requirements: The backend's requirement specs.
165        python_version: The requested Python version; None takes the lock's.
166        max_cuda_version: The backend's CUDA index ceiling.
167        lock_dir: Where locks live.
168
169    Returns:
170        The lock.
171
172    Raises:
173        VenvLockError: When the lock is missing or stale.
174    """
175    path = lock_path(name, lock_dir)
176    if not path.is_file():
177        raise VenvLockError(f"No lock for venv '{name}' at {path}. " + LOCK_MISSING_HINT.format(name=name))
178    lock = parse_lock(path, name)
179    python = python_version or lock.python
180    if python != lock.python:
181        raise VenvLockError(
182            f"Lock for venv '{name}' is for Python {lock.python}, not {python}. " + LOCK_MISSING_HINT.format(name=name)
183        )
184    expected = requirements_sha256(requirements, python, max_cuda_version)
185    if lock.requirements_sha256 != expected:
186        raise VenvLockError(
187            f"Lock for venv '{name}' was compiled from different requirements. " + LOCK_MISSING_HINT.format(name=name)
188        )
189    return lock

214def discover_backends(package_root: Optional[Path] = None) -> dict[str, Backend]:
215    """Find every ``ensure_venv`` call in senselab and resolve its arguments.
216
217    Each call's arguments are read from the calling module's attributes: a name, a call of a
218    zero-argument function, or a literal.
219
220    Args:
221        package_root: The ``senselab`` package directory; defaults to the installed one.
222
223    Returns:
224        Backends by venv name.
225
226    Raises:
227        VenvLockError: When two calls name one venv with different inputs, or an argument cannot
228            be resolved.
229    """
230    import ast
231    import importlib
232
233    import senselab
234
235    root = package_root or Path(senselab.__file__).parent
236    found: dict[str, Backend] = {}
237    for path in sorted(root.rglob("*.py")):
238        source = path.read_text()
239        if "ensure_venv(" not in source:
240            continue
241        module_name = ".".join(("senselab", *path.relative_to(root).with_suffix("").parts))
242        if module_name in _NOT_BACKENDS:
243            continue
244        calls = [
245            node
246            for node in ast.walk(ast.parse(source))
247            if isinstance(node, ast.Call)
248            and (getattr(node.func, "id", None) or getattr(node.func, "attr", None)) == "ensure_venv"
249        ]
250        if not calls:
251            continue
252        module = importlib.import_module(module_name)
253
254        def value(node: Optional[ast.expr], module: object = module) -> object:
255            if node is None:
256                return None
257            if isinstance(node, ast.Constant):
258                return node.value
259            if isinstance(node, ast.Name):
260                return getattr(module, node.id)
261            if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and not node.args:
262                return getattr(module, node.func.id)()
263            raise VenvLockError(f"{module_name}: cannot resolve ensure_venv argument {ast.unparse(node)}")
264
265        for call in calls:
266            keywords = {k.arg: k.value for k in call.keywords}
267            name = value(call.args[0] if call.args else keywords.get("name"))
268            requirements = value(call.args[1] if len(call.args) > 1 else keywords.get("requirements"))
269            python = value(call.args[2] if len(call.args) > 2 else keywords.get("python_version"))
270            cuda = value(keywords.get("max_cuda_version"))
271            if not isinstance(name, str) or not isinstance(requirements, list):
272                raise VenvLockError(f"{module_name}: ensure_venv needs a str name and a list of requirements")
273            if not isinstance(python, str):
274                raise VenvLockError(f"{module_name}: ensure_venv('{name}') must declare python_version")
275            backend = Backend(
276                name=name,
277                requirements=list(requirements),
278                python=python,
279                max_cuda_version=tuple(cuda) if cuda else None,  # type: ignore[arg-type]
280                modules=[module_name],
281            )
282            if name in found:
283                prior = found[name]
284                if (prior.requirements, prior.python, prior.max_cuda_version) != (
285                    backend.requirements,
286                    backend.python,
287                    backend.max_cuda_version,
288                ):
289                    raise VenvLockError(f"venv '{name}' has different inputs in {prior.modules} and {module_name}")
290                if module_name not in prior.modules:
291                    prior.modules.append(module_name)
292            else:
293                found[name] = backend
294    return found

304def _compile(
305    in_file: Path,
306    python_version: str,
307    constraints: Path,
308    exclude_newer: Optional[str],
309    *,
310    hashes: bool,
311    no_emit: tuple[str, ...] = (),
312) -> str:
313    cmd = [
314        _uv(),
315        "pip",
316        "compile",
317        str(in_file),
318        "--universal",
319        "--python-version",
320        python_version,
321        "--constraint",
322        str(constraints),
323        "--no-header",
324        "--quiet",
325        "--color",
326        "never",
327        "--no-sources",
328    ]
329    if exclude_newer:
330        cmd += ["--exclude-newer", exclude_newer]
331    if hashes:
332        cmd.append("--generate-hashes")
333    else:
334        cmd += ["--annotation-style", "line"]
335    for package in no_emit:
336        cmd += ["--no-emit-package", package]
337    result = subprocess.run(cmd, capture_output=True, text=True, cwd=in_file.parent)
338    if result.returncode != 0:
339        raise VenvLockError(f"uv pip compile failed for {in_file.name}:\n{result.stderr.strip()}")
340    return result.stdout

343def resolved_packages(line_annotated: str) -> dict[str, tuple[str, set[str]]]:
344    """Parse ``--annotation-style line`` output into ``{name: (version_or_url, parents)}``.
345
346    A name resolved more than once under different markers keeps its first entry and the union of
347    parents.
348    """
349    out: dict[str, tuple[str, set[str]]] = {}
350    for line in line_annotated.splitlines():
351        match = _LINE_RE.match(line)
352        if not match:
353            continue
354        name = normalize_name(match.group(1))
355        via = _VIA_RE.search(line)
356        parents = {normalize_name(p.strip()) for p in via.group(1).split(",")} if via else set()
357        if name in out:
358            out[name] = (out[name][0], out[name][1] | parents)
359        else:
360            out[name] = (match.group(4), parents)
361    return out

364def torch_only_cuda_packages(packages: dict[str, tuple[str, set[str]]]) -> list[str]:
365    """Names to leave out of the lock body.
366
367    These are ``torch``, ``torchaudio`` and the CUDA-variant packages that only they pull in.
368    """
369    family = {name for name in packages if name in TORCH_PACKAGES}
370    changed = True
371    while changed:
372        changed = False
373        for name, (_, parents) in packages.items():
374            if name not in family and parents and parents <= family:
375                family.add(name)
376                changed = True
377    return sorted(name for name in family if name in TORCH_PACKAGES or name.startswith(_CUDA_VARIANT_PREFIXES))

381    """The newest PyTorch CUDA index this venv can be routed to."""

396def _index_wheels(tag: str, package: str) -> list[str]:
397    url = f"{PYTORCH_INDEX_BASE}/{tag}/{package}/"
398    if url not in _INDEX_PAGES:
399        with urllib.request.urlopen(url, timeout=60) as response:  # noqa: S310 -- fixed https URL

401        _INDEX_PAGES[url] = re.findall(r">([^<>]+\.whl)<", page)
402    return _INDEX_PAGES[url]

405def wheel_on_index(package: str, version: str, tag: str, python_version: str, platform: tuple[str, ...]) -> bool:
406    """Whether the PyTorch index ``tag`` serves ``package==version`` for this Python and platform.
407
408    Args:
409        package: ``torch`` or ``torchaudio``.
410        version: The version, without a local tag.
411        tag: ``cu128``, ``cpu``, ...
412        python_version: ``major.minor``.
413        platform: Substrings the wheel's platform tag must all contain, e.g. :data:`LINUX_X86_64`.
414
415    Returns:
416        True when a matching wheel is listed.
417    """
418    cp = "cp" + python_version.replace(".", "")
419    for wheel in _index_wheels(tag, package):
420        if not (
421            wheel.startswith(f"{package}-{version}-")
422            or wheel.startswith(f"{package}-{version}%2B")
423            or wheel.startswith(f"{package}-{version}+")
424        ):
425            continue
426        if f"-{cp}-" in wheel and all(part in wheel for part in platform):
427            return True
428    return False

436def compile_lock(
437    name: str,
438    requirements: list[str],
439    python_version: str,
440    max_cuda_version: Optional[tuple[int, int]] = None,
441    *,
442    exclude_newer: Optional[str] = None,
443    check_torch_index: bool = True,
444    max_steps: int = 12,
445) -> str:
446    """Compile a lock's full text.
447
448    When the resolution includes ``torch``, the chosen ``torch`` / ``torchaudio`` versions must be
449    served by every index in :func:`torch_targets`; otherwise ``torch`` is bounded below the
450    rejected version and the resolution repeated.
451
452    Args:
453        name: The venv name.
454        requirements: The backend's requirement specs.
455        python_version: ``major.minor``.
456        max_cuda_version: The backend's CUDA index ceiling.
457        exclude_newer: An RFC 3339 timestamp; distributions uploaded later are ignored.
458        check_torch_index: Whether to check the PyTorch indexes (needs network).
459        max_steps: How many times ``torch`` may be bounded down.
460
461    Returns:
462        The lock text, header included.
463
464    Raises:
465        VenvLockError: When no ``torch`` within the requirements is served on every target.
466    """
467    with tempfile.TemporaryDirectory(prefix=f"venv-lock-{name}-") as tmp:
468        work = Path(tmp)
469        in_file = work / "requirements.in"
470        in_file.write_text("\n".join([*requirements, *IPC_REQUIREMENTS]) + "\n")
471        extra: list[str] = []
472        notes: list[str] = []
473        for _ in range(max_steps):
474            constraints = work / "constraints.txt"
475            constraints.write_text("\n".join([*QUARANTINED, *extra]) + "\n")
476            annotated = _compile(in_file, python_version, constraints, exclude_newer, hashes=False)
477            packages = resolved_packages(annotated)
478            pins = tuple(f"{p}=={packages[p][0]}" for p in sorted(TORCH_PACKAGES) if p in packages)
479            if not pins or not check_torch_index:
480                break
481            missing = [
482                f"{pin} on {tag}/{platform}"
483                for pin in pins
484                for tag, platform in torch_targets(max_cuda_version)
485                if not wheel_on_index(pin.split("==")[0], pin.split("==")[1], tag, python_version, platform)
486            ]
487            if not missing:
488                break
489            torch_version = packages["torch"][0] if "torch" in packages else None
490            if torch_version is None:
491                raise VenvLockError(f"'{name}': torchaudio without torch is not served: {missing}")
492            notes.append(f"torch {torch_version} skipped: not served as " + "; ".join(missing))
493            extra.append(f"torch<{torch_version}")
494        else:
495            raise VenvLockError(f"'{name}': no torch within the requirements is served on every index: {notes}")
496        no_emit = tuple(torch_only_cuda_packages(packages)) if pins else ()
497        body = _compile(in_file, python_version, constraints, exclude_newer, hashes=True, no_emit=no_emit)
498    uv_version = subprocess.run([_uv(), "--version"], capture_output=True, text=True).stdout.strip()
499    lines = [
500        f"# senselab subprocess-venv lock: {name}",
501        f"# requirements-sha256: {requirements_sha256(requirements, python_version, max_cuda_version)}",
502        f"# python: {python_version}",
503        f"# torch: {' '.join(pins) if pins else 'none'}",
504        f"# torch-index: {top_cuda_tag(max_cuda_version) if pins else 'none'}",
505        f"# exclude-newer: {exclude_newer or 'none'}",
506        f"# uv: {uv_version}",
507        *(f"# note: {note}" for note in notes),
508        f"# Regenerate: uv run python scripts/lock_subprocess_venvs.py {name}",
509    ]
510    return "\n".join(lines) + "\n" + body

def discover_backends( package_root: Optional[pathlib.Path] = None) -> dict[str, Backend]: 
215def discover_backends(package_root: Optional[Path] = None) -> dict[str, Backend]:
216    """Find every ``ensure_venv`` call in senselab and resolve its arguments.
217
218    Each call's arguments are read from the calling module's attributes: a name, a call of a
219    zero-argument function, or a literal.
220
221    Args:
222        package_root: The ``senselab`` package directory; defaults to the installed one.
223
224    Returns:
225        Backends by venv name.
226
227    Raises:
228        VenvLockError: When two calls name one venv with different inputs, or an argument cannot
229            be resolved.
230    """
231    import ast
232    import importlib
233
234    import senselab
235
236    root = package_root or Path(senselab.__file__).parent
237    found: dict[str, Backend] = {}
238    for path in sorted(root.rglob("*.py")):
239        source = path.read_text()
240        if "ensure_venv(" not in source:
241            continue
242        module_name = ".".join(("senselab", *path.relative_to(root).with_suffix("").parts))
243        if module_name in _NOT_BACKENDS:
244            continue
245        calls = [
246            node
247            for node in ast.walk(ast.parse(source))
248            if isinstance(node, ast.Call)
249            and (getattr(node.func, "id", None) or getattr(node.func, "attr", None)) == "ensure_venv"
250        ]
251        if not calls:
252            continue
253        module = importlib.import_module(module_name)
254
255        def value(node: Optional[ast.expr], module: object = module) -> object:
256            if node is None:
257                return None
258            if isinstance(node, ast.Constant):
259                return node.value
260            if isinstance(node, ast.Name):
261                return getattr(module, node.id)
262            if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and not node.args:
263                return getattr(module, node.func.id)()
264            raise VenvLockError(f"{module_name}: cannot resolve ensure_venv argument {ast.unparse(node)}")
265
266        for call in calls:
267            keywords = {k.arg: k.value for k in call.keywords}
268            name = value(call.args[0] if call.args else keywords.get("name"))
269            requirements = value(call.args[1] if len(call.args) > 1 else keywords.get("requirements"))
270            python = value(call.args[2] if len(call.args) > 2 else keywords.get("python_version"))
271            cuda = value(keywords.get("max_cuda_version"))
272            if not isinstance(name, str) or not isinstance(requirements, list):
273                raise VenvLockError(f"{module_name}: ensure_venv needs a str name and a list of requirements")
274            if not isinstance(python, str):
275                raise VenvLockError(f"{module_name}: ensure_venv('{name}') must declare python_version")
276            backend = Backend(
277                name=name,
278                requirements=list(requirements),
279                python=python,
280                max_cuda_version=tuple(cuda) if cuda else None,  # type: ignore[arg-type]
281                modules=[module_name],
282            )
283            if name in found:
284                prior = found[name]
285                if (prior.requirements, prior.python, prior.max_cuda_version) != (
286                    backend.requirements,
287                    backend.python,
288                    backend.max_cuda_version,
289                ):
290                    raise VenvLockError(f"venv '{name}' has different inputs in {prior.modules} and {module_name}")
291                if module_name not in prior.modules:
292                    prior.modules.append(module_name)
293            else:
294                found[name] = backend
295    return found
