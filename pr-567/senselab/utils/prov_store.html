senselab.utils.prov_store

An append-only provenance store, modelled on W3C PROV.

Entities are what the graph believes exists, activities are node executions, agents are what acted. Relations are PROV's own. Nothing is modified after it is added, so merging two stores is a set union and is order-independent.

 49class Agent:
 50    """What acted: a model at a resolved commit, or the software itself.
 51
 52    Attributes:
 53        id: The agent's id.
 54        agent_type: ``"model"`` or ``"software"``.
 55        model_id: The model's identifier, for a model agent.

 57        unresolved_reason: Why the commit is unknown, when it is. A provenance model that cannot say

 59        version: Software version, for a software agent.
 60    """

 70def _digest(payload: object) -> str:
 71    return hashlib.sha256(json.dumps(payload, sort_keys=True, default=str).encode()).hexdigest()[:16]

 74class ProvStore:
 75    """An append-only PROV document.
 76
 77    Args:
 78        run_id: Mixed into every id so two runs never collide.
 79    """
 80
 81    def __init__(self, run_id: str) -> None:
 82        """Create an empty store."""
 83        self.run_id = run_id
 84        self._entities: dict[str, Entity] = {}
 85        self._activities: dict[str, Activity] = {}
 86        self._agents: dict[str, Agent] = {}
 87        self._relations: list[tuple[RELATION, str, str]] = []
 88
 89    def entity(self, *, prov_type: PROV_TYPE, extent: tuple[float, float] | None, attributes: dict[str, Any]) -> str:
 90        """Add an entity.
 91
 92        Args:
 93            prov_type: What sort of thing it is.
 94            extent: ``(start, end)`` in seconds, or None for something without one.
 95            attributes: Whatever describes it.
 96
 97        Returns:
 98            Its id.
 99        """
100        eid = f"{prov_type}-{_digest([self.run_id, prov_type, extent, attributes])}"
101        self._entities[eid] = Entity(id=eid, prov_type=prov_type, extent=extent, attributes=dict(attributes))
102        return eid
103
104    def activity(self, *, node: str, step: str | None, parameters: dict[str, Any]) -> str:
105        """Add an activity.
106
107        Args:
108            node: The node executing.
109            step: Which step of it, when a node has several.
110            parameters: The values it ran with.
111
112        Returns:
113            Its id.
114        """
115        aid = f"act-{_digest([self.run_id, node, step, parameters])}"
116        self._activities[aid] = Activity(id=aid, node=node, step=step, parameters=dict(parameters))
117        return aid

119    def agent(
120        self,
121        *,
122        agent_type: AGENT_TYPE,
123        model_id: str | None = None,
124        commit_sha: str | None = None,
125        unresolved_reason: str | None = None,
126        version: str | None = None,
127    ) -> str:
128        """Add an agent.
129
130        Args:
131            agent_type: ``"model"`` or ``"software"``.
132            model_id: Required for a model agent.
133            commit_sha: A resolved 40-hex commit.
134            unresolved_reason: Why the commit is unknown, if it is.
135            version: Software version, for a software agent.
136
137        Returns:
138            Its id.
139
140        Raises:
141            ValueError: If ``commit_sha`` is not exactly 40 hex characters, if ``commit_sha`` and
142                ``unresolved_reason`` are supplied together, if ``unresolved_reason`` is empty, or if a
143                model agent is missing ``model_id`` or supplies neither a commit nor a reason it is
144                missing.
145        """
146        if commit_sha is not None and not _SHA.fullmatch(commit_sha):
147            raise ValueError(
148                f"commit_sha must be a resolved 40-hex commit, got {commit_sha!r}. A ref recorded as a "
149                "commit makes the provenance confidently wrong."
150            )
151        if commit_sha is not None and unresolved_reason is not None:
152            raise ValueError("commit_sha and unresolved_reason contradict each other; supply exactly one")
153        if unresolved_reason is not None and not unresolved_reason.strip():
154            raise ValueError("unresolved_reason must not be empty; an empty reason says nothing")
155        if agent_type == "model" and model_id is None:
156            raise ValueError("a model agent needs model_id")
157        if agent_type == "model" and commit_sha is None and unresolved_reason is None:
158            raise ValueError("a model agent needs commit_sha or unresolved_reason; silence is not a third option")
159        gid = f"agent-{_digest([self.run_id, agent_type, model_id, commit_sha, unresolved_reason, version])}"
160        self._agents[gid] = Agent(
161            id=gid,
162            agent_type=agent_type,
163            model_id=model_id,
164            commit_sha=commit_sha,
165            unresolved_reason=unresolved_reason,
166            version=version,
167        )
168        return gid
169
170    def _relate(self, relation: RELATION, source: str, target: str) -> None:
171        triple = (relation, source, target)
172        if triple not in self._relations:
173            self._relations.append(triple)
174
175    def was_generated_by(self, entity_id: str, activity_id: str) -> None:
176        """Record that an activity produced an entity."""
177        self._relate("wasGeneratedBy", entity_id, activity_id)

179    def used(self, activity_id: str, entity_id: str) -> None:
180        """Record that an activity read an entity."""
181        self._relate("used", activity_id, entity_id)
182
183    def was_associated_with(self, activity_id: str, agent_id: str) -> None:
184        """Record which agent ran an activity."""
185        self._relate("wasAssociatedWith", activity_id, agent_id)
186
187    def was_attributed_to(self, entity_id: str, agent_id: str) -> None:
188        """Record which agent is answerable for an entity."""
189        self._relate("wasAttributedTo", entity_id, agent_id)
190
191    def was_derived_from(self, entity_id: str, source_entity_id: str) -> None:
192        """Record that an entity refines or answers another, keeping both."""
193        self._relate("wasDerivedFrom", entity_id, source_entity_id)
194
195    def was_invalidated_by(self, entity_id: str, activity_id: str) -> None:
196        """Record that an entity should no longer be read as what it was. It is not removed."""
197        self._relate("wasInvalidatedBy", entity_id, activity_id)

207    def entities(self, prov_type: PROV_TYPE | None = None) -> list[Entity]:
208        """Return entities, optionally of one type."""
209        return [e for e in self._entities.values() if prov_type is None or e.prov_type == prov_type]

211    def _targets(self, relation: RELATION, source: str) -> list[str]:
212        return [t for r, s, t in self._relations if r == relation and s == source]
213
214    def generated_by(self, entity_id: str) -> str | None:
215        """Return the activity that generated an entity, or None."""
216        found = self._targets("wasGeneratedBy", entity_id)
217        return found[0] if found else None
218
219    def uses_of(self, activity_id: str) -> list[str]:
220        """Return the entities an activity read."""
221        return self._targets("used", activity_id)
222
223    def associated_with(self, activity_id: str) -> list[str]:
224        """Return the agents associated with an activity."""
225        return self._targets("wasAssociatedWith", activity_id)
226
227    def derived_from(self, entity_id: str) -> list[str]:
228        """Return the entities an entity was derived from."""
229        return self._targets("wasDerivedFrom", entity_id)

231    def is_invalidated(self, entity_id: str) -> bool:
232        """Whether an entity has been invalidated."""
233        return bool(self._targets("wasInvalidatedBy", entity_id))

235    def write_jsonl(self, path: str | Path) -> None:
236        """Write the store as one PROV-JSON-shaped record per line."""
237        lines = [
238            json.dumps({"record": "entity", **asdict(e)}, sort_keys=True, default=str) for e in self._entities.values()
239        ]
240        lines += [
241            json.dumps({"record": "activity", **asdict(a)}, sort_keys=True, default=str)
242            for a in self._activities.values()
243        ]
244        lines += [
245            json.dumps({"record": "agent", **asdict(g)}, sort_keys=True, default=str) for g in self._agents.values()
246        ]
247        lines += [
248            json.dumps({"record": "relation", "relation": r, "source": s, "target": t}, sort_keys=True)
249            for r, s, t in self._relations
250        ]
251        Path(path).write_text("\n".join(lines) + "\n")

253    @classmethod
254    def read_jsonl(cls, path: str | Path, run_id: str = "read") -> "ProvStore":
255        """Read a store back.
256
257        Raises:
258            ValueError: If a line carries an unrecognised ``record`` kind.
259        """
260        store = cls(run_id=run_id)
261        for line in Path(path).read_text().splitlines():
262            if not line.strip():
263                continue
264            rec = json.loads(line)
265            kind = rec.pop("record")
266            if kind == "entity":
267                extent = rec.pop("extent")
268                store._entities[rec["id"]] = Entity(extent=tuple(extent) if extent else None, **rec)
269            elif kind == "activity":
270                store._activities[rec["id"]] = Activity(**rec)
271            elif kind == "agent":
272                store._agents[rec["id"]] = Agent(**rec)
273            elif kind == "relation":
274                store._relations.append((rec["relation"], rec["source"], rec["target"]))
275            else:
276                raise ValueError(f"unrecognised record kind {kind!r} in {path}")
277        return store

279    @classmethod
280    def merge(cls, stores: Sequence["ProvStore"]) -> "ProvStore":
281        """Union several stores. Append-only makes this order-independent."""
282        out = cls(run_id="merged")
283        for s in stores:
284            out._entities.update(s._entities)
285            out._activities.update(s._activities)
286            out._agents.update(s._agents)
287        seen: set[tuple[RELATION, str, str]] = set()
288        for s in stores:
289            for rel in s._relations:
290                if rel not in seen:
291                    seen.add(rel)
292                    out._relations.append(rel)
293        return out

295    def fingerprint(self) -> str:
296        """A content hash that ignores insertion order."""
297        return _digest(
298            {
299                "e": sorted(self._entities),
300                "act": sorted(self._activities),
301                "ag": sorted(self._agents),
302                "r": sorted(f"{r}:{s}:{t}" for r, s, t in self._relations),
303            }
304        )
