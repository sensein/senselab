senselab.utils.tasks.cached_inference

Content-addressable caching for expensive model inference (T051).

Lifted out of scripts/analyze_audio.py so the cache contract is importable, unit-testable, and reusable by the adaptive loop rather than living in a 2500-line CLI script.

A cache entry is keyed on everything that can change the result:

(schema version, audio signature, task, model id, params, code version, senselab version)

code_version is a caller-supplied string identifying the behavior that produced an entry. It deliberately replaced an earlier wrapper_hash that was a sha256 of the CLI script's own source. Source hashing was the wrong tool: editing a comment, a docstring, or letting ruff-format rotate a line invalidated every cached model result, while the blast radius grew as more stages shared one file. Callers now pass a coarse, hand-managed identifier (see STAGE_VERSIONS in workflows/audio_analysis/stage_context.py) whose counterpart obligation is explicit: bump a stage's version when the stored shape of its outcome changes.

senselab_version still participates, and covers the larger surface — most stages are thin pass-throughs to a tasks/ API, so library-side changes are what usually matter.

Cache keys are NOT stable across senselab versions and are not intended to be: CACHE_SCHEMA_VERSION is the deliberate global invalidation lever, and sync_cache_with_schema_version() wipes stale entries automatically on every host rather than requiring anyone to delete a directory by hand.

268def serialize(obj: Any) -> Any:  # noqa: ANN401 — recursive heterogeneous serializer
269    """Convert senselab outputs (ScriptLine, tensor, etc.) to JSON-friendly types.
270
271    ``torch`` is imported lazily so that merely deriving a cache key or writing a
272    JSON sidecar does not pull in the ML stack — which in turn keeps
273    ``stage_context`` importable without torch (there is a test asserting that).
274    After the first call it is a ``sys.modules`` lookup.
275    """
276    import torch
277
278    if isinstance(obj, dict):
279        return {k: serialize(v) for k, v in obj.items()}
280    if isinstance(obj, (list, tuple)):
281        return [serialize(x) for x in obj]
282    if isinstance(obj, torch.Tensor):
283        return {
284            "_tensor_shape": list(obj.shape),
285            "_dtype": str(obj.dtype),
286            "values": obj.detach().cpu().tolist(),
287        }
288    if hasattr(obj, "model_dump"):
289        return serialize(obj.model_dump())
290    if hasattr(obj, "__dict__") and not isinstance(obj, type):
291        return {k: serialize(v) for k, v in vars(obj).items() if not k.startswith("_")}
292    if isinstance(obj, (str, int, float, bool)) or obj is None:
293        return obj
294    return repr(obj)

302@runtime_checkable
303class _HasWaveform(Protocol):
304    """Structural type for :func:`audio_signature`.
305
306    Typed structurally rather than as ``senselab.audio.data_structures.Audio``
307    on purpose: ``utils/`` must not import from ``audio/`` (that would invert the
308    dependency direction and risk an import cycle), and it lets callers pass any
309    waveform-carrying object — which the tests do.
310    """
311
312    @property
313    def waveform(self) -> Any: ...  # noqa: ANN401 — a torch.Tensor in practice
314
315    @property
316    def sampling_rate(self) -> int: ...
317
318
319def audio_signature(audio: _HasWaveform) -> str:
320    """Return a deterministic sha256 of the audio waveform PCM + sampling rate.
321
322    Two identical-sounding files produce the same signature regardless of
323    their on-disk format (e.g., WAV vs FLAC) — what matters is the post-
324    resample, post-downmix waveform that each task actually sees. Extra
325    metadata (file path, mtime, encoding) is intentionally excluded.
326
327    This is the join key between a run's ``summary.json``
328    (``passes[label].audio_signature``) and each cache entry's
329    ``provenance.audio_signature``; the adaptive loop indexes cached results on
330    it, so the two must be produced by this one function.
331    """
332    arr = audio.waveform.detach().cpu().contiguous().numpy()
333    h = hashlib.sha256()
334    h.update(str(audio.sampling_rate).encode())
335    h.update(b"|")
336    h.update(str(arr.shape).encode())
337    h.update(b"|")
338    h.update(arr.tobytes())
339    return h.hexdigest()

342def write_json(path: Path, payload: Any) -> None:  # noqa: ANN401 — heterogeneous senselab outputs
343    """Write a JSON file with senselab-aware serialization.

353def cache_key(
354    *,
355    audio_sig: str,
356    task: str,
357    model_id: str | None,
358    params: dict[str, Any],
359    code_version: str,
360    senselab_ver: str,
361    commit_sha: str | None,
362) -> str:
363    """Compute the deterministic cache key for one (audio, task, model, params, commit) combo.
364
365    ``commit_sha`` is keyword-only and has no default on purpose: an omitted argument must fail
366    mypy/type-checking rather than silently keying on ``None`` for every commit of a model. A
367    default is exactly how the staleness bug this parameter fixes would come back — a caller could
368    forget to pass it and still get a key, just one that can't tell two commits apart.
369
370    Args:
371        audio_sig: Output of :func:`audio_signature` for the clip.
372        task: The stage name, e.g. ``"asr"``.
373        model_id: The Hub id or backend name, or ``None`` for a model-less stage.
374        params: The call's other keyword arguments, canonicalized.
375        code_version: Caller-supplied wrapper-behavior version (see module docstring).
376        senselab_ver: Installed senselab version.
377        commit_sha: The immutable 40-hex commit this call resolved ``model_id`` to, or ``None``
378            when ``model_id`` names no Hub repo (a local backend name, or no model at all).
379
380    Returns:
381        A 64-character hex sha256 digest.
382    """
383    payload = {
384        "schema": CACHE_SCHEMA_VERSION,
385        "audio_signature": audio_sig,
386        "task": task,
387        "model": model_id,
388        "params": params,
389        "code_version": code_version,
390        "senselab_version": senselab_ver,
391        # Without this, an upstream push to a tracked ref loads new weights under
392        # an unchanged key and a stale result is served as current.
393        "commit_sha": commit_sha,
394    }
395    return hashlib.sha256(canonical_params(payload).encode()).hexdigest()

398def align_cache_key(
399    *,
400    audio_sig: str,
401    transcript_sha: str,
402    language: str | None,
403    aligner_model_id: str,
404    aligner_params: dict[str, Any],
405    code_version: str,
406    senselab_ver: str,
407    aligner_commit_sha: str | None,
408) -> str:
409    """Cache key for one (audio, transcript, language, aligner, commit) alignment call.
410
411    Independent from the ASR cache: an alignment cache hit replays prior
412    timestamps without invoking the aligner; an ASR-cache miss + alignment-cache
413    hit (or vice versa) is supported by construction.
414
415    ``aligner_commit_sha`` is keyword-only and has no default, mirroring :func:`cache_key`'s
416    ``commit_sha``: this payload carried only the ref-shaped ``aligner_model_id`` and no commit,
417    so an upstream push to a forced-aligner repo (e.g. an MMS or Qwen aligner checkpoint) served
418    timestamps computed by the old commit under an unchanged key. A default would let a caller
419    forget to pass it and reopen exactly that hole.
420
421    Args:
422        audio_sig: Output of :func:`audio_signature` for the clip.
423        transcript_sha: Output of :func:`transcript_signature` for the text being aligned.
424        language: ISO language code passed to the aligner, or ``None``.
425        aligner_model_id: The aligner's Hub id or backend name.
426        aligner_params: The aligner's other keyword arguments, canonicalized.
427        code_version: Caller-supplied wrapper-behavior version.
428        senselab_ver: Installed senselab version.
429        aligner_commit_sha: The immutable 40-hex commit ``aligner_model_id`` resolved to, or
430            ``None`` when it names no Hub repo.
431
432    Returns:
433        A 64-character hex sha256 digest.
434    """
435    payload = {
436        "schema": CACHE_SCHEMA_VERSION,
437        "audio_signature": audio_sig,
438        "task": "alignment",
439        "transcript_sha": transcript_sha,
440        "language": language,
441        "aligner_model": aligner_model_id,
442        "aligner_params": aligner_params,
443        "code_version": code_version,
444        "senselab_version": senselab_ver,

483def prune_unreachable_entries(cache_dir: Path, *, senselab_ver: str) -> int:
484    """Delete cache entries no current key can ever hit; return how many were removed.
485
486    ``senselab_version`` and ``code_version`` are *inside* the cache key, so a
487    senselab release orphans every entry and a ``STAGE_VERSIONS`` bump orphans that
488    stage's. Nothing previously reclaimed them: ``CACHE_SCHEMA_VERSION`` only wipes
489    on a schema change, so the directory grew monotonically across releases and a
490    cache that looked healthy could be entirely dead weight.
491
492    An entry is unreachable when its recorded ``provenance.senselab_version``
493    differs from the running one, or its ``provenance.code_version`` no longer
494    matches the declared version for that task. Entries without provenance are
495    kept — absence of evidence isn't evidence of staleness, and a hit on them is
496    still correct.
497
498    Args:
499        cache_dir: The cache directory.
500        senselab_ver: The running senselab version.
501
502    Returns:
503        Number of entries removed.
504    """
505    try:
506        from senselab.audio.workflows.audio_analysis.stage_context import stage_code_version
507    except ImportError:  # pragma: no cover — stage versions live in the audio workflow
508        return 0
509
510    removed = 0
511    for entry in cache_dir.glob("*.json"):
512        try:
513            prov = (json.loads(entry.read_text(encoding="utf-8")) or {}).get("provenance") or {}
514        except (json.JSONDecodeError, OSError):
515            continue  # corrupt entries already read as a miss; leave them to be overwritten
516        if not prov:
517            continue
518        recorded_ver = prov.get("senselab_version")
519        stale = recorded_ver is not None and recorded_ver != senselab_ver
520        if not stale:
521            task = prov.get("task")
522            recorded_code = prov.get("code_version")
523            if task and recorded_code is not None:
524                try:
525                    stale = recorded_code != stage_code_version(str(task))
526                except KeyError:
527                    stale = True  # task no longer declares a version → unreachable
528        if stale:
529            try:
530                entry.unlink()
531                removed += 1
532            except OSError:
533                continue
534    if removed:
535        print(
536            f"Cache: pruned {removed} unreachable entr{'y' if removed == 1 else 'ies'} "
537            f"(senselab/stage version drift) in {cache_dir}",
538            file=sys.stderr,
539        )
540    return removed

543def sync_cache_with_schema_version(cache_dir: Path) -> None:
544    """Keep the on-disk cache state and :data:`CACHE_SCHEMA_VERSION` in sync.
545
546    The cache directory carries a ``.schema_version`` marker file. On each run:
547
548    - If the directory is empty / missing the marker → the cache was just
549      created (or manually cleared). Write the current schema version. No
550      data wipe is needed because there's nothing to wipe.
551    - If the marker exists and matches the current code version → keep cache.
552    - If the marker exists but doesn't match → the code has bumped the
553      schema since the cache was populated. Wipe all cache entries and
554      rewrite the marker with the current version.
555
556    Bidirectional invariant: clearing the cache resets the version to current
557    automatically (since the marker is recreated); bumping the version in
558    code wipes the cache automatically (since the marker mismatch triggers
559    the wipe). The user never has to manually delete cache files when they
560    edit the schema number.
561    """
562    cache_dir.mkdir(parents=True, exist_ok=True)
563    marker = cache_dir / ".schema_version"
564    on_disk_version: int | None = None
565    if marker.exists():
566        try:
567            on_disk_version = int(marker.read_text().strip())
568        except (ValueError, OSError):
569            on_disk_version = None
570
571    # Has the cache been populated with non-marker entries?
572    has_entries = any(p.name != ".schema_version" for p in cache_dir.iterdir())
573
574    if on_disk_version == CACHE_SCHEMA_VERSION:
575        prune_unreachable_entries(cache_dir, senselab_ver=senselab_version())
576        return
577
578    if on_disk_version is None and not has_entries:
579        # Fresh / cleared cache. Write current version, no wipe needed.
580        marker.write_text(str(CACHE_SCHEMA_VERSION))
581        print(
582            f"Cache: initialized {cache_dir} at schema_version={CACHE_SCHEMA_VERSION}",
583            file=sys.stderr,
584        )
585        return
586
587    # Mismatch — wipe and rewrite the marker.
588    n_removed = 0
589    for p in cache_dir.iterdir():
590        if p.name == ".schema_version":
591            continue
592        try:
593            if p.is_dir():
594                shutil.rmtree(p)
595            else:
596                p.unlink()
597            n_removed += 1
598        except OSError:
599            continue
600    marker.write_text(str(CACHE_SCHEMA_VERSION))
601    print(
602        f"Cache: schema_version {on_disk_version} → {CACHE_SCHEMA_VERSION}; "
603        f"wiped {n_removed} stale entr{'y' if n_removed == 1 else 'ies'} in {cache_dir}",
604        file=sys.stderr,
605    )

612    name: str,
613    fn: Any,  # noqa: ANN401 — generic dispatcher
614    *args: Any,  # noqa: ANN401
615    **kwargs: Any,  # noqa: ANN401
616) -> dict[str, Any]:
617    """Run a task with timing + structured error capture.
618
619    Never raises: a failing model becomes ``{"status": "failed", ...}`` with the

624    started = time.perf_counter()

626        result = fn(*args, **kwargs)

632            "elapsed_s": round(elapsed, 3),
633            "error": repr(exc),
634            "traceback": traceback.format_exc(limit=5),
635        }
636    elapsed = time.perf_counter() - started
637    print(f"  [{name}] ok in {elapsed:.1f}s", flush=True)
638    return {"status": "ok", "elapsed_s": round(elapsed, 3), "result": result}

652

654    not invoked. On a miss the task runs and, **only if it succeeded**, the
