senselab.utils.reproduce.cache

Step keys, content-addressed result dirs, the completion marker, run claims and dated run links.

step_key is the sha256 of canonical_json over step_key_payload: everything that decides a step's result, and nothing volatile (job ids, visible GPUs, TMPDIR, HOME, the date, where the data roots or results live, the partition and limits, min_driver, the step's own name or its repository URL). native_digests and driver_env_sha256 compute two of its run-time inputs, and reproduce_source_sha256 stands for the driver's own behaviour. A step's outputs live in result_dir = <results_root>/<step_key>. The dir is complete (is_done) once its summary.json exists with "exit": 0; write_summary writes that file whole or not at all, and is the last write of a run. claim holds a SharedFileLock on <results_root>/.claims/<step_key> so two jobs never compute one step at once. link_run points <runs_root>/<YYYY-MM-DD>/<analysis>/<step> at the step's result dir with a relative symlink.

Errors are a CacheError whose message is a fixed code only.

124def native_digests(step: StepSpec, profile: Profile) -> dict[str, list[str]]:
125    """Digest the content of every native dir a step selects, for `step_key`'s ``native``.
126
127    Args:
128        step: The step.
129        profile: The caller's profile.
130
131    Returns:
132        Each name in ``step.resources.native`` mapped to one `digest.input_set_digest` hex per dir
133        (``bin_dirs`` then ``lib_dirs``), over the files the entry's ``bin_patterns`` or
134        ``lib_patterns`` select, leaving out `NATIVE_EXCLUDE` (``__pycache__`` dirs and ``*.pyc``).
135
136    Raises:
137        CacheError: ``E_KEY_NATIVE`` if the step selects a native entry the profile lacks.
138        DigestError: As `digest.input_set_digest` with ``require_nonempty``: ``E_DIGEST_EMPTY`` when a dir
139            matches nothing, ``E_DIGEST_UNREADABLE`` for a dangling symlink or unreadable file among the
140            matches.
141    """
142    if not set(step.resources.native) <= set(profile.native):
143        raise CacheError("E_KEY_NATIVE")
144    return {
145        name: [
146            input_set_digest(d, patterns, exclude_patterns=NATIVE_EXCLUDE, require_nonempty=True)[0]
147            for d, patterns in _native_dirs(profile, name)
148        ]
149        for name in step.resources.native
150    }

154def reproduce_source_sha256() -> str:
155    """The identity of the driver's behaviour: a sha256 over the installed ``senselab.utils.reproduce`` sources.
156
157    Computed once per process, over every ``*.py`` file under the package directory (this module's
158    own directory, i.e. ``Path(senselab.utils.reproduce.__file__).parent``), as sorted pairs of
159    relative POSIX path and content sha256.
160
161    Returns:
162        A hex sha256.
163    """
164    root = Path(__file__).resolve().parent
165    files = sorted(p.relative_to(root).as_posix() for p in root.rglob("*.py") if "__pycache__" not in p.parts)
166    pairs = [[rel, hashlib.sha256((root / rel).read_bytes()).hexdigest()] for rel in files]
167    return hashlib.sha256(canonical_json({"schema": REPRODUCE_SOURCE_SCHEMA, "files": pairs})).hexdigest()

175def _runtime_closure(installed: Mapping[str, Distribution]) -> set[str]:
176    """`DRIVER_PACKAGE`'s installed runtime dependency closure, transitively.
177
178    `DRIVER_PACKAGE`'s own requirements are taken with no extra; a requirement that asks for extras
179    (``dep[x]``) also brings in ``dep``'s requirements for those extras.
180    """
181    seen: set[tuple[str, str]] = set()
182    names: set[str] = set()
183    todo: list[tuple[str, frozenset[str]]] = [(DRIVER_PACKAGE, frozenset())]
184    while todo:
185        name, extras = todo.pop()
186        wanted = {"", *extras} - {extra for seen_name, extra in seen if seen_name == name}
187        if name not in installed or not wanted:
188            continue
189        seen.update((name, extra) for extra in wanted)
190        names.add(name)
191        for text in installed[name].requires or []:
192            try:
193                req = Requirement(text)
194            except InvalidRequirement:
195                continue
196            if req.marker is None or any(req.marker.evaluate({"extra": extra}) for extra in wanted):
197                todo.append((canonicalize_name(req.name), frozenset(canonicalize_name(e) for e in req.extras)))
198    return names

201def driver_env_sha256() -> str:
202    """The identity of the driver's own interpreter env, for an adapter that runs in it.
203
204    Only the interpreter's own ``purelib`` and ``platlib`` dirs are read (never ``sys.path``, the cwd or
205    ``PYTHONPATH``), and only `DRIVER_PACKAGE`'s runtime dependency closure there: its requirements
206    without an extra, followed transitively along with the extras each requirement asks for.
207    `DRIVER_PACKAGE` itself is left out; its reproduce code is `reproduce_source_sha256`.
208
209    Returns:
210        A hex sha256 over the interpreter's implementation, version and free-threading build flag
211        (``platform.python_implementation()``, ``platform.python_version()``, ``Py_GIL_DISABLED``; not
212        ``sys.version``, whose build date and compiler differ between builds of one release) and the sorted
213        ``name==version`` of that closure.
214    """
215    installed: dict[str, Distribution] = {}
216    for dist in distributions(path=_site_dirs()):
217        name = dist.metadata["Name"] if dist.read_text("METADATA") else None
218        if name:
219            installed.setdefault(canonicalize_name(name), dist)
220    closure = _runtime_closure(installed) - {DRIVER_PACKAGE}
221    names = sorted(f"{name}=={installed[name].version}" for name in closure)
222    python = [
223        platform.python_implementation(),
224        platform.python_version(),
225        bool(sysconfig.get_config_var("Py_GIL_DISABLED")),
226    ]
227    return hashlib.sha256(canonical_json({"python": python, "distributions": names})).hexdigest()

230def step_key_payload(
231    step: StepSpec,
232    profile: Profile,
233    *,
234    upstream: Mapping[str, str],
235    submodules: Mapping[str, str],
236    adapter_sha256: str,
237    adapter_env_sha256: str,
238    input_digests: Sequence[str],
239    native: Mapping[str, Sequence[str]],
240    sources_lock_sha256: str,
241    r_session_sha256: str | None = None,
242    r_module_env_sha256: str | None = None,
243    shard: int | None = None,
244) -> dict[str, Any]:
245    """The object `step_key` hashes.
246
247    Args:
248        step: The step.
249        profile: The caller's profile; only the step's keyed env (`sandbox.keyed_templates`) and its
250            ``policy`` (as `policy_sha256`) are taken from it.
251        upstream: Each name in ``step.needs`` mapped to that step's key, and nothing else.
252        submodules: Submodule path to 40-hex commit, as the checkout records it; empty when the step
253            does not use submodules.
254        adapter_sha256: The sha256 of the step's adapter script.
255        adapter_env_sha256: The identity of the interpreter env that runs the adapter: the step venv's
256            ``installed_sha256``, or `driver_env_sha256` when the adapter runs in the driver's env.
257        input_digests: One `digest.input_set_digest` hex per ``step.inputs`` entry, in manifest order.
258        native: Each name in ``step.resources.native`` mapped to `native_digests`' digests for it (one per
259            ``bin_dirs`` then ``lib_dirs`` entry of the profile's `NativeSpec`), and nothing else.
260        sources_lock_sha256: The sha256 of the manifest's sources lock (`lock.load_lock`); with the step's
261            ``lock_inputs`` ids it stands for the registered inputs, whose files are not read for the key.
262        r_session_sha256: The sha256 of the prefetch ``SESSION.json``; required for ``r`` and
263            ``ir_notebook`` steps and refused for others.
264        r_module_env_sha256: The sha256 of the captured R module env; required and refused likewise.
265        shard: The shard index of a sharded step (``0 <= shard < shards.count``), keyed as ``[shard, count]``;
266            required for a sharded step and refused for others.
267
268    Returns:
269        A JSON-serialisable dict, tagged with `STEP_KEY_SCHEMA`.
270
271    Raises:
272        CacheError: ``E_KEY_UPSTREAM`` if ``upstream`` is not exactly the needs with hex keys,
273            ``E_KEY_INPUTS`` if the input digests do not match ``step.inputs`` one to one,
274            ``E_KEY_ADAPTER`` for an adapter or adapter-env sha256 that is not hex, ``E_KEY_NATIVE`` if
275            ``native`` does not give one hex digest per selected native dir, ``E_KEY_SUBMODULES`` for a commit
276            that is not 40-hex or submodules on a step that disables them, ``E_KEY_R_SESSION`` for R
277            digests missing on an R step or given on another, ``E_KEY_NO_WHEELS`` for an env with
278            sdist builds and no recorded ``wheels.lock``, ``E_KEY_LOCK`` for a lock sha256 that is not hex,
279            ``E_KEY_SHARD`` for a shard index missing, out of range, or given for a step with no shards.
280        ValueError: As `sandbox.keyed_templates`.
281    """
282    if not _is_sha256(sources_lock_sha256):
283        raise CacheError("E_KEY_LOCK")
284    if set(upstream) != set(step.needs) or not all(_is_sha256(k) for k in upstream.values()):
285        raise CacheError("E_KEY_UPSTREAM")
286    if len(input_digests) != len(step.inputs) or not all(_is_sha256(d) for d in input_digests):
287        raise CacheError("E_KEY_INPUTS")
288    if not (_is_sha256(adapter_sha256) and _is_sha256(adapter_env_sha256)):
289        raise CacheError("E_KEY_ADAPTER")
290    selected = step.resources.native
291    if set(native) != set(selected) or not all(
292        name in profile.native
293        and len(native[name]) == len(_native_dirs(profile, name))
294        and all(_is_sha256(d) for d in native[name])
295        for name in selected
296    ):
297        raise CacheError("E_KEY_NATIVE")
298    if (submodules and not step.submodules) or not all(
299        isinstance(p, str) and p and isinstance(c, str) and _COMMIT.match(c) for p, c in submodules.items()
300    ):
301        raise CacheError("E_KEY_SUBMODULES")
302    r_digests = (r_session_sha256, r_module_env_sha256)
303    if step.kind in R_KINDS:
304        if not all(_is_sha256(d) for d in r_digests):
305            raise CacheError("E_KEY_R_SESSION")
306    elif any(d is not None for d in r_digests):
307        raise CacheError("E_KEY_R_SESSION")
308    count = None if step.shards is None else step.shards.count
309    if (count is None) != (shard is None) or (count is not None and shard is not None and not 0 <= shard < count):
310        raise CacheError("E_KEY_SHARD")
311    env = step.env
312    if env.sdist_allow and env.wheels is None:
313        raise CacheError("E_KEY_NO_WHEELS")
314    return {
315        "schema": STEP_KEY_SCHEMA,
316        "reproduce_source_sha256": reproduce_source_sha256(),
317        "commit": step.commit,
318        "submodules": dict(sorted(submodules.items())),
319        "patch_set_sha256": patch_set_sha256(step.patches),
320        "env_key": env_key(env),
321        "wheels_lock_sha256": None if env.wheels is None else env.wheels.lock_sha256,
322        "overlay_lock_sha256s": {
323            "test": None if env.test is None else env.test.lock_sha256,
324            "notebook_tools": None if env.notebook_tools is None else env.notebook_tools.lock_sha256,
325        },
326        "env": keyed_templates(step, profile),
327        "kind": step.kind,
328        "argv": list(step.argv),
329        "cwd": step.cwd,
330        "inputs": [{"root": i.root, "patterns": sorted(set(i.patterns))} for i in step.inputs],
331        "input_digests": list(input_digests),
332        "stage": [{"root": s.root, "patterns": sorted(set(s.patterns)), "dest": s.dest} for s in step.stage],
333        "outputs": list(step.outputs),
334        "native": {name: list(native[name]) for name in sorted(selected)},
335        "gpu_arch": None if step.resources.gpu_arch is None else sorted(set(step.resources.gpu_arch)),
336        "adapter_sha256": adapter_sha256,
337        "adapter_env_sha256": adapter_env_sha256,
338        "hf_weights": [
339            {
340                "repo": w.repo,
341                "repo_type": w.repo_type,
342                "revision": w.revision,
343                "files": sorted(([f.path, f.sha256] for f in w.files)),
344            }
345            for w in sorted(step.hf_weights, key=lambda w: (w.repo, w.repo_type))
346        ],
347        "url_weights": sorted([w.file, w.sha256, w.place] for w in step.url_weights),
348        "r_session_sha256": r_session_sha256,
349        "r_module_env_sha256": r_module_env_sha256,
350        "metrics_schema": step.metrics_schema.model_dump(mode="json"),
351        "expect": [e.model_dump(mode="json") for e in step.expect],
352        "extract": [e.model_dump(mode="json") for e in step.extract],
353        "replicate": step.replicate,
354        "upstream": dict(sorted(upstream.items())),
355        "policy_sha256": policy_sha256(profile),
356        "sources_lock_sha256": sources_lock_sha256,
357        "lock_inputs": sorted(step.lock_inputs),
358        "restricted": step.restricted,
359        "shard": None if shard is None else [shard, count],
360        "resumable": step.resumable,
361        "resume_dir": step.resume_dir,
362        "engine": _engine_key(step),
363        "draw": None if step.draw is None or step.draw.is_empty() else step.draw.model_dump(mode="json"),
364    }

384def step_key(
385    step: StepSpec,
386    profile: Profile,
387    *,
388    upstream: Mapping[str, str],
389    submodules: Mapping[str, str],

400
401    Args:
402        step: The step.

474def is_done(out: Path) -> bool:
475    """Whether a result dir is complete: its ``summary.json`` is a JSON object with integer ``exit`` 0.
476
477    Args:
478        out: A result dir.
479
480    Returns:
481        True only for a complete, successful run; a missing dir, missing or unreadable summary, a
482        non-zero or non-integer exit is False.
483    """
484    try:
485        summary = json.loads((out / SUMMARY).read_text())
486    except (OSError, ValueError):
487        return False
488    return isinstance(summary, dict) and type(summary.get("exit")) is int and summary["exit"] == 0

511        TypeError: For a value JSON cannot encode; nothing is written.
512        OSError: If the write or rename fails; the temporary file is removed and any earlier summary
513            is left as it was.

521            os.write(fd, text.encode())
522            os.fsync(fd)
523        finally:
524            os.close(fd)
525        os.replace(tmp, target)
526    except BaseException:
527        tmp.unlink(missing_ok=True)
528        raise

575def reset_result_dir(results_root: Path, key: str, *, keep: str | None = None, keep_attempts: bool = False) -> Path:
576    """Give a step an empty result dir, removing what an incomplete or failed run left; call under `claim`.
577
578    Args:
579        results_root: The profile's results root.
580        key: The step key.
581        keep: A dir relative to the result dir (an attempt's ``work/<resume_dir>``) kept through the wipe: moved
582            to `RESUME_KEEP` and back, only when every component is a real directory. A `RESUME_KEEP` left by a
583            reset that died is restored the same way.
584        keep_attempts: Keep `ATTEMPTS`, the per-attempt record.
585
586    Returns:
587        The step's result dir (`result_dir`), now empty but for what was kept.
588
589    A dir is wiped only when its ``summary.json`` is absent or records a non-zero integer exit.
590
591    Raises:
592        CacheError: ``E_KEY_BAD`` for a malformed key, ``E_RESULT_NOT_DIR`` if a symlink or non-directory
593            holds the path, ``E_RESULT_DONE`` if the dir is complete, ``E_RESULT_UNREADABLE`` if the
594            summary cannot be read or decoded or is not an object with an integer exit (nothing is deleted then),
595            ``E_RESULT_RESET`` if removing or creating it fails.
596    """
597    out = result_dir(results_root, key)
598    if out.is_symlink() or (os.path.lexists(out) and not out.is_dir()):
599        raise CacheError("E_RESULT_NOT_DIR")
600    if out.is_dir():
601        try:
602            text = (out / SUMMARY).read_text()
603        except FileNotFoundError:
604            text = None
605        except (OSError, ValueError):
606            raise CacheError("E_RESULT_UNREADABLE") from None
607        if text is not None:
608            try:
609                summary = json.loads(text)
610            except ValueError:
611                raise CacheError("E_RESULT_UNREADABLE") from None
612            if not isinstance(summary, dict) or type(summary.get("exit")) is not int:
613                raise CacheError("E_RESULT_UNREADABLE")
614            if summary["exit"] == 0:
615                raise CacheError("E_RESULT_DONE")
616    try:
617        if out.exists() and (keep is not None or keep_attempts):
618            _wipe_keeping(out, keep, keep_attempts)
619            return out
620        if out.exists():
621            shutil.rmtree(out)
622        out.mkdir(parents=True)
623    except OSError:
624        raise CacheError("E_RESULT_RESET") from None
625    return out

629def claim(results_root: Path, key: str, *, timeout: float = CLAIM_TIMEOUT_S) -> Iterator[tuple[Path, bool]]:
630    """Hold the exclusive claim on one step key while it is checked, computed and marked done.
631
632    The lock files are ``<results_root>/.claims/<key>.{lock,heartbeat,holder}``; the results root's own
633    mode is never changed. ``leftover`` is the lock's own takeover verdict (`SharedFileLock.took_over`): an
634    identity in ``.holder`` read before an uncontended acquire means a holder that never released, i.e. one that
635    was killed, since a clean release clears ``.holder`` and a waiter's acquire is contended.
636
637    Args:
638        results_root: The profile's results root.
639        key: The step key.
640        timeout: Seconds to wait for a live holder before giving up.
641
642    Yields:
643        ``(out, leftover)``: the step's result dir (`result_dir`), and whether a killed holder's identity was
644        left. Re-check `is_done` inside the claim, since another job may have completed the step while this
645        one waited.
646
647    Raises:
648        CacheError: ``E_KEY_BAD`` for a malformed key, ``E_CLAIM_HELD`` if another live process held
649            the claim for the whole timeout.
650    """
651    out = result_dir(results_root, key)
652    claims = results_root / CLAIMS
653    claims.mkdir(parents=True, exist_ok=True)
654    lock = SharedFileLock(claims / key, timeout=timeout, manage_dir_mode=False)
655    try:
656        lock.__enter__()
657    except TimeoutError:
658        raise CacheError("E_CLAIM_HELD") from None
659    try:
660        yield out, lock.took_over
661    finally:
662        lock.__exit__(None, None, None)
