senselab.utils.reproduce.lock

Sources locks: one file per run listing the data, models, URLs, code and env locks its steps may use.

A manifest references exactly one lock (spec.SourcesRef), <base_dir>/sources/<name>.lock.yaml, by path and sha256; the lock's sha256 joins every step key. load_lock reads it, step_findings checks a step's declared pins against it, resolve_inputs reads the registry entries of a step's lock_inputs and checks they are the ones the lock pinned, and lock_findings lints a lock alone.

Errors are a LockError whose message is a fixed code only.

112class SourcesLock(_Model):
113    """What a run may use: data and external inputs, models, URL files, code and env locks."""

115    model_config = ConfigDict(extra="forbid", populate_by_name=True)

117    lock_schema: Literal["reproduce-sources-lock-1"] = Field(alias="schema")

125    @model_validator(mode="after")
126    def _unique(self) -> SourcesLock:
127        ids = [e.id for e in self.registered()]
128        if len(set(ids)) != len(ids):
129            raise ValueError("registered input ids repeat")
130        models = [m.id for m in self.models]

132            raise ValueError("model ids repeat")
133        envs = [e.lock for e in self.envs]
134        if len(set(envs)) != len(envs):
135            raise ValueError("env locks repeat")

138    def registered(self) -> list[RegisteredEntry]:
139        """The registered entries of ``data`` and ``external``."""
140        return [e for e in self.data if isinstance(e, RegisteredEntry)] + list(self.external)

155def load_lock(base_dir: Path, ref: SourcesRef) -> tuple[SourcesLock, str]:
156    """Read the lock a manifest references.
157
158    Args:
159        base_dir: The manifest repository.
160        ref: The manifest's ``sources``.
161
162    Returns:
163        ``(lock, sha256)``.
164
165    Raises:
166        LockError: ``E_LOCK_SHA`` for a missing or unreadable file or bytes whose sha256 is not
167            ``ref.sha256``; ``E_LOCK_INVALID`` as `parse_lock`.
168    """
169    try:
170        data = (Path(base_dir) / ref.lock).read_bytes()
171    except OSError:
172        raise LockError("E_LOCK_SHA") from None
173    sha = hashlib.sha256(data).hexdigest()
174    if sha != ref.sha256:
175        raise LockError("E_LOCK_SHA")
176    return parse_lock(data), sha

185def step_findings(
186    step: StepSpec, lock: SourcesLock, submodules: Mapping[str, str] | None = None
187) -> list[tuple[str, str]]:
188    """Check a step's declared pins against a lock; nothing is read from disk.
189
190    Args:
191        step: The step.
192        lock: The manifest's lock.
193        submodules: The checkout's submodule commits, when known; compared with the code entry's.
194
195    Returns:
196        ``(code, detail)`` pairs, the detail naming the step field. ``E_LOCK_REF``: a ``lock_inputs`` id
197        that is no registered entry, a data-root input or ``stage`` source whose patterns no pattern entry of
198        its root lists, or ``hf_weights`` with no models entry of the same repository, type and revision
199        carrying each declared file with its sha256 (the step may declare a subset of the entry's files), or a
200        ``url_weights`` file with no ``urls`` entry of the same file and sha256.
201        ``E_LOCK_MISMATCH``: no code entry with the step's repository, commit and patch sha256s (and
202        submodules, when given), or an env lock missing from ``envs`` or pinned there with another sha256.
203    """
204    found: list[tuple[str, str]] = []
205    registered = {e.id for e in lock.registered()}
206    found += [("E_LOCK_REF", f"lock_inputs {i}") for i in step.lock_inputs if i not in registered]
207    listed = {(e.root, p) for e in lock.data if isinstance(e, PatternEntry) for p in e.patterns}
208    sourced = [("inputs", i, s.root, s.patterns) for i, s in enumerate(step.inputs)]
209    sourced += [("stage", i, s.root, s.patterns) for i, s in enumerate(step.stage)]
210    for field, index, root, patterns in sourced:
211        data_root = not root.startswith("up:") and root not in RESERVED_ROOT_NAMES
212        if data_root and any((root, p) not in listed for p in patterns):
213            found.append(("E_LOCK_REF", f"{field}[{index}]"))
214    for w in step.hf_weights:
215        pinned = [
216            {f.path: f.sha256 for f in m.files}
217            for m in lock.models
218            if (m.repo, m.repo_type, m.revision) == (w.repo, w.repo_type, w.revision)
219        ]
220        if not any(all(files.get(f.path) == f.sha256 for f in w.files) for files in pinned):
221            found.append(("E_LOCK_REF", f"hf_weights {w.repo}"))
222    pinned_urls = {(u.file, u.sha256) for u in lock.urls}
223    found += [
224        ("E_LOCK_REF", f"url_weights {w.file}") for w in step.url_weights if (w.file, w.sha256) not in pinned_urls
225    ]
226    patches = [p.sha256 for p in step.patches]
227    code = [c for c in lock.code if (c.repo, c.commit, c.patches) == (step.repo, step.commit, patches)]
228    if not code or (submodules is not None and all(c.submodules != dict(submodules) for c in code)):
229        found.append(("E_LOCK_MISMATCH", "code"))
230    envs = {e.lock: e.sha256 for e in lock.envs}
231    found += [("E_LOCK_MISMATCH", f"env {path}") for path, sha in _env_refs(step) if envs.get(path) != sha]
232    return found

235def resolve_inputs(base_dir: Path, lock: SourcesLock, ids: Iterable[str]) -> dict[str, inputs.RegisteredInput]:
236    """Read the registry entries of ``ids`` and check they are the entries the lock pinned.
237
238    Args:
239        base_dir: The manifest repository.
240        lock: The manifest's lock.
241        ids: Registered input ids (a step's ``lock_inputs``).
242
243    Returns:
244        Each id's registry entry. No file of an input is opened.
245
246    Raises:
247        LockError: ``E_LOCK_REF`` for an id the lock does not register; ``E_LOCK_STALE`` when the registry
248            file's sha256 or its manifest's sha256 is not the one the lock pinned; ``E_INPUT_REGISTRY`` when
249            the entry cannot be read (`inputs.read_registry_entry`).
250    """
251    pinned = {e.id: e for e in lock.registered()}
252    out: dict[str, inputs.RegisteredInput] = {}
253    for input_id in ids:
254        if input_id not in pinned:
255            raise LockError("E_LOCK_REF")
256        try:
257            entry, sha = inputs.read_registry_entry(base_dir, input_id)
258        except inputs.InputError as error:
259            raise LockError(error.code) from None
260        if sha != pinned[input_id].registry_sha256 or entry.manifest.sha256 != pinned[input_id].manifest_sha256:
261            raise LockError("E_LOCK_STALE")
262        out[input_id] = entry
263    return out

266def lock_findings(base_dir: Path, path: Path) -> list[tuple[str, str]]:
267    """Lint one lock file alone: it validates, and every registered entry is current.
268
269    Args:
270        base_dir: The manifest repository.
271        path: The lock file.
272
273    Returns:
274        ``(code, detail)`` pairs: ``E_LOCK_SHA`` (unreadable), ``E_LOCK_INVALID``, and ``E_LOCK_STALE`` or
275        ``E_INPUT_REGISTRY`` with the input id.
276    """
277    try:
278        lock = parse_lock(Path(path).read_bytes())
279    except OSError:
280        return [("E_LOCK_SHA", "unreadable")]
281    except LockError as error:
282        return [(error.code, "schema")]
283    found = []
284    for entry in lock.registered():
285        try:
286            resolve_inputs(base_dir, lock, [entry.id])
287        except LockError as error:
288            found.append((error.code, entry.id))
289    return found

203        submodules, when given), or an env lock missing from ``envs`` or pinned there with another sha256.
204    """
205    found: list[tuple[str, str]] = []
206    registered = {e.id for e in lock.registered()}
207    found += [("E_LOCK_REF", f"lock_inputs {i}") for i in step.lock_inputs if i not in registered]
208    listed = {(e.root, p) for e in lock.data if isinstance(e, PatternEntry) for p in e.patterns}
209    sourced = [("inputs", i, s.root, s.patterns) for i, s in enumerate(step.inputs)]
210    sourced += [("stage", i, s.root, s.patterns) for i, s in enumerate(step.stage)]
211    for field, index, root, patterns in sourced:
212        data_root = not root.startswith("up:") and root not in RESERVED_ROOT_NAMES
213        if data_root and any((root, p) not in listed for p in patterns):
214            found.append(("E_LOCK_REF", f"{field}[{index}]"))
215    for w in step.hf_weights:
216        pinned = [
217            {f.path: f.sha256 for f in m.files}
218            for m in lock.models
219            if (m.repo, m.repo_type, m.revision) == (w.repo, w.repo_type, w.revision)
220        ]
221        if not any(all(files.get(f.path) == f.sha256 for f in w.files) for files in pinned):
222            found.append(("E_LOCK_REF", f"hf_weights {w.repo}"))
223    pinned_urls = {(u.file, u.sha256) for u in lock.urls}
224    found += [
225        ("E_LOCK_REF", f"url_weights {w.file}") for w in step.url_weights if (w.file, w.sha256) not in pinned_urls
226    ]
227    patches = [p.sha256 for p in step.patches]
228    code = [c for c in lock.code if (c.repo, c.commit, c.patches) == (step.repo, step.commit, patches)]
229    if not code or (submodules is not None and all(c.submodules != dict(submodules) for c in code)):
230        found.append(("E_LOCK_MISMATCH", "code"))
231    envs = {e.lock: e.sha256 for e in lock.envs}
232    found += [("E_LOCK_MISMATCH", f"env {path}") for path, sha in _env_refs(step) if envs.get(path) != sha]
233    return found

def resolve_inputs( base_dir: pathlib.Path, lock: SourcesLock, ids: Iterable[str]) -> dict[str, senselab.utils.reproduce.inputs.RegisteredInput]: 
236def resolve_inputs(base_dir: Path, lock: SourcesLock, ids: Iterable[str]) -> dict[str, inputs.RegisteredInput]:
237    """Read the registry entries of ``ids`` and check they are the entries the lock pinned.
238
239    Args:
240        base_dir: The manifest repository.
241        lock: The manifest's lock.
242        ids: Registered input ids (a step's ``lock_inputs``).
243
244    Returns:
245        Each id's registry entry. No file of an input is opened.
246
247    Raises:
248        LockError: ``E_LOCK_REF`` for an id the lock does not register; ``E_LOCK_STALE`` when the registry
249            file's sha256 or its manifest's sha256 is not the one the lock pinned; ``E_INPUT_REGISTRY`` when
250            the entry cannot be read (`inputs.read_registry_entry`).
251    """
252    pinned = {e.id: e for e in lock.registered()}
253    out: dict[str, inputs.RegisteredInput] = {}
254    for input_id in ids:
255        if input_id not in pinned:
256            raise LockError("E_LOCK_REF")
257        try:
258            entry, sha = inputs.read_registry_entry(base_dir, input_id)
259        except inputs.InputError as error:
260            raise LockError(error.code) from None
261        if sha != pinned[input_id].registry_sha256 or entry.manifest.sha256 != pinned[input_id].manifest_sha256:
262            raise LockError("E_LOCK_STALE")
263        out[input_id] = entry
264    return out

def lock_findings(base_dir: pathlib.Path, path: pathlib.Path) -> list[tuple[str, str]]: 
267def lock_findings(base_dir: Path, path: Path) -> list[tuple[str, str]]:
268    """Lint one lock file alone: it validates, and every registered entry is current.
269
270    Args:
271        base_dir: The manifest repository.
272        path: The lock file.
273
274    Returns:
275        ``(code, detail)`` pairs: ``E_LOCK_SHA`` (unreadable), ``E_LOCK_INVALID``, and ``E_LOCK_STALE`` or
276        ``E_INPUT_REGISTRY`` with the input id.
277    """
278    try:
279        lock = parse_lock(Path(path).read_bytes())
280    except OSError:
281        return [("E_LOCK_SHA", "unreadable")]
282    except LockError as error:
283        return [(error.code, "schema")]
284    found = []
285    for entry in lock.registered():
286        try:
287            resolve_inputs(base_dir, lock, [entry.id])
288        except LockError as error:
289            found.append((error.code, entry.id))
290    return found
