senselab.utils.portable_audio_io

The one definition of how senselab reads and writes audio files.

Two callers implement nothing of their own: senselab.audio.data_structures.Audio.save_to_file() delegates to write_audio(), and subprocess-venv workers -- which run in isolated environments where senselab is not installed -- get this file staged next to their payload by senselab.utils.subprocess_venv.stage_portable_audio_io() and import it directly.

This module must import nothing from senselab, and nothing beyond numpy, soundfile and the standard library. src/tests/utils/portable_audio_io_test.py enforces both by AST sweep; a convenient from senselab... here breaks every worker the next time a venv is rebuilt.

Two decisions live here and nowhere else:

  • which subtype a destination gets (resolve_subtype()), and
  • what happens when the samples cannot be represented in it (apply_range_policy()).

The measurements behind both, and why the default preserves rather than clips: specs/20260819-091500-wav-subtype-sweep/design.md.

 40# Preference order per input dtype; the first subtype the destination supports wins. FLAC has no
 41# float subtype at any depth, so float data resolves to PCM_24 there and the range policy applies.
 42_PREFERENCE_BY_KIND = {
 43    "float": ("FLOAT", "DOUBLE", "PCM_32", "PCM_24", "PCM_16"),
 44    "int16": ("PCM_16", "PCM_24", "PCM_32", "FLOAT"),
 45    "int32": ("PCM_32", "FLOAT", "PCM_24", "PCM_16"),

 88def widest_subtype(fmt: str, dtype: Union[np.dtype, type, str] = np.float32) -> str:
 89    """Return the subtype of ``fmt`` that loses the least of a ``dtype`` array.
 90
 91    Args:
 92        fmt: A libsndfile format name, case-insensitive.
 93        dtype: The dtype of the array about to be written. Only the dtypes ``soundfile.write``
 94            accepts are distinguished (float, ``int32``, ``int16``); anything else is float.
 95
 96    Returns:
 97        A subtype name ``soundfile.check_format(fmt, subtype)`` accepts. For a lossy container
 98        with no PCM subtype, the container's own default.
 99
100    Raises:
101        ValueError: If ``fmt`` is not a format libsndfile can write, or has no default subtype.
102    """
103    normalized = fmt.upper()
104    try:
105        available = set(sf.available_subtypes(normalized))
106    except (TypeError, ValueError) as exc:
107        raise ValueError(f"{fmt!r} is not a format soundfile can write") from exc
108    if not available:
109        raise ValueError(f"{fmt!r} is not a format soundfile can write")
110
111    resolved = np.dtype(dtype)
112    if np.issubdtype(resolved, np.integer):
113        kind = "int16" if resolved.itemsize <= 2 else "int32"
114    else:
115        kind = "float"
116
117    for candidate in _PREFERENCE_BY_KIND[kind]:
118        if candidate in available:
119            return candidate
120
121    default = sf.default_subtype(normalized)
122    if default is None:
123        raise ValueError(f"{fmt!r} has no default subtype; name one explicitly")
124    return default

127def resolve_subtype(
128    fmt: str,
129    dtype: Union[np.dtype, type, str] = np.float32,
130    subtype: Optional[str] = None,
131) -> str:
132    """Decide the subtype for a write, validating an explicit request against the format.
133
134    Args:
135        fmt: A libsndfile format name.
136        dtype: The dtype of the array about to be written.
137        subtype: The caller's request, or None to preserve as much as ``fmt`` allows.
138
139    Returns:
140        The subtype to write.
141
142    Raises:
143        ValueError: If ``subtype`` is not one ``fmt`` can carry. ``subtype="FLOAT"`` with a FLAC
144            destination is the case worth naming: FLAC has no float subtype at any depth, so
145            "preserve exactly" and "write FLAC" are mutually exclusive and must not resolve to a
146            quiet quantization.
147    """
148    normalized = fmt.upper()
149    if subtype is None:
150        return widest_subtype(normalized, dtype)
151    requested = subtype.upper()
152    if not sf.check_format(normalized, requested):
153        raise ValueError(
154            f"{normalized} cannot carry subtype {requested}; it supports "
155            f"{sorted(sf.available_subtypes(normalized))}. "
156            f"{'FLAC is an integer codec and has no float subtype at any depth. ' if normalized == 'FLAC' else ''}"
157            "Choose a format that supports it, or a subtype this one has."
158        )
159    return requested

162def subtype_preference(
163    fmt: str,
164    preferred: Optional[str],
165    dtype: Union[np.dtype, type, str] = np.float32,
166) -> Optional[str]:
167    """Degrade a subtype *preference* to what ``fmt`` can carry.
168
169    The distinction from :func:`resolve_subtype` is the point. An explicit subtype is a demand and
170    must raise when the format cannot carry it, because a caller asking for FLOAT and silently
171    receiving PCM_16 is the defect this module exists to remove. A preference -- a codec hint
172    derived from a source stream, say -- is not a demand: the caller wants as much fidelity as the
173    container allows and has no opinion beyond that.
174
175    Args:
176        fmt: A libsndfile format name.
177        preferred: The subtype the caller would like, or None.
178        dtype: The dtype of the array about to be written.
179
180    Returns:
181        ``preferred`` when ``fmt`` can carry it, otherwise None, which leaves
182        :func:`resolve_subtype` to pick the widest subtype the format has.
183    """
184    if preferred is None:
185        return None
186    normalized, requested = fmt.upper(), preferred.upper()
187    return requested if sf.check_format(normalized, requested) else None

190def out_of_range_fraction(samples: Union[np.ndarray, Sequence[float]]) -> float:
191    """Fraction of samples a fixed-point subtype would clip, i.e. those outside ±1.
192
193    Args:
194        samples: Samples, in any shape.
195
196    Returns:
197        The fraction in ``[0, 1]``; ``0.0`` for an empty array.
198    """
199    arr = np.asarray(samples)
200    if arr.size == 0:
201        return 0.0
202    return float(np.count_nonzero(np.abs(arr) > 1.0) / arr.size)

205def apply_range_policy(
206    samples: np.ndarray,
207    fmt: str,
208    subtype: str,
209    out_of_range: str = RAISE,
210    destination: str = "",
211) -> Tuple[np.ndarray, float, float, float]:
212    """Enforce the range policy for a write that is about to happen.
213
214    The one place the "can this data be represented, and if not, what then" question is answered.
215    :func:`write_audio` calls it, and so does ``Audio.save_to_file`` on the paths where libsndfile
216    is not the writer -- so a float-incapable destination behaves identically either way.
217
218    Args:
219        samples: The samples about to be written.
220        fmt: The resolved libsndfile format name (used only in messages).
221        subtype: The resolved subtype. A float one makes this a no-op.
222        out_of_range: ``"raise"`` (default), ``"warn"`` or ``"normalize"``.
223        destination: The path, for the message.
224
225    Returns:
226        ``(samples, peak, fraction, gain)``. ``samples`` is rescaled only under ``"normalize"``,
227        and ``gain`` is the factor applied, so the original is recoverable by dividing by it.
228
229    Raises:
230        ValueError: If ``out_of_range`` is not a known policy, or if it is ``"raise"`` and the
231            samples do not fit.
232    """
233    if out_of_range not in OUT_OF_RANGE_POLICIES:
234        raise ValueError(f"out_of_range must be one of {OUT_OF_RANGE_POLICIES}, got {out_of_range!r}")
235
236    peak = float(np.abs(samples).max()) if samples.size else 0.0
237    if is_float_subtype(subtype) or peak <= 1.0:
238        return samples, peak, 0.0, 1.0
239
240    fraction = out_of_range_fraction(samples)
241    where = f" writing {destination}" if destination else ""
242    detail = (
243        f"{100 * fraction:.4g}% of samples exceed ±1 (peak {peak:.6g}) and {fmt}/{subtype} cannot represent them{where}"
244    )
245    if out_of_range == RAISE:
246        raise ValueError(
247            detail + ". Write a float-capable format (wav, aiff, w64, caf leave subtype=None and "
248            "get FLOAT), or pass out_of_range='normalize' to rescale with the gain reported, or "
249            "out_of_range='warn' to accept the clipping."
250        )
251    if out_of_range == WARN:
252        logger.warning("%s; those samples are being clipped.", detail)
253        return samples, peak, fraction, 1.0
254
255    gain = 1.0 / peak
256    logger.warning("%s; rescaling by %.6g. Divide by that gain to recover the original.", detail, gain)
257    return (samples * gain).astype(samples.dtype, copy=False), peak, fraction, gain

260def write_audio(
261    path: Union[str, os.PathLike],
262    samples: np.ndarray,
263    sampling_rate: int,
264    format: Optional[str] = None,
265    subtype: Optional[str] = None,
266    out_of_range: str = RAISE,
267    channels_first: bool = True,
268) -> AudioWriteReport:
269    """Write audio, preserving it exactly unless the destination cannot or the caller says otherwise.
270
271    Args:
272        path: Destination file. Its extension picks the format when ``format`` is None.
273        samples: 1-D samples, or 2-D shaped by ``channels_first``.
274        sampling_rate: Sample rate in Hz.
275        format: Explicit libsndfile format name, case-insensitive.
276        subtype: Explicit subtype. None resolves to the widest the format supports, which is
277            ``FLOAT`` for WAV -- so a plain ``.wav`` write preserves float samples exactly.
278        out_of_range: What to do when the resolved subtype cannot represent the samples:
279            ``"raise"`` (default), ``"warn"`` (clip, loudly) or ``"normalize"`` (rescale, with
280            the gain in the returned report).
281        channels_first: For 2-D ``samples``, whether the first axis is channels (senselab's
282            convention). Ignored for 1-D input.
283
284    Returns:
285        An :class:`AudioWriteReport` naming what was written.
286
287    Raises:
288        ValueError: If the format cannot be written, the subtype does not fit it, or the samples
289            do not fit the subtype under ``out_of_range="raise"``.
290    """
291    fmt = format_for(path, format)
292    if not libsndfile_can_write(fmt):
293        raise ValueError(f"{fmt!r} is not a format soundfile can write; got destination {os.fspath(path)!r}")
294
295    array = np.asarray(samples)
296    if array.ndim == 2 and channels_first:
297        array = array.T
298
299    resolved = resolve_subtype(fmt, array.dtype, subtype)
300    array, peak, fraction, gain = apply_range_policy(array, fmt, resolved, out_of_range, os.fspath(path))
301
302    sf.write(os.fspath(path), array, sampling_rate, subtype=resolved, format=fmt)
303    return AudioWriteReport(
304        path=os.fspath(path),
305        format=fmt,
306        subtype=resolved,
307        peak=peak,
308        out_of_range_fraction=fraction,
309        gain=gain,
310    )

313def read_audio(
314    path: Union[str, os.PathLike],
315    always_2d: bool = False,
316    channels_first: bool = True,
317) -> Tuple[np.ndarray, int]:
318    """Read audio as float32, in senselab's axis order.
319
320    Args:
321        path: Source file.
322        always_2d: Return a 2-D array even for mono.
323        channels_first: For 2-D output, put channels on the first axis. Ignored when the result
324            is 1-D.
325
326    Returns:
327        ``(samples, sampling_rate)``.
328    """
329    data, sampling_rate = sf.read(os.fspath(path), dtype="float32", always_2d=always_2d)
330    if data.ndim == 2 and channels_first:
331        data = data.T
332    return data, int(sampling_rate)

222        subtype: The resolved subtype. A float one makes this a no-op.

224        destination: The path, for the message.
225

229
230    Raises:
231        ValueError: If ``out_of_range`` is not a known policy, or if it is ``"raise"`` and the

235        raise ValueError(f"out_of_range must be one of {OUT_OF_RANGE_POLICIES}, got {out_of_range!r}")
236
237    peak = float(np.abs(samples).max()) if samples.size else 0.0

240
241    fraction = out_of_range_fraction(samples)

243    detail = (

245    )

248            detail + ". Write a float-capable format (wav, aiff, w64, caf leave subtype=None and "
249            "get FLOAT), or pass out_of_range='normalize' to rescale with the gain reported, or "
250            "out_of_range='warn' to accept the clipping."
251        )
252    if out_of_range == WARN:
253        logger.warning("%s; those samples are being clipped.", detail)
254        return samples, peak, fraction, 1.0
255
256    gain = 1.0 / peak
257    logger.warning("%s; rescaling by %.6g. Divide by that gain to recover the original.", detail, gain)
258    return (samples * gain).astype(samples.dtype, copy=False), peak, fraction, gain
