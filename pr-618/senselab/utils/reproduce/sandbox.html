senselab.utils.reproduce.sandbox

The step sandbox: user, network, mount and PID namespaces with read-only data, and an allowlisted env.

sandbox_argv wraps a step's argv so that it runs inside a user, network, mount and PID namespace in which every mount is read-only except the writable directories it names, with loopback raised; the step itself runs with every capability dropped, so it cannot undo the mounts. build_env builds the step's whole environment from the profile and the step; nothing is inherited from the caller except a few SLURM_* job ids and CUDA_VISIBLE_DEVICES, which are recorded but not keyed. A step's draw reaches it as REPRO_SEED, REPRO_DATA_ORDER_SEED and REPRO_SAMPLING. Proxies point at a dead port; an engine step also gets REPRO_ENGINE_IDENTITY, and a vllm step no_proxy=127.0.0.1,localhost and its compile caches under {tmp} (engine). canary is a Python snippet whose TCP connect must fail inside the sandbox.

 96SWEEP = """\
 97import ctypes, os, sys
 98libc = ctypes.CDLL(None, use_errno=True)
 99libc.mount.argtypes = [ctypes.c_char_p, ctypes.c_char_p, ctypes.c_char_p, ctypes.c_ulong, ctypes.c_void_p]
100KEEP = {"nosuid": 2, "nodev": 4, "noexec": 8, "noatime": 1024, "nodiratime": 2048, "relatime": 1 << 21,
101        "strictatime": 1 << 24}
102PSEUDO = set(sys.argv[1].split(","))
103skip = {os.fsencode(p) for p in sys.argv[2:] if not p.startswith("@")}
104masked = [os.fsencode(p[1:]).rstrip(b"/") + b"/" for p in sys.argv[2:] if p.startswith("@")]
105top = {}
106with open("/proc/self/mountinfo", "rb") as handle:
107    for line in handle:
108        left, _, right = line.partition(b" - ")
109        fields = left.split()
110        top[fields[4]] = (fields[5], right.split()[0].decode())
111bad = []
112for raw, (options, fstype) in top.items():
113    path = raw.decode("unicode_escape").encode("latin-1") if b"\\\\" in raw else raw
114    if path in skip or fstype == "proc" or any(path.startswith(m) for m in masked):
115        continue
116    flags = 32 | 4096 | 1
117    for option in options.split(b","):
118        flags |= KEEP.get(option.decode(), 0)
119    if libc.mount(None, path, None, flags, None) != 0 and fstype not in PSEUDO:
120        bad.append(fstype)
121if bad:
122    sys.stderr.write("reproduce sandbox: mounts left writable: " + ", ".join(sorted(set(bad))) + "\\n")
123    sys.exit(1)
124"""
125"""Run with the driver's interpreter (``-I -S``) inside the namespace, before the capability drop: remounts the
126topmost mount at every mount point read-only, keeping each mount's locked flags, except the paths named on
127its command line after the comma-separated pseudo types (the writable binds), the fresh ``/proc``, and every
128mount point under a masked dir (named with a leading ``@``), which the mask makes unreachable. A
129mount that refuses is fatal unless its type is a kernel pseudo filesystem in `PSEUDO_FS`."""

147def _seen(roots: list[str], rw: list[str], extra: list[str], visible: Iterable[str | os.PathLike[str]]) -> list[str]:
148    driver = [os.path.realpath(sys.executable), os.path.realpath(sys.prefix), os.path.realpath(sys.base_prefix)]

152def inner_tmp(
153    ro_roots: Iterable[str | os.PathLike[str]],
154    rw_dirs: Iterable[str | os.PathLike[str]],
155    ro_extra: Iterable[str | os.PathLike[str]],
156    *,

168def sandbox_script(
169    ro_roots: Iterable[str | os.PathLike[str]],
170    rw_dirs: Iterable[str | os.PathLike[str]],
171    ro_extra: Iterable[str | os.PathLike[str]] = (),
172    *,
173    netns: bool = True,
174    tmp: str | os.PathLike[str] | None = None,
175    visible: Iterable[str | os.PathLike[str]] = (),
176    masked: Iterable[str | os.PathLike[str]] = (),
177) -> str:
178    """Return the ``sh`` script `sandbox_argv` runs inside the namespace; it ends by exec'ing the step.
179
180    Everything is read-only by default: after the binds, every mount in the namespace is remounted
181    read-only (`SWEEP`), so only ``rw_dirs`` (and ``tmp``) stay writable, and a fresh private tmpfs is
182    mounted on ``/dev/shm``.
183
184    Args:
185        ro_roots: Data roots; each is bound to itself first, so a missing root aborts the script.
186        rw_dirs: The only directories that stay writable, each bound to itself before the sweep.
187        ro_extra: Further directories (the checkout, the venv) that must exist and stay read-only.
188        netns: Whether the script runs in its own network namespace and raises loopback there; False
189            only for the test-only mode that runs with the host network.
190        tmp: A writable directory also bound on ``/tmp``; when any path given here (or in ``visible``, or
191            the driver's own interpreter, which runs the sweep) lies under ``/tmp``, the bind would hide it,
192            so ``/tmp`` stays read-only and the dir is bound on `SHORT_TMP` instead (`inner_tmp` names
193            which).
194        visible: Other paths the step needs to see, checked only against the ``/tmp`` bind.
195        masked: Directories hidden under an empty tmpfs (made read-only by the sweep), when they exist; the
196            host socket dirs of `MASKED_HOST_DIRS` are always masked unless a path the sandbox needs lies
197            under one.
198
199    Returns:
200        The script. It starts with ``set -e``, so a failed mount ends it before the step runs, and it
201        execs the step through ``setpriv`` with every capability dropped. Each list of binds is written
202        shallow-first, so a nested path is bound after its parent.
203
204    Raises:
205        ValueError: If a path is relative, a read-only path lies under (or is) a writable directory, or a
206            writable directory lies under a ``ro_extra`` directory.
207    """
208    roots, rw, extra = _shallow_first(ro_roots), _shallow_first(rw_dirs), _shallow_first(ro_extra)
209    masks = [_abs(m) for m in masked]
210    if tmp is not None:
211        rw = _shallow_first([*rw, tmp])
212    for d in rw:
213        hidden = [p for p in (*roots, *extra) if _under(p, d)]
214        if hidden:
215            raise ValueError(f"read-only paths under a writable directory would be hidden by its bind: {hidden}")
216        if any(_under(d, e) for e in extra):
217            raise ValueError(f"writable directory under a read-only checkout or venv: {d}")
218    seen = _seen(roots, rw, extra, visible)
219    over_tmp = tmp is not None and _over_tmp(seen)
220    q = shlex.quote
221    skip = [*rw, *([SYSTEM_TMP] if over_tmp else [])]
222    lines = ["set -e", f"{MOUNT} --make-rprivate /"]
223    hidden = [h for h in MASKED_HOST_DIRS if not any(_under(p, h) for p in seen)]
224    lines += [_mask(h) for h in hidden]
225    lines += [f"{MOUNT} --bind {q(r)} {q(r)}" for r in roots]
226    lines += [f"{MOUNT} --bind {q(d)} {q(d)}" for d in rw]
227    lines += [f"{MOUNT} --bind {q(e)} {q(e)}" for e in extra]
228    if over_tmp:
229        lines.append(f"{MOUNT} --bind {q(_abs(tmp))} {SYSTEM_TMP}")  # type: ignore[arg-type]
230    lines += [_mask(m) for m in masks]
231    pseudo = ",".join(sorted(PSEUDO_FS))
232    under_masks = [f"@{p}" for p in (*hidden, *masks)]
233    sweep_args = [pseudo, *skip, *under_masks]
234    lines.append(" ".join([q(sys.executable), "-I", "-S", "-c", q(SWEEP), *(q(p) for p in sweep_args)]))
235    lines.append(f"if [ -d /dev/shm ]; then {MOUNT} -t tmpfs -o nosuid,nodev,mode=1777 tmpfs /dev/shm; fi")
236    if tmp is not None and not over_tmp:
237        lines += [f"/usr/bin/mkdir {SHORT_TMP}", f"{MOUNT} --bind {q(_abs(tmp))} {SHORT_TMP}"]
238    lines += [f"{IP} link set lo up"] if netns else []
239    lines += [f'exec {" ".join(DROP)} "$@"']
240    return "\n".join(lines) + "\n"

250def sandbox_argv(
251    argv: Sequence[str],
252    ro_roots: Iterable[str | os.PathLike[str]],
253    rw_dirs: Iterable[str | os.PathLike[str]],
254    ro_extra: Iterable[str | os.PathLike[str]] = (),
255    *,
256    netns: bool = True,
257    tmp: str | os.PathLike[str] | None = None,
258    visible: Iterable[str | os.PathLike[str]] = (),
259    masked: Iterable[str | os.PathLike[str]] = (),
260) -> list[str]:
261    """Wrap a step's argv to run in the read-only-by-default sandbox.
262
263    Args:
264        argv: The expanded step argv; the child receives it unchanged.
265        ro_roots: Data roots; they must exist.
266        rw_dirs: The only directories that stay writable.
267        ro_extra: The checkout and the venv; they must exist.
268        netns: False drops the network namespace (`UNSHARE_NO_NETNS`, no loopback step); test-only.
269        tmp: The step's scratch dir, writable and also bound on ``/tmp`` (see `sandbox_script`).
270        visible: Other paths the step must see, checked against the ``/tmp`` bind.
271        masked: Directories hidden under an empty read-only tmpfs (see `sandbox_script`).
272
273    Returns:
274        ``[*UNSHARE, "/bin/sh", "-c", SCRIPT, "sh", *argv]``, where `UNSHARE` is ``/usr/bin/unshare -rnm
275        --pid --fork --kill-child --mount-proc --propagation private``.
276
277    Raises:
278        ValueError: If argv is empty, or as `sandbox_script`.
279    """
280    if not argv:
281        raise ValueError("empty argv")
282    unshare = UNSHARE if netns else UNSHARE_NO_NETNS
283    script = sandbox_script(ro_roots, rw_dirs, ro_extra, netns=netns, tmp=tmp, visible=visible, masked=masked)
284    return [*unshare, SH, "-c", script, "sh", *argv]

333def sandbox_status() -> tuple[str, str]:
334    """Whether a sandbox can be built here: ``("ok", "")``, ``("no-userns", detail)`` or ``("sweep-refused", line)``.
335
336    ``sweep-refused`` means the namespaces work but a mount refused the read-only remount, so this node cannot
337    host a sandbox; the line names the refusing filesystem types.
338    """
339    try:
340        done = subprocess.run(
341            sandbox_argv(["true"], [], []),
342            env={"PATH": ":".join(SYSTEM_PATH)},
343            capture_output=True,
344            timeout=30,
345        )
346    except (OSError, subprocess.SubprocessError) as error:
347        return "no-userns", type(error).__name__
348    if done.returncode == 0:
349        return "ok", ""
350    stderr = done.stderr.decode(errors="replace") if isinstance(done.stderr, bytes) else str(done.stderr or "")
351    refusal = sweep_refusal(stderr)
352    if refusal is not None:
353        return "sweep-refused", refusal
354    return "no-userns", stderr.strip()[-200:]

372_TOKEN = re.compile(r"\{\{|\}\}|\{([^{}]*)\}|[{}]")
373"""`spec.render`'s token grammar: ``{{``, ``}}``, one placeholder, or a stray brace."""
374
375
376def _native(step: StepSpec, profile: Profile, kind: str) -> list[str]:
377    """The ``{native:<name>:<i>}`` templates of the selected entries' ``bin_dirs`` or ``lib_dirs``.
378
379    ``i`` counts an entry's ``bin_dirs`` then its ``lib_dirs``, as `cache.native_digests` does, so the keyed env
380    names each dir by entry and position and its host path is not keyed (its content is, by digest).
381    """
382    out = []
383    for name in step.resources.native:
384        entry = profile.native[name]
385        start = 0 if kind == "bin_dirs" else len(entry.bin_dirs)
386        out += [f"{{native:{name}:{start + i}}}" for i in range(len(getattr(entry, kind)))]
387    return out

389
390def _native_values(step: StepSpec, profile: Profile) -> dict[str, str]:
391    """Each ``native:<name>:<i>`` placeholder of `_native` mapped to its absolute dir."""
392    values = {}
393    for name in step.resources.native:
394        entry = profile.native[name]
395        for i, path in enumerate([*entry.bin_dirs, *entry.lib_dirs]):
396            values[f"native:{name}:{i}"] = os.fspath(path)
397    return values

399
400def _render_native(template: str, values: Mapping[str, str]) -> str:
401    """Expand the ``{native:<name>:<i>}`` placeholders only; every other token is kept for `render`."""
402
403    def expand(match: re.Match[str]) -> str:
404        name = match.group(1)
405        if name is not None and name.startswith("native:"):
406            return _escape(values[name])
407        return match.group(0)
408
409    return _TOKEN.sub(expand, template)

416def _keyed(step: StepSpec, profile: Profile) -> tuple[dict[str, str], dict[str, str]]:
417    check_step_profile(step, profile)
418    bad = sorted(k for k in profile.env if _reserved(k))
419    if bad:
420        raise ValueError(f"profile env may not set driver-owned variables: {bad}")
421    colon = [p for p in step.pythonpath if ":" in p]
422    if colon:
423        raise ValueError(f"pythonpath entries may not contain ':': {colon}")
424    templated = {
425        "PATH": ":".join(["{venv}/bin", *SYSTEM_PATH, *_native(step, profile, "bin_dirs")]),
426        "JUPYTER_PATH": "{venv}/share/jupyter",
427        "JUPYTER_DATA_DIR": "{tmp}/jupyter/data",
428        "JUPYTER_CONFIG_DIR": "{tmp}/jupyter/config",
429        "HF_HOME": "{cache}/huggingface",
430        "TORCH_HOME": "{cache}/torch",
431        "XDG_CACHE_HOME": "{tmp}/cache",
432        "CUDA_CACHE_PATH": "{tmp}/cuda-cache",
433        "REPRO_OUT": "{out}",
434        "REPRO_TMP": "{tmp}",
435    }
436    if step.hf_weights:
437        templated["HF_HOME"] = "{tmp}/hf"
438        templated["HF_HUB_CACHE"] = "{tmp}/hf-hub"
439        templated["HUGGINGFACE_HUB_CACHE"] = "{tmp}/hf-hub"
440    if step.url_weights:
441        templated["TORCH_HOME"] = "{tmp}/torch"
442    if step.pythonpath:
