senselab.utils.reproduce.inputs

Registered inputs: registry entries, the per-file manifest format and the manifest writer.

A registered input is a frozen tree under a profile root, described by a registry entry <base_dir>/inputs/<id>.yaml (RegisteredInput) that names the tree, its per-file manifest and that manifest's sha256. A manifest is a TSV: one # root=… comment line, the header MANIFEST_COLUMNS (optionally followed by sha256), then one row per file or symlink, sorted by relative path. A symlink is recorded by its target and never followed; a file by its size, mtime_ns and xxh3-128. write_manifest names the tree in that line by registry root, # root={root:<name>}/<relpath> hash=xxh3_128, with no host path, host name or date, so identical data gives the same manifest sha256 on any site; the writing time (UTC) goes in the unhashed dirs sidecar. The existing tree hasher's line (# root=<abs path> date=… host=… xxhash=…) is still read.

precheck is the gate's cheap check of a registered input against its manifest: lstat only, no file content. It also applies the link rule: a link row's target, resolved against the link's own directory, must lie under the input itself or under another registered input the step declares. With extras: check it also lists each dir that holds a row (dirs_sidecar), so an added file is a mismatch before any full verification (dir_changes).

verify re-hashes a registered input against its manifest. append_record stores the outcome in <verify dir>/<id>.json (verify_dir) under a lock, with an accepted-mtime overlay <id>.mtime-overlay.tsv when only mtimes drifted. verified_codes gives E_INPUT_UNVERIFIED unless the latest record is current, ok and for the registered manifest.

Errors are an InputError whose message is a fixed code only.

 193def read_registry_entry(base_dir: Path, input_id: str) -> tuple[RegisteredInput, str]:
 194    """Read one registry entry and the sha256 of its file.
 195
 196    Args:
 197        base_dir: The manifest repository.
 198        input_id: The input id; the file is ``inputs/<id>.yaml``.
 199
 200    Returns:
 201        ``(entry, sha256 of the file's bytes)``.
 202
 203    Raises:
 204        InputError: ``E_INPUT_REGISTRY`` for a missing or unreadable file, an entry that does not
 205            validate, or an ``id`` other than the file stem.
 206    """
 207    try:
 208        data = registry_path(base_dir, input_id).read_bytes()
 209        entry = RegisteredInput.model_validate(yaml.safe_load(data))

 211        raise InputError(CODE) from None
 212    if entry.id != input_id:
 213        raise InputError(CODE)
 214    return entry, hashlib.sha256(data).hexdigest()

 243def _parse_row(fields: Sequence[str], with_sha: bool) -> Row:
 244    rel, size, mtime, xxh, link = fields[:5]
 245    sha = fields[5] if with_sha else ""
 246    if not rel:
 247        raise ValueError("empty relpath")
 248    if link:
 249        if size or mtime or xxh or sha:
 250            raise ValueError("a link row carries only its target")
 251        return Row(rel, None, None, None, link, None)
 252    if not _XXH.match(xxh) or (with_sha and not _HEX64.match(sha)):
 253        raise ValueError("bad hash")
 254    return Row(rel, _int(size), _int(mtime), xxh, None, sha or None)

 257def read_manifest(path: Path, expected_sha256: str) -> list[Row]:
 258    """Read a manifest whose bytes must have the registered sha256.
 259
 260    Args:
 261        path: The manifest file.
 262        expected_sha256: The registry entry's ``manifest.sha256``.
 263
 264    Returns:
 265        The rows, in file order.
 266
 267    Raises:
 268        InputError: ``E_INPUT_REGISTRY`` for a missing or unreadable file, a sha256 other than
 269            ``expected_sha256``, a missing ``# root=`` line or header, a malformed row, an ``ERR:`` hash, or a
 270            relpath that is absolute, has an empty, ``.`` or ``..`` component, or is not strictly ascending.
 271    """
 272    try:
 273        data = Path(path).read_bytes()
 274    except OSError:
 275        raise InputError(CODE) from None
 276    if hashlib.sha256(data).hexdigest() != expected_sha256:
 277        raise InputError(CODE)
 278    try:
 279        lines = data.decode().split("\n")
 280        if lines and lines[-1] == "":
 281            lines.pop()
 282        if len(lines) < 2 or not lines[0].startswith("# root="):
 283            raise ValueError("no root line")
 284        header = tuple(lines[1].split("\t"))
 285        with_sha = header == (*MANIFEST_COLUMNS, SHA256_COLUMN)
 286        if header != MANIFEST_COLUMNS and not with_sha:
 287            raise ValueError("bad header")
 288        rows: list[Row] = []
 289        for line in lines[2:]:
 290            fields = line.split("\t")
 291            if len(fields) != len(header):
 292                raise ValueError("bad row")
 293            row = _parse_row(fields, with_sha)
 294            parts = row.relpath.split("/")
 295            if row.relpath.startswith("/") or any(p in ("", ".", "..") for p in parts):
 296                raise ValueError("bad relpath")
 297            if rows and row.relpath <= rows[-1].relpath:
 298                raise ValueError("relpaths not strictly ascending")
 299            rows.append(row)
 300    except (UnicodeDecodeError, ValueError):
 301        raise InputError(CODE) from None
 302    return rows

 305def _hashes(path: str, sha256: bool) -> tuple[str, str]:
 306    xxh = xxhash.xxh3_128()
 307    sha = hashlib.sha256() if sha256 else None
 308    with open(path, "rb") as handle:
 309        while chunk := handle.read(_CHUNK):
 310            xxh.update(chunk)
 311            if sha is not None:
 312                sha.update(chunk)
 313    return xxh.hexdigest(), "" if sha is None else sha.hexdigest()

 316def _row(path: str, rel: str, sha256: bool) -> tuple[str, ...]:
 317    tail = ("",) if sha256 else ()
 318    try:
 319        st = os.lstat(path)
 320        if stat.S_ISLNK(st.st_mode):
 321            return (rel, "", "", "", os.readlink(path), *tail)
 322        xxh, sha = _hashes(path, sha256)
 323        return (rel, str(st.st_size), str(st.st_mtime_ns), xxh, "", *((sha,) if sha256 else ()))
 324    except OSError as error:
 325        return (rel, "", "", f"ERR:{type(error).__name__}", "", *tail)

 334def _lstat_path(top: str, rel: str) -> tuple[str, str]:
 335    """``lstat`` each component of ``rel`` under ``top``, never following a symlink.
 336
 337    Returns:
 338        ``("link", prefix)`` at the first symlinked component, ``("missing", prefix)`` where the path
 339        stops existing (or a non-final component is not a directory), else ``("dir" | "file", rel)``.
 340
 341    Raises:
 342        OSError: When a component cannot be ``lstat``-ed for another reason.
 343    """
 344    parts = rel.split("/") if rel else []
 345    for index in range(len(parts)):
 346        prefix = "/".join(parts[: index + 1])
 347        try:
 348            st = os.lstat(os.path.join(top, prefix))
 349        except (FileNotFoundError, NotADirectoryError):
 350            return "missing", prefix
 351        if stat.S_ISLNK(st.st_mode):
 352            return "link", prefix
 353        if index < len(parts) - 1 and not stat.S_ISDIR(st.st_mode):
 354            return "missing", prefix
 355        if index == len(parts) - 1:
 356            return ("dir" if stat.S_ISDIR(st.st_mode) else "file"), rel
 357    return "dir", rel

 360def _collect(
 361    top: str,
 362    include: Sequence[str],
 363    exclude: Sequence[str],
 364    visit: Callable[[str, str], T],
 365    *,
 366    strict: bool = True,
 367) -> tuple[dict[str, T], list[str]]:
 368    """Select the files and links of a tree as `write_manifest` does, and apply ``visit(path, rel)`` to each.
 369
 370    Args:
 371        top: The tree's top.
 372        include: Selecting patterns (empty: everything).
 373        exclude: Patterns whose matches are left out.
 374        visit: Applied to each selected entry, inside the walk's threads.
 375        strict: Refuse a literal include that is missing or cannot be ``lstat``-ed; otherwise skip it.
 376
 377    Returns:
 378        ``({rel: visit result}, the paths whose listing or ``lstat`` failed)``.
 379
 380    Raises:
 381        InputError: ``E_INPUT_REGISTRY`` for a pattern outside the grammar, a top that is not a directory,
 382            or (``strict``) a literal include that is missing or cannot be ``lstat``-ed.
 383    """
 384    try:
 385        for pattern in (*include, *exclude):
 386            digest._check_pattern(pattern)
 387    except digest.DigestError:
 388        raise InputError(CODE) from None
 389    if not os.path.isdir(top):
 390        raise InputError(CODE)
 391    patterns: list[str] = []
 392    files: list[str] = []
 393    bases: set[str] = set() if include else {""}
 394    links: set[str] = set()
 395    errors: list[str] = []
 396    for pattern in include:
 397        literal = not digest._WILD.search(pattern)
 398        try:
 399            kind, at = _lstat_path(top, pattern if literal else digest._base(pattern))
 400        except OSError:
 401            if literal and strict:
 402                raise InputError(CODE) from None
 403            errors.append(pattern)
 404            continue
 405        if kind == "missing" and literal:
 406            if strict:
 407                raise InputError(CODE)
 408            continue
 409        if kind == "link":
 410            links.add(at)
 411        elif kind == "dir":
 412            bases.add(at)
 413            patterns.append(f"{pattern}/**" if literal else pattern)
 414        elif kind == "file":
 415            files.append(at)
 416            patterns.append(pattern)
 417    walks = sorted(b for b in bases if not any(o != b and (o == "" or b.startswith(o + "/")) for o in bases))
 418    found: dict[str, T] = {}
 419
 420    def selected(path: str) -> tuple[str, T] | None:
 421        rel = os.path.relpath(path, top).replace(os.sep, "/")
 422        return (rel, visit(path, rel)) if _selected(rel, patterns, exclude) else None
 423
 424    for rel in sorted(links):
 425        if _selected(rel, (), exclude):
 426            found[rel] = visit(os.path.join(top, rel), rel)
 427    for rel_file in files:
 428        if (item := selected(os.path.join(top, rel_file))) is not None:
 429            found[item[0]] = item[1]
 430    for base in walks:
 431        start = os.path.join(top, base) if base else top
 432        for item in fastio.map_files(start, selected, links_as_files=True, on_error=lambda p, e: errors.append(p)):
 433            if item is not None:
 434                found[item[0]] = item[1]
 435    return found, errors

 438def write_manifest(
 439    root: str | os.PathLike[str],
 440    out: str | os.PathLike[str],
 441    *,
 442    include: Sequence[str] = (),
 443    exclude: Sequence[str] = (),
 444    sha256: bool = False,
 445    label: str | None = None,
 446) -> dict[str, int]:
 447    """Write the manifest of a tree, atomically; nothing is printed.
 448
 449    Args:
 450        root: The tree's top; relpaths are relative to it.
 451        out: The manifest file to write.
 452        include: Patterns (the `digest` grammar) selecting what is recorded; empty records everything. A
 453            pattern with no wildcard is ``lstat``-ed directly (a literal directory is walked whole), and a
 454            wildcard pattern walks only from its wildcard-free leading directories. Each component of a
 455            literal path or a walk's base is ``lstat``-ed; a symlinked component is recorded as a link row
 456            and not followed.
 457        exclude: Patterns whose matches are left out.
 458        sha256: Also record each file's sha256, as a sixth column.
 459        label: The tree as the registry names it, ``{root:<name>}/<relpath>`` (or ``{root:<name>}`` for the
 460            root itself), written in the root line in place of the host path; `registry_codes` checks it
 461            against the entry. None writes ``unlabelled``, which no registry entry accepts.
 462
 463    Returns:
 464        ``{"entries", "links", "unreadable", "walk_errors", "bytes"}`` counts.
 465
 466    Raises:
 467        InputError: ``E_INPUT_REGISTRY`` for a pattern outside the grammar, a root that is not a directory,
 468            a literal include that is missing or cannot be ``lstat``-ed, or a ``label`` outside its form.
 469    """
 470    if label is not None and _label_tree(label) is None:
 471        raise InputError(CODE)
 472    top = os.fspath(root)
 473    found, errors = _collect(top, include, exclude, lambda path, rel: _row(path, rel, sha256))
 474    rows = [found[k] for k in sorted(found)]
 475    header = (*MANIFEST_COLUMNS, SHA256_COLUMN) if sha256 else MANIFEST_COLUMNS
 476    when = datetime.datetime.now(datetime.UTC).strftime("%Y-%m-%dT%H:%M:%SZ")
 477    text = f"# root={'unlabelled' if label is None else label}{HASH_TAG}\n"
 478    text += "\t".join(header) + "\n" + "".join("\t".join(r) + "\n" for r in rows)
 479    _atomic_write(Path(out), text.encode())
 480    _write_dirs(top, Path(out), found, include, exclude, written=when)
 481    return {
 482        "entries": len(rows),
 483        "links": sum(1 for r in rows if r[4]),
 484        "unreadable": sum(1 for r in rows if r[3].startswith("ERR:")),
 485        "walk_errors": len(errors),
 486        "bytes": sum(int(r[1]) for r in rows if r[1]),
 487    }

 490def _include_patterns(top: str, include: Sequence[str]) -> list[str]:
 491    """`_collect`'s selecting patterns: a literal include naming a real directory selects everything below it."""
 492    out = []
 493    for pattern in include:
 494        path = os.path.join(top, pattern)
 495        literal_dir = not digest._WILD.search(pattern) and os.path.isdir(path) and not os.path.islink(path)
 496        out.append(f"{pattern}/**" if literal_dir else pattern)
 497    return out

 500def _may_hold(rel: str, patterns: Sequence[str], exclude: Sequence[str]) -> bool:
 501    """Whether the dir ``rel`` could hold a selected entry.
 502
 503    It could when some pattern reaches below it and no exclude takes it whole (matches it, or matches it with
 504    ``/**`` stripped).
 505    """
 506    parts = rel.split("/")
 507    if patterns and not any(digest.reaches_below(p.split("/"), parts) for p in patterns):
 508        return False
 509    whole = [e[:-3] if e.endswith("/**") else e for e in exclude]
 510    return not any(digest.pattern_matches(rel, e) for e in whole)

 513def _row_dirs(relpaths: Iterable[str]) -> dict[str, set[str]]:
 514    """Each dir holding a row, and each ancestor up to the top (``.``), with the names of the rows directly in it."""
 515    out: dict[str, set[str]] = {".": set()}
 516    for rel in relpaths:
 517        parts = rel.split("/")
 518        for index in range(len(parts) - 1):
 519            out.setdefault("/".join(parts[: index + 1]), set())
 520        out.setdefault("/".join(parts[:-1]) or ".", set()).add(parts[-1])
 521    return out

 524def _open(reldir: str, patterns: Sequence[str]) -> bool:
 525    """Whether a name added in or below ``reldir`` could be selected.
 526
 527    It could when a pattern reaching below it has ``**``, or a wildcard component at the depth of ``reldir``'s
 528    entries or deeper. With every reaching pattern literal (a literal file include), the selected names are fixed,
 529    and listing the dir could find nothing.
 530    """
 531    if not patterns:
 532        return True
 533    rel = [] if reldir == "." else reldir.split("/")
 534    for pattern in patterns:
 535        parts = pattern.split("/")
 536        if not digest.reaches_below(parts, rel):
 537            continue
 538        if "**" in parts or any(digest._WILD.search(part) for part in parts[len(rel) :]):
 539            return True
 540    return False

 543def _join(reldir: str, name: str) -> str:
 544    return name if reldir == "." else f"{reldir}/{name}"

 547def _added(
 548    top: str, reldir: str, rows: set[str], known: set[str], patterns: Sequence[str], exclude: Sequence[str]
 549) -> bool:
 550    """Whether ``reldir`` holds an entry added since the manifest was written; never follows a link.
 551
 552    That is a selected non-dir entry that is not a row, or a subdir that may hold a selected entry and is not
 553    ``known``. Entries that went away are the row scan's to report.
 554
 555    Raises:
 556        OSError: When the dir cannot be listed.
 557    """
 558    path = top if reldir == "." else os.path.join(top, reldir)
 559    with os.scandir(path) as entries:
 560        for entry in entries:
 561            rel = _join(reldir, entry.name)
 562            if entry.is_dir(follow_symlinks=False):
 563                if rel not in known and _may_hold(rel, patterns, exclude):
 564                    return True
 565            elif entry.name not in rows and _selected(rel, patterns, exclude):
 566                return True
 567    return False

 570def _label_tree(label: str) -> tuple[str, str] | None:
 571    """``(root name, normalised relpath)`` of a ``{root:<name>}/<relpath>`` label, or None outside that form.
 572
 573    The relpath must be relative with no ``..``, no backslash and no `` hash=``; ``{root:<name>}`` alone is ``"."``.
 574    """
 575    match = _LABEL.fullmatch(label)
 576    if match is None:
 577        return None
 578    rel = match.group(2) or "."
 579    parts = rel.split("/")
 580    if rel.startswith("/") or "\\" in rel or ".." in parts or " hash=" in rel:
 581        return None
 582    return match.group(1), os.path.normpath(rel)

 585def _write_dirs(
 586    top: str,
 587    out: Path,
 588    relpaths: Iterable[str],
 589    include: Sequence[str],
 590    exclude: Sequence[str],
 591    *,
 592    written: str | None = None,
 593) -> None:
 594    """Write the dirs sidecar of a manifest (see `DIRS_SUFFIX`).
 595
 596    From the top, each open dir (`_open`) is listed and each subdir that may hold a selected entry (`_may_hold`)
 597    is recorded and, when open, walked in turn; a link is never followed. The row dirs are recorded too.
 598    """
 599    patterns = _include_patterns(top, include)
 600    known = set(_row_dirs(relpaths))
 601    pending = ["."]
 602    while pending:
 603        reldir = pending.pop()
 604        if not _open(reldir, patterns):
 605            continue
 606        path = top if reldir == "." else os.path.join(top, reldir)
 607        try:
 608            with os.scandir(path) as entries:
 609                subdirs = [_join(reldir, e.name) for e in entries if e.is_dir(follow_symlinks=False)]
 610        except OSError:
 611            continue
 612        for rel in subdirs:
 613            if _may_hold(rel, patterns, exclude):
 614                known.add(rel)
 615                pending.append(rel)
 616    head = [] if written is None else [f"{WRITTEN_PREFIX}{written}"]
 617    text = "\n".join([*head, DIRS_HEADER, *sorted(known)]) + "\n"
 618    _atomic_write(Path(os.fspath(out) + DIRS_SUFFIX), text.encode())

 632def dir_changes(inp: RegisteredInput, profile: Profile, rows: Sequence[Row]) -> bool:
 633    """Whether a dir of the input's dirs sidecar holds an entry added since the manifest was written.
 634
 635    Each listed dir is checked for a selected non-dir entry that is not one of the manifest's rows there, or a
 636    subdir that may hold a selected entry (`_may_hold`) and is not in the sidecar; an entry the input's
 637    ``include``/``exclude`` leave out is not counted, and one that went away is the row scan's to report. A dir
 638    that is gone is skipped (its rows are missing); one that cannot be listed otherwise counts as changed. No
 639    sidecar (a manifest written before it existed) checks nothing, nor does ``extras: ignore``; an unreadable or
 640    malformed one counts as changed.
 641    """
 642    if inp.extras == "ignore":
 643        return False
 644    try:
 645        lines = dirs_sidecar(inp, profile).read_text().splitlines()
 646        if lines and lines[0].startswith(WRITTEN_PREFIX):
 647            lines = lines[1:]
 648    except FileNotFoundError:
 649        return False
 650    except (OSError, UnicodeDecodeError, InputError):
 651        return True
 652    if (
 653        not lines
 654        or lines[0] != DIRS_HEADER
 655        or any(not d or d.startswith("/") or ".." in d.split("/") for d in lines[1:])
 656    ):
 657        return True
 658    known = set(lines[1:])
 659    top = os.fspath(input_top(inp, profile))
 660    patterns = _include_patterns(top, inp.include)
 661    in_dir = _row_dirs(r.relpath for r in rows)
 662    for reldir in sorted(d for d in known if _open(d, patterns)):
 663        try:
 664            if _added(top, reldir, in_dir.get(reldir, set()), known, patterns, inp.exclude):
 665                return True
 666        except (FileNotFoundError, NotADirectoryError):
 667            continue
 668        except OSError:
 669            return True
 670    return False

 673def _atomic_write(path: Path, data: bytes, mode: int | None = None) -> None:
 674    """Write ``data`` through a temporary file and a rename; ``mode`` defaults to ``0o666`` less the umask."""
 675    path.parent.mkdir(parents=True, exist_ok=True)
 676    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=f".{path.name}.", suffix=".tmp")
 677    try:
 678        with os.fdopen(fd, "wb") as handle:
 679            handle.write(data)
 680            handle.flush()
 681            os.fsync(handle.fileno())
 682        if mode is None:
 683            mask = os.umask(0)
 684            os.umask(mask)
 685            mode = 0o666 & ~mask
 686        os.chmod(tmp, mode)
 687        os.replace(tmp, path)
 688    finally:
 689        if os.path.exists(tmp):
 690            os.unlink(tmp)

 760def _parent_state(top: str, rel: str, seen: dict[str, str]) -> str:
 761    """``ok``, ``missing`` or ``changed`` (a component is a symlink or not a directory) for ``rel``'s parent."""
 762    parts = rel.split("/")[:-1]
 763    for index in range(len(parts)):
 764        prefix = "/".join(parts[: index + 1])
 765        state = seen.get(prefix)
 766        if state is None:
 767            try:
 768                st = os.lstat(os.path.join(top, prefix))
 769                state = "ok" if stat.S_ISDIR(st.st_mode) else "changed"
 770            except (FileNotFoundError, NotADirectoryError):
 771                state = "missing"
 772            except OSError:
 773                state = "changed"
 774            seen[prefix] = state
 775        if state != "ok":
 776            return state
 777    return "ok"

 780def scan(
 781    inp: RegisteredInput,
 782    profile: Profile,
 783    rows: Sequence[Row],
 784    tops: Sequence[str],
 785    overlay: Mapping[str, int] | None = None,
 786) -> Scan:
 787    """``lstat`` every manifest row under the input (and its original), reading no file content.
 788
 789    Args:
 790        inp: The registered input.
 791        profile: The site profile.
 792        rows: Its manifest rows (`manifest_rows`).
 793        tops: The trees a link may point into, besides the input itself.
 794        overlay: Accepted ``mtime_ns`` per relpath, which an mtime difference may equal.
 795
 796    Returns:
 797        The counts: ``missing`` (no entry, or a parent dir gone), ``size_diff`` (a file row whose entry is no
 798        longer a regular file under real directories, or has another size), ``mtime_diff`` (same size,
 799        another mtime, not the accepted one), ``link_diff`` (a link row whose entry is no longer that link),
 800        ``links_out`` (a link row whose recorded target lies outside the input and ``tops``), and, for an
 801        input with ``original``, ``paired`` (file rows other than the top-level ``dataset_description.json``)
 802        and ``unlinked`` (paired rows whose original is missing or another inode).
 803
 804    Raises:
 805        InputError: ``E_INPUT_REGISTRY`` when a root is not in the profile.
 806    """
 807    top = os.fspath(input_top(inp, profile))
 808    original = original_top(inp, profile)
 809    allowed = [top, *tops]
 810    accepted = overlay or {}
 811    seen: dict[str, str] = {}
 812    files = links = missing = size_diff = mtime_diff = link_diff = links_out = unlinked = paired = 0
 813    present: list[tuple[Row, str, int]] = []
 814    for row in rows:
 815        path = os.path.join(top, row.relpath)
 816        if row.link_target is not None:
 817            links += 1
 818            if not link_resolves_inside(path, row.link_target, allowed):
 819                links_out += 1
 820        else:
 821            files += 1
 822        state = _parent_state(top, row.relpath, seen)
 823        if state == "missing":
 824            missing += 1
 825            continue
 826        try:
 827            st = None if state != "ok" else os.lstat(path)
 828        except (FileNotFoundError, NotADirectoryError):
 829            missing += 1
 830            continue
 831        except OSError:
 832            st = None
 833        if row.link_target is not None:
 834            try:
 835                same = st is not None and stat.S_ISLNK(st.st_mode) and os.readlink(path) == row.link_target
 836            except OSError:
 837                same = False
 838            link_diff += not same
 839            continue
 840        if st is None or not stat.S_ISREG(st.st_mode) or st.st_size != row.size:
 841            size_diff += 1
 842            continue
 843        if st.st_mtime_ns != row.mtime_ns and accepted.get(row.relpath) != st.st_mtime_ns:
 844            mtime_diff += 1
 845        present.append((row, path, st.st_mtime_ns))
 846        if original is not None and row.relpath != DESCRIPTION:
 847            paired += 1
 848            try:
 849                other = os.lstat(os.path.join(original, row.relpath))
 850                unlinked += (other.st_dev, other.st_ino) != (st.st_dev, st.st_ino)
 851            except OSError:
 852                unlinked += 1
 853    return Scan(files, links, missing, size_diff, mtime_diff, link_diff, links_out, unlinked, paired, present)

 857    inp: RegisteredInput,
 858    profile: Profile,
 859    step_inputs: Mapping[str, RegisteredInput],
 860    overlay: Mapping[str, int] | None = None,
 861) -> list[str]:
 862    """The gate's pre-check of one registered input: ``lstat`` of every manifest row, no file content.
 863
 864    Args:
 865        inp: The registered input.
 866        profile: The site profile.
 867        step_inputs: The registered inputs the step declares (``lock_inputs``); a link may point into any of
 868            them.

 872        Sorted codes: ``E_INPUT_MISSING`` for a missing row, ``E_INPUT_MISMATCH`` for a size, mtime, link or

1008def verify(
1009    inp: RegisteredInput,
1010    profile: Profile,
1011    *,
1012    now: datetime.datetime,
1013    others: Iterable[RegisteredInput] = (),
1014) -> Verification:
1015    """Re-hash every file row of a registered input (xxh3-128) and count the differences; nothing is written.
1016
1017    Args:
1018        inp: The registered input.
1019        profile: The site profile.
1020        now: The record's date.
1021        others: The other registry entries; a link into one of them is not counted in ``links_out``.
1022
1023    Returns:
1024        The verification. ``status`` is ``ok`` iff ``missing``, ``size``, ``hash``, ``links``, ``extra`` and
1025        ``unlinked`` are 0. ``hash`` counts files whose hash differs or that cannot be read; ``mtime_only``
1026        files with the recorded hash and another mtime; ``links`` changed link rows; ``extra`` selected
1027        entries absent from the manifest (0 with ``extras: ignore``); ``links_out`` links outside the input
1028        and ``others`` (it does not change the status). An ``ok`` record with ``mtime_only`` above 0 carries
1029        the overlay of the current mtimes.
1030
1031    Raises:
1032        InputError: ``E_INPUT_REGISTRY`` when the manifest cannot be read (`manifest_rows`).
1033    """
1034    rows = manifest_rows(inp, profile)
1035    found = scan(inp, profile, rows, _tops(others, profile))
1036    hashes = list(fastio.ordered_map(lambda item: _xxh(item[1]), found.present, threads=VERIFY_THREADS))
1037    hash_diff = mtime_only = 0
1038    overlay: dict[str, int] = {}
1039    for (row, _, mtime), xxh in zip(found.present, hashes):
1040        if xxh != row.xxh3_128:
1041            hash_diff += 1
1042        elif mtime != row.mtime_ns:
1043            mtime_only += 1
1044            overlay[row.relpath] = mtime
1045    top = os.fspath(input_top(inp, profile))
