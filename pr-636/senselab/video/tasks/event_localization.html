senselab.video.tasks.event_localization

Event localization

Finds when labelled events occur in a long recording by asking a video- or audio-language model about overlapping windows, and scores the result against annotated segments. The model call and the media cutting stay with the caller; this module holds the parts that do not depend on a model.

from senselab.video.tasks.event_localization import build_prompt, localize_events, segment_list_schema

labels = ["hands flapping", "jumping", "rocking", "spinning"]
family = {"task": "List every interval in which the child shows one of the behaviours below.",
          "labels": {l: "definition" for l in labels}}
schema = segment_list_schema(labels)          # JSON schema with `label` as an enum, for constrained decoding


def ask(window):                              # window = absolute (start, end) in seconds
    clip = cut(video, *window)                # caller's media cutting
    return model(clip, build_prompt(family, window[1] - window[0]), json_schema=schema)


result = localize_events(duration_s, ask, labels, win_s=30, stride_s=20)
result.segments       # [{"label", "start_s", "end_s", "score", "n_windows", ...}] on the recording's time axis
result.counts()       # parse counts: unparsed, salvaged, bad_item, bad_label, folded, bad_order, clamped, empty
  • plan_windows / split_window / shift_segments: windows that cover the recording with a fixed overlap, halving of a refused window, window-relative to absolute times.
  • segment_list_schema and parse_reply: the closed label set as a JSON-schema enum, and validation of a reply (labels outside the set, bad or reversed times, clamping, a reply cut off mid-list). strict=True accepts only exact labels.
  • merge_windows: per label, segments that overlap or lie within gap_s are chained; the score is the fraction of the covering windows that saw the segment.
  • metrics: mAP over tIoU thresholds (ActivityNet / OpenTAD convention), segment and point-event precision / recall / F1, per-label presence with prevalence, balanced accuracy and MCC, repeat-count error, variant confusion, shuffled and whole-recording floors, rasterize for frame-wise protocols.

Design, measurements and the window time-axis findings: specs/20261105-video-event-localization/design.md.

def event_pairs( preds: Sequence[Mapping[str, Any]], gts: Sequence[Mapping[str, Any]], tol_s: float, by_label: bool = False) -> List[Tuple[Mapping[str, Any], Mapping[str, Any]]]: 
152def event_pairs(
153    preds: Sequence[Seg], gts: Sequence[Seg], tol_s: float, by_label: bool = False
154) -> List[Tuple[Seg, Seg]]:
155    """One-to-one matches of point events: segment centres within ``tol_s`` seconds, closest pairs first per video."""
156    pv: Dict[Any, List[Tuple[float, Seg]]] = defaultdict(list)
157    gv: Dict[Any, List[Tuple[float, Seg]]] = defaultdict(list)
158    for p in preds:
159        pv[p["video"]].append(((float(p["start_s"]) + float(p["end_s"])) / 2, p))
160    for g in gts:
161        gv[g["video"]].append(((float(g["start_s"]) + float(g["end_s"])) / 2, g))
162    out: List[Tuple[Seg, Seg]] = []
163    for v, ps in pv.items():
164        gl = gv.get(v, [])
165        pairs = sorted(
166            (abs(a - b), i, j)
167            for i, (a, p) in enumerate(ps)
168            for j, (b, g) in enumerate(gl)
169            if abs(a - b) <= tol_s and (not by_label or p["label"] == g["label"])
170        )
171        ui, uj = set(), set()
172        for _, i, j in pairs:
173            if i in ui or j in uj:
174                continue
175            ui.add(i)
176            uj.add(j)
177            out.append((ps[i][1], gl[j][1]))
178    return out

def event_prf( preds: Sequence[Mapping[str, Any]], gts: Sequence[Mapping[str, Any]], tol_s: float, by_label: bool = False) -> Dict[str, float]: 
181def event_prf(preds: Sequence[Seg], gts: Sequence[Seg], tol_s: float, by_label: bool = False) -> Dict[str, float]:
182    """Precision, recall and F1 of point events.
183
184    Segment centres within ``tol_s`` seconds match one-to-one, closest pairs first.
185    """
186    tp = len(event_pairs(preds, gts, tol_s, by_label))
187    fp, fn = len(preds) - tp, len(gts) - tp
188    pr, rc = _div(tp, tp + fp), _div(tp, tp + fn)
189    return {"tp": tp, "fp": fp, "fn": fn, "precision": pr, "recall": rc, "f1": _div(2 * pr * rc, pr + rc)}

def localize_events( duration_s: float, ask: Callable[[Tuple[float, float]], str], labels: Sequence[str], win_s: float = 30.0, stride_s: float = 20.0, gap_s: float = 1.0, point_labels: Sequence[str] = (), min_win_s: float = 4.0, strict: bool = False, **merge_kwargs: Any) -> LocalizationResult: 
42def localize_events(
43    duration_s: float,
44    ask: Callable[[Window], str],
45    labels: Sequence[str],
46    win_s: float = 30.0,
47    stride_s: float = 20.0,
48    gap_s: float = 1.0,
49    point_labels: Sequence[str] = (),
50    min_win_s: float = 4.0,
51    strict: bool = False,
52    **merge_kwargs: Any,  # noqa: ANN401
53) -> LocalizationResult:
54    """Ask a model about overlapping windows of a recording and merge the answers into one segment list.
55
56    Args:
57        duration_s: Length of the recording in seconds.
58        ask: Called with an absolute ``(start, end)`` window; returns the model's reply for that stretch. The reply's
59            times are relative to the window start. It raises `WindowTooLarge` for a window the model refuses by size.
60        labels: The closed label set the reply must use.
61        win_s: Window length in seconds.
62        stride_s: Distance between window starts in seconds.
63        gap_s: Gap between same-label segments that `merge_windows` bridges.
64        point_labels: Labels that are point events (zero-length segments are kept).
65        min_win_s: A refused window is halved until its halves would be shorter than twice this.
66        strict: Pass to `parse_reply`.
67        **merge_kwargs: Further `merge_windows` arguments (``min_len_s``).
68
69    Returns:
70        The merged absolute segments with the per-window replies, parse counts and the number of split / unfit windows.
71    """
72    result = LocalizationResult(segments=[])
73
74    def one(w: Window) -> None:
75        try:
76            text = ask(w)
77        except WindowTooLarge:
78            if w[1] - w[0] > 2 * min_win_s:
79                result.n_split += 1
80                for h in split_window(w, 2):
81                    one(h)
82            else:
83                result.n_unfit += 1
84            return
85        segs, counts = parse_reply(text, labels, w[1] - w[0], point_labels=point_labels, strict=strict)
86        result.windows.append(WindowReply((w[0], w[1]), text, shift_segments(segs, w[0]), counts))
87
88    for w in plan_windows(duration_s, win_s, stride_s):
89        one(w)
90    result.segments = merge_windows([(r.window, r.segments) for r in result.windows], gap_s=gap_s, **merge_kwargs)
91    return result

def map_tal( preds: Sequence[Mapping[str, Any]], gts: Sequence[Mapping[str, Any]], thresholds: Sequence[float] = (0.3, 0.4, 0.5, 0.6, 0.7), classes: Optional[Sequence[str]] = None) -> Dict[str, Any]: 
 81def map_tal(
 82    preds: Sequence[Seg],
 83    gts: Sequence[Seg],
 84    thresholds: Sequence[float] = TIOUS,
 85    classes: Optional[Sequence[str]] = None,
 86) -> Dict[str, Any]:
 87    """Per-class AP and mAP at tIoU thresholds (the ActivityNet / OpenTAD convention).
 88
 89    Predictions of a class are ranked by score over all videos and each is matched to the unmatched ground truth of the
 90    same video and class with the highest tIoU of at least the threshold; AP is the area under the all-point
 91    interpolated precision-recall curve. Classes with no ground truth are left out of the mean.
 92
 93    Returns:
 94        ``{"map": {thr: value}, "avg_map": mean over thresholds, "ap": {class: {thr: value}}, "n_gt": {class: n}}``
 95        with values as fractions.
 96    """
 97    cls = sorted(set(classes) if classes is not None else {g["label"] for g in gts})
 98    ap: Dict[str, Dict[float, float]] = {}
 99    n_gt: Dict[str, int] = {}
100    for c in cls:
101        g = [x for x in gts if x["label"] == c]
102        n_gt[c] = len(g)
103        if not g:
104            continue
105        p = [x for x in preds if x["label"] == c]
106        ap[c] = {t: _ap(p, g, t) for t in thresholds}
107    m = {t: _div(sum(ap[c][t] for c in ap), len(ap)) for t in thresholds}
108    return {"map": m, "avg_map": _div(sum(m.values()), len(m)), "ap": ap, "n_gt": n_gt}

def merge_windows( per_window: Sequence[Tuple[Tuple[float, float], Sequence[Dict[str, Any]]]], gap_s: float = 1.0, min_len_s: float = 0.0) -> List[Dict[str, Any]]: 
18def merge_windows(
19    per_window: Sequence[Tuple[Window, Sequence[Segment]]], gap_s: float = 1.0, min_len_s: float = 0.0
20) -> List[Segment]:
21    """Merge per-window segments (absolute times) into one segment list.
22
23    Per label, the segments of all windows are sorted by start and chained into one segment when they overlap or are at
24    most ``gap_s`` apart (the union of the members). ``score`` is the number of distinct windows that contributed a
25    member divided by the number of windows that cover the merged segment's midpoint, in ``(0, 1]`` (1 for a one-window
26    recording), so a segment seen in both overlapping windows outranks one seen in a single window. ``n_windows`` and
27    ``n_members`` are kept. Values of the optional ``extra`` mapping (for example an attribute the model reported) are
28    carried as the most frequent value among the members.
29
30    Args:
31        per_window: ``((t0, t1), segments)`` per window, segments with absolute ``start_s``/``end_s`` and a ``label``.
32        gap_s: Largest gap between two segments of the same label that is bridged.
33        min_len_s: Merged segments shorter than this are dropped; 0 keeps point events.
34
35    Returns:
36        Segments with ``label``, ``start_s``, ``end_s``, ``score``, ``n_windows``, ``n_members`` (and ``extra``),
37        sorted by start, end and label.
38    """
39    wins = [w for w, _ in per_window]
40    items: List[_Member] = []
41    for wi, (_, segs) in enumerate(per_window):
42        for s in segs:
43            items.append((s["label"], float(s["start_s"]), float(s["end_s"]), wi, s.get("extra")))
44    out: List[Segment] = []
45    for label in sorted({i[0] for i in items}):
46        mine = sorted((i for i in items if i[0] == label), key=lambda x: (x[1], x[2]))
47        cur: List[_Member] = []
48        cur_end = -1e18
49        groups: List[List[_Member]] = []
50        for it in mine:
51            if cur and it[1] <= cur_end + gap_s:
52                cur.append(it)
53                cur_end = max(cur_end, it[2])
54            else:
55                if cur:
56                    groups.append(cur)
57                cur, cur_end = [it], it[2]
58        if cur:
59            groups.append(cur)
60        for g in groups:
61            s0, e0 = min(x[1] for x in g), max(x[2] for x in g)
62            if e0 - s0 < min_len_s:
63                continue
64            nwin = len({x[3] for x in g})
65            cov = max(_cover(wins, (s0 + e0) / 2.0), nwin, 1)
66            seg: Segment = {
67                "label": label,
68                "start_s": round(s0, 3),
69                "end_s": round(e0, 3),
70                "score": round(nwin / cov, 4),
71                "n_windows": nwin,
72                "n_members": len(g),
73            }
74            votes: Dict[str, Dict[str, int]] = {}
75            for x in g:
76                for k, v in (x[4] or {}).items():
77                    votes.setdefault(k, {}).setdefault(v, 0)
78                    votes[k][v] += 1
79            if votes:
80                seg["extra"] = {k: max(sorted(d), key=lambda v: d[v]) for k, d in votes.items()}
81            out.append(seg)
82    out.sort(key=lambda s: (s["start_s"], s["end_s"], s["label"]))
83    return out

Values of the optional extra mapping (for example an attribute the model reported) are carried as the most frequent value among the members.

Arguments:

  • min_len_s: Merged segments shorter than this are dropped; 0 keeps point events.
Returns:

def parse_reply( text: str, labels: Sequence[str], window_s: float, aliases: Optional[Mapping[str, str]] = None, point_labels: Sequence[str] = (), strict: bool = False) -> Tuple[List[Dict[str, Any]], Dict[str, int]]: 
148def parse_reply(
149    text: str,
150    labels: Sequence[str],
151    window_s: float,
152    aliases: Optional[Mapping[str, str]] = None,
153    point_labels: Sequence[str] = (),
154    strict: bool = False,
155) -> Tuple[List[Segment], Dict[str, int]]:
156    """Validate a model reply into segments relative to the window, and count what was dropped or repaired.
157
158    A reply is read as a JSON list (optionally in a code fence, after a ``<think>`` block, or in a dict under
159    ``segments``). A reply cut off in the middle of the list keeps its complete objects (``salvaged``). Items are
160    dropped when they are not objects, have no usable time, name a label outside ``labels``, or end before they
161    start; times outside ``[0, window_s]`` are clamped. A segment of zero length is kept only for a label in
162    ``point_labels``.
163
164    Args:
165        text: The model's reply.
166        labels: The closed label set.
167        window_s: Window length in seconds.
168        aliases: Extra spellings mapped to a label (only used when ``strict`` is false).
169        point_labels: Labels that are point events (``start_s`` may equal ``end_s``).
170        strict: Accept a label only when it is spelled exactly as in ``labels`` and times only as numbers.
171
172    Returns:
173        ``(segments, counts)``. Counts: ``unparsed`` (no JSON list and not an explicit "none"), ``salvaged`` (a
174        truncated list was repaired), ``bad_item``, ``bad_label``, ``folded`` (label accepted after case/underscore
175        folding or an alias), ``bad_order``, ``clamped``, ``empty`` (the reply is an empty list). Extra string/number
176        keys of an item are carried in ``extra``.
177    """
178    counts = dict.fromkeys(COUNT_KEYS, 0)
179    body = _THINK.sub("", text or "").strip()
180    val = _first_list(body)
181    if val is None and body.lstrip("`").lstrip().lstrip("json").lstrip().startswith("["):
182        objs = [_loads(m) for m in _OBJECT.findall(body)]
183        val = [o for o in objs if isinstance(o, dict)] or None
184        counts["salvaged"] = 1 if val else 0
185    if val is None:
186        counts["empty" if _NONE_REPLY.fullmatch(body or "") else "unparsed"] = 1
187        return [], counts
188    if not val:
189        counts["empty"] = 1
190        return [], counts
191    exact = {str(x): str(x) for x in labels}
192    fold = {_norm(x): str(x) for x in labels}
193    for a, b in (aliases or {}).items():
194        fold[_norm(a)] = b
195    out: List[Segment] = []
196    for it in val:
197        if not isinstance(it, dict):
198            counts["bad_item"] += 1
199            continue
200        raw = next((it[k] for k in LABEL_KEYS if k in it), "")
201        lab = exact.get(str(raw))
202        if lab is None and not strict:
203            lab = fold.get(_norm(raw))
204            counts["folded"] += lab is not None
205        if lab is None:
206            counts["bad_label"] += 1
207            continue
208        keys_s = START_KEYS[:1] if strict else START_KEYS
209        keys_e = END_KEYS[:1] if strict else END_KEYS
210        s = next((_seconds(it[k]) for k in keys_s if k in it and _seconds(it[k]) is not None), None)
211        e = next((_seconds(it[k]) for k in keys_e if k in it and _seconds(it[k]) is not None), None)
212        if s is None and e is None:
213            counts["bad_item"] += 1
214            continue
215        s = e if s is None else s
216        e = s if e is None else e
217        assert s is not None and e is not None
218        if e < s:
219            counts["bad_order"] += 1
220            continue
221        s2, e2 = min(max(s, 0.0), window_s), min(max(e, 0.0), window_s)

224        if e2 <= s2 and lab not in point_labels:
225            counts["bad_order"] += 1
226            continue
227        seg: Segment = {"label": lab, "start_s": round(s2, 3), "end_s": round(e2, 3)}
228        skip = set(LABEL_KEYS) | set(START_KEYS) | set(END_KEYS)
229        extra = {k: str(v) for k, v in it.items() if k not in skip and isinstance(v, (str, int, float, bool))}
230        if extra:
231            seg["extra"] = extra
232        out.append(seg)
233    return out, counts

def plan_windows( duration_s: float, win_s: float = 30.0, stride_s: float = 20.0, min_tail_s: float = 5.0) -> List[Tuple[float, float]]: 
12def plan_windows(
13    duration_s: float, win_s: float = 30.0, stride_s: float = 20.0, min_tail_s: float = 5.0
14) -> List[Window]:
15    """Plan overlapping windows ``[t0, t1]`` that cover ``[0, duration_s]``.
16
17    A recording no longer than ``win_s + min_tail_s`` is one window. Otherwise windows start every ``stride_s`` and the
18    last one is pulled back so that it ends at ``duration_s`` and is ``win_s`` long, so a short remainder never
19    becomes a tiny window. Every instant lies in at least one window.
20
21    Args:
22        duration_s: Length of the recording in seconds.
23        win_s: Window length in seconds.
24        stride_s: Distance between window starts in seconds; at most ``win_s``.
25        min_tail_s: Slack beyond ``win_s`` that still counts as a single window.
26
27    Returns:
28        ``(start, end)`` pairs in seconds on the recording's own time axis; empty for a non-positive duration.
29
30    Raises:
31        ValueError: If ``win_s`` or ``stride_s`` is not positive or ``stride_s`` exceeds ``win_s``.
32    """
33    if duration_s <= 0:
34        return []
35    if win_s <= 0 or stride_s <= 0 or stride_s > win_s:
36        raise ValueError("need 0 < stride_s <= win_s")
37    if duration_s <= win_s + min_tail_s:
38        return [(0.0, float(duration_s))]
39    out: List[Window] = []
40    t0 = 0.0
41    while True:
42        t1 = min(t0 + win_s, duration_s)
43        if t1 >= duration_s - 1e-9:
44            t0 = max(0.0, duration_s - win_s)
45            if not out or out[-1][0] < t0 - 1e-9:
46                out.append((t0, float(duration_s)))
47            else:
48                out[-1] = (out[-1][0], float(duration_s))
49            break
50        out.append((t0, t1))
51        t0 += stride_s
52    return out

 86def rasterize(segments: Sequence[Segment], n_frames: int, fps: float, background: str = "background") -> List[str]:
 87    """Turn segments into one label per frame.
 88
 89    Frame ``i`` covers ``[i / fps, (i + 1) / fps)`` and takes the label of the segment that contains its centre. On an
 90    overlap the segment with the higher ``score`` wins (ties: the shorter one, then the earlier one).
 91
 92    Args:
 93        segments: Segments with ``label``, ``start_s``, ``end_s`` and an optional ``score`` (default 1).
 94        n_frames: Number of frames.
 95        fps: Frame rate.
 96        background: Label of a frame in no segment.
 97
 98    Returns:
 99        ``n_frames`` labels.
100    """
101    labels = [background] * n_frames
102    best: List[Optional[Tuple[float, float, float]]] = [None] * n_frames
103    for s in segments:
104        a, b = float(s["start_s"]), float(s["end_s"])
105        i0 = max(0, int(math.ceil(a * fps - 0.5)))
106        i1 = min(n_frames, int(math.ceil(b * fps - 0.5)))
107        key = (float(s.get("score", 1.0)), -(b - a), -a)
108        for i in range(i0, i1):
109            cur = best[i]
110            if cur is None or key > cur:
111                best[i] = key
112                labels[i] = s["label"]
113    return labels

125def seg_prf(preds: Sequence[Seg], gts: Sequence[Seg], iou: float, by_label: bool = True) -> Dict[str, float]:
126    """Micro precision, recall and F1 at one tIoU threshold.
127
128    A prediction is correct when it overlaps an unmatched ground truth of the same video (and label when ``by_label``)
129    at ``iou`` or more; predictions are visited by score and each takes its best-IoU ground truth.
130    """
131    gv: Dict[Any, List[Seg]] = defaultdict(list)
132    for g in gts:
133        gv[g["video"]].append(g)
134    used = {v: [False] * len(x) for v, x in gv.items()}
135    tp = 0
136    for p in sorted(preds, key=lambda x: -float(x.get("score", 1.0))):
137        best, bj = 0.0, -1

139            if used[p["video"]][j] or (by_label and g["label"] != p["label"]):

149    return {"tp": tp, "fp": fp, "fn": fn, "precision": pr, "recall": rc, "f1": _div(2 * pr * rc, pr + rc)}

20def tiou(a: Tuple[float, float], b: Tuple[float, float]) -> float:
21    """Temporal intersection over union of two ``(start, end)`` intervals."""
22    inter = min(a[1], b[1]) - max(a[0], b[0])
23    if inter <= 0:
24        return 0.0
25    union = (a[1] - a[0]) + (b[1] - b[0]) - inter
26    return inter / union if union > 0 else 0.0
