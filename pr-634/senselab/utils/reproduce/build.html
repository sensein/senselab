senselab.utils.reproduce.build

Stage 3 of a reproduce env: sdists built into wheels, recorded in wheels.lock and reused.

Each SdistSpec is built in its own build venv, made from its hashed build_env lock (and the env's torch lock when given), with uv build --wheel --no-build-isolation, an allowlisted environment (build_environment()), a fixed SOURCE_DATE_EPOCH and umask, inside the step sandbox's network namespace (wheel_build_argv()), so no index or host is reachable. Wheels are stored content-addressed, <wheelhouse>/<wheel sha256>/<wheel file>, and never overwritten. The first build records the wheel in wheels.lock; later calls reuse it after checking its sha256, and only rebuild builds again, adding a new wheel beside the old one.

wheels.lock has one whitespace-separated line per sdist under a column header::

# name version sdist_url sdist_sha256 build_env_sha256 patch_sha256s build_config_sha256 wheel_path wheel_sha256

patch_sha256s is a comma-joined list in manifest order, or -. build_config_sha256 is build_config_sha256(). wheel_path is relative to the wheelhouse. Reads and writes hold a flock on the sidecar <wheels.lock>senselab.utils.reproduce.lock; writes replace the file atomically. Paths in a spec are relative to the manifest repository, passed as base_dir.

 137def fetch_sdist(url: str, sha256: str, dest: Path) -> Path:
 138    """Download an sdist into ``dest`` and check its sha256; reuse a present copy that matches.
 139
 140    Args:
 141        url: The sdist URL (``https://`` or ``file://``).
 142        sha256: The manifest sha256.
 143        dest: The download directory; created if absent.
 144
 145    Returns:
 146        ``dest / <file name from the URL>``.
 147
 148    Raises:
 149        ReproduceLockError: If the downloaded bytes do not match ``sha256``; nothing is left behind.
 150    """
 151    dest.mkdir(parents=True, exist_ok=True)
 152    path = dest / _sdist_filename(url)
 153    if path.is_file() and file_sha256(path) == sha256:
 154        return path
 155    digest = hashlib.sha256()
 156    https = urllib.parse.urlsplit(url).scheme == "https"
 157    context = ssl.create_default_context(cafile=ca_bundle()) if https else None
 158    fd, part = tempfile.mkstemp(dir=dest, prefix=f".{path.name}.", suffix=".part")
 159    try:
 160        with (
 161            os.fdopen(fd, "wb") as out,
 162            urllib.request.urlopen(url, context=context, timeout=FETCH_TIMEOUT) as response,  # noqa: S310
 163        ):
 164            while chunk := response.read(1 << 20):
 165                digest.update(chunk)
 166                out.write(chunk)
 167        if digest.hexdigest() != sha256:
 168            raise ReproduceLockError(f"sdist {url}: sha256 {digest.hexdigest()} != manifest {sha256}")
 169        os.replace(part, path)
 170    finally:
 171        if os.path.exists(part):
 172            os.unlink(part)
 173    return path

 176def _copy_verified(refs: Iterable[FileRef], base_dir: Path, inputs: Path) -> None:
 177    """Copy hashed files from ``base_dir`` into ``inputs`` and check the copies' sha256s."""
 178    refs = list(refs)
 179    for ref in refs:
 180        source = base_dir / ref.file
 181        if not source.resolve().is_relative_to(base_dir.resolve()):
 182            raise ReproduceLockError(f"{ref.file} resolves outside {base_dir}")
 183        target = inputs / ref.file
 184        target.parent.mkdir(parents=True, exist_ok=True)
 185        shutil.copyfile(source, target)
 186    verify_files(refs, inputs)

 197def _tar_member(member: tarfile.TarInfo, dest: str) -> tarfile.TarInfo | None:
 198    """The ``data`` extraction filter, with a regular file's mode from :func:`_file_mode`."""
 199    filtered = tarfile.data_filter(member, dest)  # type: ignore[attr-defined,unused-ignore]
 200    if filtered is None or not member.isreg():
 201        return filtered
 202    return filtered.replace(mode=_file_mode(member.mode), deep=False)  # type: ignore[no-any-return,unused-ignore]

 205def _checked_member(member: tarfile.TarInfo, root: Path) -> tarfile.TarInfo:
 206    """The manual check used where ``tarfile.data_filter`` is missing (Python < 3.11.4).
 207
 208    Only regular files and directories pass, each with a relative name, no ``..`` component, inside ``root``.
 209
 210    Raises:
 211        ReproduceLockError: For any other member.
 212    """
 213    name = PurePosixPath(member.name)
 214    inside = (root / member.name).resolve().is_relative_to(root)
 215    if name.is_absolute() or ".." in name.parts or not inside or not (member.isreg() or member.isdir()):
 216        raise ReproduceLockError(f"tar member {member.name!r} is not a plain file or directory inside the archive")
 217    member.mode = _file_mode(member.mode) if member.isreg() else ((member.mode & 0o755) | 0o700)
 218    return member

 221def safe_extract(data: bytes, dest: Path) -> None:
 222    """Extract a tar archive's bytes into ``dest`` with the ``data`` filter, or the manual check without it.
 223
 224    Raises:
 225        tarfile.TarError: When the ``data`` filter refuses a member.
 226        ReproduceLockError: When the manual check refuses one.
 227    """
 228    dest.mkdir(parents=True, exist_ok=True)
 229    root = dest.resolve()
 230    with tarfile.open(fileobj=io.BytesIO(data)) as archive:
 231        if _HAS_DATA_FILTER:
 232            archive.extractall(dest, filter=_tar_member)
 233            return
 234        members = [_checked_member(m, root) for m in archive.getmembers()]
 235        trusted = {"filter": "fully_trusted"} if hasattr(tarfile, "fully_trusted_filter") else {}
 236        archive.extractall(dest, members=members, **trusted)  # type: ignore[arg-type,unused-ignore]

 239def _unpack(data: bytes, filename: str, dest: Path) -> Path:
 240    """Unpack sdist bytes into ``dest`` and return the single top-level directory.
 241
 242    File modes end up inside the wheel, so a regular file's mode is :func:`_file_mode` of its
 243    archive mode, whatever the caller's umask.
 244
 245    Args:
 246        data: The verified sdist bytes.
 247        filename: The sdist file name; ``.zip`` selects zip, anything else tar.
 248        dest: The unpack directory; must not exist.
 249    """
 250    dest.mkdir(parents=True)
 251    root = dest.resolve()
 252    if filename.endswith(".zip"):
 253        with zipfile.ZipFile(io.BytesIO(data)) as archive:
 254            members = archive.infolist()
 255            for member in members:
 256                if not (dest / member.filename).resolve().is_relative_to(root):
 257                    raise ReproduceLockError(f"{filename}: member {member.filename!r} leaves the unpack directory")
 258            archive.extractall(dest)
 259            for member in members:
 260                if not member.is_dir():
 261                    (dest / member.filename).chmod(_file_mode((member.external_attr >> 16) or 0o644))
 262    else:
 263        safe_extract(data, dest)
 264    tops = list(dest.iterdir())
 265    if len(tops) != 1 or not tops[0].is_dir():
 266        raise ReproduceLockError(f"{filename} does not unpack to one top-level directory")
 267    return tops[0]

 321def in_modules(sdist: SdistSpec, argv: list[str], modules_init: Path | None = None) -> list[str]:
 322    """Wrap ``argv`` so it runs with the sdist's modules loaded, without reading any dotfile.
 323
 324    Without ``build_modules`` ``argv`` is returned unchanged. With them it becomes ``bash --noprofile
 325    --norc -c '. <modules_init> && module load <m…> && exec "$@"'``; when :func:`needs_nvcc`, the
 326    script also fails unless ``nvcc`` is on ``PATH`` and exports ``CUDA_HOME`` from it.
 327
 328    Args:
 329        sdist: The sdist spec.
 330        argv: The command.
 331        modules_init: The profile's ``modules_init``.
 332
 333    Returns:
 334        The argv to run.
 335
 336    Raises:
 337        ReproduceLockError: ``E_MODULES_INIT`` when modules are needed and ``modules_init`` is missing.
 338    """
 339    if not sdist.build_modules:
 340        return argv
 341    init = modules_init_word(modules_init)
 342    loaded = " ".join(_safe_word(m, "build module") for m in sdist.build_modules)
 343    script = f". {init} && module load {loaded} && "
 344    if needs_nvcc(sdist):
 345        script += (
 346            f'if ! command -v nvcc >/dev/null; then echo "nvcc not on PATH after module load {loaded}" >&2; exit 1; fi'
 347            f" && {CUDA_HOME_FROM_NVCC} && "
 348        )
 349    script += 'exec "$@"'
 350    return ["bash", "--noprofile", "--norc", "-c", script, "bash", *argv]

 424def base_environment(uv: str, work_dir: Path) -> dict[str, str]:
 425    """The allowlisted environment for every subprocess of a build.
 426
 427    ``PATH`` is the system path; ``HOME`` and ``TMPDIR`` are under ``work_dir``; the locale is
 428    ``C.UTF-8``; ``SSL_CERT_FILE`` is :func:`ca_bundle`; ``UV_CACHE_DIR`` and
 429    ``UV_PYTHON_INSTALL_DIR`` are the caller's, resolved through uv when unset. Nothing else of the
 430    caller's environment is passed.
 431
 432    Args:
 433        uv: The uv executable.
 434        work_dir: The build work directory; ``home`` and ``tmp`` are created under it.
 435
 436    Returns:
 437        A new mapping.
 438    """
 439    home, tmp = work_dir / "home", work_dir / "tmp"
 440    home.mkdir(parents=True, exist_ok=True)
 441    tmp.mkdir(parents=True, exist_ok=True)
 442    env = {
 443        "PATH": SYSTEM_PATH,
 444        "HOME": str(home),
 445        "TMPDIR": str(tmp),
 446        "LANG": "C.UTF-8",
 447        "LC_ALL": "C.UTF-8",
 448        "UV_CACHE_DIR": _uv_location(uv, "UV_CACHE_DIR", ["cache", "dir"]),
 449        "UV_PYTHON_INSTALL_DIR": _uv_location(uv, "UV_PYTHON_INSTALL_DIR", ["python", "dir"]),
 450    }
 451    bundle = ca_bundle()
 452    if bundle is not None:
 453        env["SSL_CERT_FILE"] = bundle
 454    return env

 457def build_environment(
 458    sdist: SdistSpec, base: Mapping[str, str], venv: Path, module_env: Mapping[str, str] | None = None
 459) -> dict[str, str]:
 460    """The wheel build's environment: ``base``, module variables, the venv first on ``PATH``, declared and fixed ones.
 461
 462    The fixed variables are ``SOURCE_DATE_EPOCH``, ``TORCH_CUDA_ARCH_LIST`` (when declared) and
 463    :data:`OFFLINE_ENV`, so the build reaches no index or proxy.
 464
 465    Args:
 466        sdist: The sdist spec.
 467        base: :func:`base_environment`.
 468        venv: The build venv.
 469        module_env: :func:`module_environment`; its ``PATH`` follows the venv's ``bin``.
 470
 471    Returns:
 472        A new mapping.
 473
 474    Raises:
 475        ReproduceLockError: If a ``build_env_vars`` key would replace an allowlisted or fixed variable.
 476    """
 477    fixed = {"SOURCE_DATE_EPOCH": SOURCE_DATE_EPOCH, **OFFLINE_ENV}
 478    if sdist.torch_cuda_arch_list is not None:
 479        fixed["TORCH_CUDA_ARCH_LIST"] = sdist.torch_cuda_arch_list

 516def _store(wheel: Path, wheelhouse: Path, info: Mapping[str, object]) -> tuple[Path, str]:
 517    """Store a built wheel at ``<wheelhouse>/<sha256>/<name>``; an existing copy must hold the same bytes."""
 518    data = wheel.read_bytes()
 519    sha = hashlib.sha256(data).hexdigest()
 520    target = wheelhouse / sha / wheel.name
 521    if target.exists():
 522        if file_sha256(target) != sha:
 523            raise ReproduceLockError(f"{target} exists with other bytes than its directory name says")
 524        return target, sha
 525    target.parent.mkdir(parents=True, exist_ok=True)
 526    _place(data, target, replace=False)
 527    _place(
 528        (json.dumps(info, indent=2, sort_keys=True) + "\n").encode(), target.with_name(target.name + ".build-info.json")
 529    )
 530    return target, sha

 533def _place(data: bytes, target: Path, replace: bool = True) -> None:
 534    """Write ``data`` to a temp file beside ``target``, then move it into place in one step.
 535
 536    With ``replace=False`` the move is a hard link, which fails if ``target`` exists; nothing is overwritten.
 537    """
 538    fd, tmp = tempfile.mkstemp(dir=target.parent, prefix=f".{target.name}.", suffix=".tmp")
 539    try:
 540        with os.fdopen(fd, "wb") as handle:
 541            handle.write(data)
 542        os.chmod(tmp, 0o666 & ~_current_umask())
 543        if replace:
 544            os.replace(tmp, target)
 545        else:
 546            try:
 547                os.link(tmp, target)
 548            except FileExistsError as error:
 549                raise ReproduceLockError(f"{target} appeared while storing it; not overwritten") from error
 550    finally:
 551        if os.path.exists(tmp):
 552            os.unlink(tmp)

 555def build_wheel(
 556    sdist: SdistSpec,
 557    base_dir: Path,
 558    out_dir: Path,
 559    torch_lock: TorchSpec | None = None,
 560    *,
 561    python: str,
 562    work_dir: Path | None = None,
 563    sdist_cache: Path | None = None,
 564    modules_init: Path | None = None,
 565) -> tuple[Path, str]:
 566    """Build one sdist into a wheel in a fresh build venv.
 567
 568    In order: the build lock, torch lock and patches are copied from ``base_dir`` and the copies
 569    checked against their manifest sha256s; the sdist is fetched, its bytes read once, checked,
 570    unpacked and patched; the build venv is created and the verified torch lock, then the build
 571    lock, are installed ``--no-deps --require-hashes``; the modules' variables are captured
 572    (:func:`module_environment`); the wheel is built in the sandbox (:func:`wheel_build_argv`,
 573    :func:`build_environment`) and stored content-addressed in ``out_dir`` with a
 574    ``<wheel>.build-info.json`` beside it (uv, gcc and nvcc versions, modules, ``python``).
 575
 576    Args:
 577        sdist: The sdist spec.
 578        base_dir: The manifest repository.
 579        out_dir: The wheelhouse; the wheel goes to ``<out_dir>/<sha256>/<wheel file>``.
 580        torch_lock: The env's torch lock, installed into the build venv first when the build needs torch.
 581        python: The interpreter for the build venv (the env's ``python_exact``).
 582        modules_init: The profile's ``modules_init``, sourced before ``build_modules`` load.
 583        work_dir: Where the build venv, inputs and source go; must not exist yet. Kept when given;
 584            otherwise a temporary directory is used and removed.
 585        sdist_cache: Where the sdist is downloaded; defaults to ``work_dir / "sdist"``.
 586
 587    Returns:
 588        ``(wheel path, wheel sha256)``.
 589
 590    Raises:
 591        ReproduceLockError: If an input's sha256 differs, a patch does not apply, a command fails, the
 592            build yields other than one wheel, or the wheelhouse holds other bytes at the wheel's path.
 593    """
 594    if work_dir is None:
 595        with tempfile.TemporaryDirectory(prefix="reproduce-build-") as tmp:
 596            return build_wheel(
 597                sdist,
 598                base_dir,
 599                out_dir,
 600                torch_lock,
 601                python=python,
 602                work_dir=Path(tmp) / "w",
 603                sdist_cache=sdist_cache,
 604                modules_init=modules_init,
 605            )
 606    if work_dir.exists():
 607        raise ReproduceLockError(f"build work dir already exists: {work_dir}")
 608    inputs = work_dir / "inputs"
 609    build_lock = FileRef(file=sdist.build_env.lock, sha256=sdist.build_env.lock_sha256)
 610    _copy_verified([*_torch_ref(torch_lock), build_lock, *sdist.patches], base_dir, inputs)
 611    archive = fetch_sdist(sdist.url, sdist.sha256, sdist_cache or work_dir / "sdist")
 612    data = archive.read_bytes()
 613    if hashlib.sha256(data).hexdigest() != sdist.sha256:
 614        raise ReproduceLockError(f"sdist {archive}: sha256 changed after download")
 615
 616    uv = uv_executable()
 617    base = base_environment(uv, work_dir)
 618    src = _unpack(data, archive.name, work_dir / "src")
 619    for patch in sdist.patches:
 620        _apply_patch(inputs / patch.file, src, base)
 621
 622    venv = work_dir / "venv"
 623    build_python = venv / "bin" / "python"
 624    _run([uv, "venv", "--no-config", "--python", python, str(venv)], env=base)
 625    if torch_lock is not None:
 626        torch_file = inputs / torch_lock.lock
 627        index = torch_index_url(torch_lock, torch_file.read_text())
 628        _run(_install_argv(uv, build_python, torch_file, index, binary_only=False), env=base)
 629    _run(_install_argv(uv, build_python, inputs / sdist.build_env.lock, None, binary_only=True), env=base)
 630
 631    dist = work_dir / "dist"
 632    dist.mkdir(parents=True, exist_ok=True)
 633    env = build_environment(sdist, base, venv, module_environment(sdist, base, modules_init))
 634    env["UV_CACHE_DIR"] = str(work_dir / "tmp" / "uv-cache")
 635    writable = [src.parent, dist, work_dir / "home", work_dir / "tmp"]
 636    _run(wheel_build_argv(uv, build_python, src, dist, venv=venv, writable=writable), env=env)
 637    wheels = sorted(dist.glob("*.whl")) if dist.is_dir() else []
 638    if len(wheels) != 1:
 639        raise ReproduceLockError(f"building {sdist.name} produced {len(wheels)} wheels, expected 1")
 640    return _store(wheels[0], out_dir, _build_info(uv, sdist, python, env))

 665def _format_line(line: WheelLine) -> str:
 666    patches = ",".join(line.patch_sha256s) or "-"
 667    fields = [line.name, line.version, line.sdist_url, line.sdist_sha256, line.build_env_sha256, patches]
 668    fields += [line.build_config_sha256, line.wheel_path, line.wheel_sha256]
 669    for field in fields:

 672    return " ".join(fields)

 675def parse_wheels_lock(text: str, source: str = "wheels.lock") -> dict[str, WheelLine]:
 676    """Parse ``wheels.lock`` text.
 677
 678    Args:
 679        text: The file's contents.
 680        source: A name for error messages.
 681
 682    Returns:
 683        Lines keyed by normalised name.
 684
 685    Raises:
 686        ReproduceLockError: If a line does not have the nine columns, or a name repeats.
 687    """
 688    out: dict[str, WheelLine] = {}
 689    for number, raw in enumerate(text.splitlines(), 1):
 690        if not raw.strip() or raw.startswith("#"):
 691            continue
 692        fields = raw.split()
 693        if len(fields) != len(WheelLine._fields):
 694            raise ReproduceLockError(f"{source}:{number}: {len(fields)} columns, expected {len(WheelLine._fields)}")
 695        name, version, url, sdist_sha, build_sha, patch_field, config, wheel_path, wheel_sha = fields
 696        patches = () if patch_field == "-" else tuple(patch_field.split(","))
 697        line = WheelLine(name, version, url, sdist_sha, build_sha, patches, config, wheel_path, wheel_sha)
 698        key = normalize_name(line.name)
 699        if key in out:
 700            raise ReproduceLockError(f"{source}:{number}: {line.name} appears twice")
 701        out[key] = line
 702    return out

 711@contextmanager
 712def file_lock(path: Path, *, shared: bool = False) -> Iterator[None]:
 713    """Hold a ``flock`` on ``path`` (created if absent, with its directory), shared or exclusive.
 714
 715    The driver uses it only for shared (reader) locks; its exclusive locks are `exclusive_lock`. The file is
 716    opened read-write: on NFS ``flock`` becomes a POSIX lock, and a shared lock on a write-only descriptor fails
 717    with ``EBADF``. A lock taken on a file that is no longer at ``path`` (an `exclusive_lock` unlinks the file as
 718    it releases; on NFS the old file is silly-renamed and keeps a link) is dropped and taken again on the file now
 719    there. The check opens the path and compares ``fstat`` results (`lock_fd_matches_path`), never ``stat``,
 720    which NFS can answer from a cached dentry.
 721    """
 722    while True:
 723        path.parent.mkdir(parents=True, exist_ok=True)
 724        fd = os.open(path, os.O_RDWR | os.O_CREAT, 0o666)
 725        try:
 726            fcntl.flock(fd, fcntl.LOCK_SH if shared else fcntl.LOCK_EX)
 727            same = lock_fd_matches_path(fd, path)
 728        except BaseException:
 729            os.close(fd)
 730            raise
 731        if same:
 732            break
 733        fcntl.flock(fd, fcntl.LOCK_UN)
 734        os.close(fd)
 735    try:
 736        yield
 737    finally:
 738        fcntl.flock(fd, fcntl.LOCK_UN)
 739        os.close(fd)

 742def exclusive_lock(resource: Path) -> contextlib.AbstractContextManager[object]:
 743    """An exclusive `SharedFileLock` over ``resource``: ``flock`` on ``<resource>.lock``, the file `file_lock` takes.
 744
 745    It waits as long as a holder holds the lock (a reader's shared `file_lock` included), records its holder and
 746    heartbeat beside the lock, and leaves the directory's mode alone. It unlinks the lock file as it releases, which
 747    `file_lock` checks for; it checks the same itself after each acquire (`SharedFileLock`), so two writers never
 748    hold the old and the new file at once.
 749    """
 750    return SharedFileLock(resource, timeout=-1, manage_dir_mode=False)

 753def _locked(wheels_lock: Path, shared: bool) -> contextlib.AbstractContextManager[object]:
 754    """Hold the sidecar ``<wheels.lock>.lock``: a shared :func:`file_lock`, or an :func:`exclusive_lock`."""
 755    if shared:
 756        return file_lock(wheels_lock.with_name(wheels_lock.name + ".lock"), shared=True)

 760def _read_verified(wheels_lock: Path, expected_sha256: str | None) -> bytes | None:
 761    """Read ``wheels.lock`` once and check it against the manifest sha256 when one is given; ``None`` if absent."""
 762    if not wheels_lock.exists():
 763        if expected_sha256 is not None:
 764            raise ReproduceLockError(f"{wheels_lock} is missing; the manifest records sha256 {expected_sha256}")
 765        return None
 766    data = wheels_lock.read_bytes()
 767    actual = hashlib.sha256(data).hexdigest()
 768    if expected_sha256 is not None and actual != expected_sha256:
 769        raise ReproduceLockError(f"{wheels_lock}: sha256 {actual} != manifest {expected_sha256}")
 770    return data

 773def _write_atomic(path: Path, text: str) -> str:
 774    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=f".{path.name}.", suffix=".tmp")
 775    try:
 776        with os.fdopen(fd, "w") as handle:
 777            handle.write(text)
 778        os.chmod(tmp, 0o666 & ~_current_umask())
 779        os.replace(tmp, path)
 780    finally:
 781        if os.path.exists(tmp):
 782            os.unlink(tmp)
 783    return hashlib.sha256(text.encode()).hexdigest()

 830def record_wheel(
 831    wheels_lock: Path,
 832    sdist: SdistSpec,
 833    wheelhouse: Path,
 834    wheel: Path,
 835    wheel_sha256: str,
 836    torch_lock: LockRef | None = None,
 837    *,
 838    python: str,
 839    expected_sha256: str | None = None,
 840) -> tuple[WheelLine, str]:
 841    """Point the sdist's ``wheels.lock`` line at a stored wheel, replacing any earlier line for it.
 842
 843    The earlier wheel stays in the wheelhouse. The caller writes the returned sha256 into the
 844    manifest's ``wheels.lock_sha256``.
 845
 846    Args:
 847        wheels_lock: The file; created if absent.
 848        sdist: The sdist spec the wheel was built from.
 849        wheelhouse: The wheelhouse; ``wheel`` must lie under it.
 850        wheel: The stored wheel, ``<wheelhouse>/<sha256>/<file>``.
 851        wheel_sha256: Its sha256.
 852        torch_lock: The torch lock the build venv had, if any.
 853        python: The build venv's interpreter version.
 854        expected_sha256: The manifest's current ``wheels.lock_sha256``; the existing file is checked against it.
 855
 856    Returns:
 857        ``(line written, new wheels.lock sha256)``.
 858
 859    Raises:
 860        ReproduceLockError: If the wheel is not a wheel of ``sdist.name`` stored under its sha256, or the
 861            existing file does not match ``expected_sha256``.
 862    """
 863    parts = wheel.name.split("-")
 864    if not wheel.name.endswith(".whl") or len(parts) < 5 or normalize_name(parts[0]) != normalize_name(sdist.name):
 865        raise ReproduceLockError(f"{wheel.name} is not a wheel of {sdist.name}")
 866    relative = wheel.relative_to(wheelhouse).as_posix()
 867    if relative != f"{wheel_sha256}/{wheel.name}":
 868        raise ReproduceLockError(f"{wheel} is not stored at <wheelhouse>/<sha256>/<file>")
 869    url, sha, build_sha, patches, config = _expected(sdist, torch_lock, python)
 870    line = WheelLine(sdist.name, parts[1], url, sha, build_sha, patches, config, relative, wheel_sha256)
 871    with _locked(wheels_lock, shared=False):
 872        data = _read_verified(wheels_lock, expected_sha256)
 873        lines = {} if data is None else parse_wheels_lock(data.decode(), str(wheels_lock))
 874        lines[normalize_name(sdist.name)] = line
 875        new_sha = _write_atomic(wheels_lock, format_wheels_lock(lines.values()))
 876    return line, new_sha

 886def reuse_wheel(
 887    wheels_lock: Path,
 888    sdist: SdistSpec,
 889    wheelhouse: Path,
 890    torch_lock: LockRef | None = None,
 891    *,
 892    python: str,
 893    expected_sha256: str | None = None,
 894) -> tuple[Path, str] | None:
 895    """Return the recorded wheel for an sdist, after checking ``wheels.lock``, the line and the wheel's sha256.
 896
 897    Args:
 898        wheels_lock: The file.
 899        sdist: The sdist spec.
 900        wheelhouse: Where recorded wheels live.
 901        torch_lock: The torch lock the build venv would have, if any.
 902        python: The build venv's interpreter version.
 903        expected_sha256: The manifest's ``wheels.lock_sha256``; checked when given.
 904
 905    Returns:
 906        ``(wheel path, sha256)``, or ``None`` if ``wheels.lock`` has no line for ``sdist.name``.
 907
 908    Raises:
 909        ReproduceLockError: If ``wheels.lock`` does not match ``expected_sha256``, the line was recorded

 916def _reuse(
 917    wheels_lock: Path,
 918    sdist: SdistSpec,
 919    wheelhouse: Path,
 920    torch_lock: LockRef | None,
 921    python: str,
 922    expected_sha256: str | None,
 923) -> tuple[Path, str, str] | None:
 924    """:func:`reuse_wheel`, also returning the sha256 of the ``wheels.lock`` bytes it checked."""
 925    with _locked(wheels_lock, shared=True):
 926        data = _read_verified(wheels_lock, expected_sha256)
 927    if data is None:
 928        return None
 929    line = parse_wheels_lock(data.decode(), str(wheels_lock)).get(normalize_name(sdist.name))
 930    if line is None:
 931        return None
 932    recorded = (line.sdist_url, line.sdist_sha256, line.build_env_sha256, line.patch_sha256s, line.build_config_sha256)
 933    if recorded != _expected(sdist, torch_lock, python):
 934        raise ReproduceLockError(
 935            f"{wheels_lock}: the {sdist.name} line was recorded for other build inputs; build again with --rebuild"
 936        )
 937    wheel = _wheel_location(wheelhouse, line)
 938    if not wheel.is_file():
 939        raise ReproduceLockError(f"recorded wheel {line.wheel_path} is missing from {wheelhouse}; use --rebuild")
 940    actual = file_sha256(wheel)
 941    if actual != line.wheel_sha256:
 942        raise ReproduceLockError(f"{wheel}: sha256 {actual} != wheels.lock {line.wheel_sha256}")
 943    return wheel, actual, hashlib.sha256(data).hexdigest()

 946def stage_wheel(wheel: Path, sha256: str, dest_dir: Path) -> Path:
 947    """Copy a recorded wheel to ``dest_dir`` and check the copy's sha256, so the install reads checked bytes.
 948
 949    Args:
 950        wheel: The wheelhouse wheel.
 951        sha256: Its recorded sha256.
 952        dest_dir: A private directory, e.g. under ``{tmp}``; created if absent.
 953
 954    Returns:
 955        ``dest_dir / <wheel file name>``.
 956
 957    Raises:
 958        ReproduceLockError: If the copied bytes do not match ``sha256``.
 959    """
 960    dest_dir.mkdir(parents=True, exist_ok=True)
 961    data = wheel.read_bytes()
 962    actual = hashlib.sha256(data).hexdigest()
 963    if actual != sha256:
 964        raise ReproduceLockError(f"{wheel}: sha256 {actual} != recorded {sha256}")
 965    target = dest_dir / wheel.name
 966    target.write_bytes(data)
 967    return target

140

142        url: The sdist URL (``https://`` or ``file://``).

151    """
