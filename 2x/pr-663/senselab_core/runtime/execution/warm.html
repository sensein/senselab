senselab_core.runtime.execution.warm

senselab on pydra's warm pool: its pool tasks, its profile as worker classes and budgets, and PoolSession.

A pool job is one senselab job: HostCall or WorkerCall with the backend's model worker as its provided field and the JobDoc as its input, its checksum the job's key digest. Its one output is the job's stored outputs. PoolSession owns a WarmPool and the Submitter(worker=pool) its jobs are made under; whoever made it closes it, so its workers and their models stay resident from one run to the next, and several runs may use it at once.

 64class ItemFailed(RuntimeError):
 65    """The backend failed the job's item; `failure` is its record, with its fault class."""
 66
 67    def __init__(self, failure: FaultRecord) -> None:
 68        """Hold the failure and carry it in the message."""
 69        super().__init__(FAULT_MARK + json.dumps(failure.model_dump(mode="json"), sort_keys=True))
 70        self.failure = failure

 72    def __reduce__(self) -> tuple[type["ItemFailed"], tuple[FaultRecord]]:
 73        """Pickle by the record."""
 74        return (type(self), (self.failure,))

 77def _call(worker: Model, job: JobDoc) -> dict[str, object]:
 78    """Send `job`'s one item to its model worker and return its outputs as the store holds them.
 79
 80    Raises:
 81        ItemFailed: the backend failed the item.
 82    """
 83    started = datetime.now(timezone.utc)
 84    call = job.job.call
 85    params = {
 86        **call.params.model_dump(mode="json"),
 87        "backend_params": job.placement.backend_params.model_dump(mode="json"),
 88    }
 89    values = job.job.values
 90    [result] = worker.run([{"id": "0", "inputs": dict(values) if job.host else as_wire(values)}], params)
 91    if "error" in result:
 92        raise ItemFailed(FaultRecord.of(result["error"]))
 93    _, outputs = as_stored(call.task, as_wire(result["outputs"]), job.job.anchors)
 94    ran(job.key.digest(), started, output_identity(outputs))
 95    return outputs

120def classify(exc: BaseException, phase: str) -> PoolFault:
121    """Senselab's fault classifier for the pool: an item's own record, the worker's verdicts, then the fault table.
122
123    A worker whose env cannot use its device (`DeviceUnusable`, carried in a load's chain) is a resource fault of no
124    memory, so the job moves up its ladder or fails `Unplaceable` and no unfit load is recorded. Every exception the
125    table does not classify is an infrastructure fault (a gap), never the job's own.
126    """
127    name = type(exc).__name__
128    message = str(exc)[:2000]
129    if isinstance(exc, ItemFailed):
130        return _of(exc.failure.fault_class, exc.failure.gap, name, exc.failure.message, phase)
131    if isinstance(exc, LicenseRefused):
132        return PoolFault("policy", name, message, phase)
133    if isinstance(exc, WorkerLoadFailed) or (isinstance(exc, WorkerFatal) and exc.from_load):
134        return _at_load(exc, name, message)
135    if isinstance(exc, (WorkerLost, ProtocolError, WorkerFatal)):
136        return PoolFault("infrastructure", name, message, phase, reason="crash")
137    return _of(*classification(exc), name, message, phase)

140def _at_load(exc: WorkerLoadFailed | WorkerFatal, name: str, message: str) -> PoolFault:
141    if _device_unusable(exc):
142        return PoolFault("resource", name, message, "load")
143    return _of(*exc.carried_class, name, message, "load")

146def _device_unusable(exc: BaseException) -> bool:
147    """Whether `exc` or what it was raised from is a `DeviceUnusable`, raised here or reported by a worker."""
148    link: BaseException | None = exc
149    while link is not None:
150        if isinstance(link, DeviceUnusable) or (
151            isinstance(link, WorkerFatal) and link.error.get("type") == DeviceUnusable.__name__
152        ):
153            return True
154        link = link.__cause__
155    return False

158def _of(fault: FaultClass, gap: bool, name: str, message: str, phase: str) -> PoolFault:
159    if fault is FaultClass.RESOURCE:
160        return PoolFault("resource", name, message, phase, reason="oom")
161    if fault is FaultClass.CONTENT and phase == "call":
162        return PoolFault("content", name, message, phase)
163    return PoolFault("infrastructure", name, message, phase, gap=gap)

166def _gpus(resource: ResourceClass) -> tuple[int, str | None]:
167    """Return a class's GPUs and GPU type: from its `gres` when it names one, its `gpu_type` otherwise; an `mps` one.
168
169    Raises:
170        ValueError: `gres` is not `gpu:[<type>:]<n>`, or names another type than `gpu_type`.
171    """
172    if resource.device == "mps":
173        return 1, None
174    if resource.device != "cuda":
175        return 0, None
176    if resource.gres is None:
177        return 1, resource.gpu_type
178    found = _GRES.match(resource.gres)
179    if found is None:
180        raise ValueError(f"gres {resource.gres!r} is not gpu:[<type>:]<count>")
181    kind = found["type"] or resource.gpu_type
182    if resource.gpu_type is not None and kind != resource.gpu_type:
183        raise ValueError(f"gres {resource.gres!r} names another GPU type than gpu_type {resource.gpu_type!r}")
184    return int(found["count"]), kind

187def worker_classes(
188    profile: ExecutionProfile, placements: Mapping[str, ClassPlacement] | None = None
189) -> list[WorkerClass]:
190    """Return the profile's resource classes as pydra worker classes: devices, typed GPUs, memory, time, partition.
191
192    A class in `placements` takes its placed partition, time, worker cap and excluded nodes (`ClassPlacement.grant`);
193    any other its profile's values and the profile's excluded nodes. A class names its partition and excluded nodes
194    as substrate arguments; nothing names a QOS. Each is tagged with its device, so a job is placed on classes of its
195    own device alone.
196    """
197    found = []
198    placed = placements or {}
199    for name, given in profile.resource_classes.items():
200        resource = placed[name].grant(given) if name in placed else given
201        exclude = placed[name].exclude if name in placed else profile.substrate.exclude_nodes
202        gpus, gpu_type = _gpus(resource)
203        args = {"partition": resource.partition} if resource.partition else {}
204        if exclude:
205            args["exclude"] = ",".join(exclude)
206        found.append(
207            WorkerClass(
208                name=name,
209                device=resource.device,
210                gpus=gpus,
211                gpu_type=gpu_type,
212                cpus=resource.cpus,
213                mem=resource.mem_bytes,
214                time=None if resource.time_s is None else int(resource.time_s),
215                max_workers=resource.max_workers,
216                substrate_args=args,
217                tags=frozenset({resource.device}),
218                quantum_jobs=profile.pool.quantum_items,
219                quantum_s=profile.pool.quantum_s,
220            )
221        )
222    return found

272def substrate_of(profile: ExecutionProfile) -> Substrate:
273    """Return the substrate that starts the profile's workers: processes on this host, or one Slurm job each."""
274    if profile.substrate.kind == "local":
275        return LocalSubstrate()

281def calibration_root(profile: ExecutionProfile) -> Path | None:
282    """Return where every session of the profile keeps the pool's calibration: under its first writable measures root.
283
284    The roots are the profile's `calibration.measures_roots` (`resolve_roots`), so a later session, over any pool
285    root, reads the samples and unfit records an earlier one wrote. Under `calibration.trust: own` the directory is
286    the reader's own (named by its uid), so another user's records never steer its placement; under `group` it is
287    shared. None (logged) when no root is writable.
288    """
289    own = profile.calibration.trust == "own"
290    for root in resolve_roots(profile.calibration.measures_roots, cache_root()):
291        found = root / CALIBRATION_DIR / str(os.geteuid()) if own else root / CALIBRATION_DIR
292        try:
293            found.parent.mkdir(parents=True, exist_ok=True)
294            found.mkdir(mode=0o700 if own else 0o777, exist_ok=True)
295        except OSError as err:
296            logger.info("measures root %s takes no pool calibration: %s", root, err)
297            continue
298        if os.access(found, os.W_OK):
299            return found
300    logger.warning("no measures root is writable; the pool's calibration is kept under its session root")
301    return None

304def claim_leases(pool: Path, classes: Iterable[str]) -> list[Path]:
305    """Return the claim leases of the tickets on the queues of `classes` in the pool at `pool`.
306
307    The paths are pydra's documented pool layout (worker-protocol reference): `queues/<label>/name` holds the queue's
308    name, `<class>` or `<class>#<residency digest>`, and `queues/<label>/claims/<ticket digest>.lease/` each claim.
309    """
310    wanted = set(classes)
311    found: list[Path] = []
312    for queue in sorted((pool / "queues").glob("*/name")):
313        try:
314            name = queue.read_text(encoding="utf-8").strip()
315        except OSError:
316            continue
317        if name.split("#", 1)[0] in wanted:
318            found += sorted((queue.parent / "claims").glob("*.lease"))
319    return found

322def newest_beat(lease: Path) -> float | None:
323    """Return the newest mtime in the lease directory `lease`, its holders' token directories and their heartbeats.
324
325    A `DirectoryLease` holder beats by making and removing an entry in its token directory; None when it is gone.
326    """
327    try:
328        newest = lease.lstat().st_mtime
329        for entry in os.scandir(lease):
330            newest = max(newest, entry.stat(follow_symlinks=False).st_mtime)
331            if entry.is_dir(follow_symlinks=False):
332                for sub in os.scandir(entry.path):
333                    newest = max(newest, sub.stat(follow_symlinks=False).st_mtime)
334    except OSError:
335        return None
336    return newest

339class PoolSession:
340    """A warm pool whoever makes it closes: its `WarmPool`, its store and the submitter its jobs are made under.
341
342    `root` holds the pool (`<root>/pool`) and the jobs' work directories (`<root>/private/work`), on storage every
343    node sees; `store` is the result store's root. The pool's calibration is shared by every session of the profile
344    (`calibration_root`). The pool's workers and their models stay resident until `close`, whatever runs use it, one
345    after another or at once.
346    """
347
348    def __init__(  # noqa: PLR0913 - the profile and two roots, then keyword-only pool parts
349        self,
350        profile: ExecutionProfile,
351        root: Path,
352        store: Path,
353        *,
354        substrate: Substrate | None = None,
355        pool_settings: PoolSettings | None = None,
356        placements: Mapping[str, ClassPlacement] | None = None,
357        here: bool | None = None,
358    ) -> None:
359        """Open the pool at `root` over the store at `store`, its workers started by `substrate` (the profile's).
360
361        Its worker classes are the profile's, each class in `placements` placed with those values (`worker_classes`).
362        `here` says the workers run on this host; when None, whether the substrate is a `LocalSubstrate`.
363
364        Raises:
365            NodeLocalStorage: the substrate refuses `root` (pydra's `SlurmSubstrate.check_root`): it is on storage
366                only this node sees. Nothing is written.
367        """
368        chosen = substrate if substrate is not None else substrate_of(profile)
369        check_root = getattr(chosen, "check_root", None)
370        if check_root is not None:
371            check_root(Path(root))
372        self.here = isinstance(chosen, LocalSubstrate) if here is None else here
373        self.profile = profile
374        self.classes = worker_classes(profile, placements)
375        self.root = Path(root).absolute()
376        work = self.root / "private" / "work"
377        work.mkdir(parents=True, exist_ok=True, mode=0o700)
378        (self.root / "private").chmod(0o700)
379        self.store = open_store(Path(store), work, lease_settings(profile))
380        calibration = DirectoryCalibration(calibration_root(profile) or self.root / "pool" / "calibration")
381        self.calibration = calibration
382        self.providers = (SenselabWorker(), SenselabHost())
383        self.pool = WarmPool(
384            self.root / "pool",
385            self.classes,
386            chosen,
387            store=self.store,
388            fit=MeasuredFit(calibration, headroom=profile.sizing.memory_headroom),
389            calibration=calibration,
390            faults=budgets(profile),
391            classify=classify,
392            settings=pool_settings if pool_settings is not None else settings(profile),
393        )
394        self.submitter = Submitter(worker=self.pool, providers=list(self.providers), telemetry=False)
395        self._beats: dict[Path, float] = {}
396
397    @property
398    def epoch(self) -> int:
399        """The pool's coordinator epoch."""
400        assert self.pool.epoch is not None
401        return int(self.pool.epoch)
402
403    def working(self, classes: Iterable[str]) -> bool:
404        """Whether the pool is at work for `classes` since this was last asked.
405
406        It is when the substrate lists a worker of one of `classes` (ready or starting: a submitted Slurm job still
407        queued is starting), or when a claim lease on one of their queues (`claim_leases`) has beaten: its newest
408        mtime (`newest_beat`) differs from the one seen at the last call, or it is seen for the first time. A claim
409        seen at the last call and gone now ended its ticket, whose outcome is on its way, so it counts too.
410        A listing that raises (an `squeue` that times out) counts as working, logged.
411        """
412        wanted = set(classes)
413        try:
414            listed = self.pool.substrate.live()
415        except Exception as err:  # noqa: BLE001 - an unknown state is not a stall; the next look is a bound later
416            logger.warning("listing the substrate's workers failed; the pool counts as working: %s", err)
417            return True
418        if any(h.worker_class in wanted for h in listed):
419            return True
420        beats = {lease: newest_beat(lease) for lease in claim_leases(self.pool.root, wanted)}
421        current = {lease: newest for lease, newest in beats.items() if newest is not None}
422        moved = set(self._beats) != set(current) or any(self._beats[k] != v for k, v in current.items())
423        self._beats = current
424        return moved
425
426    def job(self, doc: JobDoc, loader: Loader, name: str) -> PydraJob:
427        """Return the pool job of `doc`, its model worker loaded by `loader`, asking for its plan class's resources."""
428        task = (WorkerCall if loader.provider == WORKER else HostCall)(worker=loader.spec, job=doc)
429        resources = request(self.profile, doc.placement.resource_class)
430        return PydraJob(task, submitter=self.submitter, name=name, resources=resources)
431
432    def close(self) -> None:
433        """Close the pool, stopping the workers it started."""
434        with contextlib.suppress(Exception):
435            self.submitter.close()
436        self.pool.close()
437
438    def __enter__(self) -> "PoolSession":
439        """Return the session, closed when the block ends."""
440        return self
441
442    def __exit__(self, *_: object) -> None:
443        """Close the session."""
444        self.close()

470def default_session(
471    profile: ExecutionProfile, store: Path, placements: Mapping[str, ClassPlacement] | None = None
472) -> PoolSession:
473    """Return this process's default session for `profile` over `store`, its classes placed by `placements`.
474
475    It counts one more live run on it; the run hands it back with `release_default`. Its root is a directory of its
476    own under `<cache root>/2x/pools`; it is closed at exit by the process that opened it, and a forked child opens
477    its own. Every run started without a session, and every call, uses it, so a model stays resident from one to the
478    next for the profile's idle window. A process holds one current session: a run of another profile, store or
479    worker classes retires it (no new run goes to it), and a retired session is closed once its last live run ends.
480    Callers that ask at once for a key with no session wait for the one that opens it.
481    """
482    key = (profile.canonical_digest(), str(Path(store).absolute()), repr(worker_classes(profile, placements)))
483    idle: list[PoolSession] = []
484    try:
485        with _DEFAULTS_LOCK:
486            found = _DEFAULTS.get(key)
487            if found is None:
488                for held in _DEFAULTS.values():
489                    (_RETIRED if _LIVE.get(id(held), 0) else idle).append(held)
490                _DEFAULTS.clear()
491                root = cache_root() / POOLS_DIR / f"{os.getpid()}-{secrets.token_hex(4)}"
492                found = PoolSession(profile, root, Path(store), placements=placements)
493                if not _OPENER:
494                    atexit.unregister(close_default_sessions)
495                    atexit.register(close_default_sessions)
496                _OPENER[:] = [os.getpid()]
497                _DEFAULTS[key] = found

404    def working(self, classes: Iterable[str]) -> bool:
405        """Whether the pool is at work for `classes` since this was last asked.
406
407        It is when the substrate lists a worker of one of `classes` (ready or starting: a submitted Slurm job still
408        queued is starting), or when a claim lease on one of their queues (`claim_leases`) has beaten: its newest
409        mtime (`newest_beat`) differs from the one seen at the last call, or it is seen for the first time. A claim
410        seen at the last call and gone now ended its ticket, whose outcome is on its way, so it counts too.
411        A listing that raises (an `squeue` that times out) counts as working, logged.
412        """
413        wanted = set(classes)

416        except Exception as err:  # noqa: BLE001 - an unknown state is not a stall; the next look is a bound later

418            return True
419        if any(h.worker_class in wanted for h in listed):
420            return True
421        beats = {lease: newest_beat(lease) for lease in claim_leases(self.pool.root, wanted)}
422        current = {lease: newest for lease, newest in beats.items() if newest is not None}
423        moved = set(self._beats) != set(current) or any(self._beats[k] != v for k, v in current.items())
