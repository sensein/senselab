senselab_worker.canonical

Canonical byte forms of JSON values and numpy arrays (ADR-006).

jcs_dumps writes RFC 8785 (JCS) JSON with strings in their canonical form (canonical_text); canonical_npy_bytes writes .npy version 1.0, C order, little-endian, one byte form per value; Tokeniser writes a value outside JSON as JSON with a one-key token object per family. Every encoder accepts an exact-type whitelist, refuses everything else, and walks containers through one Walk.

 44def _unassigned_pattern() -> "re.Pattern[str]":
 45    text = resources.files("senselab_worker").joinpath(UNICODE_TABLE).read_text(encoding="ascii")
 46    header: dict[str, str] = {}
 47    gaps: list[str] = []
 48    following = 0
 49    for line in text.splitlines():
 50        if not line or line.startswith("#"):
 51            continue
 52        first, *rest = line.split(" ")
 53        if not all(c in "0123456789ABCDEF" for c in first):
 54            header[first] = " ".join(rest)
 55            continue
 56        start, end = int(first, 16), int(rest[0], 16)
 57        if start > following:
 58            gaps.append(f"\\U{following:08x}-\\U{start - 1:08x}")
 59        following = end + 1
 60    if header.get("unicode_version") != UNICODE_VERSION:
 61        raise RuntimeError(f"{UNICODE_TABLE} is for Unicode {header.get('unicode_version')}, not {UNICODE_VERSION}")
 62    if following <= sys.maxunicode:
 63        gaps.append(f"\\U{following:08x}-\\U0010ffff")
 64    return re.compile("[" + "".join(gaps) + "]")

 77def canonical_text(text: str) -> str:
 78    """Return `text`'s canonical form, the same on every supported Python.
 79
 80    Text whose code points are all assigned in `UNICODE_VERSION` is put in NFC. Text holding any
 81    other code point is returned unchanged: it is never normalised, and it is told apart from every
 82    normalised text by that code point, which no NFC output over assigned code points holds.
 83
 84    Raises:
 85        InvalidText: if `text` holds a lone surrogate.
 86    """
 87    if text.isascii():
 88        return text
 89    require_unicode(text)
 90    if _UNASSIGNED.search(text) is not None:
 91        return text
 92    return unicodedata.normalize("NFC", text)

 95def require_unicode(text: str) -> None:
 96    """Refuse `text` if it holds a lone surrogate.
 97
 98    Raises:
 99        InvalidText: naming the first lone surrogate and its index.
100    """
101    found = None if text.isascii() else _SURROGATE.search(text)
102    if found is not None:
103        raise InvalidText(f"U+{ord(found.group()):04X} at index {found.start()} is a lone surrogate")

106def is_portable(text: str) -> bool:
107    """Whether every code point of `text` is assigned in `UNICODE_VERSION`, so `canonical_text` normalises it."""
108    return text.isascii() or _UNASSIGNED.search(text) is None

111def canonical_scalar(v: object) -> Scalar:
112    """Return `v` as the exact Python JSON scalar it encodes, dispatching on `type(v)` exactly.
113
114    `None`, `bool`, `int`, `float` and `str` pass unchanged. A numpy scalar passes as its `.item()`
115    only if its type is defined by numpy itself, its dtype kind is one of `b i u f U`, and its
116    `.item()` is one of those Python types (so `longdouble` is refused).
117
118    Raises:
119        TypeError: on any other value: subclasses (including `Enum` members with a scalar mix-in),
120            `datetime64`, `timedelta64`, `void`, complex, bytes and object scalars.
121    """
122    t = type(v)
123    if t in _PY_SCALARS:
124        return v  # type: ignore[return-value]
125    if isinstance(v, numpy.generic) and t.__module__ == "numpy" and v.dtype.kind in _CANONICAL_KINDS:
126        item = v.item()
127        if type(item) in _PY_SCALARS:
128            return item  # type: ignore[no-any-return]
129    raise TypeError(f"not a canonical JSON scalar: {t.__module__}.{t.__qualname__}")

149def in_json_domain(scalar: Scalar) -> bool:
150    """Whether an exact JSON scalar is one `jcs_dumps` writes: a finite float, an integer within +/-MAX_SAFE_INT."""

158def require_depth(depth: int) -> None:
159    """Refuse a container at `depth`, counted from its line's root (the root is at depth 1).
160
161    Raises:
162        ValueError: if `depth` exceeds `MAX_DEPTH`.
163    """
164    if depth > MAX_DEPTH:
165        raise ValueError(f"nesting deeper than {MAX_DEPTH} containers")

168_JSON_BRACKETS = re.compile(rb'"(?:[^"\\]|\\[\s\S])*(?:"|\\?\Z)|[\[\]{}]')

171def require_json_depth(text: bytes) -> None:
172    """Refuse JSON text whose containers nest deeper than `MAX_DEPTH`, before any parser sees it.
173
174    The bound is counted over the brackets outside strings, so it is the same on every Python; no parser's
175    recursion limit is consulted.
176
177    Raises:
178        ValueError: if the text nests deeper than `MAX_DEPTH` containers.
179    """
180    depth = 0
181    for token in _JSON_BRACKETS.finditer(text):
182        char = token.group()
183        if char in (b"[", b"{"):
184            depth += 1
185            require_depth(depth)
186        elif char in (b"]", b"}"):
187            depth -= 1

190@contextmanager
191def entered(container: object, active: set[int], base: int = 0) -> Iterator[None]:
192    """Mark `container` as being encoded for the duration of the block, on the path held in `active`.
193
194    Every encoder that recurses into JSON containers enters each one through this, so a cycle or
195    excess nesting raises the same error wherever values are encoded; a child shared by two parents
196    is not a cycle. `base` is the number of containers enclosing the encoded value's root, so the
197    container's depth is `base + len(active) + 1`.
198
199    Raises:
200        ValueError: if `container` is already on the path, that is, it contains itself, or if its
201            depth exceeds `MAX_DEPTH` (`require_depth`).
202    """
203    if id(container) in active:
204        raise ValueError(f"a {type(container).__name__} contains itself; JSON has no cycles")
205    require_depth(base + len(active) + 1)
206    active.add(id(container))
207    try:
208        yield
209    finally:
210        active.discard(id(container))

213class WorkBudgetExceeded(ValueError):
214    """An encoding would write or replay more than its `Walk`'s budget of characters."""
215
216
217class Walk:
218    """One encoder's pass over a value: the shared cycle and depth guard, a work budget and a memo.
219
220    `enter` builds a container's result inside `entered` and keeps it by identity and depth, so a
221    container shared by many parents is built once per depth. The memo holds each container it
222    keys, so no identity is reused while the pass lives. `builds` counts the results built, not reused.
223    """
224
225    def __init__(self, base: int = 0, budget: int = MAX_CANONICAL_CHARS) -> None:
226        """Start a pass whose root is enclosed by `base` containers and may spend `budget` characters."""
227        self.active: set[int] = set()
228        self.base = base
229        self.budget = budget
230        self.spent = 0
231        self.builds = 0
232        self._done: dict[tuple[int, int], tuple[object, object]] = {}
233
234    def charge(self, size: int) -> None:
235        """Spend `size` characters.
236
237        Raises:
238            WorkBudgetExceeded: once the pass has spent more than its budget.
239        """
240        self.spent += size
241        if self.spent > self.budget:
242            raise WorkBudgetExceeded(f"canonical form exceeds {self.budget} characters")
243
244    def enter(self, container: object, build: Callable[[], T]) -> T:
245        """Return `build()` for `container`, or the result kept from building it at this depth before.
246
247        Raises:
248            ValueError: if `container` contains itself or nests deeper than `MAX_DEPTH` (`entered`).
249        """
250        key = (id(container), len(self.active))
251        held = self._done.get(key)
252        if held is not None:
253            return cast(T, held[1])
254        with entered(container, self.active, self.base):
255            result = build()
256        self.builds += 1
257        self._done[key] = (container, result)
258        return result
259
260    def remember(self, value: object, build: Callable[[], T]) -> T:
261        """Return `build()` for `value`, kept by identity and depth like `enter`, without entering it."""
262        key = (id(value), len(self.active))
263        held = self._done.get(key)
264        if held is not None:
265            return cast(T, held[1])
266        result = build()
267        self.builds += 1
268        self._done[key] = (value, result)
269        return result

272def jcs_dumps(obj: object, walk: Walk | None = None) -> bytes:
273    """Serialise a JSON value as RFC 8785 canonical UTF-8 bytes.
274
275    Containers are exactly `list`, `tuple` and `dict` with `str` keys; scalars are what
276    `canonical_scalar` accepts. Strings and keys are in their canonical form (`canonical_text`); keys
277    sort by UTF-16 code units. `walk`, when given, is the pass's `Walk`, which the caller reads afterwards.
278
279    Raises:
280        ValueError: on a NaN or infinite float, an integer outside +/-(2**53 - 1), a lone surrogate
281            (`InvalidText`), two keys of one object with one canonical form, a container
282            that contains itself, nesting deeper than `MAX_DEPTH`, or text longer than
283            `MAX_CANONICAL_CHARS` (`WorkBudgetExceeded`).
284        TypeError: on any other type, including container subclasses, non-`str` keys and arrays.
285    """
286    writer = _Writer(walk)
287    writer.write(obj)
288    return "".join(writer.parts).encode("utf-8")
289
290
291class _Writer:
292    def __init__(self, walk: Walk | None) -> None:
293        self.parts: list[str] = []
294        self.walk = Walk() if walk is None else walk
295
296    def out(self, text: str) -> None:
297        self.walk.charge(len(text))
298        self.parts.append(text)
299
300    def write(self, obj: object) -> None:
301        t = type(obj)
302        if t is list or t is tuple or t is dict:
303            before = len(self.parts)
304            start, end, size = self.walk.enter(obj, lambda: self._container(obj))
305            if start < before:
306                self.walk.charge(size)
307                self.parts.append("".join(self.parts[start:end]))
308            return
309        scalar = canonical_scalar(obj)
310        if scalar is None:
311            self.out("null")
312        elif scalar is True:
313            self.out("true")
314        elif scalar is False:
315            self.out("false")
316        elif type(scalar) is int:
317            self.out(_int_text(scalar))
318        elif type(scalar) is float:
319            self.out(_float_text(scalar))
320        else:
321            self.out(_json_string(canonical_text(cast(str, scalar))))
322
323    def _container(self, obj: object) -> tuple[int, int, int]:
324        start, spent = len(self.parts), self.walk.spent
325        if type(obj) is dict:
326            self._object(cast("dict[object, object]", obj))
327        else:
328            self.out("[")
329            for i, item in enumerate(cast("list[object] | tuple[object, ...]", obj)):
330                if i:
331                    self.out(",")
332                self.write(item)
333            self.out("]")
334        return start, len(self.parts), self.walk.spent - spent
335
336    def _object(self, obj: dict[object, object]) -> None:
337        members: dict[str, object] = {}
338        for key, value in obj.items():
339            if type(key) is not str:
340                raise TypeError(f"JSON object key is not a str: {type(key).__qualname__}")
341            text = canonical_text(key)
342            if text in members:
343                raise ValueError(f"duplicate JSON object key after NFC: {text!r}")
344            members[text] = value
345        self.out("{")
346        for i, key in enumerate(sorted(members, key=utf16_sort_key)):
347            if i:
348                self.out(",")
349            self.out(_json_string(key))
350            self.out(":")
351            self.write(members[key])
352        self.out("}")

359def jcs_loads(data: bytes | str) -> object:
360    """Parse JCS text back to the value `jcs_dumps` wrote it from.
361
362    An integer literal outside +/-(2**53 - 1) can only come from a float, so it parses as that
363    float. Only text `jcs_dumps` returns is accepted, so `jcs_dumps(jcs_loads(b)) == b`.
364
365    Raises:
366        NotCanonicalJSON: on text that is not UTF-8 JSON, repeats a key within one object, holds a
367            `NaN` or `Infinity` literal or a number that overflows to infinity, nests deeper than
368            `MAX_DEPTH`, or is not the canonical form of the value it parses to.
369    """
370    raw = data.encode("utf-8", "surrogatepass") if isinstance(data, str) else bytes(data)
371    try:
372        require_json_depth(raw)
373    except ValueError as err:
374        raise NotCanonicalJSON(f"JSON nests deeper than {MAX_DEPTH} containers") from err
375    try:
376        value = json.loads(
377            raw.decode("utf-8"),
378            parse_int=_parse_int,
379            parse_float=finite_float,
380            parse_constant=refuse_constant,
381            object_pairs_hook=unique_object,
382        )
383    except NotCanonicalJSON:
384        raise
385    except ValueError as err:
386        raise NotCanonicalJSON(f"not JSON: {err}") from err
387    try:
388        canonical = jcs_dumps(value)
389    except (ValueError, TypeError) as err:
390        raise NotCanonicalJSON(f"JSON jcs_dumps cannot write: {err}") from err
391    if canonical != raw:
392        raise NotCanonicalJSON("JSON is not in its canonical form")
393    return value

439_ECMA_MAX_POINT = 21
440_ECMA_MIN_POINT = -6
441"""The decimal-point bounds of ECMA-262 Number::toString: outside (-6, 21] a number is written with an exponent."""
442
443
444def _float_text(value: float) -> str:
445    """Format a finite float as ECMAScript Number.prototype.toString does."""
446    if not in_json_domain(value):
447        raise ValueError(f"non-finite number is not JSON: {value}")
448    if value == 0.0:
449        return "0"
450    sign = "-" if value < 0 else ""
451    mantissa, _, exp_text = repr(abs(value)).partition("e")
452    int_part, _, frac_part = mantissa.partition(".")
453    digits = int_part + frac_part
454    point = len(int_part) + int(exp_text or 0)
455    stripped = digits.lstrip("0")
456    point -= len(digits) - len(stripped)
457    digits = stripped.rstrip("0")
458    k = len(digits)
459    if k <= point <= _ECMA_MAX_POINT:
460        return sign + digits + "0" * (point - k)
461    if 0 < point <= _ECMA_MAX_POINT:
462        return sign + digits[:point] + "." + digits[point:]
463    if _ECMA_MIN_POINT < point <= 0:
464        return sign + "0." + "0" * -point + digits
465    exponent = point - 1
466    exp_sign = "+" if exponent >= 0 else "-"
467    head = digits[0] + ("." + digits[1:] if k > 1 else "")
468    return f"{sign}{head}e{exp_sign}{abs(exponent)}"

475def npy_bytes(a: numpy.ndarray) -> bytes:
476    """Serialise an array as `.npy` version 1.0, C order, little-endian, keeping every value's bits.
477
478    The header is numpy's version 1.0 dict, space-padded so the data starts on a 64-byte boundary.
479
480    Raises:
481        TypeError: if `a` is not exactly a `numpy.ndarray` (masked arrays and other subclasses).
482        ValueError: on a dtype whose kind is not one of `b i u f U`, or on `longdouble`, whose
483            itemsize and byte layout differ between platforms.
484    """
485    header = npy_header(a)
486    data = a.astype(a.dtype.newbyteorder("<"), order="C", copy=False)
487    return header + data.tobytes(order="C")

490def npy_header(a: numpy.ndarray) -> bytes:
491    """Return the bytes `npy_bytes(a)` writes before the data: magic, version 1.0, header length and header.
492
493    Raises:
494        TypeError, ValueError: as `npy_bytes`.
495    """
496    if type(a) is not numpy.ndarray:
497        raise TypeError(f"not a plain numpy.ndarray: {type(a).__module__}.{type(a).__qualname__}")
498    if a.dtype.kind not in _CANONICAL_KINDS or a.dtype.fields is not None:
499        raise ValueError(f"dtype has no canonical form: {a.dtype}")
500    if a.dtype.char in _PLATFORM_DTYPE_CHARS:
501        raise ValueError(f"platform-specific dtype has no canonical form: {a.dtype.name}")
502    shape = tuple(int(n) for n in a.shape)
503    header = f"{{'descr': {a.dtype.newbyteorder('<').str!r}, 'fortran_order': False, 'shape': {shape!r}, }}"
504    padding = -(len(_NPY_MAGIC) + 2 + len(header) + 1) % _NPY_ALIGN
505    header_bytes = (header + " " * padding + "\n").encode("latin1")
506    return _NPY_MAGIC + struct.pack("<H", len(header_bytes)) + header_bytes

509def canonical_npy_bytes(a: numpy.ndarray) -> bytes:
510    """Serialise an array as canonical `.npy`: `npy_bytes` of its values, each in its one canonical form.
511
512    Every NaN becomes the positive quiet NaN with no payload (`QUIET_NAN_BITS`), `-0.0` becomes
513    `0.0`, every nonzero bool byte becomes 1, and every string is in its canonical form
514    (`canonical_text`), at a `U` width no narrower than the array's.
515
516    Raises:
517        TypeError: if `a` is not exactly a `numpy.ndarray` (masked arrays and other subclasses).
518        ValueError: on a dtype `npy_bytes` refuses, or a string holding a lone surrogate (`InvalidText`).
519    """
520    if type(a) is not numpy.ndarray:
521        raise TypeError(f"not a plain numpy.ndarray: {type(a).__module__}.{type(a).__qualname__}")
522    if a.dtype.fields is None and a.dtype.char not in _PLATFORM_DTYPE_CHARS:
523        if a.dtype.kind == "f":
524            a = _canonical_floats(a)
525        elif a.dtype.kind == "b":
526            a = _canonical_bools(a)
527        elif a.dtype.kind == "U":
528            a = _canonical_strings(a)
529    return npy_bytes(a)

532def _canonical_floats(a: numpy.ndarray) -> numpy.ndarray:
533    nan = numpy.isnan(a)
534    negative_zero = (a == 0) & numpy.signbit(a)

638    def __init__(self, spec: Tokeniser, walk: Walk) -> None:
639        self.spec = spec

642    def value(self, value: object, path: str) -> object:
643        t = type(value)
644        if t is dict or t is list or t is tuple or t is numpy.ndarray:
645            try:
646                return self.walk.enter(value, lambda: self.container(value, path))
647            except NoCanonicalForm:
648                raise
649            except (ValueError, TypeError) as err:
650                raise self.spec.error(path, value, str(err)) from err
651        if self.spec.extra is not None:

653            if found is not None:
654                return self.extra(value, found[0], found[1], path)
655        return self.scalar(value, path)

657    def scalar(self, value: object, path: str) -> object:
658        """A scalar in the JSON domain as itself, an integer too large for it or a non-finite float as a token."""
659        try:
660            scalar = canonical_scalar(value)
661        except TypeError as err:
662            raise self.spec.error(path, value, self.spec.error.refusal) from err
663        if in_json_domain(scalar):
664            return scalar

667                return {INT_TOKEN: str(scalar)}
668            except ValueError as err:
669                raise self.spec.error(path, value, str(err)) from err
670        number = cast(float, scalar)
671        return {FLOAT_TOKEN: "NaN" if math.isnan(number) else ("Infinity" if number > 0 else "-Infinity")}
672
673    def extra(self, value: object, token: str | None, payload: object, path: str) -> object:
674        try:
675            if token is None:
676                return self.walk.remember(value, lambda: self.value(payload, path))
677            return self.walk.enter(value, lambda: {token: self.value(payload, path)})
678        except NoCanonicalForm:
679            raise
680        except ValueError as err:
681            raise self.spec.error(path, value, str(err)) from err
682
683    def container(self, value: Any, path: str) -> object:  # noqa: ANN401
684        t = type(value)
685        if t is dict:
686            out: dict[str, object] = {}
687            seen: set[str] = set()
688            for key, item in value.items():
689                if type(key) is not str:
690                    raise self.spec.error(f"{path}[{key!r}]", key, "is not a str object key")
691                try:
692                    text = canonical_text(key)
693                except InvalidText as err:
694                    raise self.spec.error(f"{path}[{key!r}]", key, str(err)) from err
695                if text in seen:
696                    raise self.spec.error(f"{path}.{key}", key, "is a key equal after NFC to another key of its object")
697                seen.add(text)
698                out[escaped_key(key)] = self.value(item, f"{path}.{key}")
699            return out
