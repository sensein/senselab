senselab_worker.codec

Wire framing and array references of the worker protocol.

A message is one JSON object per newline-terminated UTF-8 line. Every numpy array travels as {"__array__": {"path", "sha256", "dtype", "shape"}}, naming an .npy file (npy_bytes) in the array directory.

 45MAX_LINE_BYTES: Final = 16 * 2**20  # invariant: the wire protocol line limit, bound to its format tag
 46MAX_ID_CHARS = 1024

 58class Deadline:
 59    """A point in monotonic time by which a sequence of steps must finish; `None` seconds never expires."""
 60
 61    def __init__(self, seconds: float | None, now: Callable[[], float] = time.monotonic) -> None:
 62        """Start the clock: the deadline falls `seconds` from now, as `now` reads it."""
 63        self._seconds, self._now = seconds, now
 64        self._end = None if seconds is None else now() + max(seconds, 0.0)
 65
 66    def remaining(self) -> float | None:
 67        """Return the seconds left, at least 0, or `None` if the deadline never expires."""
 68        return None if self._end is None else max(self._end - self._now(), 0.0)
 69
 70    def expired(self) -> bool:
 71        """Return whether no time is left."""
 72        return self._end is not None and self._now() >= self._end
 73
 74    def check(self, what: str) -> None:
 75        """Raise `TimeoutError` naming `what` if the deadline has passed."""
 76        if self.expired():
 77            raise TimeoutError(f"{what} did not finish within {self._seconds} s")

 80def encode(msg: dict[str, Any], kind: Kind, *, array_dir: Path) -> bytes:
 81    """Return `msg` as one wire line of `kind`, writing its arrays into `array_dir`.
 82
 83    Raises:
 84        WireValueError: on a value with no wire form (see `to_wire`).

 87    """

 91def decode(line: bytes, kind: Kind, *, array_dir: Path) -> dict[str, Any]:
 92    """Return the `kind` message on a wire line, with its array references loaded.

111def dump_line(msg: dict[str, Any], kind: Kind, *, limit: int = MAX_LINE_BYTES) -> bytes:
112    """Return a message already in wire form as one line, after checking it is a valid `kind` message.
113
114    Raises:
115        ProtocolError: if `validate_message` refuses it, it nests deeper than `MAX_DEPTH`, or it is
116            not JSON text.
117        LineTooLarge: if the line, newline included, exceeds `limit` bytes.
118    """
119    validate_message(msg, kind)
120    _check_message(msg)
121    try:
122        line = _json_bytes(msg) + b"\n"
123    except (ValueError, TypeError) as exc:
124        raise ProtocolError(f"{kind} message has no wire form: {type(exc).__name__}: {exc}") from exc
125    if len(line) > limit:
126        raise LineTooLarge(len(line), limit)
127    return line

143def _check_message(msg: object) -> None:
144    try:
145        _height(msg, set(), {})
146    except ValueError as exc:
147        raise ProtocolError(f"wire message: {exc}") from exc

150def _height(value: object, active: set[int], heights: dict[int, int]) -> int:
151    """Return the containers on the tallest path down from `value`, each distinct container walked once.
152
153    Raises:
154        ValueError: if a container is deeper than `MAX_DEPTH`, or a key or string holds a lone surrogate.
155    """
156    if isinstance(value, str):
157        require_unicode(value)
158        return 0
159    if isinstance(value, dict):
160        for key in value:
161            if isinstance(key, str):
162                require_unicode(key)
163        children: Any = value.values()
164    elif isinstance(value, (list, tuple)):
165        children = value
166    else:
167        return 0
168    known = heights.get(id(value))
169    if known is not None:
170        require_depth(len(active) + known)
171        return known
172    with entered(value, active):
173        height = 1 + max((_height(child, active, heights) for child in children), default=0)
174    heights[id(value)] = height
175    return height

178def parse_line(line: bytes) -> dict[str, Any]:
179    """Parse one wire line into a dict, leaving array references as they are.
180
181    Raises:
182        ProtocolError: if the line exceeds `MAX_LINE_BYTES`, is not UTF-8 JSON, holds `NaN`,
183            `Infinity` or a number that overflows to infinity (`1e999`), repeats a key within one
184            object, nests deeper than `MAX_DEPTH` containers, holds a lone surrogate (`U+D800`-`U+DFFF`)
185            in a key or string, or is not an object.
186    """
187    if len(line) > MAX_LINE_BYTES:
188        raise ProtocolError(f"wire line exceeds {MAX_LINE_BYTES} bytes")
189    try:
190        require_json_depth(line)
191    except ValueError as exc:
192        raise ProtocolError(f"wire line nests deeper than {MAX_DEPTH} containers") from exc
193    try:
194        msg = json.loads(
195            line.decode("utf-8"),
196            object_pairs_hook=unique_object,
197            parse_float=finite_float,
198            parse_constant=refuse_constant,
199        )
200    except ValueError as exc:
201        raise ProtocolError(f"wire line is not JSON: {exc}") from exc
202    if type(msg) is not dict:
203        raise ProtocolError(f"wire line is a JSON {type(msg).__name__}, not an object")
204    _check_message(msg)
205    return msg

208def to_wire(value: object, *, array_dir: Path, at: Slot, limit: int = MAX_LINE_BYTES) -> Any:  # noqa: ANN401
209    """Return `value` as plain JSON data, writing each numpy array into `array_dir` as a reference.
210
211    `at` names where the value sits on its line, and so how many containers enclose it
212    (`ENVELOPE`): its containers' depths are counted from the line's root, as the reader counts
213    them. Containers are exactly `dict` with `str` keys, `list` and `tuple` (which becomes a
214    list); scalars are what `canonical_scalar` accepts, so a numpy scalar becomes its `.item()`;
215    arrays are exactly `numpy.ndarray`, and a reference is `ARRAY_REF_HEIGHT` containers tall.
216
217    Raises:
218        WireValueError: on any other type or a subclass, a non-finite float, an int with more decimal
219            digits than `sys.get_int_max_str_digits()` allows (the reader's limit), a string or key that
220            is not valid Unicode, an object or structured array, a dict whose only key is
221            `__array__`, a container that contains itself, or a container deeper than `MAX_DEPTH`
222            on the line.
223        ValueTooLarge: once a lower bound on the value's encoded size exceeds `limit` bytes.
224    """
225    encoder = _Encoder(Path(array_dir), ENVELOPE[at], limit)
226    try:
227        return encoder.encode(value)[0]
228    except (WireValueError, ValueTooLarge):
229        raise
230    except (ValueError, TypeError) as exc:
231        raise WireValueError(f"no wire form: {exc}") from exc
232
233
234class _Encoder:
235    """The state of one `to_wire` call."""
236
237    def __init__(self, array_dir: Path, base: int, limit: int) -> None:
238        self.array_dir = array_dir
239        self.base = base
240        self.limit = limit
241        self.active: set[int] = set()
242        self.done: dict[int, tuple[Any, int, int]] = {}
243        self.spent = 0
244
245    def charge(self, size: int) -> None:
246        self.spent += size
247        if self.spent > self.limit:
248            raise ValueTooLarge(f"wire form exceeds {self.limit} bytes; pass large data as arrays")
249
250    def encode(self, value: object) -> tuple[Any, int]:
251        """Return the wire form of `value` and its height in containers (0 for a scalar)."""
252        t = type(value)
253        if not (t is dict or t is list or t is tuple or isinstance(value, numpy.ndarray)):
254            return self.scalar(value), 0
255        found = self.done.get(id(value))
256        if found is not None:
257            wire, height, size = found
258            require_depth(self.base + len(self.active) + height)
259            self.charge(size)
260            return wire, height
261        start = self.spent
262        wire, height = self.array(value) if isinstance(value, numpy.ndarray) else self.container(value)
263        self.done[id(value)] = (wire, height, self.spent - start)
264        return wire, height
265
266    def array(self, a: numpy.ndarray) -> tuple[Any, int]:
267        require_depth(self.base + len(self.active) + ARRAY_REF_HEIGHT)
268        ref = _write_array(a, self.array_dir)
269        self.charge(len(ref["path"]) + len(ref["sha256"]))
270        return {ARRAY_KEY: ref}, ARRAY_REF_HEIGHT
271
272    def container(self, value: Any) -> tuple[Any, int]:  # noqa: ANN401
273        with entered(value, self.active, self.base):
274            self.charge(1 + max(len(value), 1))
275            height = 0
276            if type(value) is dict:
277                if list(value) == [ARRAY_KEY]:
278                    raise ValueError(f"a dict whose only key is {ARRAY_KEY!r} is reserved for array references")
279                out: Any = {}
280                for key, item in value.items():
281                    if type(key) is not str:
282                        raise TypeError(f"JSON object key is not a str: {type(key).__qualname__}")
283                    self.charge(len(key) + 3)
284                    key.encode("utf-8")
285                    out[key], child = self.encode(item)
286                    height = max(height, child)
287            else:
288                out = []
289                for item in value:
290                    wire, child = self.encode(item)
291                    out.append(wire)
292                    height = max(height, child)
293            return out, height + 1
294
295    def scalar(self, value: object) -> object:
296        scalar = canonical_scalar(value)
297        if type(scalar) is str:
298            self.charge(len(scalar) + 2)
299            scalar.encode("utf-8")
300        elif type(scalar) is int:
301            self.charge(len(str(scalar)))
302        else:
303            self.charge(1)
304        if type(scalar) is float and not math.isfinite(scalar):
305            raise ValueError(f"non-finite number is not JSON: {scalar}; pass it inside an array")
306        return scalar

309def _write_array(a: numpy.ndarray, array_dir: Path) -> dict[str, Any]:
310    data = npy_bytes(a)
311    sha = _raw_digest.bytes_digest("npy@1", data)
312    root = array_dir.absolute()
313    path = root / f"{sha}.npy"
314    tmp = root / f".{uuid.uuid4().hex}.tmp"
315    fd = os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
316    try:
317        with os.fdopen(fd, "wb") as f:
318            f.write(data)
319        os.replace(tmp, path)
320    except BaseException:
321        remove_owned(root, tmp, pattern=ARRAY_TEMP_NAME)
322        raise
323    return {"path": str(path), "sha256": sha, "dtype": a.dtype.newbyteorder("<").str, "shape": list(a.shape)}

326def _map_refs(value: object, on_ref: Callable[[object], object]) -> Any:  # noqa: ANN401
327    if isinstance(value, dict):
328        if len(value) == 1 and ARRAY_KEY in value:
329            return on_ref(value[ARRAY_KEY])
330        return {key: _map_refs(item, on_ref) for key, item in value.items()}
331    if isinstance(value, list):
332        return [_map_refs(item, on_ref) for item in value]
333    return value

336def from_wire(
337    value: object,
338    *,
339    array_dir: Path,
340    deadline: Deadline | None = None,
341    shared: dict[str, numpy.ndarray] | None = None,
342) -> Any:  # noqa: ANN401
343    """Return wire data with each array reference replaced by its array, loaded into memory.
344
345    Each file is opened without following a symlink and without blocking, must be a regular file,
346    and is read in chunks of `READ_CHUNK_BYTES`, checking `deadline` between them. Each distinct sha256
347    is read once per `shared` (one message's decoded arrays by sha256; this call's own when None): every
348    reference to it is the same array, and an array referenced more than once is made read-only, so the
349    references decoded before the repeat see that too. An array referenced once stays writable.
350
351    Raises:
352        ArrayIntegrityError: if a reference is malformed, points outside `array_dir`, or its file
353            is not a regular file or differs from its sha256, dtype or shape; `ArrayReadError`, also an
354            `OSError` with the failed call's errno, if the file cannot be opened or read.
355        TimeoutError: if `deadline` passes while reading.
356    """
357    root = Path(array_dir).resolve()
358    arrays: dict[str, numpy.ndarray] = {} if shared is None else shared
359    return _map_refs(value, lambda ref: _read_array(ref, root, deadline or Deadline(None), arrays))

362_NOT_NPY_BYTES = "array file is not npy_bytes of its array: version 1.0, C order, little-endian"
363_NPY_HEAD_MAX = 10 + 0xFFFF
364"""The longest a version 1.0 `.npy` preamble can be: magic, version and length (10 bytes) and a 16-bit header."""
365
366
367def _read_file(path: Path, deadline: Deadline) -> bytearray:
368    fd = os.open(path, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
369    try:
370        if not stat.S_ISREG(os.fstat(fd).st_mode):
371            raise ValueError(f"array file is not a regular file: {path}")
372        data = bytearray()
373        while True:
374            deadline.check(f"reading array file {path}")
375            chunk = os.read(fd, READ_CHUNK_BYTES)
376            if not chunk:
377                return data
378            data += chunk
379    finally:
380        os.close(fd)

383def _npy_view(data: bytearray) -> numpy.ndarray:
384    """Return the array a version 1.0 `.npy` in `data` holds, as a writable view of `data`, not a copy."""
385    head = io.BytesIO(bytes(memoryview(data)[:_NPY_HEAD_MAX]))
386    if numpy.lib.format.read_magic(head) != (1, 0):
387        raise ValueError(_NOT_NPY_BYTES)
388    shape, fortran_order, dtype = numpy.lib.format.read_array_header_1_0(head)
389    if fortran_order or dtype.itemsize == 0:
390        loaded: numpy.ndarray = numpy.load(io.BytesIO(data), allow_pickle=False)
391        return loaded
392    return numpy.frombuffer(data, dtype=dtype, count=math.prod(shape), offset=head.tell()).reshape(shape)

395def _verified_npy(data: bytearray, sha: str) -> numpy.ndarray:
396    if _raw_digest.bytes_digest("npy@1", data) != sha:
397        raise ValueError(f"array file bytes do not match sha256 {sha}")
398    a = _npy_view(data)
399    header = npy_header(a)
400    if not (data.startswith(header) and len(data) == len(header) + a.nbytes):
401        raise ValueError(_NOT_NPY_BYTES)
402    return a

405_REF_KEYS = frozenset({"path", "sha256", "dtype", "shape"})
406

408def _checked_ref(ref: object) -> dict[str, Any]:
409    """Return `ref` if it is an array reference with exactly `_REF_KEYS` and a list of `int` as its shape."""
410    if not isinstance(ref, dict):
411        raise TypeError(f"array reference is a {type(ref).__name__}, not an object")
412    if ref.keys() != _REF_KEYS:
413        raise ValueError(f"array reference keys are {sorted(ref)}, not {sorted(_REF_KEYS)}")
414    shape = ref["shape"]
415    if type(shape) is not list or any(type(n) is not int for n in shape):
416        raise TypeError(f"array reference shape is not a list of int: {shape!r}")
417    return ref

420def _read_array(ref: object, root: Path, deadline: Deadline, arrays: dict[str, numpy.ndarray]) -> numpy.ndarray:
421    try:
422        ref = _checked_ref(ref)
423        sha = ref["sha256"]
424        if not (isinstance(sha, str) and SHA256_HEX.fullmatch(sha)):
425            raise ValueError(f"sha256 is not 64 lowercase hex digits: {sha!r}")
426        named = Path(ref["path"])
427        if named.name != f"{sha}.npy" or named.parent.resolve() != root:
428            raise ValueError(f"array path is not {sha}.npy in {root}: {ref['path']}")
429        a = arrays.get(sha)
430        if a is not None:
431            a.flags.writeable = False
432        else:
433            a = arrays[sha] = _verified_npy(_read_file(root / f"{sha}.npy", deadline), sha)
434        if a.dtype.str != ref["dtype"] or list(a.shape) != ref["shape"]:
435            raise ValueError(f"array is {a.dtype.str} {list(a.shape)}, reference says {ref['dtype']!r} {ref['shape']}")
436    except TimeoutError:
437        raise
438    except OSError as exc:
439        what = exc.strerror or str(exc)
440        where = "" if exc.filename is None else f": {exc.filename}"
441        raise ArrayReadError(exc.errno, f"bad array reference: {type(exc).__name__}: {what}{where}") from exc
442    except Exception as exc:
443        raise ArrayIntegrityError(f"bad array reference: {type(exc).__name__}: {exc}") from exc
444    return a

516def _request(msg: dict[str, Any]) -> None:
517    _identifier(msg, "id", "request")
518    items = _typed(msg, "items", "request", (list,))
519    ids: set[str] = set()
520    for item in items:
521        if type(item) is not dict:
522            raise _fail("request", f"an item is a {type(item).__name__}, not an object")
523        item_id = _identifier(item, "id", "request")
524        _object(item, "inputs", "request")
525        if item_id in ids:
526            raise _fail("request", f"item id {item_id!r} is not unique")
527        ids.add(item_id)
528    _object(msg, "params", "request")

535def _result(msg: dict[str, Any]) -> None:
536    _identifier(msg, "id", "result")
537    _identifier(msg, "item", "result")
538    if ("outputs" in msg) == ("error" in msg):
