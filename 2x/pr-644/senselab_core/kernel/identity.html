 86def _pinned_identity_error(source: str, identity: str) -> str | None:
 87    pattern, form = (COMMIT_HEX, "a 40-hex commit") if source == "hf" else (SHA256_HEX, "a sha256")
 88    if pattern.fullmatch(identity):
 89        return None
 90    return f"a pinned {source} model's identity is {form}, got {identity!r}"

 93def _text_fields(text: str) -> tuple[Literal["hf", "url", "local"], str, str]:
 94    """Return the `(source, id, ref)` a reference's text form names; raise `ValueError` if it names none."""
 95    if text.startswith("local:"):
 96        return "local", text[len("local:") :], ""
 97    if text.startswith("url:"):
 98        url, sep, digest = text[len("url:") :].partition(_URL_HASH)
 99        return "url", url, digest if sep else ""
100    model_id, sep, ref = text.partition("@")
101    if not sep:
102        raise ValueError(f"{text!r} needs an explicit revision: '{text}@<ref>', or a 'url:' or 'local:' prefix")
103    return "hf", model_id, ref

106class ModelRef(Frozen):
107    """A model reference: hub id and revision, a URL, or a local path.
108
109    Text forms, read by `parse` and written by `str`:
110    - `<id>@<ref>`: source `hf`. `<id>` is `name` or `org/name` (letters, digits, `.`, `_`, `-`);
111      `<ref>` is a branch, tag or commit: `/`-separated non-empty components without whitespace, `@` or a backslash,
112      none starting with `.`, and no `..` anywhere.
113    - `url:<url>#sha256=<64 hex>`: source `url`. `<url>` is the id and must be an absolute URL with a
114      scheme and a host; `ref` holds the declared weight hash (pip-style), which `resolve` requires and
115      which is checked against the downloaded bytes at load. `ref` is empty only for a url with no
116      declared hash, which `resolve` refuses.
117    - `local:<path>`: source `local`. The rest of the string is the id verbatim, any non-empty path;
118      `ref` is empty.
119
120    Every reference has a text form: one whose `str` does not parse back to it is refused.
121    """
122
123    source: Literal["hf", "url", "local"]
124    id: NonEmptyStr
125    ref: str
126
127    @model_validator(mode="after")
128    def _check_form(self) -> "ModelRef":
129        self._check_source_form()
130        if _text_fields(str(self)) != (self.source, self.id, self.ref):
131            raise ValueError(f"{self.source} reference {self.id!r} has no text form that parses back to it")
132        return self
133
134    def _check_source_form(self) -> None:
135        if self.source == "hf":
136            if not _HF_ID.fullmatch(self.id):
137                raise ValueError(f"hub id must be 'name' or 'org/name', got {self.id!r}")
138            if not _HF_REF.fullmatch(self.ref):
139                raise ValueError(
140                    f"hub reference {self.id!r} needs a revision of '/'-separated components without whitespace, "
141                    f"'@', '\\', a leading '.' or '..', got {self.ref!r}"
142                )
143            return
144        if self.source == "local" and self.ref:
145            raise ValueError(f"local references carry no revision, got {self.ref!r}")
146        if self.source == "url":
147            if self.ref and not SHA256_HEX.fullmatch(self.ref):
148                raise ValueError(f"a url reference's declared hash is 64 lowercase hex, got {self.ref!r}")
149            parts = urlsplit(self.id)
150            if not parts.scheme or not parts.netloc or re.search(r"\s", self.id):
151                raise ValueError(f"url reference must be an absolute URL with a host, got {self.id!r}")
152
153    @classmethod
154    def parse(cls, text: str) -> "ModelRef":
155        """Parse `org/name@rev`, `url:<url>` or `local:<path>`; raise `ValueError` on anything else."""
156        source, model_id, ref = _text_fields(text)
157        return cls(source=source, id=model_id, ref=ref)
158
159    def __str__(self) -> str:
160        """Return the text form that `parse` reads."""
161        if self.source == "hf":
162            return f"{self.id}@{self.ref}"
163        return (
164            f"url:{self.id}{_URL_HASH}{self.ref}" if self.source == "url" and self.ref else f"{self.source}:{self.id}"
165        )

168class ResolvedModel(Frozen):
169    """A reference bound to the identity of its weights.
170
171    A pinned (`unresolved_reason is None`) hub model's identity is its 40-hex commit; a pinned url or
172    local model's is the sha256 of its weight set (for a url, the hash declared in the reference).
173    A source that cannot be pinned carries a non-empty `unresolved_reason`, and its identity is
174    unconstrained; none exists in Phase 1.
175    """
176
177    ref: ModelRef
178    identity: NonEmptyStr
179    unresolved_reason: NonEmptyStr | None = None
180    license: License | None = None
181
182    @model_validator(mode="after")
183    def _check_identity(self) -> "ResolvedModel":
184        if self.unresolved_reason is None and (error := _pinned_identity_error(self.ref.source, self.identity)):
185            raise ValueError(error)
186        return self
187
188    @property
189    def pinned(self) -> bool:
190        """Whether the identity pins the weights."""
191        return self.unresolved_reason is None
192
193    def key_identity(self) -> "ModelIdentity":
194        """Return the model's identity as a logical key holds it.
195
196        Raises:
197            UnpinnedModel: if the model is not pinned.
198        """
199        if not self.pinned:
200            raise UnpinnedModel(f"{self.ref} is not pinned ({self.unresolved_reason}); it cannot enter a key")
201        model_id = self.ref.id if self.ref.source == "hf" else None
202        return ModelIdentity(id=model_id, identity=self.identity)

361def acceptable(
362    stored: ExecutionContext,
363    current: ExecutionContext,
364    policy: ReusePolicy | str,
365    *,
366    backend: str,
367    equivalence: Equivalence = NO_EQUIVALENCE,
368) -> bool:
369    """Return whether a result `backend` computed in `stored` may be reused in `current` under `policy`.
370
371    `policy` is coerced with `ReusePolicy(policy)`. `exact` needs equal fingerprints; `equivalent`
372    also accepts two contexts whose (numerics class, env lock) members (`member_key`) are in one of the backend's
373    classes in `equivalence`; both keep a result to its env lock unless a class measured that pair. `tolerant`
374    accepts any context.
375
376    Raises:
377        ValueError: if `policy` is not a `ReusePolicy` value.
378    """
379    policy = ReusePolicy(policy)
380    a, b = stored.fingerprint(), current.fingerprint()
381    match policy:
382        case ReusePolicy.TOLERANT:
383            return True
384        case ReusePolicy.EXACT:
385            return a == b
386        case ReusePolicy.EQUIVALENT:
387            na, nb = member_key(stored.numerics, stored.env_lock), member_key(current.numerics, current.env_lock)
388            return a == b or any(na in members and nb in members for members in equivalence.get(backend, {}).values())
389        case _:
390            assert_never(policy)
