 27def _finite_vector(v: object) -> numpy.ndarray:

167class Transcript(Formatted):
168    """A tree of transcript nodes: `segments`, in order, each dividing into words, over a recorded `extent`.
169
170    Every span lies inside `extent`, on its clock; a transcript with no span may declare no extent.
171    """
172
173    format: FormatTag = "transcript@2.0"
174
175    segments: tuple[TranscriptNode, ...]
176    extent: Span | None = None
177
178    @model_validator(mode="after")
179    def _spans_within_extent(self) -> "Transcript":
180        spans = [n.span for s in self.segments for n in s.nodes() if n.span is not None]
181        if spans and self.extent is None:
182            raise ValueError("a transcript with timed spans declares its extent")
183        for span in spans:
184            if self.extent is not None and span.clock_id != self.extent.clock_id:
185                raise ValueError(f"a span on clock {span.clock_id!r} in a transcript on clock {self.extent.clock_id!r}")
186            if (
187                self.extent is not None
188                and not self.extent.start_ns <= span.start_ns <= span.end_ns <= self.extent.end_ns
189            ):
190                raise ValueError(
191                    f"a span [{span.start_ns}, {span.end_ns}) lies outside the declared extent "
192                    f"[{self.extent.start_ns}, {self.extent.end_ns})"
193                )
194        return self

197class TextSpans(Formatted):
198    """Columnar labelled character spans `[start[i], end[i])` over the text whose content identity is `text`.
199
200    Offsets count code points of the text's canonical content, of `length` code points, with
201    `0 <= start <= end <= length`; `score` is an optional finite score per span.
202    """
203
204    format: FormatTag = "text-spans@1.0"
205
206    text: TextIdentity
207    length: Annotated[int, Strict(), Field(ge=0, description="Length of the text in code points.")]
208    start: OffsetArray
209    end: OffsetArray
210    label: LabelArray
211    score: (
212        Annotated[
213            numpy.ndarray,
214            BeforeValidator(_finite_column),
215            ArrayForm(("number",), 1, None, "One finite score per span."),
216        ]
217        | None
218    ) = None
219
220    @model_validator(mode="after")
221    def _check_columns(self) -> "TextSpans":
222        _check_lengths(start=self.start, end=self.end, label=self.label, score=self.score)
223        if numpy.any(self.start < 0) or numpy.any(self.end < self.start) or numpy.any(self.end > self.length):
224            raise ValueError(f"a span lies outside [0, {self.length}] or ends before it starts")
225        return self
226
227    @classmethod
228    def over(cls, text: Text, *, start: object, end: object, label: object, score: object = None) -> "TextSpans":
229        """Return the spans over `text`, naming its identity and length."""
230        return cls.model_validate(
231            {"text": text.identity, "length": len(text.content), "start": start, "end": end, "label": label}
232            | ({} if score is None else {"score": score})
233        )

265class Metric(Formatted):
266    """The value of metric `name`, in `unit`, computed over `items` items.
267
268    A ratio metric carries the `numerator` and `denominator` of its `value`, so that the aggregate over several
269    results is the ratio of their sums; a metric that is not a ratio of sums carries neither. A zero denominator
270    (with a zero numerator) is a result with nothing to score; `empty_value` is what the metric declares such a
271    result, and the value of an aggregate whose summed denominator is 0 (None: undefined).
272    """
273
274    format: FormatTag = "metric@2.0"
275
276    name: NonEmptyStr
277    value: FiniteFloat
278    unit: NonEmptyStr
279    items: Annotated[int, Strict(), Field(ge=1)]
280    numerator: FiniteFloat | None = None
281    denominator: Annotated[float, Field(ge=0, allow_inf_nan=False)] | None = None
282    empty_value: FiniteFloat | None = None
283
284    @model_validator(mode="after")
285    def _ratio_parts_come_together(self) -> "Metric":
286        if (self.numerator is None) != (self.denominator is None):
287            raise ValueError("a metric carries both its numerator and its denominator, or neither")
288        if self.denominator == 0 and self.numerator != 0:
289            raise ValueError("a metric of denominator 0 has numerator 0")
290        if self.empty_value is not None and self.denominator is None:
291            raise ValueError("a metric declares an empty value only with its numerator and denominator")
292        return self
