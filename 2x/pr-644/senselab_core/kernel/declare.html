101def check_canonical_params(model: type[BaseModel]) -> None:
102    """Check that every field of `model` has a type whose values have one canonical JSON form.
103
104    Allowed, recursively: `None`, `bool`, `int`, `float`, `str`; `Literal` of those; an `Enum` whose
105    values are all `str`; `list[X]`, `tuple[X, ...]` and `tuple[X, Y, ...]`; `dict[str, X]` and
106    `Mapping[str, X]`; `Optional` and unions of allowed types; nested `DeclaredParams`; and
107    `Annotated` constraints over any of these.
108
109    Raises:
110        NonCanonicalParams: naming the first field and type outside that set, such as `set`,
111            `frozenset`, `Decimal`, `datetime`, `Path`, `bytes`, `Any` or `object`.
112    """
113    _check_model(model, model.__name__, set())
114
115
116def _check_model(model: type[BaseModel], where: str, seen: set[type[BaseModel]]) -> None:
117    if model in seen:
118        return
119    seen.add(model)
120    for name, field in model.model_fields.items():
121        _check_type(field.annotation, f"{where}.{name}", seen)
122
123
124def _check_type(t: object, where: str, seen: set[type[BaseModel]]) -> None:
125    origin = typing.get_origin(t)
126    if origin is None:
127        _check_plain(t, where, seen)
128    else:
129        _check_generic(origin, typing.get_args(t), t, where, seen)
130
131
132def _check_generic(origin: object, args: tuple[object, ...], t: object, where: str, seen: set[type[BaseModel]]) -> None:
133    if origin is Annotated:
134        _check_type(args[0], where, seen)
135    elif origin is Literal:
136        bad = [a for a in args if type(a) not in _SCALARS]
137        if bad:
138            raise NonCanonicalParams(f"{where}: Literal values must be JSON scalars, got {bad!r}")
139    elif origin in _UNIONS or (origin in _SEQUENCES and args):
140        for arg in args:
141            if arg is not Ellipsis:
142                _check_type(arg, where, seen)
143    elif origin in _MAPPINGS and len(args) == 2:  # noqa: PLR2004 - a mapping type has a key and a value parameter
144        if args[0] is not str:
145            raise NonCanonicalParams(f"{where}: mapping keys must be str, got {args[0]!r}")
146        _check_type(args[1], where, seen)
147    else:
148        raise NonCanonicalParams(f"{where}: {t!r} has no canonical JSON form; declare a canonical type")
149
150
151def _check_plain(t: object, where: str, seen: set[type[BaseModel]]) -> None:
152    if t in _SCALARS:
153        return
154    if isinstance(t, type) and issubclass(t, Enum):
155        if not all(type(m.value) is str for m in t):
156            raise NonCanonicalParams(f"{where}: an Enum param must have str values, {t.__qualname__} does not")
157    elif isinstance(t, type) and issubclass(t, DeclaredParams):
158        _check_model(t, where, seen)
159    else:
160        raise NonCanonicalParams(f"{where}: {t!r} has no canonical JSON form; declare a canonical type")

183class TaskSpec(Frozen):
184    """A declared task: its name and version, kind, typed ports, params model, constraints and profile slices.
185
186    `inputs` and `outputs` map port names to kernel type names. `constraints` name only declared ports and params.
187    `profile_slices` names the profile slices a call of the task reads; each enters its key by identity.
188    """
189
190    name: NonEmptyStr
191    version: NonEmptyStr
192    kind: TaskKind
193    inputs: StrMap
194    outputs: StrMap
195    params: type[TaskParams]
196    constraints: tuple[Constraint, ...] = ()
197    profile_slices: tuple[SliceName, ...] = ()
198
199    @model_validator(mode="after")
200    def _constraints_name_declared_things(self) -> "TaskSpec":
201        for c in self.constraints:
202            names, declared = (
203                (c.ports, self.inputs) if isinstance(c, InputsAgree) else ((c.low, c.high), self.params.model_fields)
204            )
205            unknown = [n for n in names if n not in declared]
206            if unknown:
207                raise ValueError(f"{self.name}: constraint {c.kind} names {unknown}, which the task does not declare")
208        return self
209
210    def signature(self) -> tuple[str, frozenset[tuple[str, str]], frozenset[tuple[str, str]]]:
211        """Return the declared operation: kind, input ports and output ports with their types."""
212        return self.kind, frozenset(self.inputs.items()), frozenset(self.outputs.items())
213
214    def check_inputs(self, values: Mapping[str, object]) -> None:
215        """Check each `InputsAgree` constraint against the input values by port.
216
217        Raises:
218            InconsistentInputs: two inputs differ in a property the task declares they share.
219        """
220        for c in self.constraints:
221            if isinstance(c, InputsAgree):
222                first, second = (getattr(values[p], c.property) for p in c.ports)
223                if first != second:
224                    raise InconsistentInputs(
225                        f"inputs {c.ports[0]!r} and {c.ports[1]!r} of {self.name!r} differ in {c.property}: "
226                        f"{first!r} and {second!r}"
227                    )
228
229    def check_params(self, params: BaseModel) -> None:
230        """Check each `ParamsOrdered` constraint against the validated params.
231
232        Raises:
233            InconsistentInputs: a `low` param exceeds its `high` param.
234        """
235        for c in self.constraints:
236            if isinstance(c, ParamsOrdered):
237                low, high = getattr(params, c.low), getattr(params, c.high)
238                if low is not None and high is not None and low > high:
239                    raise InconsistentInputs(f"param {c.low!r} ({low}) exceeds param {c.high!r} ({high})")

305class BackendSpec(Frozen):
306    """A declared implementation of a task.
307
308    `models` are the catalogue models it serves; `env` where it runs; `batch_invariant` whether an
309    item's outputs are independent of its batch-mates; `tunables` the declared values of each
310    tunable; `worker` the module serving `load(init)` and `handle(state, items, params)`; `worker_sources` the
311    source file of each module of an isolated backend's worker, by dotted name, `worker` among them (empty
312    for a host or server backend and for a worker the runtime itself provides), which `ensure_env` places in the env;
313    `load_stub` the `init` under which `load` builds the fake stack's model in place of weights; `params` its
314    `BackendParams` model; `server` the engine configuration of an `Env.server` backend, whose `worker` module runs in
315    this process and is given the running server (`load` takes it in `init["server"]`); `requirements` the
316    requirement strings a host backend needs from the senselab lock (`ensure_env` refuses an `Env.host()` backend
317    unless that lock covers each). `output_affecting` names the components of `OUTPUT_AFFECTING` that change this
318    backend's outputs and so enter its logical key, held sorted and once each; `chunk_grid` names the grid its worker
319    cuts an item into when that grid is not a backend param; `min_input_ns` the shortest audio, in ns, it accepts, which
320    its worker refuses shorter than with a `ContentFault` before the model runs (`None` for no minimum).
321    """
322
323    task: NonEmptyStr
324    name: NonEmptyStr
325    version: NonEmptyStr
326    stack: NonEmptyStr
327    models: tuple[NonEmptyStr, ...] = ()
328    env: Env
329    devices: tuple[NonEmptyStr, ...] = Field(min_length=1)
330    batch_invariant: bool
331    tunables: Tunables = FrozenMapping({})
332    output_affecting: tuple[str, ...] = ()
333    worker: ModuleName
334    worker_sources: StrMap = FrozenMapping({})
335    load_stub: Params
336    params: type[BackendParams] = BackendParams
337    server: ServerConfig | None = None
338    requirements: tuple[NonEmptyStr, ...] = ()
339    chunk_grid: NonEmptyStr | None = None
340    min_input_ns: int | None = Field(default=None, gt=0)
341
342    @field_validator("output_affecting")
343    @classmethod
344    def _check_output_affecting(cls, v: tuple[str, ...]) -> tuple[str, ...]:
345        if not set(v) <= OUTPUT_AFFECTING:
346            raise ValueError(f"output_affecting may name only {sorted(OUTPUT_AFFECTING)}, got {sorted(v)}")
347        return tuple(sorted(set(v)))
348
349    @field_validator("worker_sources")
350    @classmethod
351    def _check_worker_sources(cls, v: Mapping[str, str]) -> Mapping[str, str]:
352        bad = [m for m in v if not re.fullmatch(r"[A-Za-z_]\w*(\.[A-Za-z_]\w*)*", m)]
353        if bad:
354            raise ValueError(f"worker_sources names {bad}, not dotted module names")
355        return v

357    @model_validator(mode="after")
358    def _check_worker_placement(self) -> "BackendSpec":
359        if self.worker_sources and self.worker not in self.worker_sources:
360            raise ValueError(f"worker_sources must hold the worker module {self.worker!r}")
361        if self.worker_sources and self.env.kind == "host":
362            raise ValueError("a host backend runs in the senselab venv and places no worker sources")
363        return self

365    @model_validator(mode="after")
366    def _check_server(self) -> "BackendSpec":
367        if (self.env.kind == "server") != (self.server is not None):
368            raise ValueError("a server backend declares `server`, and no other backend does")
369        if self.env.kind == "server" and self.worker_sources:
370            raise ValueError("a server backend's worker runs in the senselab venv and places no worker sources")
371        return self

380class Registry:
381    """Declared tasks and their backends, in declaration order.
382
383    Refuses a repeated task name, a repeated backend name within a task, a backend of an undeclared
384    task, and a second task with the same kind and ports (P-37).
385    """
386
387    def __init__(self) -> None:
388        """Start empty."""
389        self._tasks: dict[str, TaskSpec] = {}
390        self._backends: dict[str, dict[str, BackendSpec]] = {}
391
392    def add_task(self, spec: TaskSpec) -> None:
393        """Declare `spec`.
394
395        Raises:
396            NonCanonicalParams: if its params model has a field with no canonical JSON form.
397            DuplicateDeclaration: if its name or its signature is already declared.
398        """
399        check_canonical_params(spec.params)
400        if spec.name in self._tasks:
401            raise DuplicateDeclaration(f"task {spec.name!r} is already declared", self._tasks[spec.name])
402        for existing in self._tasks.values():
403            if existing.signature() == spec.signature():
404                raise DuplicateDeclaration(
405                    f"task {spec.name!r} declares the same kind and ports as {existing.name!r}; "
406                    "make it a param, strategy or backend of that task",
407                    existing,
408                )
409        self._tasks[spec.name] = spec
410        self._backends[spec.name] = {}
411
412    def add_backend(self, spec: BackendSpec) -> None:
413        """Declare `spec` for its task.
414
415        Raises:
416            NonCanonicalParams: if its params model has a field with no canonical JSON form.
417            UnknownTask: if its task is not declared.
418            DuplicateDeclaration: if the task already has a backend of that name.
419        """
420        check_canonical_params(spec.params)
421        backends = self._task_backends(spec.task)
422        if spec.name in backends:
423            raise DuplicateDeclaration(
424                f"backend {spec.name!r} of task {spec.task!r} is already declared", backends[spec.name]
425            )
426        backends[spec.name] = spec
427
428    def snapshot(self) -> tuple[dict[str, TaskSpec], dict[str, dict[str, BackendSpec]]]:
429        """Return a copy of the declarations, for `restore`."""
430        return dict(self._tasks), {t: dict(b) for t, b in self._backends.items()}
431
432    def restore(self, snapshot: tuple[dict[str, TaskSpec], dict[str, dict[str, BackendSpec]]]) -> None:
433        """Make the declarations those `snapshot` took, dropping any declared since."""
434        tasks, backends = snapshot
435        self._tasks = dict(tasks)
436        self._backends = {t: dict(b) for t, b in backends.items()}
437
438    def tasks(self) -> list[TaskSpec]:
439        """Return the declared tasks."""
440        return list(self._tasks.values())
441
442    def spec(self, task: str) -> TaskSpec:
443        """Return the declared task `task`, or raise `UnknownTask`."""
444        self._task_backends(task)
445        return self._tasks[task]
446
447    def backends(self, task: str) -> list[BackendSpec]:
448        """Return the backends of `task`, or raise `UnknownTask`."""
449        return list(self._task_backends(task).values())
450
451    def _task_backends(self, task: str) -> dict[str, BackendSpec]:
452        try:
453            return self._backends[task]
454        except KeyError:
455            raise UnknownTask(f"no task {task!r} is declared; declared: {sorted(self._tasks)}") from None

466def kernel_types() -> dict[str, type]:
467    """Return the kernel value types a port may name, by name: every `Recording` and `Formatted` subclass and protocol.
468
469    The protocols are `PORT_PROTOCOLS`. A subclass whose `port_value` is False is left out.
470    """
471    found: dict[str, type] = {Recording.__name__: Recording} | {p.__name__: p for p in PORT_PROTOCOLS}
472    pending: list[type] = [Recording, Formatted]
473    while pending:
474        for sub in pending.pop().__subclasses__():
475            if getattr(sub, "port_value", True):
476                found[sub.__name__] = sub
477            pending.append(sub)
478    return found

484def task(  # noqa: PLR0913 - a task declares its name, version, kind, constraints, slices and registry
485    *,
486    name: str,
487    version: str,
488    kind: TaskKind,
489    constraints: tuple[Constraint, ...] = (),
490    profile_slices: tuple[str, ...] = (),
491    registry: Registry | None = None,
492) -> Callable[[C], C]:
493    """Set the decorated class's `spec` to its `TaskSpec`, and declare it in `registry` when one is given.
494
495    A package declares such classes in the registry its `register` is given, with `declare_module`.
496
497    The class holds `inputs` and `outputs`, each mapping port names to kernel value types (a port protocol, a
498    `Recording` or a `Formatted` result) or to `list[T]` of one, and may hold `Params`, a `TaskParams` subclass
499    (`TaskParams` when absent).
500
501    Raises:
502        TypeError: if `inputs` or `outputs` is missing or maps a port to anything but a kernel value type or a list
503            of one.
504    """
505
506    def declare(cls: C) -> C:
507        spec = TaskSpec(
508            name=name,
509            version=version,
510            kind=kind,
511            inputs=_port_types(cls, "inputs"),
512            outputs=_port_types(cls, "outputs"),
513            params=getattr(cls, "Params", TaskParams),
514            constraints=constraints,
515            profile_slices=profile_slices,
516        )
517        if registry is not None:
518            registry.add_task(spec)
519        cls.spec = spec  # type: ignore[attr-defined]
520        return cls
521
522    return declare

525def _port_type(t: object) -> str | None:
526    """Return the port type name of `t`, a kernel value type or `list[T]` of one, or None for anything else."""
527    if typing.get_origin(t) is list:
528        (item,) = typing.get_args(t) or (None,)
529        named = _port_type(item)
530        return None if named is None or named.startswith("list[") else f"list[{named}]"
531    if (
532        isinstance(t, type)
533        and (t in PORT_PROTOCOLS or issubclass(t, (Recording, Formatted)))
534        and getattr(t, "port_value", True)
535    ):
536        return t.__name__
537    return None

540def _port_types(cls: type, ports: str) -> dict[str, str]:

Return the logical key of calling backend for task (ADR-002).

params is an instance of the task's params model and backend_params of the backend's; inputs maps each declared input port, and no other, to its content: or output: identity; model is the resolved model, which enters the key as its key_identity(); profile_slices maps each profile slice the call reads to its profile: identity. batch_formation and chunk_grid are given exactly when the backend declares them output-affecting. service is the attestation of the service answering the call, given exactly when the backend is a server backend. The runtime builds keys with senselab_core.runtime.keys.build_key.

Raises:
  • ValueError: if the backend implements another task, params, backend params or inputs do not match the declarations, an output-affecting component is given undeclared or missing, or service is given for a backend that is not a server backend or missing for one that is.
  • UnpinnedModel: if model is not pinned.
  • pydantic.ValidationError: if an identity is not of its typed form.
