 94class SourceStamp(Frozen):
 95    """What a source file was when a reader opened it: its stat fields, a digest of its sound header, and its content.
 96
 97    `header` is `value_digest("soundfile.info", {frames, channels, samplerate, format, subtype})`;
 98    `frames` and `sampling_rate` repeat header fields a group's layout is checked against; `channels` is the channel
 99    layout the stamp was taken under, so a stamp of the same file under another layout is another stamp.
100    `chunks` are the sha256 of each consecutive `chunk_size`-byte chunk of the file and `content` their root, as
101    `content_identity` gives them; `content` is the identity root of a recording of that file. A windowed read
102    verifies only the chunks it touches. Stamps are taken and checked by a `SourceReader`
103    (`senselab_core.runtime.audio_io`).
104    """
105
106    size: int = Field(ge=0)
107    mtime_ns: int
108    inode: int = Field(ge=0)
109    device: int = Field(ge=0)
110    frames: int = Field(ge=0)
111    channels: tuple[ChannelInfo, ...] = Field(min_length=1)
112    sampling_rate: Rate
113    header: NonEmptyStr
114    chunk_size: int = Field(ge=1)
115    chunks: tuple[Hex64, ...]
116    content: Hex64

150class RegularGroup(Frozen):
151    """Channels sampled together at one rate, held in `data` `(channels, samples)` or read from the sound file `source`.
152
153    Exactly one of `data` and `source` is given. A source group carries `source_stamp`, the stamp a
154    `SourceReader` took of the file; validation reads no file. The stamp's layout and rate must match the
155    declared channel count and rate. Samples are decoded only by `read` through a reader, which
156    checks the file still has the stamp, content included. A held group carries a file's stamp only as
157    `decoded_from` builds it: with `decoded`, the `npy@1` digest of the samples the reader decoded, which
158    the held samples must still have; it is then identified as that file is.
159    """
160
161    channels: tuple[ChannelInfo, ...] = Field(min_length=1)
162    sampling_rate: Rate
163    data: SignalArray | None = None
164    source: SourcePath | None = None
165    source_stamp: SourceStamp | None = None
166    decoded: Annotated[str, StringConstraints(pattern=r"^[0-9a-f]{64}$")] | None = None
167
168    @model_validator(mode="after")
169    def _check_layout(self) -> "RegularGroup":
170        if (self.data is None) == (self.source is None):
171            raise ValueError("a regular group needs exactly one of data and source")
172        stamp = self.source_stamp
173        if self.source is not None and stamp is None:
174            raise ValueError("a source group carries the stamp a reader took of its file")
175        if self.decoded is not None and (self.data is None or stamp is None):
176            raise ValueError("`decoded` belongs only to held samples carrying their file's stamp")
177        if stamp is not None and stamp.channels != self.channels:
178            raise ValueError("declared channels differ from the channel layout the stamp was taken under")
179        if stamp is not None and stamp.sampling_rate != self.sampling_rate:
180            raise ValueError(f"declared rate {self.sampling_rate} Hz, file rate {stamp.sampling_rate} Hz")
181        if self.data is not None:
182            require_finite(self.data, "a regular group's samples")
183            if stamp is not None and self.data.shape != (len(stamp.channels), stamp.frames):
184                raise ValueError(f"held samples {self.data.shape} are not the stamped file's channels and frames")
185            if stamp is not None and self.decoded != held_npy_digest(self.data):
186                raise ValueError("held samples carry a file's stamp only as decoded from it (`decoded_from`)")
187            rows = self.data.shape[0]
188        else:
189            assert stamp is not None
190            rows = len(stamp.channels)
191        _check_channels(self.channels, rows)
192        return self
193
194    @classmethod
195    def decoded_from(cls, group: "RegularGroup", reader: SourceReader) -> "RegularGroup":
196        """Return a source group's samples decoded by `reader`, held with the file's stamp and their digest.
197
198        Raises:
199            ValueError: if `group` holds its samples, or as `read`.
200            StaleSource: if the file changed since it was stamped.
201        """
202        if group.source is None:
203            raise ValueError("only a source group is decoded; a held group is its own samples")
204        data = group.read(reader=reader)
205        fields = {"channels": group.channels, "sampling_rate": group.sampling_rate, "data": data}
206        return cls(**fields, source_stamp=group.source_stamp, decoded=held_npy_digest(data))
207
208    @property
209    def frames(self) -> int:
210        """Number of samples per channel, from the held data or the source stamp."""
211        if self.data is not None:
212            return int(self.data.shape[1])
213        assert self.source_stamp is not None
214        return self.source_stamp.frames
215
216    def read(
217        self, start_ns: int | None = None, end_ns: int | None = None, reader: SourceReader | None = None
218    ) -> numpy.ndarray:
219        """Return the `(channels, samples)` samples whose times lie in `[start_ns, end_ns)` from the group start.
220
221        Omitted bounds mean the start and the end; a range running past the end stops there. Held
222        data is returned as a read-only view; a source group is decoded by `reader` to float32 for
223        the requested frames only, after the reader checks the file still matches `source_stamp`.
224
225        Raises:
226            TypeError: if a bound is not an integer.
227            ValueError: if `start_ns` is negative or `end_ns < start_ns`, a source group is read
228                without a reader, or a decoded sample is not finite or not of the group's layout.
229            StaleSource: if the source file changed since it was stamped.
230        """
231        start = 0 if start_ns is None else require_ns(start_ns, "start_ns")
232        if start < 0:
233            raise ValueError(f"start_ns must be non-negative, got {start}")
234        end = None if end_ns is None else require_ns(end_ns, "end_ns")
235        if end is not None and end < start:
236            raise ValueError(f"end_ns {end} is before start_ns {start}")
237        frames = self.frames
238        first = min(_frame_at(start, self.sampling_rate), frames)
239        stop = frames if end is None else min(_frame_at(end, self.sampling_rate), frames)
240        if self.data is not None:
241            return self.data[:, first:stop]
242        assert self.source is not None and self.source_stamp is not None
243        if reader is None:
244            raise ValueError(f"{self.source} is a source group; read it through a SourceReader")
245        samples = reader.decode(Path(self.source), self.source_stamp, first, stop - first)
246        expected = (len(self.channels), stop - first)
247        if samples.shape != expected or samples.dtype != AUDIO_DTYPE:
248            raise ValueError(f"decoded {samples.dtype} {samples.shape} from {self.source}, expected float32 {expected}")
249        return require_finite(samples, f"samples decoded from {self.source}")

310class Recording(Frozen):
311    """Named channel groups sharing one clock, anchored by `anchor`; `groups` is read-only.
312
313    `identity` is derived from the content by validation: omitted, it is filled in; supplied, it must
314    equal the derived one. It is `content:` and `value_digest(IDENTITY_TYPE, {"groups": ...})`, each stamped
315    group standing as its channels, rate and content identity root.
316    `IDENTITY_TYPE` is a declared token, so moving, renaming or subclassing the class keeps identities.
317    """
318
319    IDENTITY_TYPE: ClassVar[str] = "recording"
320
321    groups: Groups
322    anchor: TimeAnchor
323    identity: ContentIdentity
324
325    if TYPE_CHECKING:
326
327        def __init__(
328            self,
329            *,
330            groups: Mapping[str, "RegularGroup | IrregularGroup"],
331            anchor: TimeAnchor,
332            identity: str | None = None,
333        ) -> None:
334            """Validate `groups` and `anchor`, deriving `identity` or checking the one supplied."""
335
336    @model_validator(mode="wrap")
337    @classmethod
338    def _derive_identity(cls, data: object, handler: ModelWrapValidatorHandler[Self]) -> Self:
339        if isinstance(data, Mapping) and "groups" in data:
340            groups = _GROUPS.validate_python(data["groups"])
341            derived = cls.identity_of(groups)
342            supplied = data.get("identity")
343            if supplied is not None and supplied != derived:
344                raise ValueError(f"identity {supplied!r} is not the content identity {derived!r}")
345            data = {**data, "groups": groups, "identity": derived}
346        return handler(data)

348    @classmethod
349    def identity_of(cls, groups: Mapping[str, "RegularGroup | IrregularGroup"]) -> str:
350        """Return the identity a recording of this type over `groups` has.
351
352        Raises:
353            ValueError: if `groups` is not a layout this type holds.
354        """
355        fields = {name: _group_content(g) for name, g in groups.items()}
356        return "content:" + value_digest(cls.IDENTITY_TYPE, {"groups": fields})

365def _group_content(group: RegularGroup | IrregularGroup) -> object:
366    if isinstance(group, RegularGroup) and group.source_stamp is not None:
367        held = group.model_dump(exclude={"source", "source_stamp", "decoded"})

372def _audio_group(groups: Mapping[str, RegularGroup | IrregularGroup]) -> RegularGroup:
373    group = groups.get(AUDIO_GROUP)
374    if list(groups) != [AUDIO_GROUP] or not isinstance(group, RegularGroup):
375        raise ValueError(f"audio has exactly one regular group named {AUDIO_GROUP!r}, got {list(groups)}")
376    channels = group.channels
377    if any(c.type != "audio" for c in channels):
378        raise ValueError(f"audio channels must have type 'audio', got {[c.type for c in channels]}")
379    if len({c.units for c in channels}) != 1:
380        raise ValueError(f"audio channels must share units, got {[c.units for c in channels]}")
381    if group.data is not None and group.data.dtype != AUDIO_DTYPE:
382        raise ValueError(f"audio samples are held as float32 full scale, got {group.data.dtype}")
383    return group

396class Audio(Recording):
397    """A recording of one regular group named `audio` whose channels are all of type `audio`, in one unit.
398
399    Held samples are float32 (`AUDIO_DTYPE`). An in-memory audio's identity is `"content:" +
400    identity_digest("audio@4", {"npy": npy_digest("npy@1", <the (channels, samples) data>),
401    "sampling_rate": rate, "channels": <each ChannelInfo's model_dump()>})`; a stamped one's (file-backed, or held
402    with its file's stamp) is `"content:" + identity_digest("audio-file@1", {"content": <the root of the file's content
403    identity>, "channels": <each ChannelInfo's model_dump()>})`, so relabelled channels are another identity.
404    """
405
406    if TYPE_CHECKING:
407
408        def __init__(
409            self,
410            *,
411            groups: Mapping[str, "RegularGroup | IrregularGroup"],
412            anchor: TimeAnchor,
413            identity: str | None = None,

417    @classmethod
418    def identity_of(cls, groups: Mapping[str, "RegularGroup | IrregularGroup"]) -> str:
419        """Return the identity of audio over `groups`, after checking they are an audio layout.
420
421        Raises:
422            ValueError: if `groups` is not one regular `audio` group of float32 audio channels in one unit.
423        """
424        group = _audio_group(groups)
425        if group.source_stamp is not None or group.data is None:
426            file_fields = {"content": _source_root(group), "channels": [c.model_dump() for c in group.channels]}
427            return "content:" + identity_digest("audio-file@1", file_fields)
428        npy = held_npy_digest(group.data)
429        channels = [c.model_dump() for c in group.channels]
430        fields = {"npy": npy, "sampling_rate": group.sampling_rate, "channels": channels}
431        return "content:" + identity_digest("audio@4", fields)

461    @classmethod
462    def from_group(cls, group: RegularGroup) -> "Audio":
463        """Return the audio of one regular group of audio channels, its clock named by its identity.
464
465        Raises:
466            ValueError: if `group` is not float32 audio channels in one unit.
467        """
468        identity = cls.identity_of({AUDIO_GROUP: group})
469        anchor = TimeAnchor(clock_id=identity, absolute_start_ns=None)

472    @classmethod
473    def from_array(cls, a: numpy.ndarray, sampling_rate: float) -> "Audio":
474        """Hold a float32 full-scale copy of samples shaped `(samples,)` or `(channels, samples)`.
475
476        Integer PCM is divided by its full scale (`SIGNAL_DTYPES`) as the decoder divides it; floats
477        are cast. The integer conversion is done in float32 (round to float32, then scale by a power of
478        two) and is bit-identical to dividing in float64 and casting to float32. The recording's clock
479        is named by its identity.
480
481        Raises:
482            ValueError: if `a` is neither 1-D nor 2-D, is empty, holds a NaN or infinite sample or one
483                that overflows float32, or its dtype is not in `SIGNAL_DTYPES` (complex, boolean, object,
484                datetime and other integer widths are refused).
485            pydantic.ValidationError: if `sampling_rate` is not a `Rate`.
486        """
487        if isinstance(a, numpy.ndarray) and type(a) is not numpy.ndarray:
488            raise ValueError(f"not a plain numpy.ndarray: {type(a).__qualname__}")
489        x = numpy.asarray(a)
490        if x.ndim == 1:
491            x = x.reshape(1, -1)
492        if x.ndim != 2:  # noqa: PLR2004 - the (channels, samples) layout has two axes
493            raise ValueError(f"audio samples must be 1-D or (channels, samples), got shape {x.shape}")
494        if x.size == 0:
495            raise ValueError(f"audio samples must not be empty, got shape {x.shape}")
496        if x.dtype.name not in SIGNAL_DTYPES:
497            raise ValueError(f"audio samples must be one of {sorted(SIGNAL_DTYPES)}, got {x.dtype}")
498        scale = SIGNAL_DTYPES[x.dtype.name]
499        with numpy.errstate(over="ignore", invalid="ignore"):
500            held = x.astype(AUDIO_DTYPE) if scale is None else x.astype(AUDIO_DTYPE) / AUDIO_DTYPE.type(scale)
501        held = require_finite(held, "audio samples, as float32")
502        group = RegularGroup(channels=audio_channels(x.shape[0]), sampling_rate=sampling_rate, data=held)
503        return cls.from_group(group)
