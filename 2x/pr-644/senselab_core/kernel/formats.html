128def _name_and_major(tag: str) -> tuple[str, str]:
129    name, version = tag.split("@")
130    return name, version.split(".")[0]
131

133def readable(written: str, found: object) -> bool:
134    """Whether `found` is a format tag of `written`'s name and major version: one a reader of `written` reads."""
135    return (
136        isinstance(found, str)
137        and re.fullmatch(FORMAT_PATTERN, found) is not None
138        and _name_and_major(found) == _name_and_major(written)
139    )

142class FormatVerdict(str, Enum):
143    """What a reader of one format makes of the tag a record declares (`check`)."""
144
145    ACCEPTED = "accepted"
146    """The written format's name and major: the reader reads it, ignoring fields of a later minor."""
147    RETIRED = "retired"
148    """A retired `<name>@<major>`: refused with the migration statement, never treated as corrupt."""
149    UNKNOWN = "unknown"
150    """Another format tag, such as a later major: version skew, not corruption."""
151    MALFORMED = "malformed"
152    """Not a format tag at all: the record is corrupt."""
153
154
155def check(written: str, found: object, retired: Collection[str] = ()) -> FormatVerdict:
156    """Return the verdict of a reader of `written` on a record declaring `found`, `retired` as `<name>@<major>`."""
157    if not isinstance(found, str) or re.fullmatch(FORMAT_PATTERN, found) is None:
158        return FormatVerdict.MALFORMED
159    if readable(written, found):
160        return FormatVerdict.ACCEPTED
161    return FormatVerdict.RETIRED if "@".join(_name_and_major(found)) in retired else FormatVerdict.UNKNOWN
162
163
164def later_minor(written: str, found: str) -> bool:
165    """Whether `found`, of `written`'s name and major, is a later minor than `written`."""
166    return int(found.rsplit(".", 1)[1]) > int(written.rsplit(".", 1)[1])

169@contextmanager
170def reading(written: str, found: object) -> Iterator[None]:
171    """Scope in which a record declaring `found` is validated by a reader of `written`.
172
173    Under a later minor of `written`'s name and major, every `Frozen` validated inside drops the fields it does not
174    declare, at every nesting level, and records what it dropped (`Frozen.read_under`); the newer minor is logged the
175    first time it is met. Under any other tag unknown fields are refused. A `Formatted` decides from its own tag; a
176    bare `Frozen` inherits the scope it is validated in.
177    """
178    newer = check(written, found) is FormatVerdict.ACCEPTED and later_minor(written, str(found))
179    parent, current = _READING.get(), _Reading(str(found), []) if newer else None
180    token = _READING.set(current)
181    try:
182        yield
183    finally:
184        _READING.reset(token)
185        if current is not None:
186            if parent is not None:
187                parent.dropped.extend(current.dropped)
188            if (written, current.tag) not in _REPORTED:
189                _REPORTED.add((written, current.tag))
190                _LOG.info("reading %s as %s: ignoring %s", current.tag, written, sorted(set(current.dropped)))

193class FieldsOfLaterMinor(ValueError):
194    """A value read under a later minor lost fields: it is not part of an identity, and it is not rewritten."""
195
196    def __init__(self, read: ReadUnder) -> None:
197        """Name the minor and the fields."""
198        super().__init__(f"read under {read.tag}, which declares fields this version dropped: {list(read.dropped)}")
199        self.read = read
200
201
202def require_writable(value: Frozen) -> None:
203    """Refuse a value that was read under a later minor and lost fields.
204
205    Raises:
206        FieldsOfLaterMinor: naming the newer minor and the fields.
207    """
208    read = value.read_under()
209    if read is not None and read.dropped:
210        raise FieldsOfLaterMinor(read)

213def validated(model: type[F], record: Mapping[str, Any], written: str, found: str) -> F:
214    """Return `record` as `model`, read under `found` by a reader of `written`.
215
216    Raises:
217        FieldsOfLaterMinor: if `record` held fields `model` does not declare, so the value is not the whole of what
218            was written and its digest would differ.
219    """
220    with reading(written, found):
221        value = model.model_validate(record)
222    require_writable(value)
223    return value

226class Formatted(Frozen):
227    """A model carrying `format: "<name>@<major>.<minor>"`.
228
229    A subclass declares the tag it writes as the default, `format: FormatTag = "<name>@<major>.<minor>"`,
230    and then reads only tags with that name and major version. Under a later minor, unknown fields are ignored
231    at every nesting level (`reading`); under its own or an earlier minor they are refused.
232    """
233
234    format: FormatTag
235
236    port_value: ClassVar[bool] = True
237    """Whether a task port may name this type; a document that never flows on a port, such as a graph, sets False."""
238
239    @classmethod
240    def _reading(cls, data: object) -> AbstractContextManager[object]:
241        written = cls.model_fields["format"].default
242        if isinstance(data, Mapping) and isinstance(written, str):
243            return reading(written, data.get("format", written))
244        return nullcontext()
245
246    @classmethod
247    def __pydantic_init_subclass__(cls, **kwargs: object) -> None:
248        """Refuse a subclass whose declared default tag is malformed."""
249        super().__pydantic_init_subclass__(**kwargs)
250        written = cls.model_fields["format"].default
251        if isinstance(written, str) and not re.match(FORMAT_PATTERN, written):
252            raise TypeError(f"{cls.__name__} declares a malformed format {written!r}")
253
254    @model_validator(mode="after")
255    def _check_readable(self) -> "Formatted":
256        written = type(self).model_fields["format"].default
257        if isinstance(written, str) and not readable(written, self.format):
258            raise ValueError(f"{type(self).__name__} reads {written} and its minor versions, got {self.format}")
259        return self

Under any other tag unknown fields are refused. A Formatted decides from its own tag; a bare Frozen inherits the scope it is validated in.
