 38class FrozenMapping(Mapping[K, V]):
 39    """A read-only mapping that pickles and deep-copies.
 40
 41    Item assignment and deletion raise `TypeError`; attribute assignment raises `AttributeError`; a
 42    second call of `__init__` on a built mapping raises `TypeError`.
 43    """
 44
 45    __slots__ = ("_items",)
 46    _items: MappingProxyType[K, V]
 47
 48    def __init__(self, items: Mapping[K, V]) -> None:
 49        """Hold a copy of `items`; a mapping is initialised once."""
 50        try:
 51            object.__getattribute__(self, "_items")
 52        except AttributeError:
 53            object.__setattr__(self, "_items", MappingProxyType(dict(items)))
 54            return
 55        raise TypeError("a FrozenMapping is initialised once")
 56
 57    def __setattr__(self, name: str, value: object) -> None:
 58        raise AttributeError(f"FrozenMapping is read-only; cannot set {name!r}")
 59
 60    def __delattr__(self, name: str) -> None:
 61        raise AttributeError(f"FrozenMapping is read-only; cannot delete {name!r}")
 62
 63    def __getitem__(self, key: K) -> V:
 64        return self._items[key]
 65
 66    def __iter__(self) -> Iterator[K]:
 67        return iter(self._items)
 68
 69    def __len__(self) -> int:
 70        return len(self._items)
 71
 72    def __repr__(self) -> str:
 73        return f"FrozenMapping({dict(self._items)!r})"

 79def is_frozen(a: numpy.ndarray) -> bool:
 80    """Whether `a` is a plain C-order little-endian array whose `.base` chain ends in an immutable `bytes` object."""
 81    if type(a) is not numpy.ndarray or not a.flags.c_contiguous or a.dtype.byteorder == ">":
 82        return False
 83    node: object = a
 84    while isinstance(node, numpy.ndarray):
 85        if node.flags.writeable:
 86            return False
 87        node = node.base
 88    return type(node) is bytes

 91def frozen_array(a: object) -> numpy.ndarray:
 92    """Return `a` if it `is_frozen`, otherwise a C-order little-endian copy whose buffer is an immutable `bytes` object.
 93
 94    Neither the returned array nor any array on its `.base` chain can be made writeable.
 95
 96    Raises:
 97        ValueError: if `a` is an ndarray subclass (a masked array, for example), whose extra state
 98            a plain copy would drop, or holds Python objects.
 99    """
100    if isinstance(a, numpy.ndarray) and type(a) is not numpy.ndarray:
101        raise ValueError(f"not a plain numpy.ndarray: {type(a).__module__}.{type(a).__qualname__}")
102    if isinstance(a, numpy.ndarray) and is_frozen(a):
103        return a
104    x = numpy.asarray(a)
105    if x.dtype.hasobject:
106        raise ValueError(f"an array of Python objects has no bytes to own: {x.dtype}")
107    dtype = x.dtype.newbyteorder("<")
108    data = numpy.ascontiguousarray(x, dtype=dtype).tobytes(order="C")
109    return numpy.frombuffer(data, dtype=dtype).reshape(x.shape)

121def held_npy_digest(a: numpy.ndarray) -> str:
122    """Return `npy_digest("npy@1", a)`, computed once per frozen array (`is_frozen`) and reused while it lives."""
123    if not is_frozen(a):
124        return npy_digest("npy@1", a)
125    key = id(a)
126    held = _NPY_DIGESTS.get(key)
127    if held is not None:
128        return held
129    digest = npy_digest("npy@1", a)
130    if _NPY_DIGESTS.setdefault(key, digest) is digest:
131        weakref.finalize(a, _forget, key)
132    return digest

160def value_digest(type_name: str, fields: Mapping[str, object]) -> str:
161    """Return `identity_digest("value@2", {"type": type_name, "fields": <fields in tokens>})`.
162
163    `fields` is written by the shared `canonical.Tokeniser`, which gives each family outside JSON its
164    own token: an array is `{"$array": <sha256 of its canonical .npy>}`, an integer outside the
165    I-JSON range `{"$int": "<decimal>"}`, a non-finite float `{"$f": ...}`, a type
166    `{"$type": "<module>.<qualname>"}` and a `Frozen` value `{"$value": {"type": ..., "fields": ...}}`;
167    any other mapping is an object, and a key that starts with `$` gains one more `$`. A set, an enum, a path,
168    bytes and a `Decimal` have no canonical form.
169
170    Raises:
171        NoCanonicalForm: on a value of no family, a lone surrogate, nesting deeper than `MAX_DEPTH`, or a
172            container that contains itself.
173        ValueError: if the tokens, inside the `{"type", "fields"}` envelope, nest deeper than `MAX_DEPTH` or
174            exceed `MAX_CANONICAL_CHARS`.
175    """
176    return identity_digest(_DIGEST_TAG, {"type": type_name, "fields": _TOKENISER(fields, "fields")})

184def _held_frozen(x: object) -> Iterator["Frozen"]:
185    """Yield each `Frozen` held in `x` or in the containers inside it, not entering a `Frozen` or a container twice."""
186    seen: set[int] = set()
187    pending = [x]
188    while pending:
189        item = pending.pop()
190        if isinstance(item, Frozen):
191            yield item
192        elif isinstance(item, (Mapping, list, tuple)) and id(item) not in seen:
193            seen.add(id(item))
194            pending.extend(reversed(list(item.values() if isinstance(item, Mapping) else item)))
195

197def _children(value: "Frozen") -> Iterator["Frozen"]:
198    for name in type(value).model_fields:
199        yield from _held_frozen(getattr(value, name))

202def _nodes(value: "Frozen") -> list["Frozen"]:
203    """Return `value` and every `Frozen` it holds, depth first, in field order, each object once."""
204    found: dict[int, Frozen] = {}
205    pending = [value]
206    while pending:
207        node = pending.pop()
208        if id(node) not in found:
209            found[id(node)] = node
210            pending.extend(reversed(list(_children(node))))
211    return list(found.values())

218def _carry(reads: Sequence["ReadUnder | None"], built: F) -> F:
219    """Give `built`, a rebuild of the value `reads` came from, the same read of itself and of each value it holds."""
220    nodes = _nodes(built)
221    if len(nodes) != len(reads):
222        nodes, reads = nodes[:1], reads[:1]
223    for node, read in zip(nodes, reads, strict=True):
224        held = object.__getattribute__(node, "__pydantic_private__")
225        object.__setattr__(node, "__pydantic_private__", MappingProxyType({**held, "_read": read}))
226    return built

229def _rebuild(cls: type[F], fields: dict[str, Any], reads: Sequence["ReadUnder | None"] = ()) -> F:
230    built = cls.model_validate(fields)
231    return _carry(reads, built) if any(reads) else built

234def thawed(value: object, walk: Walk | None = None) -> Any:  # noqa: ANN401
235    """Return `value` with every `Mapping` as a `dict` and every `list` or `tuple` as a `list`.
236
237    A container shared by several parents is thawed once per depth, through the shared guard.
238
239    Raises:
240        ValueError: if a container contains itself or nests deeper than `MAX_DEPTH`.
241    """
242    walk = Walk() if walk is None else walk
243    if isinstance(value, Mapping):
244        items = value
245        return walk.enter(value, lambda: {k: thawed(x, walk) for k, x in items.items()})
246    if isinstance(value, (list, tuple)):
247        elements: Sequence[object] = value
248        return walk.enter(value, lambda: [thawed(x, walk) for x in elements])
249    return value

256def _seal(v: "Frozen") -> None:
257    """Make a freshly validated value read-only: its fields-set and its derived private values."""
258    if _is_sealed(v):
259        return

261    object.__setattr__(v, "__pydantic_fields_set__", frozenset(object.__getattribute__(v, "__pydantic_fields_set__")))
262    object.__setattr__(v, "__pydantic_private__", private)

293def _declared(cls: type["Frozen"], data: Mapping[Any, Any]) -> dict[Any, Any]:
294    """Return `data` without the keys `cls` declares no field for, by name or alias."""
295    names = set(cls.model_fields)
296    for field in cls.model_fields.values():
297        names.update(a for a in (field.alias, field.validation_alias) if isinstance(a, str))
298    return {k: v for k, v in data.items() if k in names}

301def _held_fields(cls: type["Frozen"], data: object) -> object:
302    """Return what `cls` validates: a sealed instance of exactly `cls` as itself, any other model as its fields.
303
304    Raises:
305        ValueError: for anything that is neither a mapping nor a pydantic model.
306    """
307    if type(data) is cls and _is_sealed(data):
308        return data
309    if isinstance(data, BaseModel):
310        held = object.__getattribute__(data, "__dict__")
311        return {name: held[name] for name in cls.model_fields if name in held}
312    if isinstance(data, Mapping):
313        return data
314    raise ValueError(f"{cls.__name__} is built from a mapping of its fields, not from a {type(data).__qualname__}")

317class Frozen(BaseModel):
318    """Immutable pydantic model that may hold numpy arrays.
319
320    Every array a value holds is a `frozen_array`. A value is built only by validating a mapping of
321    its fields (`__init__`, `model_validate`, `model_validate_json`, nested validation). A sealed
322    instance of exactly the type passes as itself; any other model (an unsealed one, one made by
323    `BaseModel.model_construct`, a subclass) is validated again from its fields as the type; any other
324    object, `from_attributes` included, is refused. Values compare and hash by `canonical_digest`,
325    computed when the value is built. Every copy (pickle, `copy`, `deepcopy`, `model_copy`,
326    `__setstate__` on a blank instance) is rebuilt through `model_validate`, so it runs every
327    validator. `model_construct` is refused. Validation errors do not repeat their input.
328
329    A built value is read-only: attribute assignment and deletion raise `ValidationError`, its private
330    values are a read-only mapping, `__setstate__` on it raises `TypeError`, and the class's `model_config`
331    is a read-only mapping.
332    """
333
334    model_config = ConfigDict(frozen=True, arbitrary_types_allowed=True, extra="forbid", hide_input_in_errors=True)
335
336    clock_fields: ClassVar[tuple[str, ...] | None] = None
337    """The fields a re-bind to another clock shifts; a type with a `clock_id` must declare them (`()` for none)."""
338
339    @classmethod
340    def __get_pydantic_core_schema__(cls, source: type[BaseModel], handler: GetCoreSchemaHandler) -> Any:  # noqa: ANN401
341        """Wrap the model's schema so it accepts only a mapping or an instance, and seals what it builds."""
342        inner = handler(source)
343
344        def build(data: object, validate: ValidatorFunctionWrapHandler) -> object:
345            value = validate(_held_fields(cls, data))
346            if isinstance(value, Frozen):
347                _seal(value)
348            return value
349
350        return {"type": "function-wrap", "function": {"type": "no-info", "function": build}, "schema": inner}
351
352    @classmethod
353    def _reading(cls, data: object) -> AbstractContextManager[object]:  # noqa: ARG003
354        """Return the scope `data` is validated in; a type that declares its own format enters its reading rules."""
355        return nullcontext()
356
357    @model_validator(mode="wrap")
358    @classmethod
359    def _drop_unknown_fields(cls, data: object, handler: ModelWrapValidatorHandler[Self]) -> Self:
360        """Validate `data` in its reading scope, minus the undeclared fields while that scope is a later minor."""
361        with cls._reading(data):
362            read = _READING.get()
363            if read is None or not isinstance(data, Mapping):
364                return handler(data)
365            kept, start = _declared(cls, data), len(read.dropped)
366            read.dropped.extend(f"{cls.__name__}.{k}" for k in data if k not in kept)
367            value = handler(kept)
368            object.__setattr__(
369                value, "__pydantic_private__", {"_read": ReadUnder(read.tag, tuple(read.dropped[start:]))}
370            )
371            return value

373    @classmethod
374    def __pydantic_init_subclass__(cls, **kwargs: Any) -> None:  # noqa: ANN401
375        """Hold the subclass's config as a read-only mapping, and require a clock type to declare its clock fields.
376
377        Raises:
378            TypeError: if a type with a `clock_id` field has no `clock_fields`, or names one that is not a field.
379        """
380        super().__pydantic_init_subclass__(**kwargs)
381        unknown = [name for name in cls.clock_fields or () if name not in cls.model_fields]
382        if unknown or ("clock_id" in cls.model_fields and cls.clock_fields is None):
383            raise TypeError(
384                f"{cls.__name__}: clock_fields must name its clock-relative fields, got {unknown or 'none declared'}"
385            )
386        cls.model_config = MappingProxyType(dict(cls.model_config))  # type: ignore[assignment]

388    def _derived(self) -> Mapping[str, object]:
389        """Return the private values a built value holds; a subclass extends this mapping."""
390        held = object.__getattribute__(self, "__pydantic_private__") or {}
391        read = held.get("_read")
392        if read is None:
393            below = [r for child in _children(self) if (r := child.read_under()) is not None]
394            read = ReadUnder(below[0].tag, tuple(k for r in below for k in r.dropped)) if below else None
395        return {"_canonical": digest_of(self), "_read": read}

397    def read_under(self) -> ReadUnder | None:
398        """Return what reading this value under a later minor cost, None when it was read under its own."""
399        read: ReadUnder | None = (self.__pydantic_private__ or {}).get("_read")
400        return read

402    def canonical_digest(self) -> str:
403        """Return `value_digest("<module>.<qualname>", <each field's held value>)`, computed when the value was built.
404

406        held value and a mapping of its fields differ. A file-backed value's digest covers its path

423    def __setstate__(self, state: dict[Any, Any]) -> None:
424        """Initialise a blank instance from `state["__dict__"]` through `model_validate`.

426        Raises:

432        built = type(self).model_validate(dict(state.get("__dict__", {})))

434            object.__setattr__(self, name, object.__getattribute__(built, name))

456        copied = _rebuild(type(self), self.model_dump(), _reads(self))

461    def model_copy(self, *, update: Mapping[str, Any] | None = None, deep: bool = False) -> Self:
462        """Return `model_validate` of the dump merged with `update`, so every validator runs; `deep` has no effect.
463
464        Raises:
465            ValueError: if `update` names a field the type does not have.
466        """
467        unknown = sorted(set(update or {}) - set(type(self).model_fields), key=str)
468        if unknown:
469            raise ValueError(f"{type(self).__name__} has no fields {unknown}; known: {sorted(type(self).model_fields)}")
470        return _carry(_reads(self), type(self).model_validate({**self.model_dump(), **(update or {})}))

502def require_finite(a: numpy.ndarray, what: str, *, nan_is_missing: bool = False) -> numpy.ndarray:
503    """Return a frozen copy of `a` with no infinite value and, unless `nan_is_missing`, no NaN.
504
505    With `nan_is_missing`, a NaN marks a missing sample and is held as the one quiet NaN without payload
506    (`canonical.QUIET_NAN_BITS`), whatever its sign or payload.
507
508    Raises:
509        ValueError: naming `what`, on an infinite value, or on a NaN unless `nan_is_missing`.
510    """
511    held = frozen_array(a)
512    if held.dtype.kind != "f":
513        return held
514    if numpy.isinf(held).any():
515        raise ValueError(f"{what} must be finite; infinite values are refused")
516    nan = numpy.isnan(held)
517    if not nan.any():
518        return held
519    if not nan_is_missing:
520        raise ValueError(f"{what} must be finite; NaN values are refused")
521    out = numpy.array(held, copy=True)
522    out.view(f"<u{out.dtype.itemsize}")[nan] = QUIET_NAN_BITS[out.dtype.itemsize]
523    return frozen_array(out)
