senselab_core.kernel.dataset

Datasets: records of named inputs, the Dataset protocol, and the input manifest input-manifest@1.0.

A Record names its inputs, either files under a dataset's root or values. A Dataset is read through records, entities, filter and shard. InputManifest lists each record's id, entities and the content identity of every input a graph binds; InputManifest.digest identifies it and does not depend on where the root is.

 89class InputManifest(Formatted):
 90    """An `input-manifest@1.0` document: rows sorted by id, one per record, identified by `digest`."""

 93    port_value: ClassVar[bool] = False

 95
 96    @model_validator(mode="after")
 97    def _sorted_unique(self) -> "InputManifest":
 98        ids = [r.id for r in self.rows]
 99        if ids != sorted(set(ids)):
100            raise ValueError("manifest rows are sorted by id, each id once")
101        return self
102
103    def digest(self) -> str:
104        """Return `identity_digest("input-manifest@1", {"rows": ...})`.
105
106        Raises:
107            ValueError: if the manifest declares a later minor than `input-manifest@1.0`.
108        """
109        if later_minor(INPUT_MANIFEST_FORMAT, self.format):
110            raise ValueError(f"a {self.format} manifest has fields a {INPUT_MANIFEST_FORMAT} reader does not hold")
111        return identity_digest(MANIFEST_DIGEST_TAG, {"rows": self.model_dump(mode="json")["rows"]})

1968def _no_init_or_replace_init(self, *args, **kwargs):
1969    cls = type(self)
1970
1971    if cls._is_protocol:
1972        raise TypeError('Protocols cannot be instantiated')
1973
1974    # Already using a custom `__init__`. No need to calculate correct
1975    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1976    if cls.__init__ is not _no_init_or_replace_init:
1977        return
1978
1979    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1980    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1981    # searches for a proper new `__init__` in the MRO. The new `__init__`
1982    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1983    # instantiation of the protocol subclass will thus use the new
1984    # `__init__` and no longer call `_no_init_or_replace_init`.
1985    for base in cls.__mro__:
1986        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1987        if init is not _no_init_or_replace_init:
1988            cls.__init__ = init
1989            break
1990    else:
1991        # should not happen
1992        cls.__init__ = object.__init__
1993
1994    cls.__init__(self, *args, **kwargs)
