 29def mount_data(groups: "argparse._SubParsersAction[argparse.ArgumentParser]") -> None:
 30    """Add the ``data`` command group to a parser's subparsers."""
 31    data = groups.add_parser("data", help="versioned datasets")
 32    commands = data.add_subparsers(dest="command", required=True)
 33    fetch_cmd = commands.add_parser("fetch", help="download and verify the files of a data manifest")
 34    fetch_cmd.add_argument("manifest", type=Path, help="path to a data-manifest@2.0 YAML file")
 35    fetch_cmd.add_argument(
 36        "--cache",
 37        type=Path,
 38        default=None,
 39        help="cache directory (default: $SENSELAB_DATA_CACHE or ~/.cache/senselab/data)",
 40    )
 41    fetch_cmd.add_argument(
 42        "--profile", default=None, help="the execution profile whose retry policy and fetch timeout apply"
 43    )
 44    fetch_cmd.add_argument(
 45        "--verify", action="store_true", help="re-hash every cached file instead of checking its size"
 46    )
 47    data.set_defaults(run=run_data)

 50def run_data(args: argparse.Namespace) -> int:
 51    """Run a parsed ``data fetch``; return the exit code (0 on success, 1 on a reported error)."""
 52    try:
 53        manifest = DataManifest.load(args.manifest)
 54        profile = _profile(args.profile)
 55        paths = fetch(
 56            manifest,
 57            cache_root=args.cache or default_cache_root(),
 58            verify="full" if args.verify else "size",
 59            retry=profile.retry,
 60            network=profile.network,
 61            timeout_s=profile.timeouts.fetch_s,
 62        )
 63    except (ManifestError, FetchError) as err:
 64        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
 65        return 1
 66    except OSError as err:
 67        cause = " ".join(str(err).split())
 68        print(
 69            f"senselab: error: {cause}; check the paths exist and the cache directory is writable (--cache DIR)",
 70            file=sys.stderr,
 71        )
 72        return 1
 73    for name, path in paths.items():
 74        print(f"{name}\t{path}")
 75    return 0

 78def _run_directory(name: str) -> Path:
 79    """Return the run directory `name` names: a directory, or a plan hash or its unique prefix.
 80
 81    Raises:
 82        FileNotFoundError: no run matches, or a prefix matches several.
 83    """
 84    if Path(name).is_dir():
 85        return Path(name)
 86    root = runs_root()
 87    listed = sorted(root.iterdir()) if root.is_dir() else []
 88    found = [p for p in listed if p.is_dir() and name and p.name.startswith(name)]
 89    if len(found) != 1:
 90        raise FileNotFoundError(f"{'no run' if not found else 'several runs'} match {name!r} under {root}")
 91    return found[0]

 98def _placement_lines(events: list[Event], resource_class: str | None = None) -> list[str]:
 99    """Return one line per class placement in `events` (of `resource_class` when given): values, layers, verdict."""
100    lines = []
101    for event in placements(events):
102        found = event.placement
103        if found is None or resource_class not in (None, found.resource_class):
104            continue
105        if found.closing is not None:
106            asked = found.request
107            lines.append(
108                f"placement {found.resource_class} epoch {event.epoch}: closed; "
109                f"wall_s {found.closing.wall_s} (requested {None if asked is None else asked.wall_s}), "
110                f"gpu_hours {found.closing.gpu_hours} (requested {None if asked is None else asked.gpu_hours}), "
111                f"device peak bytes {found.closing.device_peak_bytes} "
112                f"(requested {None if asked is None else asked.device_memory_bytes})"
113            )
114            continue
115        layers = ", ".join(f"{k} by {v}" for k, v in found.used.layers.items())
116        note = "" if found.note is None else f"; {found.note}"
117        lines.append(
118            f"placement {found.resource_class} epoch {event.epoch}: {found.verdict}; partition {found.used.partition}, "
119            f"time_s {found.used.time_s}, max_workers {found.used.max_workers}, {len(found.exclude)} excluded "
120            f"({layers}){note}"
121        )
122    return lines

125def run_status(args: argparse.Namespace) -> int:
126    """Run a parsed ``status``: the jobs of a run by state, its faults and its broken envs, from its events."""
127    try:
128        run = Run(_run_directory(args.run_ref))
129        status = run.status()
130    except (OSError, ValueError) as err:
131        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
132        return 1
133    state = "cancelled" if status.cancelled else "done" if status.done else "not done"
134    print(f"run {run.manifest().get('plan_hash', run.run_dir.name)}: {state}; last event {status.last_event_ns}")
135    print("jobs: " + ", ".join(f"{k} {v}" for k, v in status.counts.items()))
136    print("faults: " + (", ".join(f"{k} {v}" for k, v in sorted(status.faults.items())) or "none"))
137    if status.reclaimed:
138        print(f"reclaimed claims: {status.reclaimed}")
139    if status.quarantined:
140        print(f"quarantined event lines: {status.quarantined}")
141    for message in status.unplaceable:
142        print(f"unplaceable: {' '.join(message.split())}")
143    for broken in status.broken:
144        print(f"broken env {broken.env}: {broken.failures} failures; {broken.fault.type}: {broken.fault.message}")
145    for node, counts in status.nodes.items():
146        print(f"node {node}: " + ", ".join(f"{k} {v}" for k, v in counts.items() if v))
147    for line in _placement_lines(run.events()):
148        print(line)
149    if args.check_provenance:
150        report = check_provenance(run.run_dir)
151        print(
152            f"provenance: {report.committed} committed keys, {report.executions} executions, {report.reuses} reuses: "
153            + ("complete" if report.complete else "INCOMPLETE")
154        )
155        for problem in report.problems:
156            print(f"provenance: {problem}")
157        return 0 if report.complete else 1
158    return 0

161def _event_line(event: Event) -> str:
162    element = "" if event.job is None else str(list(event.job.element))
163    fault = "" if event.fault is None else f" fault {event.fault}"
164    key = "" if event.key is None else f" key {event.key[:12]}"
165    return f"  {event.t_ns} {event.kind} {event.node}{element} attempt {event.attempt} seq {event.seq}{key}{fault}"

168def _placement(run_dir: Path, events: list[Event]) -> PlanPlacement:
169    """Return the placement of the attempt that wrote the newest of `events`, else the latest attempt's."""
170    for event in reversed(events):
171        path = run_dir / PLACEMENTS_DIR / f"{event.attempt}.json"
172        if path.is_file():
173            return read_placement(path)
174    return latest_placement(run_dir)

177def _failure_text(failure: FaultRecord) -> str:
178    """Return a failure's lines for `explain`: an unplaceable plan's own headline first, then its class and tail."""
179    head = f"unplaceable: {' '.join(failure.message.split())}\n" if failure.type == UNPLACEABLE else ""
180    return f"{head}fault {failure.fault}: {failure.type}\n{_tail(failure.message)}"

183def run_explain(args: argparse.Namespace) -> int:
184    """Run a parsed ``explain``: why a node or key failed, was skipped, hit the cache or was placed where it was."""
185    try:
186        run = Run(_run_directory(args.run_ref))
187        document = json.loads((run.run_dir / PLAN_FILE).read_text(encoding="utf-8"))
188        logical = LogicalPlan.model_validate(document["logical"])
189        events = run.events()
190    except (OSError, ValueError, KeyError) as err:
191        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
192        return 1
193    tasks = {n.id: n.task for n in logical.graph.nodes}
194    matched = [e for e in events if e.node == args.target or (e.key is not None and e.key.startswith(args.target))]
195    nodes = sorted({e.node for e in matched if e.node is not None} | ({args.target} if args.target in tasks else set()))
196    if not nodes:
197        print(f"senselab: error: no node or key {args.target!r} in run {run.run_dir.name}", file=sys.stderr)
198        return 1
199    for node in nodes:
200        shown = [e for e in matched if e.node == node] or [e for e in events if e.node == node]
201        try:
202            placement = _placement(run.run_dir, shown)
203        except (OSError, ValueError) as err:
204            print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
205            return 1
206        print(node_line(node, tasks[node], placement) if node in placement.nodes else f"node {node}")
207        if node in placement.nodes:
208            for line in _placement_lines(events, placement.nodes[node].resource_class):
209                print(line)
210        print(f"events ({len(shown)}):")
211        for event in shown:
212            print(_event_line(event))
213        for event in (e for e in shown if e.failure is not None):
214            assert event.failure is not None
215            print(_failure_text(event.failure))
216        for event in (e for e in shown if e.kind in ("blocked", "cancelled")):
217            why = "an upstream job failed" if event.kind == "blocked" else "the run was cancelled before it was sent"
218            print(f"{event.kind} {event.node}{list(event.job.element) if event.job else ''}: {why}")
219    return 0

222def _profile(argument: str | None) -> ExecutionProfile:
223    """Return the profile `argument` names: a packaged profile, a profile YAML file, or a whole profile (`.json`)."""
224    if argument is None:

228    return ExecutionProfile.load(argument)

240def run_plan_request(args: argparse.Namespace) -> int:
241    """Run a parsed ``plan``: print what each resource class of a run needs, as `placement-request@1.0` JSON.
242
243    Nothing is computed, no model downloaded and no worker started.
244    """
245    try:
246        profile = _profile(args.profile)
247        placed = _with_placement(run_plan(_run_directory(args.run_ref), profile), args)
248        document = placed.placement_request().document()
249    except (OSError, ValueError) as err:
250        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
251        return 1
252    print(json.dumps(document, indent=2, sort_keys=True))
253    return 0

260def run_coordinate(args: argparse.Namespace) -> int:
261    """Run a parsed ``coordinate``: take the run's next epoch and send its jobs to its queues, for pool workers.
262
263    A plan whose unit fits no resource class prints the unit, each class tried and the numbers, and exits 3.
264
265    The profile's substrate starts the workers unless ``--external`` says they are started on their own.
266    """
267    try:
268        profile = _profile(args.profile)
269        run_dir = _run_directory(args.run_ref)
270        substrate = (
271            ExternalWorkers() if args.external else substrate_of(profile, run_dir, run_plan(run_dir, profile).hash)
272        )
273        outcome = coordinate(
274            run_dir,
275            substrate=substrate,
276            profile=profile,
277            overrides=load_overrides(args.placement),
278            hook=args.placement_hook,
279        )
280    except Unplaceable as err:
281        print(f"senselab: unplaceable: {' '.join(str(err).split())}", file=sys.stderr)
282        return UNPLACEABLE_EXIT
283    except (OSError, ValueError, StaleCoordinator, QueueHeld, PlacementHookFailed) as err:
284        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
285        return 1
286    blocked = sum(len(e) for e in outcome.blocked.values())
287    failed = sum(r.failure is not None for refs in outcome.results.values() for r in refs)
288    print(f"run {outcome.plan_hash}: {'done' if outcome.done else 'not done'}; {failed} failed, {blocked} blocked")
289    return 0 if outcome.done else 1

292def run_execute(args: argparse.Namespace) -> int:
293    """Run a parsed ``execute``: serve a run's queues of the named resource classes as one pool worker."""
294    pool = Path(args.pool)
295    classes = [c for c in args.classes.split(",") if c]
296    try:
297        profile = _profile(args.profile)
298        return serve_pool(pool, classes, profile=profile, store=run_store(pool.parent, profile))
299    except (OSError, ValueError) as err:
300        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
301        return 1

313def run_unfit(args: argparse.Namespace) -> int:
314    """Run a parsed ``unfit``: list the units recorded as not fitting a class, or ``--clear`` one unit's records.
315
316    A unit is named by the digest label ``explain`` prints, or any prefix of its digest; ``--clear`` needs the prefix
317    to name one unit. A record binds the code that
318    wrote it, so a fixed loader needs no clearing; this is for a record of a load a busy card, not the unit, failed.
319    """
320    try:
321        profile = _profile(args.profile)
322        roots = measure_roots(profile)
323        found = [
324            (unit, records)
325            for unit in roots.units()
326            if unit.digest().startswith(args.unit or "") and (records := roots.unfit_records(unit))
327        ]
328        if args.clear and not args.unit:
329            print("senselab: error: --clear names no unit; give the unit's digest label", file=sys.stderr)
330            return 1
331        if args.clear and len(found) > 1:
332            listed = ", ".join(u.label() for u, _ in found)
333            print(
334                f"senselab: error: {args.unit!r} matches {len(found)} units ({listed}); give more of the label",
335                file=sys.stderr,
336            )
337            return 1
338        for unit, records in found:
339            if args.clear:
340                print(f"unit {unit.label()}: removed {roots.remove_unfit(unit)} record(s)")
341                continue
342            for record in records:
343                print(
344                    f"unit {unit.label()} ({unit.task} {unit.backend}): a load ran out of {record.resource} memory in "
345                    f"a grant of {record.grant_bytes} bytes"
346                )
347    except (OSError, ValueError) as err:
348        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
349        return 1
350    return 0

353def mount_runs(groups: "argparse._SubParsersAction[argparse.ArgumentParser]") -> None:
354    """Add the ``status``, ``explain``, ``coordinate`` and ``execute`` commands to a parser's subparsers."""
355    status = groups.add_parser("status", help="the jobs, faults and broken envs of a run")
356    status.add_argument(
357        "run_ref",
358        metavar="run",
359        help="a run directory, or a plan hash (or its unique prefix) under $SENSELAB_CACHE/2x/runs",
360    )
361    status.add_argument(
362        "--check-provenance",
363        action="store_true",
364        help=(
365            "also check ADR-010: each committed key has an execution or a Reuse, and no writer records more Reuse "
366            "of a key than the run has jobs with it; exit 1 if not"
367        ),
368    )
369    status.set_defaults(run=run_status)
370    planning = groups.add_parser("plan", help="what each resource class of a run needs, as placement-request@1.0")
371    planning.add_argument("run_ref", metavar="run", help="a run directory, or a plan hash (or its unique prefix)")
372    planning.add_argument("--json", action="store_true", help="print the request as JSON (the only output form)")
373    planning.add_argument("--profile", default=None, help="the execution profile the run was placed under")
374    _placement_arguments(planning)
375    planning.set_defaults(run=run_plan_request)
376    explain = groups.add_parser("explain", help="the events, fault and placement of a node or key of a run")
377    explain.add_argument("run_ref", metavar="run", help="a run directory, or a plan hash (or its unique prefix)")
378    explain.add_argument("target", help="a node id, or a key digest or its prefix")
379    explain.set_defaults(run=run_explain)
380    coordinating = groups.add_parser("coordinate", help="coordinate a run on its queues, as the next epoch")
381    coordinating.add_argument("run_ref", metavar="run", help="a run directory, or a plan hash (or its unique prefix)")
382    coordinating.add_argument("--profile", default=None, help="the execution profile the run was placed under")
383    _placement_arguments(coordinating)
384    coordinating.add_argument(
385        "--external", action="store_true", help="start no workers: they are started on their own (execute --pool)"
386    )
387    coordinating.set_defaults(run=run_coordinate)
388    unfit = groups.add_parser("unfit", help="the units recorded as not fitting a resource class; --clear removes one's")
389    unfit.add_argument(
390        "unit", nargs="?", default=None, help="a unit's digest label (or digest prefix); all when omitted"
391    )
392    unfit.add_argument("--clear", action="store_true", help="remove the named unit's records")
393    unfit.add_argument("--profile", default=None, help="the execution profile whose measures roots are read")
394    unfit.set_defaults(run=run_unfit)
395    executing = groups.add_parser("execute", help="serve a run's queues as one pool worker")
396    executing.add_argument("--pool", required=True, help="the run's queues directory, <run dir>/queues")
397    executing.add_argument("--classes", required=True, help="the resource classes to serve, comma-separated")
398    executing.add_argument(
399        "--profile", default=None, help="the execution profile the run was placed under: a packaged name, or a file"
400    )
401    executing.set_defaults(run=run_execute)

404def main(argv: Sequence[str] | None = None) -> int:
405    """Run the ``data`` group and the ``status``, ``explain``, ``coordinate`` and ``execute`` commands."""
406    parser = argparse.ArgumentParser(prog="senselab")
407    groups = parser.add_subparsers(dest="group", required=True)
408    mount_data(groups)
409    mount_runs(groups)
410    args = parser.parse_args(argv)
411    return int(args.run(args))

185    """Run a parsed ``explain``: why a node or key failed, was skipped, hit the cache or was placed where it was."""
186    try:

205            print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
206            return 1

208        if node in placement.nodes:

210                print(line)
211        print(f"events ({len(shown)}):")
212        for event in shown:

214        for event in (e for e in shown if e.failure is not None):

def run_unfit(args: argparse.Namespace) -> int: 
314def run_unfit(args: argparse.Namespace) -> int:
315    """Run a parsed ``unfit``: list the units recorded as not fitting a class, or ``--clear`` one unit's records.
316
317    A unit is named by the digest label ``explain`` prints, or any prefix of its digest; ``--clear`` needs the prefix
318    to name one unit. A record binds the code that
319    wrote it, so a fixed loader needs no clearing; this is for a record of a load a busy card, not the unit, failed.
320    """
321    try:
322        profile = _profile(args.profile)
323        roots = measure_roots(profile)
324        found = [
325            (unit, records)
326            for unit in roots.units()
327            if unit.digest().startswith(args.unit or "") and (records := roots.unfit_records(unit))
328        ]
329        if args.clear and not args.unit:
330            print("senselab: error: --clear names no unit; give the unit's digest label", file=sys.stderr)
331            return 1
332        if args.clear and len(found) > 1:
333            listed = ", ".join(u.label() for u, _ in found)
334            print(
335                f"senselab: error: {args.unit!r} matches {len(found)} units ({listed}); give more of the label",
336                file=sys.stderr,
337            )
338            return 1
339        for unit, records in found:
340            if args.clear:
341                print(f"unit {unit.label()}: removed {roots.remove_unfit(unit)} record(s)")
342                continue
343            for record in records:
344                print(
345                    f"unit {unit.label()} ({unit.task} {unit.backend}): a load ran out of {record.resource} memory in "
346                    f"a grant of {record.grant_bytes} bytes"
347                )
348    except (OSError, ValueError) as err:
349        print(f"senselab: error: {' '.join(str(err).split())}", file=sys.stderr)
350        return 1
351    return 0
