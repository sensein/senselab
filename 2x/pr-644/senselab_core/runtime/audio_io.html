 41class Memo:
 42    """A thread-safe mapping that keeps at most `maxsize` entries, evicting the least recently used.
 43
 44    A miss is always safe: callers recompute what was evicted.
 45    """
 46
 47    def __init__(self, maxsize: int) -> None:
 48        """Start empty, keeping at most `maxsize` (at least 1) entries."""
 49        if maxsize < 1:
 50            raise ValueError(f"a memo keeps at least one entry, got {maxsize}")
 51        self.maxsize = maxsize
 52        self._items: OrderedDict[Hashable, object] = OrderedDict()
 53        self._lock = threading.Lock()
 54
 55    def get(self, key: Hashable) -> object | None:
 56        """Return the value held for `key`, marking it recently used, or `None`."""
 57        with self._lock:
 58            if key not in self._items:
 59                return None
 60            self._items.move_to_end(key)
 61            return self._items[key]
 62
 63    def put(self, key: Hashable, value: object) -> None:
 64        """Hold `value` for `key`, evicting the least recently used entry beyond `maxsize`."""
 65        with self._lock:
 66            self._items[key] = value
 67            self._items.move_to_end(key)
 68            while len(self._items) > self.maxsize:
 69                self._items.popitem(last=False)
 70
 71    def clear(self) -> None:
 72        """Drop every entry; callers recompute on their next miss."""
 73        with self._lock:
 74            self._items.clear()
 75
 76    def __len__(self) -> int:
 77        """Return the number of entries held."""
 78        with self._lock:
 79            return len(self._items)

103@contextmanager
104def _opened(
105    path: str | os.PathLike[str],
106    chunk_size: int = DEFAULT_CHUNK_SIZE,
107    channels: tuple[ChannelInfo, ...] | None = None,
108) -> Iterator[tuple[soundfile.SoundFile, SourceStamp, int]]:
109    """Open `path` once and yield the sound file, the stamp of that open file (content read in full) and its ctime.
110
111    The stamp records `channels`, or audio channels of the file's count when None.
112
113    Raises:
114        StaleSource: if the file's stat fields changed while its content was read.
115        UndecodableSource: if libsndfile cannot decode it, truncated or not a sound file.
116        OSError: if it cannot be read, a libsndfile system error among them.
117        ValueError: if `channels` is not as many as the file's.
118    """
119    with open(path, "rb") as raw:
120        st = os.fstat(raw.fileno())
121        content = content_identity(raw, chunk_size=chunk_size)
122        raw.seek(0)
123        after = os.fstat(raw.fileno())
124        if (after.st_size, after.st_mtime_ns, after.st_ctime_ns) != (st.st_size, st.st_mtime_ns, st.st_ctime_ns):
125            raise StaleSource(f"{path} changed while it was read")
126        try:
127            sound = soundfile.SoundFile(raw)
128        except soundfile.LibsndfileError as exc:
129            if exc.code == _SYSTEM_ERROR:
130                raise OSError(f"{path} cannot be opened: {exc}") from exc
131            raise UndecodableSource(f"{path} is not a sound file: {exc}") from exc
132        with sound as f:
133            if channels is not None and len(channels) != f.channels:
134                raise ValueError(f"{len(channels)} channels declared, {f.channels} channels in the file")
135            stamp = SourceStamp(
136                size=st.st_size,
137                mtime_ns=st.st_mtime_ns,
138                inode=st.st_ino,
139                device=st.st_dev,
140                frames=f.frames,
141                channels=audio_channels(f.channels) if channels is None else channels,
142                sampling_rate=f.samplerate,
143                header=_header(f),
144                chunk_size=chunk_size,
145                chunks=tuple(content["chunks"]),
146                content=content["root"],
147            )
148            yield f, stamp, st.st_ctime_ns

151def stamp(
152    path: str | os.PathLike[str],
153    *,
154    chunk_size: int = DEFAULT_CHUNK_SIZE,
155    channels: tuple[ChannelInfo, ...] | None = None,
156) -> SourceStamp:
157    """Open the sound file at `path`, read every byte and its header, and return its current stamp.
158
159    `chunk_size` is the grid its content is hashed on; `channels` the layout the stamp records (audio channels of
160    the file's count when None). The stamp is remembered, so `check` of it at this path is an `os.stat`.
161
162    Raises:
163        StaleSource: if the file changed while it was read.
164        UndecodableSource: if the file is not a sound file.
165        ValueError: if `channels` is not as many as the file's.
166    """
167    with _opened(path, chunk_size, channels) as (_, taken, ctime_ns):
168        pass
169    _remember(path, taken, ctime_ns)
170    return taken

173def check(recorded: SourceStamp, path: str | os.PathLike[str]) -> None:
174    """Check that the file at `path` still has the stamp `recorded`.
175
176    A stamp this process took from the file at `path` is checked by `os.stat` alone, its ctime included,
177    so a rewrite that restores size and mtime is caught; any other stamp is checked as a decode checks it
178    (`_verified`), reading only the chunks the sound header lies in.
179
180    Raises:
181        StaleSource: if the file differs from the stamp.
182    """
183    ctime_ns = VERIFIED.get((os.fspath(path), recorded.canonical_digest()))
184    if ctime_ns is not None:
185        st = os.stat(path)
186        if (*_stat_fields(st), st.st_ctime_ns) != (*_stamped_fields(recorded), ctime_ns):
187            raise StaleSource(f"{path} changed since it was opened: recorded {recorded!r}")
188        return
189    with _verified(path, recorded) as (_, ctime_ns):
190        pass
191    _remember(path, recorded, ctime_ns)

194class _ChunkChecked:
195    """A read-only binary file over `fd` that serves bytes only from chunks whose digest the stamp records.
196
197    Each chunk a read touches is read whole by `os.pread`, hashed and compared before any of its bytes is served.
198    Two chunks are kept: the one read last, and the one the last seek landed in, which a decoder searching for a
199    frame returns to; so a read of any length holds two chunks at most. A chunk that differs ends the read short and
200    is held in `stale`, since the decoder's callback cannot raise through it.
201    """
202
203    def __init__(self, fd: int, stamp: SourceStamp, path: object) -> None:
204        self._fd, self._stamp, self._path = fd, stamp, path
205        self._pos = 0
206        self._chunks: dict[int, bytes] = {}
207        self._anchor = 0
208        self.stale: StaleSource | None = None
209
210    def seek(self, offset: int, whence: int = os.SEEK_SET) -> int:
211        base = {os.SEEK_SET: 0, os.SEEK_CUR: self._pos, os.SEEK_END: self._stamp.size}[whence]
212        target = max(base + offset, 0)
213        if target != self._pos:
214            self._anchor = target // self._stamp.chunk_size
215        self._pos = target
216        return self._pos
217
218    @property
219    def held(self) -> int:
220        """How many verified chunks are kept."""
221        return len(self._chunks)
222
223    def tell(self) -> int:
224        return self._pos
225
226    def readinto(self, buffer: memoryview) -> int:
227        view = memoryview(buffer).cast("B")
228        done = 0
229        while done < len(view) and self._pos < self._stamp.size and self.stale is None:
230            index, offset = divmod(self._pos, self._stamp.chunk_size)
231            try:
232                chunk = self._verified(index)
233            except StaleSource as exc:
234                self.stale = exc
235                break
236            take = min(len(view) - done, len(chunk) - offset)
237            view[done : done + take] = chunk[offset : offset + take]
238            done += take
239            self._pos += take
240        return done
241
242    def _verified(self, index: int) -> bytes:
243        if index not in self._chunks:
244            size = self._stamp.chunk_size
245            parts: list[bytes] = []
246            while (got := sum(map(len, parts))) < size and (part := os.pread(self._fd, size - got, index * size + got)):
247                parts.append(part)
248            chunk = b"".join(parts)
249            if chunk_digest(chunk) != self._stamp.chunks[index]:
250                raise StaleSource(f"{self._path} changed since it was opened: chunk {index} differs")
251            self._chunks = {i: c for i, c in self._chunks.items() if i == self._anchor}
252            self._chunks[index] = chunk
253        return self._chunks[index]

256@contextmanager
257def _verified(path: str | os.PathLike[str], stamp: SourceStamp) -> Iterator[tuple[soundfile.SoundFile, int]]:
258    """Open `path` as sound through `_ChunkChecked`, checked against `stamp`, and yield it with the file's ctime.
259
260    The stat fields and sound header are checked against `stamp` and the stamp's content against the root of its
261    chunk digests; every chunk read, by the header parse or in the body, is verified before it is decoded.
262
263    Raises:
264        StaleSource: if the file differs from `stamp` in its stat fields, its header or a chunk read, or the stamp's
265            content is not the root of its chunks.
266    """
267    fd = os.open(path, os.O_RDONLY)
268    try:
269        st = os.fstat(fd)
270        rooted = content_root(stamp.chunk_size, stamp.chunks) == stamp.content
271        if _stat_fields(st) != _stamped_fields(stamp) or not rooted:
272            raise StaleSource(f"{path} changed since it was opened: recorded {stamp!r}")
273        checked = _ChunkChecked(fd, stamp, path)
274        try:
275            with soundfile.SoundFile(checked) as f:
276                if _header(f) != stamp.header:
277                    raise StaleSource(f"{path} changed since it was opened: its sound header differs")
278                yield f, st.st_ctime_ns
279        except soundfile.LibsndfileError:
280            if checked.stale is None:
281                raise
282        if checked.stale is not None:
283            raise checked.stale
284    finally:
285        os.close(fd)

329def open_audio(path: str | os.PathLike[str]) -> Audio:
330    """Open a sound file lazily: its bytes are hashed once and its header read; no sample is decoded.
331
332    Its identity is its content identity, and the recording's clock is named by its identity. Its
333    samples are read by `group.read(..., reader=READER)`.
334
335    Raises:
336        StaleSource: if the file changed while it was read.
337        UndecodableSource: if the file is not a sound file.
338    """
339    taken = stamp(path)
340    return Audio.from_group(source_group(taken.channels, taken.sampling_rate, path, taken))

Open a sound file lazily: its bytes are hashed once and its header read; no sample is decoded.

Its identity is its content identity, and the recording's clock is named by its identity. Its samples are read by group.read(..., reader=READER).

Raises:
  • StaleSource: if the file changed while it was read.
  • UndecodableSource: if the file is not a sound file.
