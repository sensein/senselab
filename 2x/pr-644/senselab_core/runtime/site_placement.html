senselab_core.runtime.site_placement

Site placement: what a run needs, explicit placement values, and the site's hook that answers.

senselab holds no site scheduling policy and runs no scheduler query. A run's needs are a PlacementRequest (placement-request@1.0); a site's PlacementHook answers with a PlacementAnswer (placement-answer@1.0); check_answer refuses an answer that could not finish the work. The values a class is placed with come from three layers, in order: the profile, the explicit overrides, the hook's answer (resolve_placement, Placer).

151def check_values(name: str, values: PlacementValues) -> None:
152    """Raise `PlacementAnswerRefused` for a time that is not positive or fewer than one worker in `values`."""
153    if values.time_s is not None and values.time_s <= 0:
154        raise PlacementAnswerRefused(name, "time_s", values.time_s, 0, "a time must be positive")
155    if values.max_workers is not None and values.max_workers < 1:
156        raise PlacementAnswerRefused(name, "max_workers", values.max_workers, 1, "at least one worker")

159def check_answer(request: PlacementRequest, answer: PlacementAnswer) -> str | None:
160    """Refuse `answer` unless each of its values lets the work in `request` finish; return a note on what was unchecked.
161
162    A time must hold the longest worker (`time_per_worker_s`), else the load plus one item (`time_floor_s`), else
163    it is accepted unchecked and the note says so. A worker count is at least 1 and at most `workers_max`.
164
165    Raises:
166        PlacementAnswerRefused: a field names a class the request does not hold, or a value is refused.
167    """
168    notes = []
169    for name, values in answer.classes.items():
170        needs = request.held(name)
171        if needs is None:
172            raise PlacementAnswerRefused(
173                name,
174                "class",
175                name,
176                sorted(c.resource_class for c in request.classes),
177                "the request holds no such class",
178            )
179        check_values(name, values)
180        if values.time_s is not None:
181            need, label = (needs.time_per_worker_s, "time_per_worker_s")
182            if need is None:
183                need, label = needs.time_floor_s, "time_floor_s"
184            if need is None:
185                notes.append(f"{name}: time_s accepted unchecked, the worker time is unmeasured")
186            elif values.time_s < need:
187                raise PlacementAnswerRefused(
188                    name, "time_s", values.time_s, need, f"below {label}: a worker could not finish"
189                )
190        if values.max_workers is not None and values.max_workers > needs.workers_max:
191            raise PlacementAnswerRefused(
192                name, "max_workers", values.max_workers, needs.workers_max, "above workers_max"
193            )
194    return "; ".join(notes) or None

229def hook_ref(hook: PlacementHook | str) -> HookRef:
230    """Return the path of `hook` and the version of the distribution that provides its module.
231
232    A lambda or a function defined inside another has a path no import can resolve; the reference says so.
233    """
234    path = hook if isinstance(hook, str) else f"{hook.__module__}:{getattr(hook, '__qualname__', type(hook).__name__)}"
235    module = path.partition(":")[0].partition(".")[0]
236    try:
237        owner = next(iter(metadata.packages_distributions().get(module, ())), None)
238        version = None if owner is None else metadata.version(owner)
239    except metadata.PackageNotFoundError:
240        version = None
241    return HookRef(path=path, version=version, importable=_HOOK_PATH.match(path) is not None and "<" not in path)

244def load_hook(path: str) -> PlacementHook:
245    """Return the callable `module:callable` names.
246
247    Raises:
248        PlacementHookFailed: the path is malformed, its module does not import, or it names no callable.
249    """
250    if not _HOOK_PATH.match(path):
251        raise PlacementHookFailed(f"placement hook {path!r} is not a `module:callable` path")
252    module, _, name = path.partition(":")
253    try:
254        found: Any = importlib.import_module(module)
255        for part in name.split("."):
256            found = getattr(found, part)
257    except (ImportError, AttributeError) as err:
258        raise PlacementHookFailed(f"placement hook {path!r} cannot be imported: {err}") from err
259    if not callable(found):
260        raise PlacementHookFailed(f"placement hook {path!r} is not callable")
261    return found  # type: ignore[no-any-return]

264def ask(hook: PlacementHook, request: PlacementRequest, *, bound_s: float, clock: Clock) -> PlacementAnswer:
265    """Call `hook(request)` on a thread, waiting up to `bound_s` seconds on `clock`.
266
267    A hook that outlasts the bound is left running on its daemon thread; its answer is dropped.
268
269    Raises:
270        PlacementHookFailed: the hook raised (chained), returned another type, or had not answered in `bound_s`.
271    """
272    done = threading.Event()
273    box: list[PlacementAnswer | BaseException] = []
274
275    def run() -> None:
276        try:
277            box.append(hook(request))
278        except BaseException as err:  # noqa: BLE001 - handed to the caller as the failure
279            box.append(err)
280        finally:
281            done.set()
282
283    threading.Thread(target=run, name="senselab-placement-hook", daemon=True).start()
284    if not clock.wait(done, bound_s):
285        raise PlacementHookFailed(f"the placement hook did not answer within {bound_s:g} s")
286    [result] = box
287    if isinstance(result, BaseException):
288        raise PlacementHookFailed(f"the placement hook raised {type(result).__name__}: {result}") from result
289    if not isinstance(result, PlacementAnswer):
290        raise PlacementHookFailed(f"the placement hook returned {type(result).__name__}, not a PlacementAnswer")
291    return result

294def site_names(partitions: Iterable[str | None] = (), nodes: Iterable[str] = ()) -> dict[str, Label]:
295    """Return each of `partitions` and `nodes`, as written and expanded, and the accounts of the environment, labelled.
296
297    An account is the one `SBATCH_ACCOUNT` or `SLURM_ACCOUNT` holds.
298    """
299    names: dict[str, Label] = {
300        os.environ[v]: "<account>" for v in ("SBATCH_ACCOUNT", "SLURM_ACCOUNT") if os.environ.get(v)
301    }
302    labelled: list[tuple[str, Label]] = [(p, "<partition>") for p in partitions if p]
303    for given, label in labelled + [(n, "<node>") for n in nodes]:
304        names.update({name: label for name in (given, expand_variables(given)) if name})
305    return names

313def _layered(
314    name: str, base: ResourceClass, overrides: Mapping[str, PlacementValues], profile_exclude: tuple[str, ...] = ()
315) -> ClassPlacement:
316    given = overrides.get(name, PlacementValues())
317    check_values(name, given)
318    values = {f: getattr(base, f) for f in ("partition", "time_s", "max_workers")}
319    layers: dict[str, Layer] = dict.fromkeys(FIELDS, "profile")
320    for field in FIELDS:
321        if getattr(given, field) is not None:
322            layers[field] = "override"
323            if field != "exclude":
324                values[field] = getattr(given, field)
325    exclude = given.exclude if given.exclude is not None else profile_exclude
326    names = site_names((base.partition, given.partition), (*profile_exclude, *(given.exclude or ())))
327    return ClassPlacement(values=CurrentValues(**values, layers=layers), exclude=tuple(exclude), names=names)

344class Placer:
345    """Resolves each class's placement once per epoch, calling the site's hook the first time a class is asked for.
346
347    The hook is `hook`, else the profile's `substrate.placement_hook`; it is called with the request narrowed to the
348    class, bounded by `substrate.placement_hook_s` seconds on `clock`. An answer is checked (`check_answer`); a
349    failed hook falls back to the explicit and profile values and says so. Every resolution, a refused one too, is
350    handed to `record` as `(epoch, class, the class's needs, the hook's reference, its placement)`. The profile's
351    layer of `exclude` is `substrate.exclude_nodes`; senselab adds none.
352
353    Args:
354        profile: The execution profile: its classes, excluded nodes, hook and the hook's bound.
355        overrides: The explicit values per class.
356        hook: The site's hook, a callable or its `module:callable` path; None takes the profile's.
357        request: Builds the run's request; called once per epoch, when the first class of it needs the hook.
358        clock: Times the hook's bound.
359        record: Told of each resolution; set before the first `for_class`.
360        inert: Said in each recorded note when the substrate applies only the worker cap.
361    """
362
363    def __init__(  # noqa: PLR0913 - the profile and layers, then keyword-only call settings
364        self,
365        profile: ExecutionProfile,
366        overrides: Mapping[str, PlacementValues] | None,
367        hook: PlacementHook | str | None,
368        *,
369        request: Callable[[], PlacementRequest],
370        clock: Clock = SystemClock(),  # noqa: B008 - stateless
371        record: Callable[[int, str, ClassNeeds | None, HookRef | None, ClassPlacement], None] | None = None,
372        inert: str | None = None,
373    ) -> None:
374        """Resolve placements under `profile`; nothing is called until `for_class`.
375
376        Raises:
377            PlacementAnswerRefused: an override names no class of the profile or holds a refused value.
378            PlacementHookFailed: the hook's path does not import.
379        """
380        self.classes, self.overrides = dict(profile.resource_classes), dict(overrides or {})
381        self._exclude = tuple(profile.substrate.exclude_nodes)
382        current_values(self.classes, self.overrides)
383        chosen = hook if hook is not None else profile.substrate.placement_hook
384        self._hook = None if chosen is None else load_hook(chosen) if isinstance(chosen, str) else chosen
385        self._ref = None if chosen is None else hook_ref(chosen)
386        self._request = request
387        self._built: dict[int, PlacementRequest] = {}
388        self.hook_s, self._clock, self.record, self._inert = profile.substrate.placement_hook_s, clock, record, inert
389        self._done: dict[tuple[int, str], ClassPlacement] = {}
390        self._lock = threading.Lock()
391
392    def _needs(self, name: str, epoch: int) -> PlacementRequest:
393        if epoch not in self._built:
394            self._built[epoch] = self._request()
395        return self._built[epoch].only(name)
396
397    def for_class(self, name: str, epoch: int) -> ClassPlacement:
398        """Return class `name`'s placement for `epoch`, asking the hook the first time this epoch.
399
400        Raises:
401            PlacementAnswerRefused: the hook's answer is refused; nothing is remembered, so the next ask calls again.
402            ValueError: `name` is no class of the profile.
403        """
404        with self._lock:
405            if (epoch, name) in self._done:
406                return self._done[epoch, name]
407            if name not in self.classes:
408                raise ValueError(f"no resource class {name!r}; the profile has {sorted(self.classes)}")
409            placed, needs, refusal = _layered(name, self.classes[name], self.overrides, self._exclude), None, None
410            if self._hook is not None:
411                request = self._needs(name, epoch)
412                needs = request.held(name)
413                if needs is not None:
414                    placed = placed.model_copy(
415                        update={"names": {**placed.names, **site_names([needs.current.partition])}}
416                    )
417                placed, refusal = self._asked(self._hook, name, request, placed)
418            if self._inert is not None:
419                placed = placed.model_copy(update={"note": "; ".join(n for n in (placed.note, self._inert) if n)})
420            if self.record is not None:
421                self.record(epoch, name, needs, self._ref, placed)
422            if refusal is not None:
423                raise refusal
424            self._done[epoch, name] = placed
425            return placed
426
427    def _asked(
428        self, hook: PlacementHook, name: str, request: PlacementRequest, placed: ClassPlacement
429    ) -> tuple[ClassPlacement, PlacementAnswerRefused | None]:
430        try:
431            answer = ask(hook, request, bound_s=self.hook_s, clock=self._clock)
432        except PlacementHookFailed as err:
433            logger.warning("%s; class %s keeps its explicit and profile values", err, name)
434            return placed.model_copy(update={"verdict": "failed", "note": str(err)}), None
435        names = {**placed.names, **_answered(answer)}
436        try:
437            note = check_answer(request, answer)
438        except PlacementAnswerRefused as err:
439            update = {"verdict": "refused", "note": str(err), "answer": answer, "names": names}
440            return placed.model_copy(update=update), err
441        given = answer.classes.get(name, PlacementValues())
442        update = {f: getattr(given, f) for f in ("partition", "time_s", "max_workers") if getattr(given, f) is not None}
443        layers = {**placed.values.layers, **{f: "hook" for f in FIELDS if getattr(given, f) is not None}}
444        values = placed.values.model_copy(update={**update, "layers": layers})
445        exclude = given.exclude if given.exclude is not None else placed.exclude
446        accepted = ClassPlacement(
447            values=values, exclude=tuple(exclude), verdict="accepted", note=note, answer=answer, names=names
448        )
449        return accepted, None

458def resolve_placement(
459    profile: ExecutionProfile,
460    overrides: Mapping[str, PlacementValues] | None,
461    hook: PlacementHook | str | None,
462    *,
463    request: PlacementRequest,
464    clock: Clock = SystemClock(),  # noqa: B008 - stateless
465) -> ResolvedPlacement:
466    """Return each class's placement: the profile's values, the explicit `overrides`, then `hook`'s answer.
467
468    The classes are those `request` holds. `hook` (a callable or its path) defaults to
469    the profile's `substrate.placement_hook`; it is called per class with `request` narrowed to the class.
470
471    Raises:
472        PlacementAnswerRefused: an override or the hook's answer is refused.
473    """
474    placer = Placer(profile, overrides, hook, request=lambda: request, clock=clock)
475    return ResolvedPlacement(classes={c.resource_class: placer.for_class(c.resource_class, 0) for c in request.classes})

478def rendered(text: str, names: Mapping[str, str]) -> str:
479    """Return `text` as a record other users read: the account `--account` names and `names` as labels, then paths.
480
481    The one helper of every text that may name a site: the refusal of a submission and the placement record. `names`
482    maps each site name to its label (`site_names`, `ClassPlacement.names`). The operator's own text goes to the log
483    (`stored_message` renders paths, user and host).
484    """
485    from senselab_core.runtime.execution.executor import stored_message  # noqa: PLC0415 - executor imports the planner
486
487    shown = re.sub(r"(--account[= ])\S+", r"\1<account>", text)
488    for name, label in sorted(names.items(), key=lambda n: len(n[0]), reverse=True):
489        shown = re.sub(rf"(?<![\w-]){re.escape(name)}(?![\w-])", label, shown)
490    if shown != text:
491        logger.info("a placement text was rendered for the record; the operator's text was: %s", text)
492    return stored_message(shown)

265def ask(hook: PlacementHook, request: PlacementRequest, *, bound_s: float, clock: Clock) -> PlacementAnswer:
266    """Call `hook(request)` on a thread, waiting up to `bound_s` seconds on `clock`.
267
268    A hook that outlasts the bound is left running on its daemon thread; its answer is dropped.
269

272    """

279        except BaseException as err:  # noqa: BLE001 - handed to the caller as the failure

399        """Return class `name`'s placement for `epoch`, asking the hook the first time this epoch.
400
401        Raises:
402            PlacementAnswerRefused: the hook's answer is refused; nothing is remembered, so the next ask calls again.
403            ValueError: `name` is no class of the profile.
404        """
405        with self._lock:
406            if (epoch, name) in self._done:
407                return self._done[epoch, name]
408            if name not in self.classes:
409                raise ValueError(f"no resource class {name!r}; the profile has {sorted(self.classes)}")
410            placed, needs, refusal = _layered(name, self.classes[name], self.overrides, self._exclude), None, None
411            if self._hook is not None:
412                request = self._needs(name, epoch)
413                needs = request.held(name)
414                if needs is not None:
415                    placed = placed.model_copy(
416                        update={"names": {**placed.names, **site_names([needs.current.partition])}}
417                    )

419            if self._inert is not None:

421            if self.record is not None:
422                self.record(epoch, name, needs, self._ref, placed)
423            if refusal is not None:

425            self._done[epoch, name] = placed

428    def _asked(
429        self, hook: PlacementHook, name: str, request: PlacementRequest, placed: ClassPlacement
430    ) -> tuple[ClassPlacement, PlacementAnswerRefused | None]:
431        try:
432            answer = ask(hook, request, bound_s=self.hook_s, clock=self._clock)
433        except PlacementHookFailed as err:
434            logger.warning("%s; class %s keeps its explicit and profile values", err, name)
435            return placed.model_copy(update={"verdict": "failed", "note": str(err)}), None
436        names = {**placed.names, **_answered(answer)}
437        try:
438            note = check_answer(request, answer)
439        except PlacementAnswerRefused as err:
440            update = {"verdict": "refused", "note": str(err), "answer": answer, "names": names}
441            return placed.model_copy(update=update), err
442        given = answer.classes.get(name, PlacementValues())
443        update = {f: getattr(given, f) for f in ("partition", "time_s", "max_workers") if getattr(given, f) is not None}
444        layers = {**placed.values.layers, **{f: "hook" for f in FIELDS if getattr(given, f) is not None}}
445        values = placed.values.model_copy(update={**update, "layers": layers})
446        exclude = given.exclude if given.exclude is not None else placed.exclude
447        accepted = ClassPlacement(
448            values=values, exclude=tuple(exclude), verdict="accepted", note=note, answer=answer, names=names
449        )
450        return accepted, None

Resolves each class's placement once per epoch, calling the site's hook the first time a class is asked for.

The hook is hook, else the profile's substrate.placement_hook; it is called with the request narrowed to the class, bounded by substrate.placement_hook_s seconds on clock. An answer is checked (check_answer); a failed hook falls back to the explicit and profile values and says so. Every resolution, a refused one too, is handed to record as (epoch, class, the class's needs, the hook's reference, its placement). The profile's layer of exclude is substrate.exclude_nodes; senselab adds none.

Arguments:
  • profile: The execution profile: its classes, excluded nodes, hook and the hook's bound.
  • overrides: The explicit values per class.
  • hook: The site's hook, a callable or its module:callable path; None takes the profile's.
  • request: Builds the run's request; called once per epoch, when the first class of it needs the hook.
  • clock: Times the hook's bound.
  • record: Told of each resolution; set before the first for_class.
  • inert: Said in each recorded note when the substrate applies only the worker cap.
