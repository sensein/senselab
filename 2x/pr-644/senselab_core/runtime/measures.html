senselab_core.runtime.measures

What a measure is of, where its observations are kept, and where a unit's estimate is read from.

A MeasureUnit names the task, backend, parameters, model, env, service and hardware a measure holds for; its digest is the key every measures root is looked up by. MeasureRoots is the layered store of measures@1 roots: write-once observation and record files under a trust filter by owner uid, a hardware index, and a probe lease. MeasureSource reads a unit's Estimate from the roots (pooled); a unit no root holds is unmeasured.

 228class MeasureSource:
 229    """Reads a unit's estimate: the roots' pooled observations, estimated once."""
 230
 231    def __init__(
 232        self,
 233        sources: Sequence[MeasuresRoot],
 234        *,
 235        estimate: Callable[[MeasureUnit, Sequence[Observation]], Estimate],
 236    ) -> None:
 237        """Read `sources` in order; `estimate` derives an estimate from the pooled observations."""
 238        self._sources = tuple(sources)
 239        self._estimate = estimate
 240
 241    def get(self, unit: MeasureUnit) -> Estimate | None:
 242        """Return the estimate of `unit` over the observations of every root, None when it is not measured."""
 243        pooled = [o for root in self._sources for o in root.observations(unit)]
 244        if pooled and (found := self._estimate(unit, pooled)).status == "measured":
 245            return found
 246        return None

 284def _git_dir(start: Path) -> Path | None:
 285    """Return the git directory of the checkout whose work tree holds senselab-core and `start`; None otherwise.
 286
 287    A `.git` found above `start` is another project's (a virtual environment inside its repository) unless
 288    `packages/senselab-core/src` lies under the same work tree and `start` lies under it.
 289    """
 290    for parent in (start, *start.parents):
 291        dot = parent / ".git"
 292        found = None
 293        if dot.is_dir():
 294            found = dot
 295        elif dot.is_file() and (named := dot.read_text().strip()).startswith("gitdir:"):
 296            found = (parent / named.removeprefix("gitdir:").strip()).resolve()
 297        if found is not None:
 298            return found if start.is_relative_to(parent / PACKAGE_IN_CHECKOUT) else None
 299    return None

 302def _ref_commit(git: Path, ref: str) -> str | None:
 303    """Return the commit of `ref`, loose or packed, in `git` or the common directory a worktree's names."""
 304    common = git / "commondir"
 305    base = (git / common.read_text().strip()).resolve() if common.is_file() else git
 306    loose = base / ref
 307    if loose.is_file():
 308        return loose.read_text().strip()
 309    packed = base / "packed-refs"
 310    lines = packed.read_text().splitlines() if packed.is_file() else []
 311    return next((line.split()[0] for line in lines if line.endswith(f" {ref}")), None)

 314def checkout_commit(root: Path | None = None) -> str | None:
 315    """Return the 40-hex commit `HEAD` names in the git checkout this package lives in (`root`: where it lives).
 316
 317    Read from the git directory, not by running git. None unless the package lies inside the work tree of that
 318    checkout, so a package installed into a virtual environment inside another repository gets none of that
 319    repository's; and None where it cannot be read.
 320    """
 321    try:
 322        git = _git_dir(Path(__file__).resolve().parent if root is None else root.resolve())
 323        if git is None:
 324            return None
 325        head = (git / "HEAD").read_text().strip()
 326        found = _ref_commit(git, head.removeprefix("ref:").strip()) if head.startswith("ref:") else head
 327    except OSError:
 328        return None
 329    return found if found is not None and COMMIT_HEX.fullmatch(found) else None

 391def resolve_roots(raw: Sequence[str], cache: Path) -> tuple[Path, ...]:
 392    """Return the roots `raw` names, in order: `cache` for the cache-root token, then `expand`.
 393
 394    A root naming a variable that is unset or empty, a root that is not absolute, a blank one and a repeat are left
 395    out, logged and never raised.
 396    """
 397    roots: list[Path] = []
 398    for entry in raw:
 399        text = entry.replace(CACHE_ROOT_TOKEN, str(cache))
 400        if empty := [n for m in _VARIABLE.finditer(text) for n in m.groups() if n and os.environ.get(n) == ""]:
 401            logger.info("measures root %r is not a root: %s is empty", entry, empty[0])
 402            continue
 403        try:
 404            path = Path(expand(text))
 405        except ValueError as err:
 406            logger.info("measures root %r is not a root: %s", entry, err)
 407            continue
 408        if not entry.strip() or path in roots:
 409            continue
 410        if not path.is_absolute():
 411            logger.info("measures root %r is not a root: %s is not absolute", entry, path)
 412            continue
 413        roots.append(path)
 414    return tuple(roots)

 509class MeasureRoots:
 510    """Layered `measures@1` roots: read pooled under a trust filter, written to the first writable root.
 511
 512    Every file is write-once: staged in `tmp/<uuid>/` and linked onto a name no other writer uses. A file's writer is
 513    its owner uid; `trust` `own` reads only files of `uid`, `group` reads every file. An `OSError` on a root is
 514    logged once per run and never raised: reads go on over the other roots and a write moves to the next writable root,
 515    else the measure is not persisted. A file of an unknown format is skipped. A malformed or misfiled file is
 516    quarantined by its owner in a writable root and skipped by anyone else. `collect` removes only files `uid` owns.
 517    """
 518
 519    def __init__(  # noqa: PLR0913 - the roots, the reader, then keyword-only staleness and clock
 520        self,
 521        roots: Sequence[Path],
 522        policy: StorePolicy,
 523        trust: Trust,
 524        uid: int,
 525        *,
 526        stale_after_s: float,
 527        lease: JobLease | None = None,
 528        clock: Clock = SystemClock(),  # noqa: B008 - stateless
 529    ) -> None:
 530        """Hold `roots` in order, the reader's `trust` and `uid`; probe leases go stale after `stale_after_s`.
 531
 532        `lease` is the profile's lease section the probe leases are taken under (the packaged profile's when None).
 533        """

 613        if root not in self._probed:

 644    def _load(  # noqa: PLR0913 - where the file is, what it must be, how it is parsed, and whether trust filters it
 645        self,
 646        root: Path,
 647        directory: StorageDir,
 648        name: str,
 649        expected: str,
 650        parse: Callable[[Mapping[str, Any], os.stat_result], Any],
 651        *,
 652        filtered: bool = True,
 653    ) -> _File | None:
 654        info = directory.lstat(name)
 655        if info is None or (filtered and not self._admits(info.st_uid)):
 656            return None
 657        try:
 658            raw = directory.read_regular(name, limit=self.policy.measures_file_bytes_max)
 659        except (WrongKindError, ValueError) as err:
 660            self._malformed(root, directory, name, info.st_uid, type(err).__name__)
 661            return None
 662        if raw is None:
 663            return None
 664        path = directory.path / name
 665        try:
 666            require_json_depth(raw)
 667            document = json.loads(raw)
 668        except ValueError:
 669            document = None
 670        found = document.get("format") if isinstance(document, dict) else None
 671        verdict = check(expected, found)
 672        if verdict in (FormatVerdict.UNKNOWN, FormatVerdict.RETIRED):
 673            self._say(
 674                (path,),
 675                "measures file %s (owner %s) has a format not read here; skipped",
 676                path,
 677                self._owner(info.st_uid),
 678            )
 679            return None
 680        try:
 681            if verdict is FormatVerdict.MALFORMED:
 682                raise ValueError("no format tag")
 683            with reading(expected, found):
 684                return _File(parse(document, info), info.st_uid, info.st_mtime)
 685        except (ValueError, KeyError, TypeError, ValidationError) as err:
 686            self._malformed(root, directory, name, info.st_uid, type(err).__name__)
 687            return None

 718    def _unit_status(
 719        self, root: Path, parent: StorageDir, unit_dir: StorageDir, digest: str
 720    ) -> Literal["ok", "missing", "bad"]:
 721        if (root, digest) in self._verified:
 722            return "ok"
 723        found = self._load(root, unit_dir, UNIT_FILE, MEASURE_UNIT_FORMAT, _parse_unit, filtered=False)

 727            self._misfiled(root, parent, digest)
 728            return "bad"

 730        return "ok"

 952    def _lease_age(self, unit_dir: StorageDir) -> float | None:
 953        try:
 954            return Lease(
 955                unit_dir.path / PROBE_LEASE,
 956                settings=self._lease,
 957                stale_after=self.stale_after_s,
 958                directory=unit_dir,
 959                clock=self._clock,
 960            ).age()
 961        except PermissionError:
 962            pass
 963        except OSError as err:
 964            if err.errno != errno.EROFS:
 965                raise
 966        self._say(
 967            ("local-clock", unit_dir.path.parents[3]),
 968            "lease liveness in %s is judged on the local clock",
 969            unit_dir.path.parents[3],
 970        )
 971        newest = 0.0
 972        with unit_dir.child(PROBE_LEASE + ".lease") as lease:
 973            for token in lease.listdir():
 974                with lease.child(token) as held:
 975                    beats = (i.st_mtime for n in held.listdir() if (i := held.lstat(n)) is not None)
 976                    newest = max([newest, os.fstat(held.fd).st_mtime, *beats])
 977        return self._clock.local_now() - newest if newest else None

 998    def probe_lease(self, unit: MeasureUnit) -> Lease | PeerProbing | Unleased:
 999        """Return the probe lease of `unit` in the write root, not taken, or why none is to be held.
1000
1001        `PeerProbing`: an admitted peer holds one in some root; wait for its observations up to `peer_wait_s`.
1002        `Unleased`: probe without a lease and do not wait. A `Lease` is taken with `with lease` (`LeaseTimeout`: a peer
1003        won the race; `LeaseUnavailable`: a stale lease this uid cannot break), renewed with `keep_alive` and
1004        `prove`d before the probe's observations are written.
1005        """
1006        if self.peer_holds(unit):
1007            return PeerProbing()
1008        digest = self._digest(unit)
1009
1010        def take(root: Path) -> Lease | PeerProbing | Unleased:
1011            with self._unit_dir(root, unit, digest) as unit_dir:
1012                if (owner := self._lease_owner(unit_dir)) is not None:
1013                    return PeerProbing() if self._admits(owner) else Unleased("foreign lease")
1014                path = unit_dir.path / PROBE_LEASE
1015            return Lease(
1016                path,
1017                settings=self._lease,
1018                timeout=0,
1019                stale_after=self.stale_after_s,
1020                mode=self._dir_mode,
1021                clock=self._clock,
1022            )
1023
1024        taken = self._in_first_writable(take)
1025        return Unleased("no writable root") if taken is None else taken

1027    def collect(self, kept: int) -> int:
1028        """Remove what `uid` owns and no reader needs; return how many.
1029
1030        That is staging and quarantine older than the policy's ages on the clock's storage time, and the
1031        observation files of probes a trusted `supersede` record names in `outdates`, beyond the newest `kept`.
1032        Another user's file is never removed.
1033        """
1034        removed = 0
1035        ages = (("tmp", self.policy.staging_stale_s), ("quarantine", self.policy.quarantine_retention_s))
1036        for root in list(self._writable()):
1037            with self._guard(root, "written"), self._open(root, create=False) as m:
1038                now = self._clock.storage_now(m)
1039                for sub, age in ages:
1040                    with self._guard(root, "written"), m.child(sub) as directory:
