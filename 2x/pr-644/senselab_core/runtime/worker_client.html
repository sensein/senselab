  72@dataclass(frozen=True)
  73class ShutdownGraces:
  74    """How long each step of stopping a worker may take, in seconds.
  75
  76    Every grace is a finite real number: `term` lies in `[0, stop]`, the others are positive, and `bound`
  77    is at most `threading.TIMEOUT_MAX`.
  78
  79    Raises:
  80        ValueError: on construction with a grace outside those ranges, or one that is not a real number.
  81    """
  82
  83    stop: float = 10.0
  84    """The graceful stop window, from the stop line to SIGKILL."""
  85    term: float = 3.0
  86    """The last part of `stop` left to SIGTERM."""
  87    kill: float = 5.0
  88    """After SIGKILL, for the group to be gone."""
  89    join: float = 2.0
  90    """For the pipe readers to stop."""
  91    exit_code_wait: float = 1.0
  92    """After the worker's output closes, for its exit code."""
  93
  94    def __post_init__(self) -> None:
  95        """Refuse graces outside their ranges."""
  96        for f in fields(self):
  97            value = getattr(self, f.name)
  98            try:
  99                finite = type(value) in (int, float) and math.isfinite(value)
 100            except OverflowError:
 101                finite = False
 102            if not finite:
 103                raise ValueError(f"grace {f.name} must be a finite number of seconds, got {value!r}")
 104            least = "at least 0" if f.name == "term" else "positive"
 105            if value < 0 or (value == 0 and f.name != "term") or value > threading.TIMEOUT_MAX:
 106                raise ValueError(f"grace {f.name} must be {least} and at most {threading.TIMEOUT_MAX} s, got {value}")
 107        if self.term > self.stop:
 108            raise ValueError(f"grace term ({self.term}) is the last part of stop ({self.stop}) and cannot exceed it")
 109        if self.bound > threading.TIMEOUT_MAX:
 110            raise ValueError(f"graces bound {self.bound} s exceeds threading.TIMEOUT_MAX ({threading.TIMEOUT_MAX})")
 111
 112    @property
 113    def bound(self) -> float:
 114        """The longest `close` and every host fault take to stop the group and release the client."""
 115        return self.stop + self.kill + self.join

 182def _wait_exit(pid: int, *, block: bool = True) -> bool:
 183    """Wait until child `pid` exits, without reaping it, or with `block=False` only ask; return whether it has exited.
 184
 185    Uses `waitid(WNOWAIT)` on Linux and a kqueue `NOTE_EXIT` on macOS, which also returns for a
 186    leader that exited before the event was registered. The exit code is read when the client reaps.
 187
 188    Raises:
 189        ChildProcessError: if `pid` is not an unreaped child (waitid only).
 190        NotImplementedError: on a platform that is neither Linux nor macOS.
 191    """
 192    if sys.platform == "linux" and hasattr(os, "waitid"):
 193        return os.waitid(os.P_PID, pid, os.WEXITED | os.WNOWAIT | (0 if block else os.WNOHANG)) is not None
 194    if sys.platform == "darwin":
 195        kq = select.kqueue()
 196        try:
 197            flags = select.KQ_EV_ADD | select.KQ_EV_ONESHOT
 198            event = select.kevent(pid, filter=select.KQ_FILTER_PROC, flags=flags, fflags=select.KQ_NOTE_EXIT)
 199            return bool(kq.control([event], 1, None if block else 0))
 200        finally:
 201            kq.close()
 202    raise NotImplementedError(f"no exit notification for platform {sys.platform!r}")

 205def _exit_seen(pid: int, exited: threading.Event) -> bool:
 206    """Whether the leader's exit is observed: by the reaper thread, else by asking `_wait_exit` now.
 207
 208    An exit seen by asking is recorded in `exited`. A child no longer unreaped counts as exited; a
 209    platform or call that cannot tell counts as running.
 210    """
 211    if not exited.is_set():
 212        try:
 213            seen = _wait_exit(pid, block=False)
 214        except ChildProcessError:
 215            seen = True
 216        except (OSError, NotImplementedError):
 217            seen = False
 218        if seen:
 219            exited.set()
 220    return exited.is_set()

 223def _wait_for_exit(pid: int, exited: threading.Event, timeout: float, clock: Clock, poll_max_s: float) -> bool:
 224    """Wait up to `timeout` seconds on `clock` for `_exit_seen`; return whether the exit was observed.
 225
 226    The polls double up to `poll_max_s` seconds apart.
 227    """
 228    deadline = Deadline(timeout, clock.now)
 229    delay = 0.001
 230    while not _exit_seen(pid, exited):
 231        remaining = deadline.remaining() or 0.0
 232        if remaining <= 0:
 233            return False
 234        clock.wait(exited, min(delay, remaining))
 235        delay = min(delay * 2, poll_max_s)
 236    return True

 251def _leader_held(pid: int, exited: threading.Event, timeout: float) -> bool | None:
 252    """Whether child `pid` is unreaped, so its pid and group id are still reserved; `None` if that cannot be told.
 253
 254    A leader whose exit has not been observed is running, hence held. After its exit the answer comes
 255    from `waitid(WNOHANG|WNOWAIT)` where the platform has it, else from `ps -o ppid=` run within
 256    `timeout` seconds.
 257    """
 258    if not _exit_seen(pid, exited):
 259        return True
 260    if sys.platform == "linux" and hasattr(os, "waitid"):
 261        try:
 262            _wait_exit(pid, block=False)
 263        except ChildProcessError:
 264            return False
 265        return True
 266    try:
 267        out = subprocess.run(
 268            [_PS, "-o", "ppid=", "-p", str(pid)], capture_output=True, text=True, timeout=timeout, check=False
 269        )
 270    except (OSError, subprocess.SubprocessError) as exc:
 271        logger.warning("cannot tell whether worker %d is still this process's child: %s", pid, exc)
 272        return None
 273    return out.stdout.strip() == str(os.getpid())

 287def _members_gone(  # noqa: PLR0913 - the group, its exit, the bound, the liveness test, the clock and the poll
 288    pgid: int, exited: threading.Event, deadline: Deadline, live: Callable[[int], bool], clock: Clock, poll_max_s: float
 289) -> bool:
 290    """Wait on `clock` until the leader has exited and group `pgid` has no member `live` names, or `deadline`.
 291
 292    The polls double up to `poll_max_s` seconds.
 293    """
 294    delay = 0.001
 295    while True:
 296        seen = _exit_seen(pgid, exited)
 297        if seen and not live(pgid):
 298            return True
 299        remaining = deadline.remaining()
 300        if remaining is not None and remaining <= 0:
 301            return False
 302        pause = delay if remaining is None else min(delay, remaining)
 303        if not seen:
 304            clock.wait(exited, pause)
 305        else:
 306            clock.sleep(pause)
 307        delay = min(delay * 2, poll_max_s)

 315@contextlib.contextmanager
 316def _signals_deferred(*, escalate: bool, request_hint: Callable[[], None] | None = None) -> Iterator[None]:
 317    """On the main thread, hold `DEFERRED_SIGNALS` for the body, then restore their handlers and raise each held one.
 318
 319    With `escalate`, the first SIGINT is held, a second raises `KeyboardInterrupt` at once, and the first SIGINT
 320    held in this process calls `request_hint`, which must take no lock (`Lines.request_hint`). A signal whose
 321    handler was not installed from Python cannot be restored, so it is not held. Held signals are not raised again
 322    when a wakeup fd is set, since its writer already saw them.
 323    """
 324    if threading.current_thread() is not threading.main_thread():
 325        yield
 326        return
 327    held: list[int] = []
 328    previous: dict[int, Any] = {}
 329
 330    def hold(signum: int, frame: object) -> None:
 331        if escalate and signum == signal.SIGINT:
 332            if signal.SIGINT in held:
 333                held[:] = [s for s in held if s != signal.SIGINT]
 334                raise KeyboardInterrupt
 335            if request_hint is not None and not _HINTED[0]:
 336                _HINTED[0] = True
 337                request_hint()
 338        held.append(signum)
 339
 340    try:
 341        for sig in DEFERRED_SIGNALS:
 342            if signal.getsignal(sig) is not None:
 343                previous[sig] = signal.signal(sig, hold)
 344        yield
 345    finally:
 346        _restore_signals(previous, held)
 347
 348
 349def _restore_signals(previous: dict[int, Any], held: list[int]) -> None:
 350    """Restore each handler in `previous`, then raise each signal in `held` once, unless a wakeup fd is set."""
 351    for number, handler in previous.items():
 352        signal.signal(number, handler)
 353    wakeup = signal.set_wakeup_fd(-1)
 354    signal.set_wakeup_fd(wakeup)
 355    if wakeup == -1:
 356        for number in dict.fromkeys(held):
 357            signal.raise_signal(number)

 374class WorkerClient:
 375    """A worker process running `python -I -m senselab_worker.launch <module>` as `launch` starts it, over the wire.
 376
 377    `module` defines `load` and `handle` (see `senselab_worker.launch`). The client makes its own
 378    `array_dir` under `array_parent` (the system temporary directory if `None`) and deletes only the
 379    array files in it, never other names: after each request every array file except those `init`
 380    references, and on close all of them and then the directory if it is empty. The worker runs in
 381    its own process group, which `close` and every host fault stop within its graces' `bound`; a
 382    descendant that calls `setsid` leaves the group and is reached only by an adopting supervisor (`Supervisor`),
 383    once the worker is released to it. One `run` may be in flight at
 384    a time. Host faults (`ProtocolMismatch`, `ProtocolError`, `WorkerLost`, `WorkerFatal`,
 385    `WorkerTimeout`, `WorkerOOMKilled`) discard the worker and carry its exit code and last stderr lines as a note.
 386    A worker that exits by SIGKILL while `memory_events` (this process's cgroup v2 `memory.events` when None) records
 387    more `oom_kill`s than at the client's start is `WorkerOOMKilled`, a resource fault, not `WorkerLost`.
 388    `hello` holds the worker's hello.
 389
 390    Shutdown is two steps, stopping the process group and then removing the files, each recorded
 391    only once it has completed and each run once. On the main thread SIGINT and SIGTERM are held
 392    while it runs and raised after it; a second SIGINT, or anything else that interrupts it,
 393    SIGKILLs the group, reaps it and removes the files before propagating. A copy of the client in
 394    a forked child is foreign: closing or collecting it closes only the child's descriptors.
 395    """
 396
 397    def __init__(  # noqa: PLR0913, PLR0915 - keyword-only options; starting a process and rolling it back on failure
 398        self,
 399        python: str | os.PathLike[str],
 400        module: str,
 401        *,
 402        init: dict[str, Any],
 403        array_parent: Path | None = None,
 404        expect_runtime: str | None = None,
 405        load_timeout: float | None = None,
 406        request_timeout: float | None = None,
 407        settings: WorkerSettings | None = None,
 408        graces: ShutdownGraces = DEFAULT_GRACES,
 409        max_line_bytes: int = MAX_LINE_BYTES,
 410        variables: Mapping[str, str] | None = None,
 411        memory_events: Path | None = None,
 412        supervisor: Supervisor | None = None,
 413        clock: Clock = SystemClock(),  # noqa: B008 - stateless
 414    ) -> None:
 415        """Start the worker, check its hello and wait until it has loaded `init`; stop it within `graces`.
 416
 417        `max_line_bytes` bounds every line the client sends and every reply line it accepts after the hello;
 418        a longer reply is `LineTooLarge`, a `ProtocolError`.
 419        `variables` are set on the worker process over the launch's allowlist (`launch`). The worker is launched
 420        through `supervisor` (`current()` when None), which also judges whether its group is live. Every timeout and
 421        grace is read and waited on `clock`. `settings` is the profile's `worker` section: the hello wait, the probe of
 422        the group leader and the longest group poll (the packaged profile's when None).
 423
 424        Raises:
 425            ValueError: if `init` uses the reserved key `array_dir` or has no wire form, or `max_line_bytes` is
 426                not an int in `[1, MAX_LINE_BYTES]`.
 427            TypeError: if `graces` is not a `ShutdownGraces`.
 428            ProtocolMismatch: if the worker's first message is not a hello, its wire or runtime differ
 429                from the host's (`expect_runtime` overrides the host's runtime identity), or no hello
 430                arrives within `settings.hello_s`.
 431            WorkerLost: if the worker exits or closes its output before it is ready.
 432            WorkerFatal: if loading fails.
 433            WorkerTimeout: if sending `init` and loading exceed `load_timeout`.
 434            WorkerOOMKilled: if the worker is killed for memory while it loads.
 435        """
 436        self._closed = True
 437        self._owner_pid = os.getpid()
 438        _check_arguments(init, graces, max_line_bytes)
 439        self._max_line_bytes, self._request_timeout = max_line_bytes, request_timeout
 440        self._settings = settings if settings is not None else ExecutionProfile.packaged().worker
 441        self._supervisor = supervisor if supervisor is not None else current()
 442        self._memory_events = memory_events if memory_events is not None else cgroup_memory_events()
 443        self._oom_baseline = oom_kills(self._memory_events)
 444        self._graces, self._clock = graces, clock
 445        self._stderr: collections.deque[str] = collections.deque(maxlen=STDERR_TAIL_LINES)
 446        self._lines = Lines(clock, on_hint=lambda: logger.warning(ESCALATE_HINT))
 447        self._proc: subprocess.Popen[bytes] | None = None
 448        self._pumps: list[Pump] = []
 449        self._reaped = WakeEvent()
 450        self._exited = WakeEvent()
 451        self._run_lock = threading.Lock()
 452        self._run_owner: int | None = None
 453        self._constructing: int | None = None
 454        self._closed_while_starting = False
 455        self._run_idle = WakeEvent()
 456        self._run_idle.set()
 457        self._channel_lock = threading.Lock()
 458        self._shutdown_lock = threading.Lock()
 459        self._shutting_down: set[int] = set()
 460        self._closed_by_caller = False
 461        self._stopped = self._released = False
 462        self._init_names: set[str] = set()
 463        self.hello: dict[str, Any] = {}
 464        self._wake = WakeEvent()
 465        try:
 466            self._array_dir = Path(tempfile.mkdtemp(prefix="senselab-arrays-", dir=array_parent)).absolute()
 467        except BaseException:
 468            self._close_wake()
 469            self._stopped = self._released = True
 470            raise
 471        self._closed = False
 472        self._constructing = threading.get_ident()
 473        try:
 474            wire_init = to_wire(
 475                {"array_dir": str(self._array_dir), **init},
 476                array_dir=self._array_dir,
 477                at="init",
 478                limit=self._max_line_bytes,
 479            )
 480            self._init_names = self._owned(array_paths(wire_init))
 481            self._start(python, module, variables or {})
 482            self.hello = self._handshake(expect_runtime or runtime_identity())
 483            deadline = Deadline(load_timeout, clock.now)
 484            self._send(dump_line({"init": wire_init}, "init", limit=self._max_line_bytes), deadline)
 485            self._receive(deadline, "loading the backend", "ready")
 486            if self._closed_while_starting:
 487                raise WorkerLost("the client was closed while it was starting")
 488        except BaseException as exc:
 489            raised = self._abort_start(exc)
 490            if raised is exc:
 491                raise
 492            raise raised from exc
 493        finally:
 494            self._constructing = None
 495
 496    def _abort_start(self, exc: BaseException) -> BaseException:
 497        """Stop the worker a constructor failed to start with `exc`; return what the constructor raises."""
 498        raised = exc
 499        if self._closed_by_caller and isinstance(exc, Exception) and not isinstance(exc, WorkerLost):
 500            raised = WorkerLost("the client was closed while it was starting")
 501        try:
 502            self._fail(raised)
 503        finally:
 504            if not (self._stopped and self._released):
 505                self._shutdown(graceful=False)
 506        return self._typed(raised)

 532    def run(self, items: list[dict[str, Any]], params: dict[str, Any]) -> list[dict[str, Any]]:
 533        """Run one request and return one result per item, in order.
 534
 535        Each result is `{"id", "outputs"}` with arrays loaded, or `{"id", "error"}` with the error
 536        object's `fault` as a `FaultClass`. An item whose inputs have no wire form fails alone, with
 537        its content error (`ValueTooLarge`, a resource error, if they cannot fit a line), and is not
 538        sent. One `request_timeout` covers sending the request, receiving the response and reading
 539        its arrays.
 540
 541        Raises:
 542            ConcurrentUse: if another `run` on this client is in flight; that call is unaffected.
 543            ValueError: if `items` or `params` are malformed or `params` has no wire form.
 544            ValueTooLarge: if `params` cannot fit one wire line, or the items that each fit alone do
 545                not fit one line together; an item that cannot fit alone fails alone instead.
 546            ProtocolError, WorkerLost, WorkerFatal: infrastructure faults; the worker is discarded.
 547            WorkerTimeout: if the request exceeds `request_timeout`; the worker is killed.
 548            WorkerOOMKilled: if the worker is killed for memory during the request.
 549        """
 550        if self._foreign():
 551            raise WorkerLost(f"this client belongs to process {self._owner_pid}; start a new WorkerClient")
 552        if not self._run_lock.acquire(blocking=False):
 553            raise ConcurrentUse("run is already in flight on this client; use one client per thread")
 554        self._run_owner = threading.get_ident()
 555        self._run_idle.clear()
 556        try:
 557            return self._run(items, params)
 558        except Exception as exc:
 559            if self._closed_by_caller and not isinstance(exc, WorkerLost):
 560                lost = WorkerLost("the client was closed during the run")
 561                self._fail(lost)
 562                raise lost from exc
 563            raise
 564        finally:
 565            try:
 566                if not self._closed:
 567                    self._remove_arrays(keep=self._init_names)
 568            finally:
 569                self._run_owner = None
 570                try:
 571                    if self._closed:
 572                        self._shutdown(graceful=False)
 573                finally:
 574                    self._run_idle.set()
 575                    self._run_lock.release()
 576
 577    def close(self) -> None:
 578        """Stop the worker's process group and remove this client's files, within `graces.bound`.
 579
 580        Safe to call twice and from another thread; a `run` in flight then raises `WorkerLost`.
 581        Called again on a thread already closing this client, as from a signal handler, it returns at
 582        once and the outer call finishes. Called on the thread running `run`, as from a signal
 583        handler, it marks the client closed and returns; that `run`, if it has not finished, stops
 584        the worker and raises `WorkerLost`. In a forked child it closes only the child's copies of
 585        the client's descriptors.
 586        """
 587        if self._foreign():
 588            self._disown()
 589            return
 590        if not self._closed:
 591            self._closed_by_caller = True
 592        if threading.get_ident() in (self._run_owner, self._constructing):
 593            self._closed_while_starting = self._closed_while_starting or self._constructing is not None
 594            self._closed = True
 595            self._lines.put(None)
 596            self._wake.set()
 597            return
 598        self._shutdown(graceful=self._run_owner is None)

 628    def _disown(self) -> None:
 629        """In a forked child, close the child's copies of the client's descriptors; touch nothing the owner has."""
 630        self._closed = True

 632        proc = self._proc

 634        for stream in streams:
 635            if stream is not None:
 636                try:
 637                    stream.close()
 638                except (OSError, ValueError):
 639                    pass
 640        for pump in self._pumps:
 641            pump.close()
 642        self._close_wake()

 649    def _run(self, items: list[dict[str, Any]], params: dict[str, Any]) -> list[dict[str, Any]]:
 650        if self._closed:
 651            raise WorkerLost("worker was discarded or closed; start a new WorkerClient", returncode=self.returncode)
 652        request_id = str(uuid.uuid4())
 653        try:
 654            validate_message({"id": request_id, "items": items, "params": params}, "request")
 655        except ProtocolError as exc:
 656            raise ValueError(f"malformed request: {exc}") from exc
 657        wire_params = to_wire(params, array_dir=self._array_dir, at="params", limit=self._max_line_bytes)
 658        results: list[dict[str, Any]] = [{} for _ in items]
 659        sent: list[tuple[int, dict[str, Any]]] = []
 660        for i, item in enumerate(items):
 661            try:
 662                sent.append(
 663                    (
 664                        i,
 665                        {
 666                            "id": item["id"],
 667                            "inputs": to_wire(
 668                                item["inputs"], array_dir=self._array_dir, at="inputs", limit=self._max_line_bytes
 669                            ),
 670                        },
 671                    )
 672                )
 673            except Exception as exc:
 674                results[i] = {"id": item["id"], "error": _mapped(error_record(exc))}
 675        sent, line = _request_line(request_id, sent, wire_params, results, self._max_line_bytes)
 676        if not sent:
 677            return results
 678        request = {"id": request_id, "items": [wire for _, wire in sent]}
 679        deadline = Deadline(self._request_timeout, self._clock.now)
 680        try:
 681            self._send(line, deadline)
 682            replies = self._read_replies(request, deadline)
 683        except BaseException as exc:
 684            self._fail(exc)
 685            typed = self._typed(exc)
 686            if typed is exc:
 687                raise
 688            raise typed from exc
 689        for (i, _), reply in zip(sent, replies):
 690            results[i] = reply
 691        return results

 693    def _start(self, python: str | os.PathLike[str], module: str, variables: Mapping[str, str]) -> None:
 694        pipe = subprocess.PIPE
 695        started = launch(python, variables)

 699                raise WorkerLost("the client was closed before its worker started")

 701        assert proc.stdin is not None and proc.stdout is not None and proc.stderr is not None

 703        tail = self._stderr

 705        def keep(line: bytes | None) -> None:

 713    def _handshake(self, expected_runtime: str) -> dict[str, Any]:
 714        deadline = Deadline(self._settings.hello_s, self._clock.now)
 715        hello: dict[str, Any] | None = None
 716        while True:
 717            try:
 718                line = self._lines.get(deadline)
 719            except queue.Empty:
 720                break
 721            if line is None:
 722                self._lines.put(None)
 723                raise WorkerLost("the worker exited or closed its output before its hello")
 724            try:
 725                msg = parse_line(line)
 726            except ProtocolError:
 727                logger.warning("worker wrote a non-JSON line before its hello: %r", line[:_LOGGED_LINE_BYTES])
 728                continue
 729            try:
 730                hello = validate_message(msg, "hello")["hello"]
 731            except ProtocolError as exc:
 732                raise ProtocolMismatch(
 733                    f"first message is not a hello: {exc}; host expects wire {WIRE_VERSION!r}"
 734                ) from exc
 735            break
 736        got = (hello.get("wire"), hello.get("runtime")) if hello is not None else None
 737        if hello is None or got != (WIRE_VERSION, expected_runtime):
 738            raise ProtocolMismatch(
 739                f"worker hello gave (wire, runtime) = {got!r}; host expects ({WIRE_VERSION!r}, {expected_runtime!r})"
 740            )
 741        return hello

 743    def _send(self, line: bytes, deadline: Deadline, *, stopping: bool = False) -> None:
 744        """Write `line` by non-blocking writes, each under the channel lock and only while the client is open.
 745
 746        With `stopping`, the line is written though the client is closed, for as long as its stdin is open.
 747        """
 748        assert self._proc is not None and self._proc.stdin is not None
 749        stdin = self._proc.stdin
 750        view = memoryview(line)
 751        with self._wake.held() as wake, selectors.DefaultSelector() as sel:
 752            if wake >= 0:
 753                sel.register(wake, selectors.EVENT_READ)
 754            watched = False
 755            while view:
 756                with self._channel_lock:
 757                    if (self._closed and not stopping) or stdin.closed:
 758                        raise WorkerLost("the client was closed while sending to the worker")
 759                    fd = stdin.fileno()
 760                    if not watched:
 761                        sel.register(fd, selectors.EVENT_WRITE)
 762                        watched = True
 763                    try:
 764                        view = view[os.write(fd, view) :]
 765                        continue
 766                    except BlockingIOError:
 767                        pass
 768                    except OSError as exc:
 769                        raise WorkerLost(f"cannot write to the worker: {exc}") from exc
 770                deadline.check("sending to the worker")
 771                self._clock.select(sel, deadline.remaining())

 788    def _receive(self, deadline: Deadline, what: str, *kinds: Kind) -> dict[str, Any]:
 789        """Return the next line as a message of one of `kinds`, the first that has its key; a `fatal` raises."""
 790        assert self._proc is not None
 791        try:
 792            line = self._lines.get(deadline)
 793        except queue.Empty:
 794            deadline.check(what)
 795            raise TimeoutError(f"{what} did not finish in time") from None
 796        if line is None:
 797            self._lines.put(None)
 798            if self._closed_by_caller:
 799                raise WorkerLost(f"the client was closed while {what}")
 800            raise WorkerLost(f"worker closed its output while {what}")
 801        if len(line) > self._max_line_bytes:
 802            raise LineTooLarge(len(line), self._max_line_bytes)
 803        msg = parse_line(line)
 804        if "fatal" in msg:
 805            raise WorkerFatal(_fatal_record(validate_message(msg, "fatal")["error"]), from_load="ready" in kinds)
 806        kind = next((k for k in kinds if k in msg), kinds[-1])
 807        return validate_message(msg, kind)

 809    def _read_replies(self, request: dict[str, Any], deadline: Deadline) -> list[dict[str, Any]]:
 810        """Read one result line per item, in request order, then `done`."""
 811        replies = []
 812        shared: dict[str, numpy.ndarray] = {}
 813        for item in request["items"]:
 814            reply = self._receive(deadline, "running the request", "result")
 815            if (reply["id"], reply["item"]) != (request["id"], item["id"]):
 816                raise ProtocolError(
 817                    f"expected the result of item {item['id']!r} of request {request['id']!r}, "
 818                    f"got item {reply['item']!r} of request {reply['id']!r}"
 819                )
 820            replies.append(self._decode_result(reply, deadline, shared))
 821        done = self._receive(deadline, "running the request", "done")
 822        if (done["id"], done["done"]) != (request["id"], len(request["items"])):
 823            raise ProtocolError(
 824                f"expected done for {len(request['items'])} items of request {request['id']!r}, got {done!r}"
 825            )
 826        device = (done["device_peak_bytes"], done["device_allocated_bytes"]) if "device_peak_bytes" in done else None
 827        request_reports.record(
 828            len(request["items"]), since_load=done.get("since_load"), load_s=done.get("load_s"), device=device
 829        )
 830        return replies

 833        self, result: dict[str, Any], deadline: Deadline, shared: dict[str, numpy.ndarray]
 834    ) -> dict[str, Any]:
 835        if "error" in result:
 836            return item_error(result["item"], result["error"])
 837        try:
 838            outputs = from_wire(
 839                result["outputs"],
 840                array_dir=self._array_dir,

 842                shared=shared,

 844        except ArrayIntegrityError as exc:
 845            return {"id": result["item"], "error": _mapped(error_record(exc))}
 846        return {"id": result["item"], "outputs": outputs}

 867    def _fail(self, exc: BaseException) -> None:
 868        """Stop the worker for the host fault `exc`, then give `exc` the exit code and the stderr tail."""
 869        self._closed = True
 870        if isinstance(exc, WorkerLost) and not self._closed_by_caller:
 871            self._clock.wait(self._exited, self._graces.exit_code_wait)
 872        self._shutdown(graceful=isinstance(exc, ProtocolMismatch))
 873        if isinstance(exc, WorkerLost) and exc.returncode is None:
 874            exc.returncode = self.returncode
 875        if isinstance(exc, (InfrastructureFault, TimeoutError)):
 876            exc.add_note(
 877                f"worker exit code {_code_text(self.returncode)}; stderr, last {len(self._stderr)} lines:\n"
 878                + "\n".join(self._stderr)
 879            )

 881    def _typed(self, exc: BaseException) -> BaseException:
 882        """Return the resource fault a host fault is, its notes kept: a timeout, or a SIGKILL under an OOM kill."""
 883        if isinstance(exc, TimeoutError) and not isinstance(exc, ResourceFault):
 884            typed: BaseException = WorkerTimeout(str(exc))
 885        elif isinstance(exc, WorkerLost) and exc.returncode == -signal.SIGKILL and self._oom_killed():
 886            typed = WorkerOOMKilled(f"the worker was killed for memory: {exc}", returncode=exc.returncode)
 887        else:
 888            return exc
 889        for note in getattr(exc, "__notes__", ()):
 890            typed.add_note(note)
 891        return typed

 898    def _shutdown(self, *, graceful: bool) -> None:
 899        """Run each shutdown step not yet completed: stop the group, then, once no run is in flight, remove files.
 900
 901        If the steps are interrupted, the group is SIGKILLed and the steps are finished, with signals held,
 902        before the interruption propagates.
 903        """
 904        if self._foreign():
 905            self._disown()
 906            return
 907        me = threading.get_ident()
 908        if me in self._shutting_down:
 909            return
 910        try:
 911            self._shutting_down.add(me)
 912            self._closed = True
 913            try:
 914                with _signals_deferred(escalate=True, request_hint=self._lines.request_hint):
 915                    self._steps(graceful=graceful, me=me)
 916            except BaseException:
 917                with _signals_deferred(escalate=False):
 918                    try:
 919                        self._kill_group_now()
 920                        self._steps(graceful=False, me=me)
 921                    except BaseException:  # noqa: BLE001 - the first interruption is the one raised
 922                        logger.exception("finishing an interrupted shutdown failed")
 923                raise
 924        finally:
 925            self._shutting_down.discard(me)
 926
 927    def _steps(self, *, graceful: bool, me: int) -> None:
 928        bound = Deadline(self._graces.bound, self._clock.now)
 929        if self._acquire_shutdown(bound):
 930            try:
 931                if not self._stopped:
 932                    self._stop(graceful=graceful)
 933            finally:
 934                self._shutdown_lock.release()
 935        if self._run_owner not in (None, me):
 936            self._clock.wait(self._run_idle, bound.remaining() or 0.0)
 937        if self._acquire_shutdown(bound):
 938            try:
 939                if not self._released and self._run_owner in (None, me):
 940                    self._release()
 941            finally:
 942                self._shutdown_lock.release()
 943
 944    def _acquire_shutdown(self, bound: Deadline) -> bool:
 945        if self._shutdown_lock.acquire(timeout=max(bound.remaining() or 0.0, 0.0)):
 946            return True
 947        logger.error("another thread's shutdown of this client did not finish within %s s", self._graces.bound)
 948        return False

 951        try:
 952            self._terminate(graceful=graceful)
 953        except Exception:  # noqa: BLE001 - shutdown is total
 954            logger.exception("stopping the worker failed")
 955            self._kill_group_now()
 956            return
 957        self._stopped = True

 959    def _kill_group_now(self) -> None:
 960        """SIGKILL the group if its leader is still this client's unreaped child."""
 961        proc = self._proc
 962        if proc is None or proc.returncode is not None:
 963            return
 964        if not _exit_seen(proc.pid, self._exited) or _leader_held(
 965            proc.pid, self._exited, self._settings.leader_probe_s
 966        ):
 967            self._sigkill_group(proc.pid)

 987    def _signal_group_until_gone(self, pgid: int, stop: Deadline) -> None:
 988        exited = self._exited
 989        _signal_group(pgid, signal.SIGTERM, leader_exited=_exit_seen(pgid, exited))
 990        _signal_group(pgid, signal.SIGCONT, leader_exited=_exit_seen(pgid, exited))
 991        live = self._supervisor.group_live
 992        poll_max_s = self._settings.group_poll_max_s
 993        if not _members_gone(pgid, exited, stop, live, self._clock, poll_max_s):
 994            self._sigkill_group(pgid)
 995            if not _members_gone(
 996                pgid, exited, Deadline(self._graces.kill, self._clock.now), live, self._clock, poll_max_s
 997            ):
 998                logger.error("worker process group %d still has members after SIGKILL", pgid)

1000    def _stop_group(self, proc: subprocess.Popen[bytes], stop: Deadline) -> None:
1001        """Signal the unreaped leader's group until it is gone within `stop`, then reap the leader if it exited."""
1002        pgid = proc.pid
1003        held = _leader_held(pgid, self._exited, self._settings.leader_probe_s)
1004        if held:
1005            try:
1006                self._signal_group_until_gone(pgid, stop)
1007            except Exception:
1008                logger.exception("stopping worker process group %d failed; sending SIGKILL", pgid)
1009                self._sigkill_group(pgid)
1010                _wait_for_exit(pgid, self._exited, self._graces.kill, self._clock, self._settings.group_poll_max_s)
1011        elif held is None:
1012            logger.warning(
1013                "worker %d: cannot tell whether it is still unreaped; its process group is not signalled", pgid
1014            )
1015        else:
1016            logger.warning("worker %d was reaped outside this client; its process group is not signalled", pgid)
1017            self._supervisor.released(pgid)
1018        if self._exited.is_set() and proc.returncode is None:
1019            self._supervisor.released(pgid)
1020            proc.wait()

1022    def _terminate(self, *, graceful: bool) -> None:
1023        """Stop the group and reap its leader, then stop the pipe readers; each part may be run again."""
1024        proc = self._proc
1025        if proc is None:
1026            return
1027        assert proc.stdout is not None and proc.stderr is not None
1028        pgid = proc.pid
1029        stop = Deadline(self._graces.stop, self._clock.now)
1030        if graceful and not self._exited.is_set():
1031            self._write_stop()
1032        self._retire_channel()
1033        if graceful:
1034            _wait_for_exit(
1035                pgid,
1036                self._exited,
1037                max((stop.remaining() or 0.0) - self._graces.term, 0.0),
1038                self._clock,
1039                self._settings.group_poll_max_s,
1040            )
1041        if proc.returncode is None:
1042            self._stop_group(proc, stop)
1043        join = Deadline(0.0 if sys.is_finalizing() else self._graces.join, self._clock.now)
1044        for pump in self._pumps:
1045            pump.finish()
1046        ended = [pump.ended for pump in self._pumps] + ([self._reaped] if self._pumps else [])
1047        for event in ended:
1048            self._clock.wait(event, join.remaining() or 0.0)
1049        self._lines.put(None)
1050        if not all(event.is_set() for event in ended):
1051            if not sys.is_finalizing():
1052                logger.warning(
1053                    "worker %d: pipe readers did not stop within %s s; its pipes stay open", pgid, self._graces.join
1054                )
1055            return
1056        for pump in self._pumps:
1057            pump.close()
1058        proc.stdout.close()
1059        proc.stderr.close()
