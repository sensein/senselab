senselab_core.runtime.provenance

A run's provenance in PROV-O JSON-LD (ADR-010): one document per writer, read back as their union by @id.

Each writer holds the records it adds (senselab_core.kernel.prov) and replaces its own provenance@3.0 document, <run>/prov/<writer uuid4 hex>.jsonld, atomically. A reader takes the union of every writer's document; two different nodes under one @id are refused, every node read is rebuilt through the record builders, and a file of another provenance format is refused (ProvenanceFormatRefused).

131def pathless(name: str, operation: Callable[[], T]) -> T:
132    """Return `operation()`, raising a `RunIOError` in place of any OS, lease or wrong-kind error.
133
134    The original error, which may name a path, is logged for the operator as a warning naming `name`.
135
136    Raises:
137        RunIOError: if `operation` raised one of those; it carries the errno text only, with no cause or context.
138    """
139    try:
140        return operation()
141    except (OSError, WrongKindError, LeaseUnavailable) as err:
142        logger.warning("%s failed: %s", name, err)
143        failure = RunIOError(err)
144    raise failure

147class ProvStore:
148    """A set of PROV records keyed by `@id`: one writer's records, or the union of a run's writers.
149
150    `add` holds a record and the `senselab:File` entities it names; `write` replaces this writer's document,
151    `<run>/prov/<writer>.jsonld`, atomically; `read` returns the union of every writer's document in a run.
152    Accessors return records in `@id` order.
153
154    A store used after a `fork` is a new writer in the child, so each process writes its own document.
155    """
156
157    def __init__(self, records: Iterable[kprov.Record] = ()) -> None:
158        """Hold `records`.
159
160        Raises:
161            ConflictingNode: if two of them are different nodes under one `@id`.
162        """
163        self._writer = (os.getpid(), uuid.uuid4().hex)
164        self._records: dict[str, kprov.Record] = {}
165        self.add(*records)
166
167    @property
168    def writer(self) -> str:
169        """This process's writer uuid4 hex: the name of the document `write` replaces."""
170        if self._writer[0] != os.getpid():
171            self._writer = (os.getpid(), uuid.uuid4().hex)
172        return self._writer[1]
173
174    def restart(self) -> None:
175        """Forget every record held and take a new writer id, so the next `write` makes a document of its own."""
176        self._writer = (os.getpid(), uuid.uuid4().hex)
177        self._records = {}
178
179    def add(self, *records: kprov.Record) -> None:
180        """Hold each record and the files it names; a record already held under its `@id` is kept once.
181
182        Nothing is added unless every record is.
183
184        Raises:
185            ConflictingNode: if a record differs from the one held, or another one given, under its `@id`.
186            ValueError: if a record's `senselab:fileLabel` or `senselab:modelSource` is one the boundary refuses.
187        """
188        added: dict[str, kprov.Record] = {}
189        for r in (r for record in records for r in (record, *record.files)):
190            if (problem := _boundary_problem(r)) is not None:
191                raise ValueError(problem)
192            held = self._records.get(r.id) or added.setdefault(r.id, r)
193            if held.jcs != r.jcs:
194                raise ConflictingNode(r.id)
195        self._records.update(added)
196
197    @classmethod
198    def merge(cls, stores: Iterable[ProvStore]) -> ProvStore:
199        """Return the union of `stores` by `@id`; the result does not depend on their order.
200
201        Raises:
202            ConflictingNode: if two stores hold different nodes under one `@id`.
203        """
204        return cls(r for store in stores for r in store.records())
205
206    def records(self) -> list[kprov.Record]:
207        """Return every record, in `@id` order."""
208        return [self._records[k] for k in sorted(self._records)]
209
210    def get(self, node_id: str) -> kprov.Record:
211        """Return the record under `node_id`.
212
213        Raises:
214            KeyError: if none is held.
215        """
216        return self._records[node_id]
217
218    def _of(self, *prefixes: str) -> list[kprov.Record]:
219        return [r for r in self.records() if r.id.partition(":")[0] in prefixes]
220
221    def entities(self) -> list[kprov.Record]:
222        """Return the content-addressed entities (`content:`, `output:`), in `@id` order."""
223        return self._of("content", "output")
224
225    def activities(self, logical_key: object = None) -> list[kprov.Record]:
226        """Return the activities, or those whose `senselab:logicalKey` is `logical_key`, in `@id` order."""
227        out = self._of("activity")
228        if logical_key is None:
229            return out
230        wanted = jcs_dumps(logical_key)
231        return [
232            r
233            for r in out
234            if "senselab:logicalKey" in (node := r.node())
235            and jcs_dumps(kprov.literal_value(node["senselab:logicalKey"])) == wanted
236        ]
237
238    def plans(self) -> list[kprov.Record]:
239        """Return the plans (`plan:<plan hash>`), in `@id` order."""
240        return self._of("plan")
241
242    def agents(self) -> list[kprov.Record]:
243        """Return the agents, in `@id` order."""
244        return self._of("agent")
245
246    def environments(self) -> list[kprov.Record]:
247        """Return the environments, in `@id` order."""
248        return self._of("environment")
249
250    def document(self) -> dict[str, object]:
251        """Return the JSON-LD document of every record, nodes in `@id` order."""
252        return kprov.document(self.records())
253
254    def write(self, run: str | Path) -> None:
255        """Replace this writer's document, `<run>/prov/<writer>.jsonld`, atomically with `document()` as JCS.
256
257        Raises:
258            RunIOError: if the document cannot be written; the error names no path.
259        """
260        directory = Path(run) / PROV_DIR
261        data = jcs_dumps(self.document())
262
263        def write() -> None:
264            directory.mkdir(exist_ok=True)
265            write_atomically(directory / f"{self.writer}.jsonld", data)
266
267        pathless("writing the provenance document", write)

269    @classmethod
270    def read(cls, run: str | Path, vocabulary: kprov.Vocabulary | None = None) -> ProvStore:
271        """Return the union of the writers' documents in `<run>/prov/`, every node rebuilt over `vocabulary`.
272
273        Each node is rebuilt through the record builders over `vocabulary` (the core vocabulary when None) and
274        must be byte-identical to its rebuild, so its `@id`, terms and literals are all checked. Every relation
275        endpoint and every record an attribute names must be a node of the union. Temporaries of an interrupted
276        write are skipped.
277
278        Raises:
279            ProvenanceFormatRefused: on a file in `prov/` that declares a format other than `provenance@3`.
280            MalformedDocument: on a file in `prov/` that is not a writer's document or cannot be read, a document
281                of another shape, a node that differs from its rebuild, a `senselab:fileLabel` that is not
282                `file-<n>`, a local path as `senselab:modelSource`, or attribute references nested too deep.
283            ConflictingNode: if two writers hold different nodes under one `@id`.
284            DanglingReference: if a node names an `@id` the union does not hold.
285        """
286        return cls.from_documents(cls.read_documents(run), vocabulary)
287
288    @staticmethod
289    def read_documents(run: str | Path) -> dict[str, bytes]:
290        """Return `{file name: bytes}` of every document in `<run>/prov/`, temporaries of an interrupted write skipped.
291
292        Raises:
293            MalformedDocument: `prov/` cannot be listed, holds an entry that is not a regular file, or a file that
294                cannot be read; the message names an error class, never a path.
295        """
296        directory = Path(run) / PROV_DIR
297        documents: dict[str, bytes] = {}
298        failure: MalformedDocument | None = None
299        try:
300            for path in sorted(directory.iterdir()) if directory.is_dir() else []:
301                if TEMPORARY_NAME.fullmatch(path.name):
302                    continue
303                if not path.is_file() or path.is_symlink():
304                    raise MalformedDocument(None, f"{PROV_DIR}/ holds an entry that is not a regular file")
305                writer = path.name.removesuffix(".jsonld") if WRITER_DOCUMENT.fullmatch(path.name) else None
306                try:
307                    documents[path.name] = path.read_bytes()
308                except OSError as err:
309                    failure = MalformedDocument(writer, f"the document cannot be read ({type(err).__name__})")
310                    break
311        except OSError as err:
312            failure = MalformedDocument(None, f"{PROV_DIR}/ cannot be read ({type(err).__name__})")
313        if failure is not None:
314            raise failure
315        return documents

317    @classmethod
318    def from_documents(cls, documents: Mapping[str, bytes], vocabulary: kprov.Vocabulary | None = None) -> ProvStore:
319        """Return the union of `documents` (`read_documents`' result), every node rebuilt over `vocabulary`.
320
321        Each node is rebuilt through the record builders over `vocabulary` (the core vocabulary when None) and
322        must be byte-identical to its rebuild, so its `@id`, terms and literals are all checked. Every relation
323        endpoint and every record an attribute names must be a node of the union.
324
325        Raises:
326            ProvenanceFormatRefused: on a document that declares a format other than `provenance@3`.
327            MalformedDocument: on a file that is not a writer's document, a document of another shape, a node
328                that differs from its rebuild, a `senselab:fileLabel` that is not `file-<n>`, or attribute
329                references nested deeper than `MAX_DEPTH`.
330            ConflictingNode: if two writers hold different nodes under one `@id`.
331            DanglingReference: if a node names an `@id` the union does not hold.
332        """
333        nodes: dict[str, tuple[bytes, str]] = {}
334        for name, data in sorted(documents.items()):
335            writer = name.removesuffix(".jsonld") if WRITER_DOCUMENT.fullmatch(name) else None
336            found = _declared_format(data)
337            if (
338                found is not None
339                and check(PROVENANCE_FORMAT, found, RETIRED_PROVENANCE_FORMATS) is not FormatVerdict.ACCEPTED
340            ):
341                raise ProvenanceFormatRefused(found)
342            if writer is None:
343                raise MalformedDocument(None, f"{PROV_DIR}/ holds a file that is not a writer's document")
344            for node_id, raw in _writer_nodes(writer, data):
345                if nodes.setdefault(node_id, (raw, writer))[0] != raw:
346                    raise ConflictingNode(node_id)
347        return cls(_Rebuild(nodes, vocabulary or core_vocabulary()).all())

357def _boundary_problem(record: kprov.Record) -> str | None:
358    """Name what in `record` could carry a path: a bad `senselab:fileLabel` or `senselab:modelSource`."""
359    node = record.node()
360    label, source = node.get("senselab:fileLabel"), node.get("senselab:modelSource")
361    if label is not None and not (isinstance(label, str) and FILE_LABEL.fullmatch(label)):
362        return "a senselab:fileLabel is file-<n>"
363    if isinstance(source, str) and not MODEL_SOURCE.fullmatch(source):
364        return "a senselab:modelSource is a source kind and an identity, never a local path"
365    return None

368def _declared_format(data: bytes) -> str | None:
369    """Return the format tag a file declares as a JSON object's `format`, None if it declares none.
370
371    The whole file is read as one JSON document, so an indented one declares its format; a JSON-lines file declares
372    it on its first line.
373    """
374    for candidate in (data, data.partition(b"\n")[0]):
375        try:
376            require_json_depth(candidate)
377            document = json.loads(candidate)
378        except ValueError:
379            continue
380        found = document.get("format") if isinstance(document, dict) else None
381        return found if isinstance(found, str) and re.fullmatch(FORMAT_PATTERN, found) else None
382    return None

385def _writer_nodes(writer: str, data: bytes) -> list[tuple[str, bytes]]:
386    failure: MalformedDocument | None = None
387    try:
388        doc = jcs_loads(data)
389    except ValueError:
390        failure = MalformedDocument(writer, "the document is not canonical JSON")
391    if failure is not None:
392        raise failure
393    graph = doc.get("@graph") if isinstance(doc, dict) else None
394    if not (isinstance(doc, dict) and doc.keys() == _DOCUMENT_KEYS and isinstance(graph, list)):
395        raise MalformedDocument(
396            writer, f"a document is {{'@context': {kprov.CONTEXT_URL!r}, 'format': ..., '@graph': [...]}}"
397        )
398    if check(PROVENANCE_FORMAT, doc["format"]) is not FormatVerdict.ACCEPTED:
399        raise MalformedDocument(writer, f"a document's format is {PROVENANCE_FORMAT}")
400    if doc["@context"] != kprov.CONTEXT_URL:
401        raise MalformedDocument(writer, f"a document's @context is {kprov.CONTEXT_URL}")
402    out = []
403    for node in graph:
404        node_id = node.get("@id") if isinstance(node, dict) else None
405        if not (isinstance(node_id, str) and kprov.NODE_ID.fullmatch(node_id)):
406            raise MalformedDocument(writer, f"a node's @id is not of the form {kprov.NODE_ID.pattern}")
407        out.append((node_id, jcs_dumps(node)))
408    return out

419class _Rebuild:
420    """Rebuild a union of raw nodes through the record builders, checking each against its bytes."""
421
422    def __init__(self, nodes: Mapping[str, tuple[bytes, str]], vocabulary: kprov.Vocabulary) -> None:
423        self.nodes = nodes
424        self.vocabulary = vocabulary
425        self.done: dict[str, kprov.Record] = {}
426        self.pending: set[str] = set()
427
428    def all(self) -> list[kprov.Record]:
429        return [self.record(node_id, node_id) for node_id in sorted(self.nodes)]
430
431    def checked_build(self, node_id: str, node: dict[str, Any], writer: str) -> kprov.Record:
432        try:
433            return self.build(node_id, node)
434        except (DanglingReference, MalformedDocument):
435            raise
436        except _REBUILD_ERRORS as err:
437            refused = type(err).__name__
438        raise MalformedDocument(writer, f"{node_id} is not what the builders write ({refused})")
439
440    def record(self, node_id: str, named_by: str) -> kprov.Record:
441        if node_id in self.done:
442            return self.done[node_id]
443        if node_id not in self.nodes:
444            raise DanglingReference(named_by, node_id)
445        raw, writer = self.nodes[node_id]
446        if node_id in self.pending:
447            raise MalformedDocument(writer, f"{node_id} names itself through its attributes")
448        if (too_deep := _depth_refusal(len(self.pending) + 1)) is not None:
449            raise MalformedDocument(writer, f"{node_id}: attribute references: {too_deep}")
450        self.pending.add(node_id)
451        node = cast("dict[str, Any]", jcs_loads(raw))
452        rebuilt = self.checked_build(node_id, node, writer)
453        if rebuilt.jcs != raw:
454            raise MalformedDocument(writer, f"{node_id} differs from its rebuild")
455        for target in sorted(_endpoints(node)):
456            if target not in self.nodes:
457                raise DanglingReference(node_id, target)
458        self.pending.discard(node_id)
459        self.done[node_id] = rebuilt
460        return rebuilt
461
462    def build(self, node_id: str, node: dict[str, Any]) -> kprov.Record:
463        record = self.make(node_id, node)
464        if (problem := _boundary_problem(record)) is not None:
465            raise MalformedDocument(self.nodes[node_id][1], f"{node_id}: {problem}")
466        return record
467
468    def make(self, node_id: str, node: dict[str, Any]) -> kprov.Record:
469        kind = node_id.partition(":")[0]
470        types = node["@type"]
471        if kind == "activity":
472            return self.activity(node_id, node)
473        attributes = self.attributes(node_id, node, _NODE_KEYS)
474        if kind == "agent":
475            return kprov.agent(self.vocabulary, types=types, attributes=attributes)
476        if kind == "environment":
477            return kprov.environment(self.vocabulary, attributes)
478        if kind == "plan":
479            return kprov.plan(node_id.partition(":")[2], self.vocabulary)
480        return kprov.entity(node_id, self.vocabulary, types=types, attributes=attributes)
481
482    def activity(self, node_id: str, node: dict[str, Any]) -> kprov.Record:
483        usages = [
484            kprov.Usage(q["prov:entity"]["@id"], port=_role(q, "senselab:port"), time=_time(q.get("prov:atTime")))
485            for q in node.get("prov:qualifiedUsage", [])
486        ]
487        generations = [
488            kprov.Generation(
489                q["@reverse"]["prov:qualifiedGeneration"]["@id"],
490                port=_role(q, "senselab:port"),
491                time=_time(q.get("prov:atTime")),
492            )
493            for q in node.get("@reverse", {}).get("prov:activity", [])
494        ]
495        usages += [kprov.Usage(e) for e in _ids(node, "prov:used") - {u.entity for u in usages}]
496        generations += [kprov.Generation(e) for e in _ids(node, "prov:generated") - {g.entity for g in generations}]
497        associations = [
498            kprov.Association(
499                q["prov:agent"]["@id"] if "prov:agent" in q else None, _role(q, "senselab:nodeId"), _plan_of(q)
500            )
501            for q in node.get("prov:qualifiedAssociation", [])
502        ]
503        return kprov.activity(
504            self.vocabulary,
505            types=node["@type"],
506            attributes=self.attributes(node_id, node, _ACTIVITY_KEYS),
507            used=usages,
508            generated=generations,
509            associations=associations,
510            started=_time(node.get("prov:startedAtTime")),
511            ended=_time(node.get("prov:endedAtTime")),
512        )
513
514    def attributes(self, node_id: str, node: dict[str, Any], structural: frozenset[str]) -> dict[str, object]:
515        out: dict[str, object] = {}
516        for term, value in node.items():
517            if term in structural:
518                continue
519            prop = self.vocabulary.property(term)

def read_documents(run: str | pathlib._local.Path) -> dict[str, bytes]: 
289    @staticmethod
290    def read_documents(run: str | Path) -> dict[str, bytes]:
291        """Return `{file name: bytes}` of every document in `<run>/prov/`, temporaries of an interrupted write skipped.
292
293        Raises:
294            MalformedDocument: `prov/` cannot be listed, holds an entry that is not a regular file, or a file that
295                cannot be read; the message names an error class, never a path.
296        """
297        directory = Path(run) / PROV_DIR
298        documents: dict[str, bytes] = {}
299        failure: MalformedDocument | None = None
300        try:
301            for path in sorted(directory.iterdir()) if directory.is_dir() else []:
302                if TEMPORARY_NAME.fullmatch(path.name):
303                    continue
304                if not path.is_file() or path.is_symlink():
305                    raise MalformedDocument(None, f"{PROV_DIR}/ holds an entry that is not a regular file")
306                writer = path.name.removesuffix(".jsonld") if WRITER_DOCUMENT.fullmatch(path.name) else None
307                try:
308                    documents[path.name] = path.read_bytes()
309                except OSError as err:
310                    failure = MalformedDocument(writer, f"the document cannot be read ({type(err).__name__})")
311                    break
312        except OSError as err:
313            failure = MalformedDocument(None, f"{PROV_DIR}/ cannot be read ({type(err).__name__})")
314        if failure is not None:
315            raise failure
316        return documents

319    def from_documents(cls, documents: Mapping[str, bytes], vocabulary: kprov.Vocabulary | None = None) -> ProvStore:
320        """Return the union of `documents` (`read_documents`' result), every node rebuilt over `vocabulary`.
321
322        Each node is rebuilt through the record builders over `vocabulary` (the core vocabulary when None) and
323        must be byte-identical to its rebuild, so its `@id`, terms and literals are all checked. Every relation
324        endpoint and every record an attribute names must be a node of the union.
325
326        Raises:
327            ProvenanceFormatRefused: on a document that declares a format other than `provenance@3`.
328            MalformedDocument: on a file that is not a writer's document, a document of another shape, a node
329                that differs from its rebuild, a `senselab:fileLabel` that is not `file-<n>`, or attribute
330                references nested deeper than `MAX_DEPTH`.
331            ConflictingNode: if two writers hold different nodes under one `@id`.
332            DanglingReference: if a node names an `@id` the union does not hold.
333        """
334        nodes: dict[str, tuple[bytes, str]] = {}
335        for name, data in sorted(documents.items()):
336            writer = name.removesuffix(".jsonld") if WRITER_DOCUMENT.fullmatch(name) else None
337            found = _declared_format(data)
338            if (
339                found is not None
340                and check(PROVENANCE_FORMAT, found, RETIRED_PROVENANCE_FORMATS) is not FormatVerdict.ACCEPTED
341            ):
342                raise ProvenanceFormatRefused(found)
343            if writer is None:
344                raise MalformedDocument(None, f"{PROV_DIR}/ holds a file that is not a writer's document")
345            for node_id, raw in _writer_nodes(writer, data):
346                if nodes.setdefault(node_id, (raw, writer))[0] != raw:
347                    raise ConflictingNode(node_id)
348        return cls(_Rebuild(nodes, vocabulary or core_vocabulary()).all())
