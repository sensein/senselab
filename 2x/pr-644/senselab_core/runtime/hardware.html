 71@dataclass(frozen=True)
 72class HardwareInfo:
 73    """Host hardware facts, gathered without importing ``torch``.
 74
 75    ``gpu_probe_error`` is None when the GPU probe found nothing wrong, including when ``nvidia-smi`` is
 76    absent. It carries a message when ``nvidia-smi`` could not run, exited non-zero, timed out or printed

 82    """

 96    """Run ``cmd`` and return its stdout; raise on a missing binary, a timeout or a non-zero exit.

101        from senselab_core.runtime.profiles import ExecutionProfile  # noqa: PLC0415 - profiles import this module

107def _parse_gpus(output: str, *, with_capability: bool) -> tuple[GpuInfo, ...]:
108    """Parse the ``nvidia-smi`` CSV lines; raise ``ValueError`` on any malformed line."""
109    width = 6 if with_capability else 5
110    gpus = []
111    for line in output.splitlines():
112        if not line.strip():
113            continue
114        fields = [f.strip() for f in line.split(",")]
115        if len(fields) != width or not fields[0].isdigit() or not fields[2] or not fields[3].isdigit():
116            raise ValueError(f"unparseable nvidia-smi line: {line!r}")
117        capability = fields[5] if with_capability and re.fullmatch(r"\d+\.\d+", fields[5]) else None
118        gpus.append(
119            GpuInfo(
120                name=fields[2],
121                memory_bytes=int(fields[3]) * MIB,
122                driver=fields[4] or None,
123                compute_capability=capability,
124                uuid=fields[1] or None,
125                index=int(fields[0]),
126            )
127        )
128    return tuple(gpus)

131def _probe_gpus(run: Callable[[list[str]], str]) -> tuple[tuple[GpuInfo, ...], Optional[str]]:
132    note: Optional[str] = None
133    query, with_capability = _NVIDIA_SMI_GPU_QUERY, True
134    try:
135        try:
136            output = run(query)
137        except subprocess.CalledProcessError:
138            query, with_capability = _NVIDIA_SMI_GPU_QUERY_NO_CAP, False
139            output = run(query)
140            note = "nvidia-smi rejected compute_cap; compute capability unknown"
141            logger.debug(note)
142    except FileNotFoundError:
143        return (), None
144    except PermissionError as exc:
145        message = f"nvidia-smi could not be executed: {exc}"
146    except subprocess.CalledProcessError as exc:
147        message = f"nvidia-smi exited {exc.returncode}"
148    except subprocess.TimeoutExpired:
149        message = "nvidia-smi timed out"
150    else:
151        try:
152            return _parse_gpus(output, with_capability=with_capability), note
153        except ValueError as exc:
154            message = str(exc)
155    logger.warning("GPU probe found no GPUs: %s", message)
156    return (), message

159def _physical_memory_bytes() -> tuple[int, Optional[str]]:
160    try:

215def probe_hardware(
216    *,
217    timeout_s: Optional[float] = None,
218    run: Optional[Callable[[list[str]], str]] = None,
219    cpuinfo: Callable[[], str] = _read_cpuinfo,
220    cpus: Callable[[], int] = process_cpus,
221) -> HardwareInfo:
222    """Probe CPU, memory, GPUs, Apple chip and platform; never imports ``torch``.
223
224    ``run`` executes a command (the ``nvidia-smi`` query, ``sysctl`` on Apple silicon) and returns
225    stdout, by default bounded by ``timeout_s`` seconds (the packaged profile's ``sizing.device_query_s`` when
226    None), raising on a missing binary, a timeout or a non-zero exit; ``cpuinfo`` reads ``/proc/cpuinfo``;
227    ``cpus`` counts the process's CPUs; tests inject all three.
228    """
229    if run is None:
230        run = functools.partial(_run, timeout_s=timeout_s)
231    system = platform.system()
232    machine = platform.machine()
233    gpus, gpu_error = _probe_gpus(run)
234    memory, memory_error = _physical_memory_bytes()
235    apple_silicon = system == "Darwin" and machine == "arm64"
236    chip, chip_error = _probe_chip(run) if apple_silicon else (None, None)
237    errors = [e for e in (memory_error, chip_error) if e]
238    return HardwareInfo(
239        cpu_count=cpus(),
240        memory_bytes=memory,
241        gpus=gpus,
242        apple_silicon=apple_silicon,
243        platform=f"{system.lower()}-{machine}",
244        gpu_probe_error=gpu_error,
245        chip=chip,
246        probe_error="; ".join(errors) or None,
247        cpu_model=chip or _cpu_model(run, system, cpuinfo),
248    )

251def visible_gpus(info: HardwareInfo, environ: Optional[Mapping[str, str]] = None) -> list[GpuInfo]:
252    """The GPUs torch would enumerate: all of them, or those ``CUDA_VISIBLE_DEVICES`` names, in its order.
253
254    Raises:
255        DeviceResolutionError: an entry of ``CUDA_VISIBLE_DEVICES`` names no GPU, or several.
256    """
257    environ = os.environ if environ is None else environ
258    raw = environ.get("CUDA_VISIBLE_DEVICES")
259    if raw is None:
260        return list(info.gpus)
261    visible = []
262    for entry in (e.strip() for e in raw.split(",") if e.strip()):
263        if entry.isdigit():
264            matches = [g for g in info.gpus if g.index == int(entry)]
265        elif entry.upper().startswith("GPU-"):
266            matches = [g for g in info.gpus if g.uuid and g.uuid.upper().startswith(entry.upper())]
267        else:
268            raise DeviceResolutionError(f"CUDA_VISIBLE_DEVICES entry {entry!r} is neither an index nor a GPU uuid")
269        if len(matches) != 1:
270            raise DeviceResolutionError(f"CUDA_VISIBLE_DEVICES entry {entry!r} matches {len(matches)} GPUs")
271        visible.append(matches[0])
272    return visible

301def device_free_memory(
302    devices: Optional[Sequence[str]] = None,
303    environ: Optional[Mapping[str, str]] = None,
304    *,
305    run: Callable[[list[str]], str] = _run,
306) -> Optional[tuple[DeviceMemory, ...]]:
307    """Return the free and total memory of `devices` (``CUDA_VISIBLE_DEVICES`` entries; None: those this process sees).
308
309    Read from ``nvidia-smi --query-gpu``; empty without ``nvidia-smi``; None, logged, where it fails or its output
310    cannot be parsed.
311
312    Raises:
313        DeviceResolutionError: a device names no GPU, or several.
314    """
315    try:
316        output = run(_NVIDIA_SMI_FREE_QUERY)
317    except FileNotFoundError:
318        return ()
319    except (OSError, subprocess.SubprocessError) as exc:
320        logger.warning("device memory unreadable: %s", exc)
321        return None
322    rows = [[f.strip() for f in line.split(",")] for line in output.splitlines() if line.strip()]
323    if any(len(r) != len(_FREE_FIELDS) or not (r[0].isdigit() and r[2].isdigit() and r[3].isdigit()) for r in rows):
324        logger.warning("device memory unreadable: %r", output)
325        return None
326    gpus = [
327        GpuInfo(name="", memory_bytes=int(r[3]) * MIB, driver=None, uuid=r[1] or None, index=int(r[0])) for r in rows
328    ]
329    free = {int(r[0]): int(r[2]) * MIB for r in rows}
330    environ = os.environ if environ is None else environ
331    named = environ if devices is None else {"CUDA_VISIBLE_DEVICES": ",".join(devices)}
332    info = HardwareInfo(cpu_count=1, memory_bytes=0, gpus=tuple(gpus), apple_silicon=False, platform="")
333    return tuple(
334        DeviceMemory(g.uuid or str(g.index), free[g.index], g.memory_bytes)
335        for g in visible_gpus(info, named)
336        if g.index is not None
337    )

346def numerics_class(
347    info: HardwareInfo, device: str, precision: str, *, environ: Optional[Mapping[str, str]] = None
348) -> dict[str, str]:
349    """Portable numerics facts for an execution context: device kind, precision, architecture.
350
351    ``device`` is a torch-style string (``cpu``, ``cuda``, ``cuda:1``, ``mps``). ``cuda:N`` indexes the
352    GPUs visible under ``CUDA_VISIBLE_DEVICES`` (read from ``environ``, default ``os.environ``);
353    ``gpu_architecture`` is ``sm_<major><minor>``, or the product name when the compute capability is
354    unknown. On ``mps`` it is the Apple chip, omitted when the chip is unknown. The result never feeds a
355    logical key.
356
357    Raises:
358        DeviceResolutionError: ``cuda:N`` is out of range, malformed, or names a visible-device entry that
359            cannot be mapped.
360    """
361    kind, _, index = device.partition(":")
362    facts = {"device_kind": kind, "precision": precision}
363    if kind == "cuda":
364        if index and not index.isdigit():
365            raise DeviceResolutionError(f"unparseable device {device!r}")
366        visible = visible_gpus(info, environ)
367        position = int(index) if index else 0
368        if position >= len(visible):
369            raise DeviceResolutionError(f"{device!r} is out of range: {len(visible)} visible GPU(s)")
370        gpu = visible[position]
371        capability = gpu.compute_capability
372        facts["gpu_architecture"] = f"sm_{capability.replace('.', '')}" if capability else gpu.name
373    elif kind == "mps" and info.chip:
374        facts["gpu_architecture"] = info.chip
375    return facts

382def cgroup_memory_events(proc_cgroup: Path = Path("/proc/self/cgroup")) -> Optional[Path]:
383    """Return the `memory.events` file of this process's cgroup v2, None where there is none."""
384    try:
385        lines = proc_cgroup.read_text(encoding="utf-8").splitlines()
386    except OSError:
387        return None
388    for line in lines:
389        if line.startswith("0::"):
390            path = CGROUP_ROOT / line[3:].lstrip("/") / "memory.events"
391            return path if path.is_file() else None
392    return None

395def oom_kills(memory_events: Optional[Path]) -> Optional[int]:
396    """Return the `oom_kill` count a cgroup `memory.events` file records, None when it cannot be read."""
397    if memory_events is None:
398        return None

403    counts = dict(line.split(maxsplit=1) for line in lines if len(line.split()) == 2)  # noqa: PLR2004 - name, count

408def memory_bytes(proc_cgroup: Path = Path("/proc/self/cgroup")) -> tuple[Optional[int], int]:
409    """Return this process's memory now (None where unreadable) and its high-water, in bytes.
410
411    From its cgroup v2 (`memory.current`, `memory.peak`), which counts every process of the cgroup; elsewhere no
412    current figure and the largest resident set of this process and of its waited-for children.
413    """
414    events = cgroup_memory_events(proc_cgroup)
415    if events is not None:
416        try:
417            current = int((events.parent / "memory.current").read_text(encoding="utf-8"))
418            peak = int((events.parent / "memory.peak").read_text(encoding="utf-8"))
419        except (OSError, ValueError):
420            pass
421        else:
422            return current, max(current, peak)
423    unit = 1 if platform.system() == "Darwin" else _KIB
424    return None, unit * max(
425        resource.getrusage(who).ru_maxrss for who in (resource.RUSAGE_SELF, resource.RUSAGE_CHILDREN)
426    )

429def proc_stats() -> Iterator[tuple[int, list[bytes]]]:
430    """Yield each process's pid and the fields of its `/proc/<pid>/stat` after the command name; none off `/proc`."""
431    for entry in os.listdir("/proc") if os.path.isdir("/proc/self") else ():
432        if not entry.isdigit():
433            continue
434        try:
435            with open(f"/proc/{entry}/stat", "rb") as f:
436                data = f.read()
437        except OSError:
438            continue
439        yield int(entry), data[data.rfind(b")") + 2 :].split()

442def process_tree(root: Optional[int] = None) -> set[int]:
443    """Return the pid `root` (this process by default) and its descendants from `/proc`; `root` alone elsewhere."""
444    root = os.getpid() if root is None else root
445    parents: dict[int, list[int]] = {}
446    for pid, fields in proc_stats():
447        if len(fields) > _STAT_PPID:
448            parents.setdefault(int(fields[_STAT_PPID]), []).append(pid)
449    found, todo = {root}, [root]
450    while todo:
451        for child in parents.get(todo.pop(), ()):
452            if child not in found:
453                found.add(child)
454                todo.append(child)
455    return found

481def device_parties(
482    uuids: Collection[str],
483    workers: Mapping[int, Collection[int]],
484    own: Collection[int],
485    *,
486    timeout_s: float,
487    run: Callable[[list[str], float], str] = _bounded,
488) -> Optional[int]:
489    """Return the parties with compute processes on the GPUs `uuids`, the most on any one; None where unread.
490
491    A party is a pool worker: the pids of `workers` (a worker's pid and its process tree, by worker) and of `own` (the
492    caller's) each count once however many processes they hold; a pid of neither is a party of its own. The caller
493    counts once though it may hold no process on the GPU yet. Read from `nvidia-smi --query-compute-apps` within
494    `timeout_s`.
495    """
496    try:
497        output = run(_NVIDIA_SMI_PLACED_QUERY, timeout_s)
498    except (OSError, subprocess.SubprocessError):
499        return None
500    rows = [[p.strip() for p in line.split(",")] for line in output.splitlines()]
501    placed = [(int(r[0]), r[1]) for r in rows if len(r) == len(_PLACED_FIELDS) and r[0].isdigit()]
502    mine = set(own)
503
504    def party(pid: int) -> object:
505        if pid in mine:
506            return "self"
507        return next((w for w, pids in workers.items() if pid in pids), ("pid", pid))
508
509    return max((len({party(p) for p, u in placed if u == uuid} | {"self"}) for uuid in uuids), default=1)
