senselab_core.runtime.licenses

SPDX license expressions, the license records derived from them, and a user's license policy (P-23).

parse_expression, ids_of and allows read an expression; derive_flags reads its flags from the packaged table data/spdx_licenses.yaml; license_record builds a License whose flags are those; check_license refuses a record whose flags contradict its expression; backend_license derives a backend's code license from its env lock's package records, which lock_records_from reads from an env-licenses document. LicensePolicy is a run's own allowed and refused license ids; policy_document and policy_of write and read it as the run's licenses.json, and laxer_than says in which field one policy allows what another refuses.

 38class _Parser:
 39    def __init__(self, text: str) -> None:
 40        self.text, self.tokens, self.pos = text, _TOKEN.findall(text), 0
 41
 42    def peek(self) -> str | None:
 43        return self.tokens[self.pos] if self.pos < len(self.tokens) else None
 44
 45    def take(self) -> str:
 46        if self.pos >= len(self.tokens):
 47            raise ValueError(f"expression ends early: {self.text!r}")
 48        self.pos += 1
 49        return str(self.tokens[self.pos - 1])
 50
 51    def atom(self) -> Any:  # noqa: ANN401
 52        if self.peek() == "(":
 53            self.take()
 54            inner = self.either()
 55            if self.peek() != ")":
 56                raise ValueError(f"unbalanced parentheses in {self.text!r}")
 57            self.take()
 58            return inner
 59        if self.peek() is None or self.peek() in (")", "AND", "OR", "WITH"):
 60            raise ValueError(f"expected a license id in {self.text!r}")
 61        name = self.take()
 62        if self.peek() == "WITH":
 63            self.take()
 64            name = f"{name} WITH {self.take()}"
 65        return name
 66
 67    def both(self) -> Any:  # noqa: ANN401
 68        node = self.atom()
 69        while self.peek() == "AND":
 70            self.take()
 71            node = ("and", node, self.atom())
 72        return node
 73
 74    def either(self) -> Any:  # noqa: ANN401
 75        node = self.both()
 76        while self.peek() == "OR":
 77            self.take()
 78            node = ("or", node, self.both())
 79        return node

 82def parse_expression(text: str) -> Any:  # noqa: ANN401
 83    """Parse an SPDX expression into nested tuples: `("or", a, b)`, `("and", a, b)` or an id string.
 84
 85    Raises:
 86        ValueError: if the expression is empty, unbalanced, ends early or has trailing text.
 87    """
 88    parser = _Parser(text)
 89    tree = parser.either()
 90    if parser.peek() is not None:
 91        raise ValueError(f"trailing text in {text!r}")
 92    return tree

 95def ids_of(tree: Any) -> list[str]:  # noqa: ANN401
 96    """Every license id in a parsed expression."""
 97    return [tree] if isinstance(tree, str) else [i for part in tree[1:] for i in ids_of(part)]

100def allows(tree: Any, allowed: frozenset[str]) -> bool:  # noqa: ANN401
101    """Whether the expression is satisfied by the `allowed` lower-case ids: `AND` needs both sides, `OR` either."""
102    if isinstance(tree, str):
103        return tree.lower() in allowed
104    left, right = allows(tree[1], allowed), allows(tree[2], allowed)
105    return (left and right) if tree[0] == "and" else (left or right)

108@cache
109def spdx_table() -> Mapping[str, Flags]:
110    """Return the packaged table, `lower-case id -> Flags`.
111
112    Raises:
113        ValueError: if the file lacks its format, a group lacks a derivation, or a group's flags are not valid.
114    """
115    raw = load_yaml(
116        resources.files("senselab_core.runtime").joinpath("data", "spdx_licenses.yaml"), "spdx_licenses.yaml"
117    )
118    if raw.get("format") != SPDX_TABLE_FORMAT:
119        raise ValueError(f"spdx_licenses.yaml must carry format {SPDX_TABLE_FORMAT!r}")
120    table: dict[str, Flags] = {}
121    for group in raw["groups"]:
122        if not str(group.get("derivation", "")).strip():
123            raise ValueError(f"a license group needs a derivation: {group.get('ids')}")
124        commercial = {True: "yes", False: "no"}.get(group["commercial_use"], group["commercial_use"])
125        if commercial not in _COMMERCIAL or group["copyleft"] not in _COPYLEFT:
126            raise ValueError(f"a license group has an unknown flag value: {group['ids']}")
127        flags = Flags(commercial, group["copyleft"], bool(group["attribution"]))
128        for license_id in group["ids"]:
129            if table.setdefault(license_id.lower(), flags) is not flags:
130                raise ValueError(f"license id {license_id!r} is in two groups")
131    return table
132

134def _flags_of(tree: Any, table: Mapping[str, Flags], expression: str) -> Flags:  # noqa: ANN401
135    if isinstance(tree, str):
136        if tree.lower() not in table:
137            raise ValueError(f"license id {tree!r} in {expression!r} is not in the license table")
138        return table[tree.lower()]
139    a, b = _flags_of(tree[1], table, expression), _flags_of(tree[2], table, expression)
140    if tree[0] == "and":
141        return Flags(
142            min(a.commercial_use, b.commercial_use, key=_COMMERCIAL.index),
143            max(a.copyleft, b.copyleft, key=_COPYLEFT.index),
144            a.attribution or b.attribution,
145        )
146    return Flags(
147        max(a.commercial_use, b.commercial_use, key=_COMMERCIAL.index),
148        min(a.copyleft, b.copyleft, key=_COPYLEFT.index),
149        a.attribution and b.attribution,
150    )

153def derive_flags(expression: str) -> Flags:
154    """Return the flags of `expression`: `AND` takes the stricter side of each flag, `OR` the more permissive.

156    Raises:

159    return _flags_of(parse_expression(expression), spdx_table(), expression)

162def check_license(record: License) -> License:
163    """Return `record`, or raise `ValueError` when its flags differ from those its expression derives."""
164    derived = derive_flags(record.expression)
165    given = Flags(record.commercial_use, record.copyleft, record.attribution)
166    if given != derived:
167        raise ValueError(f"license flags {tuple(given)} contradict {record.expression!r}, which gives {tuple(derived)}")
168    return record

171def license_record(
172    expression: str, *, source: str, verified: Mapping[str, str], gated: bool, flags: Mapping[str, Any] | None = None
173) -> License:
174    """Return the `License` of `expression` with its flags derived; `flags`, when given, must equal them.
175
176    Raises:
177        ValueError: if the expression is malformed, names an unknown id, or `flags` contradict it.
178    """
179    derived = derive_flags(expression)
180    record = License.model_validate(
181        {
182            **derived._asdict(),
183            **(flags or {}),
184            "expression": expression,
185            "source": source,
186            "verified": verified,
187            "gated": gated,
188        }
189    )
190    return check_license(record)

193def backend_license(backend: BackendSpec, lock_records: Mapping[str, License | None]) -> License:
194    """Return the code license of `backend`: every package of its env lock, `AND`-joined, flags derived.
195
196    `lock_records` maps each package name of the lock (the root lock for `Env.host`) to its record. The
197    record is `manual` verified unless every package record was read as `pep639`, and dated by the latest.
198
199    Raises:
200        MissingLicense: if a package has no record.
201        ValueError: if `lock_records` is empty or a package record's flags contradict its expression.
202    """
203    if not lock_records:
204        raise ValueError(f"{backend.task}/{backend.name}: no lock records to derive a license from")
205    subject = f"{backend.task}/{backend.name}"
206    records = {
207        name: check_license(require_license(r, f"package {name} of {subject}")) for name, r in lock_records.items()
208    }
209    expressions = sorted({r.expression for r in records.values()})
210    how = "pep639" if {r.verified["how"] for r in records.values()} == {"pep639"} else "manual"
211    return license_record(
212        " AND ".join(f"({e})" if len(expressions) > 1 else e for e in expressions),
213        source=f"env lock of {subject}: {len(records)} packages",
214        verified={"date": max(r.verified["date"] for r in records.values()), "how": how},
215        gated=any(r.gated for r in records.values()),
216    )

230def lock_records_from(text: str) -> LockRecords:
231    """Return the provider of ``package -> License`` over the ``env-licenses`` document `text`.
232
233    The document is written by ``scripts/check_licenses.py --lock NAME`` (or ``--host``) from package metadata; its
234    envs are the lock names, and ``host`` for the root lock.
235
236    Raises:
237        ValueError: if the format is not ``env-licenses@1.0``, or (from the provider) a backend's env has no records.
238    """
239    raw = load_yaml(text, "env licenses")
240    if raw.get("format") != ENV_LICENSES_FORMAT:
241        raise ValueError(f"env licenses must carry format {ENV_LICENSES_FORMAT!r}")
242    envs = raw["envs"]
243
244    def records(backend: BackendSpec) -> dict[str, License | None]:
245        name = env_name(backend)
246        if name not in envs:
247            raise ValueError(f"no license records for env {name!r}; run scripts/check_licenses.py --lock {name}")
248        return {
249            package: license_record(
250                entry["expression"], source=entry["source"], verified=entry["verified"], gated=False
251            )
252            for package, entry in envs[name].items()
253        }
254
255    return records

258def packaged_lock_records(package: str) -> LockRecords:
259    """Return the provider over ``data/env_licenses.yaml`` of `package`, read on its first use."""
260    read: list[LockRecords] = []
261
262    def records(backend: BackendSpec) -> dict[str, License | None]:
263        if not read:
264            read.append(
265                lock_records_from(resources.files(package).joinpath("data", "env_licenses.yaml").read_text("utf-8"))

276@dataclass(frozen=True)
277class LicensePolicy:
278    """A run's license policy: the SPDX ids it allows and those it refuses, matched ignoring case.
279
280    `allow` None allows every id `refuse` does not name. A license is accepted when its expression is satisfied by
281    the allowed ids that are not refused (`allows`); a subject with no license record is refused. A stored result
282    whose context records no env license is served only with `allow_unlicensed_hits`.
283    """
284
285    allow: frozenset[str] | None = None
286    refuse: frozenset[str] = frozenset()
287    allow_unlicensed_hits: bool = False
288
289    def refusal(self, subject: str, record: License | None) -> str | None:
290        """Return why the policy refuses `record`, the license of `subject`, or None when it accepts it."""
291        if record is None:
292            return f"{subject} has no license record"
293        tree = parse_expression(record.expression)
294        allowed = {i.lower() for i in (ids_of(tree) if self.allow is None else self.allow)}
295        if allows(tree, frozenset(allowed - {i.lower() for i in self.refuse})):
296            return None
297        return f"{subject} is licensed {record.expression!r}, which the run's license policy does not allow"
298
299    def check(self, subject: str, record: License | None) -> None:
300        """Raise `LicenseRefused` when the policy refuses `record`, the license of `subject`."""
301        if (reason := self.refusal(subject, record)) is not None:
302            raise LicenseRefused(reason)

309class LaxerLicensePolicy(ValueError):
310    """A run was given a license policy that allows what the policy it was planned under refuses.
311
312    Attributes:

314        ids: The license ids the given policy allows that the stored one refuses, empty when the field names no id.
315    """

317    def __init__(self, field: str, ids: tuple[str, ...]) -> None:
318        """Name the field and ids."""
319        named = f" ({', '.join(ids)})" if ids else ""
320        super().__init__(
321            f"the license policy given is laxer than the run's in {field}{named}; a run's policy can be tightened, "
322            "never loosened: plan the run again in a fresh run directory to loosen it"
323        )
324        self.field, self.ids = field, ids

327def policy_document(policy: LicensePolicy | None) -> dict[str, Any]:
328    """Return `licenses.json` for `policy`: ids lower-cased and sorted, the absent policy as null."""
329    body = None
330    if policy is not None:
331        allow = None if policy.allow is None else sorted({i.lower() for i in policy.allow})
332        body = {
333            "allow": allow,
334            "refuse": sorted({i.lower() for i in policy.refuse}),
335            "allow_unlicensed_hits": policy.allow_unlicensed_hits,
336        }
337    return {"format": LICENSES_FORMAT, "policy": body}

345def policy_of(document: object) -> LicensePolicy | None:
346    """Return the policy a `licenses.json` document holds.
347
348    Raises:
349        ValueError: it is not a `run-licenses@1` document.
350    """
351    if not isinstance(document, dict) or document.get("format") != LICENSES_FORMAT or "policy" not in document:
352        raise ValueError(f"not a {LICENSES_FORMAT} document")
353    found = document["policy"]
354    if found is None:
355        return None
356    try:
357        allow, refuse, hits = found["allow"], found["refuse"], found["allow_unlicensed_hits"]
358    except (KeyError, TypeError) as err:
359        raise ValueError(f"a {LICENSES_FORMAT} policy has allow, refuse and allow_unlicensed_hits") from err
360    return LicensePolicy(
361        allow=None if allow is None else frozenset(allow),
362        refuse=frozenset(refuse),
363        allow_unlicensed_hits=bool(hits),
364    )

367def _normal(policy: LicensePolicy | None) -> tuple[frozenset[str] | None, frozenset[str], bool]:
368    """Return `policy` as `(allow, refuse, allow_unlicensed_hits)`; the absent one allows all, no unlicensed hit."""
369    if policy is None:
370        return None, frozenset(), False
371    allow = None if policy.allow is None else frozenset(i.lower() for i in policy.allow)
372    return allow, frozenset(i.lower() for i in policy.refuse), policy.allow_unlicensed_hits

375def laxer_than(given: LicensePolicy | None, stored: LicensePolicy | None) -> LaxerLicensePolicy | None:
376    """Return the error naming how `given` allows what `stored` refuses, None when it is at least as strict.
377
378    A policy accepts the ids it allows (every id when `allow` is None) that it does not refuse, and serves an
379    unlicensed stored result only with `allow_unlicensed_hits`; the absent policy is `(every id, none refused, no
380    unlicensed hit)`. `given` is at least as strict when it accepts no id `stored` does not and serves no hit
381    `stored` does not.
382    """
383    q, p = _normal(given), _normal(stored)
384    if (found := _loosened(q[0], q[1], p[0], p[1])) is not None:
385        return LaxerLicensePolicy(*found)
386    return LaxerLicensePolicy("allow_unlicensed_hits", ()) if q[2] and not p[2] else None

389def _loosened(
390    q_allow: frozenset[str] | None, q_refuse: frozenset[str], p_allow: frozenset[str] | None, p_refuse: frozenset[str]
391) -> tuple[str, tuple[str, ...]] | None:
392    """Return the field and ids by which `(q_allow, q_refuse)` accepts an id `(p_allow, p_refuse)` does not, if any."""
393    if p_allow is not None and q_allow is None:
394        return "allow", ()
395    if p_allow is None:
396        bad = p_refuse - q_refuse if q_allow is None else (q_allow - q_refuse) & p_refuse
397        return ("refuse", tuple(sorted(bad))) if bad else None
398    assert q_allow is not None
399    bad = (q_allow - q_refuse) - (p_allow - p_refuse)
400    if unlisted := bad - p_allow:
401        return "allow", tuple(sorted(unlisted))
402    return ("refuse", tuple(sorted(bad))) if bad else None
