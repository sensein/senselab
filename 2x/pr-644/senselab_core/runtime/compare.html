senselab_core.runtime.compare

Output agreement by standard comparisons: hash equality, exact equality, and isclose within named thresholds.

runtime/data/comparison.yaml (comparison@1.0) declares, per kernel type, how each field of its wire form compares (COMPARISONS); the thresholds its rules name are the run config's (RunConfigFile.load, over the packaged runtime/data/run_config.yaml). compare decides whether two outputs agree and returns the field-level differences, each with the threshold it was judged by. Outputs are compared as their flattened leaves (flatten).

 53def _numeric(value: object) -> bool:
 54    if isinstance(value, (bool, str)):
 55        return False
 56    if isinstance(value, numpy.ndarray):
 57        return value.dtype.kind in "biuf"
 58    try:
 59        numpy.asarray(value, dtype=float)
 60    except (TypeError, ValueError):
 61        return False
 62    return True

 69def flatten(value: object, prefix: str = "") -> Fields:
 70    """Return the leaves of a wire form `value` by dotted path (`FLATTEN_VERSION`).
 71
 72    Numbers, None and numeric arrays are float lists, a None (or NaN) where a value is absent; a non-numeric
 73    array is flattened element by element, like a list; a string or bool is its JSON.
 74
 75    Raises:
 76        NotWireForm: a leaf is none of these (a kernel value compares as its `wire_form`); it names the path.
 77    """
 78    if isinstance(value, numpy.ndarray) and not _numeric(value):
 79        return flatten(value.tolist(), prefix)
 80    if isinstance(value, Mapping):
 81        out: Fields = {}
 82        for key in sorted(value, key=str):
 83            out |= flatten(value[key], f"{prefix}.{key}" if prefix else str(key))
 84        return out
 85    if isinstance(value, (list, tuple)) and not _numeric(value):
 86        out = {}
 87        for n, element in enumerate(value):
 88            out |= flatten(element, f"{prefix}[{n}]")
 89        return out
 90    if _numeric(value):
 91        floats = numpy.asarray(value, dtype=float).ravel().tolist()
 92        return {prefix: [None if math.isnan(x) else x for x in floats]}
 93    if isinstance(value, (str, bool)):
 94        return {prefix: json.dumps(value)}
 95    raise NotWireForm(f"{prefix or 'the value'} is a {type(value).__name__}, not a wire form")

137    def load(cls, override: Mapping[str, Any] | Path | None = None) -> "RunConfigFile":
138        """Return the packaged run config with `override` (a run config mapping or YAML file) deep-merged over it.
139
140        A threshold of `override` may set `rtol`, `atol` or both; `calibration` may set `trust`.

154        _known("calibration key", merged.get("calibration"), RunCalibration.model_fields)

163def _known(what: str, given: object, known: Iterable[str]) -> None:
164    if not isinstance(given, Mapping):
165        return
166    names = sorted(known)
167    for name in sorted(set(map(str, given)) - set(names)):
168        closest = difflib.get_close_matches(name, names, n=1)
169        raise UnknownThreshold(f"unknown {what} {name!r}" + (f"; did you mean {closest[0]!r}?" if closest else ""))

192        """Return the rule of the first pattern matching the index-free `relative` path, or None."""

197def _pattern(text: str) -> re.Pattern[str]:
198    return re.compile(".*".join(map(re.escape, text.split("*"))))

201def _rule(where: str, raw: object) -> Rule:
202    if raw == "exact":
203        return Rule("exact")
204    if raw == "optional":
205        return Rule("optional")
206    if isinstance(raw, Mapping) and set(raw) == {"close"} and isinstance(raw["close"], str) and raw["close"]:
207        return Rule("close", raw["close"])
208    raise ValueError(
209        f"{COMPARISON_FORMAT} {where}: a rule is 'exact', 'optional' or {{close: <threshold name>}}, got {raw!r}"
210    )

213def parse_comparisons(text: str) -> Mapping[str, TypeComparison]:
214    """Return the per-type comparisons a `comparison@1.0` document declares, keyed by `<name>@<major>`.
215
216    Raises:
217        ValueError: the document is not of that format and form.
218    """
219    raw = load_yaml(text, "comparison table")
220    if raw.get("format") != COMPARISON_FORMAT or not isinstance(raw.get("types"), dict):
221        raise ValueError(f"a comparison table is a mapping of format {COMPARISON_FORMAT!r} and 'types'")
222    out = {}
223    for name, entry in raw["types"].items():
224        if (
225            not isinstance(entry, dict)
226            or not isinstance(entry.get("fields"), dict)
227            or set(entry) - {"fields", "match_labels"}
228        ):
229            raise ValueError(f"{COMPARISON_FORMAT} type {name}: a mapping of 'fields' and an optional 'match_labels'")
230        labels = entry.get("match_labels")
231        if labels is not None and (not isinstance(labels, dict) or set(labels) != {"label", "start", "end"}):
232            raise ValueError(f"{COMPARISON_FORMAT} type {name}: match_labels names 'label', 'start' and 'end'")
233        out[str(name)] = TypeComparison(
234            fields=tuple((str(p), _rule(f"{name}.{p}", r)) for p, r in entry["fields"].items()),
235            match_labels=None if labels is None else (labels["label"], labels["start"], labels["end"]),
236        )
237    return out

266def _type_of(leaf: Value | None, comparisons: Comparisons) -> str | None:
267    if not isinstance(leaf, str):
268        return None
269    try:
270        tag = json.loads(leaf)
271    except ValueError:
272        return None
273    if not isinstance(tag, str) or re.fullmatch(FORMAT_PATTERN, tag) is None:
274        return None
275    key = tag.split(".", 1)[0]
276    if key in comparisons:
277        return key
278    if key in _KERNEL_TYPES:
279        raise UndeclaredComparison(tag)
280    return None

283def typed_values(fields: Mapping[str, Value], comparisons: Comparisons | None = None) -> dict[str, str]:
284    """Return the path prefixes holding a kernel value's wire form, each with its type's `<name>@<major>`.
285
286    A kernel value is found by its `format` leaf under any path; the root's prefix is `""`. `comparisons` defaults to
287    `COMPARISONS`.
288
289    Raises:
290        UndeclaredComparison: a kernel value's type has no entry in `comparisons`.
291    """
292    table = COMPARISONS if comparisons is None else comparisons
293    out = {}
294    for path, leaf in fields.items():
295        prefix, _, name = path.rpartition(".")
296        if name == "format" and (kind := _type_of(leaf, table)) is not None:
297            out[prefix] = kind
298    return out

306def best_assignment(weights: Sequence[Sequence[float]]) -> list[tuple[int, int]]:
307    """Return the one-to-one (row, column) pairs of largest total weight, as many as the smaller side.
308
309    With a tie it returns the optimum whose sorted `(row, column)` sequence is lexically smallest.
310    """
311    n, m = len(weights), len(weights[0]) if weights else 0
312    wanted = min(n, m)
313    best = _total(weights, _hungarian(weights))
314    chosen: list[tuple[int, int]] = []
315    for i in range(n):
316        if len(chosen) == wanted:
317            break
318        used = {c for _, c in chosen}
319        free = [j for j in range(m) if j not in used]
320        for j in free:
321            rest_rows = list(range(i + 1, n))
322            rest_cols = [c for c in free if c != j]
323            if min(len(rest_rows), len(rest_cols)) != wanted - len(chosen) - 1:
324                continue
325            sub = [[weights[r][c] for c in rest_cols] for r in rest_rows]
326            total = _total(weights, chosen) + weights[i][j] + _total(sub, _hungarian(sub))
327            if math.isclose(total, best, rel_tol=1e-12, abs_tol=1e-12):
328                chosen.append((i, j))
329                break
330    if len(chosen) != wanted:
331        raise ValueError(f"no optimal assignment of {wanted} pairs found in the weights")
332    return chosen

335def _total(weights: Sequence[Sequence[float]], pairs: Sequence[tuple[int, int]]) -> float:
336    return math.fsum(weights[i][j] for i, j in pairs)

339def _hungarian(weights: Sequence[Sequence[float]]) -> list[tuple[int, int]]:
340    """Return one optimal assignment of `weights` (Hungarian method), as many pairs as the smaller side."""
341    n, m = len(weights), len(weights[0]) if weights else 0
342    size = max(n, m)
343    top = max((w for row in weights for w in row), default=0.0)
344    cost = [[top - (weights[i][j] if i < n and j < m else 0.0) for j in range(size)] for i in range(size)]
345    u, v, match, way = [0.0] * (size + 1), [0.0] * (size + 1), [0] * (size + 1), [0] * (size + 1)
346    for i in range(1, size + 1):
347        match[0], column = i, 0
348        least, used = [float("inf")] * (size + 1), [False] * (size + 1)
349        while match[column]:
350            used[column], row, delta, nxt = True, match[column], float("inf"), 0
351            for j in range(1, size + 1):
352                if not used[j]:
353                    reduced = cost[row - 1][j - 1] - u[row] - v[j]
354                    if reduced < least[j]:
355                        least[j], way[j] = reduced, column
356                    if least[j] < delta:
357                        delta, nxt = least[j], j
358            for j in range(size + 1):
359                if used[j]:
360                    u[match[j]] += delta
361                    v[j] -= delta
362                else:
363                    least[j] -= delta
364            column = nxt
365        while column:
366            previous = way[column]
367            match[column] = match[previous]
368            column = previous
369    return sorted((match[j] - 1, j - 1) for j in range(1, size + 1) if match[j] - 1 < n and j - 1 < m)

372def _relabel_one(
373    expected: Mapping[str, Value], actual: Mapping[str, Value], prefix: str, columns: tuple[str, str, str]
374) -> Fields:
375    base = f"{prefix}." if prefix else ""
376    label = re.compile(re.escape(f"{base}{columns[0]}") + r"\[(\d+)\]")
377    times = (f"{base}{columns[1]}", f"{base}{columns[2]}")
378
379    def keys(fields: Mapping[str, object]) -> list[str]:
380        found = [(int(m.group(1)), k) for k in fields if (m := label.fullmatch(k))]
381        return [k for _, k in sorted(found)]
382
383    ek, ak = keys(expected), keys(actual)
384    if not ek or not ak or any(t not in expected or t not in actual for t in times):
385        return dict(actual)
386    elabel = [json.loads(str(expected[k])) for k in ek]
387    alabel = [json.loads(str(actual[k])) for k in ak]
388    rows, cols = sorted(set(elabel)), sorted(set(alabel))
389    overlap = [[0.0] * len(cols) for _ in rows]
390    spans = [[list(f[t]) for t in times] for f in (expected, actual)]
391    if any(len(s) != len(k) for s, k in ((spans[0][0], ek), (spans[1][0], ak))):
392        return dict(actual)
393    for i, (es, ee) in enumerate(zip(*spans[0], strict=True)):
394        for j, (a_s, a_e) in enumerate(zip(*spans[1], strict=True)):
395            if None not in (es, ee, a_s, a_e):
396                overlap[rows.index(elabel[i])][cols.index(alabel[j])] += max(0.0, min(ee, a_e) - max(es, a_s))
397    mapped = {cols[c]: rows[r] for r, c in best_assignment(overlap) if overlap[r][c] > 0}
398    out = dict(actual)
399    for k, name in zip(ak, alabel, strict=True):
400        out[k] = json.dumps(mapped.get(name, f"<unmatched:{name}>"))
401    return out

404def relabel(
405    expected: Mapping[str, Value], actual: Mapping[str, Value], comparisons: Comparisons | None = None
406) -> Fields:
407    """Return `actual` with the labels of each kernel value that declares `match_labels` renamed to `expected`'s.
408
409    Labels are arbitrary per run, so they are mapped one to one by most shared time (the map DER scores under); a
410    label with no counterpart (no shared time) is renamed to one no expected label equals. `actual` is unchanged
411    where either side has no labelled spans.
412    """
413    table = COMPARISONS if comparisons is None else comparisons
414    out = dict(actual)
415    typed = typed_values(expected, table) | typed_values(actual, table)
416    for prefix, kind in sorted(typed.items()):
417        columns = table[kind].match_labels
418        if columns is not None:
419            out = _relabel_one(expected, out, prefix, columns)
420    return out

456SUMMARY_VALUES = 4

462def _summary(leaf: Value | None) -> str:
463    if leaf is None:
464        return "absent"
465    if isinstance(leaf, str):
466        return leaf if len(leaf) <= SUMMARY_CHARS else f"{leaf[:SUMMARY_CHARS]}..."
467    if len(leaf) <= SUMMARY_VALUES:
468        return repr(leaf)
469    present = [x for x in leaf if x is not None]
470    return f"{len(leaf)} values in [{min(present):g}, {max(present):g}]" if present else f"{len(leaf)} absent values"

473def _gap(a: Value, b: Value) -> tuple[float | None, float | None]:
474    """Return the largest absolute and relative difference over the paired numbers of `a` and `b`, None if undefined."""
475    if isinstance(a, str) or isinstance(b, str) or len(a) != len(b):
476        return None, None
477    pairs = [(x, y) for x, y in zip(a, b, strict=True) if x is not None and y is not None]
478    if not pairs:
479        return None, None
480    want, got = (numpy.asarray(side, dtype=float) for side in zip(*pairs, strict=True))
481    with numpy.errstate(invalid="ignore"):
482        gap = numpy.abs(got - want)
483        nonzero = want != 0
484        absolute = float(gap.max())
485        relative = float((gap[nonzero] / numpy.abs(want[nonzero])).max()) if nonzero.any() else None
486    return tuple(x if x is not None and math.isfinite(x) else None for x in (absolute, relative))  # type: ignore[return-value]

489def _exact(a: Value, b: Value) -> str | None:
490    if a == b:
491        return None
492    if isinstance(a, str) or isinstance(b, str):
493        return "not equal"
494    if len(a) != len(b):
495        return f"{len(a)} values against {len(b)}"
496    unequal = [n for n, (x, y) in enumerate(zip(a, b, strict=True)) if x != y]
497    return f"{len(unequal)} of {len(a)} values not equal, first at [{unequal[0]}]"

500def _close(a: Value, b: Value, bound: Threshold) -> str | None:
501    if isinstance(a, str) or isinstance(b, str):
502        return None if a == b else "not equal"
503    if len(a) != len(b):
504        return f"{len(a)} values against {len(b)}"
505    if any((x is None) != (y is None) for x, y in zip(a, b, strict=True)):
506        return "an absent value against a number"
507    pairs = [(x, y) for x, y in zip(a, b, strict=True) if x is not None and y is not None]
508    if not pairs:
509        return None
510    want, got = (numpy.asarray(side, dtype=float) for side in zip(*pairs, strict=True))
511    close = numpy.isclose(got, want, rtol=bound.rtol, atol=bound.atol)
512    if close.all():
513        return None
514    gap = numpy.nan_to_num(numpy.abs(got - want), nan=numpy.inf)
515    largest = float(numpy.where(close, 0.0, gap).max())
516    return (
517        f"{int((~close).sum())} of {len(pairs)} values beyond rtol {bound.rtol:g} atol {bound.atol:g},"
518        f" largest |difference| {largest:g}"
519    )

522def _judge(path: str, a: Value | None, b: Value | None, rule: Rule, thresholds: Thresholds) -> Difference | None:
523    if a is None or b is None:
524        return Difference(
525            path=path,
526            rule="missing",
527            threshold=None,
528            bound=None,
529            expected=_summary(a),
530            actual=_summary(b),
531            absolute=None,
532            relative=None,
533            detail="present on one side only",
534        )
535    bound = thresholds.by_name[rule.threshold] if rule.threshold is not None else None
536    why = _close(a, b, bound) if bound is not None else _exact(a, b)
537    if why is None:
538        return None
539    absolute, relative = _gap(a, b)
540    return Difference(
541        path=path,
542        rule=rule.kind,
543        threshold=rule.threshold,
544        bound=bound,
545        expected=_summary(a),
546        actual=_summary(b),
547        absolute=absolute,

549        detail=why,

553def _optional_gaps(
554    expected: Mapping[str, Value], actual: Mapping[str, Value], rules: Mapping[str, Rule]
555) -> dict[str, Difference]:
556    """Return a difference per `optional` parent that is a `[None]` leaf on one side and nested leaves on the other."""
557    out = {}
558    for path, rule in rules.items():
559        if rule.kind != "optional" or (path in expected) == (path in actual):
560            continue
561        absent, present = (expected, actual) if path in expected else (actual, expected)
562        if absent[path] != [None] or not any(k.startswith(f"{path}.") for k in present):
563            continue
564        out[path] = Difference(
565            path=path,
566            rule="optional",
567            threshold=None,
568            bound=None,
569            expected="absent" if absent is expected else "present",
570            actual="absent" if absent is actual else "present",
571            absolute=None,
572            relative=None,
573            detail="absent on one side, present on the other",
574        )
575    return out

578def compare_fields(
579    expected: Mapping[str, Value],
580    actual: Mapping[str, Value],
581    thresholds: Thresholds,
582    comparisons: Comparisons | None = None,
583) -> Comparison:
584    """Return whether flattened outputs agree, leaf by leaf under each kernel type's declared comparison.
585
586    A leaf under a kernel value (found by its `format` leaf) is judged by its type's first matching rule in
587    `comparisons` (default `COMPARISONS`); any other leaf is exact. Labels a type matches are mapped first
588    (`relabel`). Equal leaves agree before any rule is looked up. A leaf on one side only is a difference; an
589    `optional` parent absent on one side and present on the other is one difference at the parent.
590
591    Raises:
592        UndeclaredComparison: a kernel value's type has no entry in `comparisons`, or a leaf that differs lies under
593            a type that declares no rule for its path.
594        KeyError: a rule names a threshold `thresholds` lacks.
595    """
596    table = COMPARISONS if comparisons is None else comparisons
597    digest = thresholds.canonical_digest()
598    typed = typed_values(expected, table) | typed_values(actual, table)
599    if expected == actual:
600        return Comparison(agree=True, differences=(), thresholds=digest)
601    actual = relabel(expected, actual, table)
602    prefixes = sorted(typed, key=len, reverse=True)
603    paths = sorted(expected.keys() | actual.keys())
604    owners = {p: _owner(p, prefixes) for p in paths}
605    rules: dict[str, Rule] = {}
606    for path in paths:
607        owner = owners[path]
608        found_rule = (
609            Rule("exact") if owner is None else table[typed[owner]].rule(path[len(owner) + 1 :] if owner else path)
610        )
611        if found_rule is None:
612            if expected.get(path) == actual.get(path):
613                continue
614            raise UndeclaredComparison(typed[owner or ""], path)
615        rules[path] = found_rule
616    gaps = _optional_gaps(expected, actual, rules)
617    found = list(gaps.values())
618    for path in paths:
619        if path not in rules or path in gaps or any(path.startswith(f"{p}.") for p in gaps):
620            continue
621        a, b = expected.get(path), actual.get(path)
622        if a is not None and a == b:
623            continue
624        if (difference := _judge(path, a, b, rules[path], thresholds)) is not None:
625            found.append(difference)
626    found.sort(key=lambda d: d.path)
627    return Comparison(agree=not found, differences=tuple(found), thresholds=digest)

630def compare(
631    expected: Mapping[str, object],
632    actual: Mapping[str, object],
633    thresholds: Thresholds,
634    comparisons: Comparisons | None = None,
635) -> Comparison:
636    """Return whether two outputs agree: equal hashes agree at once; otherwise `compare_fields` of their leaves.
637
638    Every kernel value in either output must have a comparison in `comparisons`, equal hashes or not.
639
640    Raises:
641        NotWireForm: either output holds a value that is not a wire form (`flatten`).
642        UndeclaredComparison: a kernel value's type has no entry in `comparisons` (default `COMPARISONS`).
643        KeyError: a rule names a threshold `thresholds` lacks.
644    """
645    leaves = flatten(expected), flatten(actual)
646    for side in leaves:
647        typed_values(side, comparisons)
648    try:
649        if value_digest("output", dict(expected)) == value_digest("output", dict(actual)):
650            return Comparison(agree=True, differences=(), thresholds=thresholds.canonical_digest())
651    except ValueError:
652        pass
653    return compare_fields(*leaves, thresholds, comparisons)

def best_assignment(weights: Sequence[Sequence[float]]) -> list[tuple[int, int]]: 
307def best_assignment(weights: Sequence[Sequence[float]]) -> list[tuple[int, int]]:
308    """Return the one-to-one (row, column) pairs of largest total weight, as many as the smaller side.
309
310    With a tie it returns the optimum whose sorted `(row, column)` sequence is lexically smallest.
311    """
312    n, m = len(weights), len(weights[0]) if weights else 0
313    wanted = min(n, m)
314    best = _total(weights, _hungarian(weights))
315    chosen: list[tuple[int, int]] = []
316    for i in range(n):
317        if len(chosen) == wanted:
318            break
319        used = {c for _, c in chosen}
320        free = [j for j in range(m) if j not in used]
321        for j in free:
322            rest_rows = list(range(i + 1, n))
323            rest_cols = [c for c in free if c != j]
324            if min(len(rest_rows), len(rest_cols)) != wanted - len(chosen) - 1:
325                continue
326            sub = [[weights[r][c] for c in rest_cols] for r in rest_rows]
327            total = _total(weights, chosen) + weights[i][j] + _total(sub, _hungarian(sub))
328            if math.isclose(total, best, rel_tol=1e-12, abs_tol=1e-12):
329                chosen.append((i, j))
330                break
331    if len(chosen) != wanted:
332        raise ValueError(f"no optimal assignment of {wanted} pairs found in the weights")
333    return chosen

580    expected: Mapping[str, Value],
581    actual: Mapping[str, Value],
582    thresholds: Thresholds,
583    comparisons: Comparisons | None = None,
584) -> Comparison:
585    """Return whether flattened outputs agree, leaf by leaf under each kernel type's declared comparison.
586
587    A leaf under a kernel value (found by its `format` leaf) is judged by its type's first matching rule in
588    `comparisons` (default `COMPARISONS`); any other leaf is exact. Labels a type matches are mapped first
589    (`relabel`). Equal leaves agree before any rule is looked up. A leaf on one side only is a difference; an
590    `optional` parent absent on one side and present on the other is one difference at the parent.
591
592    Raises:
593        UndeclaredComparison: a kernel value's type has no entry in `comparisons`, or a leaf that differs lies under
594            a type that declares no rule for its path.
595        KeyError: a rule names a threshold `thresholds` lacks.
596    """
597    table = COMPARISONS if comparisons is None else comparisons
598    digest = thresholds.canonical_digest()
599    typed = typed_values(expected, table) | typed_values(actual, table)
600    if expected == actual:
601        return Comparison(agree=True, differences=(), thresholds=digest)
602    actual = relabel(expected, actual, table)
603    prefixes = sorted(typed, key=len, reverse=True)
604    paths = sorted(expected.keys() | actual.keys())
605    owners = {p: _owner(p, prefixes) for p in paths}
606    rules: dict[str, Rule] = {}
607    for path in paths:
608        owner = owners[path]
609        found_rule = (
610            Rule("exact") if owner is None else table[typed[owner]].rule(path[len(owner) + 1 :] if owner else path)
611        )
612        if found_rule is None:
613            if expected.get(path) == actual.get(path):
614                continue
615            raise UndeclaredComparison(typed[owner or ""], path)
616        rules[path] = found_rule
617    gaps = _optional_gaps(expected, actual, rules)
618    found = list(gaps.values())
619    for path in paths:
620        if path not in rules or path in gaps or any(path.startswith(f"{p}.") for p in gaps):
621            continue
622        a, b = expected.get(path), actual.get(path)
623        if a is not None and a == b:
624            continue
625        if (difference := _judge(path, a, b, rules[path], thresholds)) is not None:
626            found.append(difference)
627    found.sort(key=lambda d: d.path)
628    return Comparison(agree=not found, differences=tuple(found), thresholds=digest)
