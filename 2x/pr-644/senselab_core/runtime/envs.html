senselab_core.runtime.envs

Locked envs: ensure_env builds an Env.isolated or Env.server env from its lock and describes Env.host.

An isolated env is a uv venv on a uv-managed interpreter (--python-preference only-managed, the lock's named version, t for free-threaded), installed from its lock: a lock with torch: pins installs them from the PyTorch wheel index routed to the host's CUDA (pick_torch_index), then the lock body with --no-deps. Under root, env <name> lives in <name>[-<tag>]-<uuid>/, and <name>[-<tag>].json (env-record@1.0) names the current build; both are written under a Lease on <name>[-<tag>], kept alive by beats while the build runs. A builder whose lease was lost, or cannot prove it right before publishing (Lease.prove), does not publish its record. A build is never moved or removed here, so a broken lease costs a duplicate build, never a mixed env, and a superseded build stays until an explicit clean-up. The worker runtime, and the worker modules a backend declares, are copied into every env as an overlay that leaves its wheel set unchanged.

141@dataclass(frozen=True)
142class EnvLock:
143    """A lock file's header: the interpreter version, the torch pins and the newest CUDA index they are served on."""
144
145    path: Path
146    python: str
147    torch_pins: tuple[str, ...]
148    max_cuda: tuple[int, int]
149
150    @classmethod
151    def read(cls, path: Path) -> "EnvLock":
152        """Parse the `# key: value` header of the lock at `path`.
153
154        Raises:
155            EnvLockError: `python:` or `torch:` is missing, or a value is not one the builder knows.
156        """
157        header: dict[str, str] = {}
158        for line in Path(path).read_text("utf-8").splitlines():
159            if not line.startswith("#"):
160                break
161            key, sep, value = line[1:].partition(":")
162            if sep:
163                header.setdefault(key.strip(), value.strip())
164        missing = [k for k in ("python", "torch") if k not in header]
165        if missing:
166            raise EnvLockError(f"lock {path} has no {missing} header line")
167        if not _PYTHON_VERSION.fullmatch(header["python"]):
168            raise EnvLockError(f"lock {path} names python {header['python']!r}, not <major>.<minor>[t]")
169        ceiling = header.get("torch-index", "none")
170        if ceiling not in (*CUDA_INDEXES, "cpu", "none"):
171            raise EnvLockError(f"lock {path} names torch-index {ceiling!r}; known: {[*CUDA_INDEXES, 'cpu']}")
172        torch = header["torch"]
173        return cls(
174            path=Path(path),
175            python=header["python"],
176            torch_pins=tuple(torch.split()) if torch != "none" else (),
177            max_cuda=CUDA_INDEXES.get(ceiling, (0, 0) if ceiling == "cpu" else DEFAULT_MAX_CUDA),
178        )

232@dataclass(frozen=True)
233class Uv:
234    """The `uv` command line every build runs through."""
235
236    command: tuple[str, ...]
237
238    @classmethod
239    def find(cls) -> "Uv":
240        """Return `uv` from `PATH`, else `~/.local/bin/uv` or `~/.cargo/bin/uv`.
241
242        Raises:
243            FileNotFoundError: none of them is an executable.
244        """
245        home = Path.home()
246        for candidate in (shutil.which("uv"), home / ".local" / "bin" / "uv", home / ".cargo" / "bin" / "uv"):
247            if candidate and os.access(candidate, os.X_OK):
248                return cls((str(candidate),))
249        raise FileNotFoundError("uv not found on PATH, in ~/.local/bin or in ~/.cargo/bin")

251    def __call__(self, *args: str) -> subprocess.CompletedProcess[str]:
252        """Run `uv <args>`, capturing its output; the caller reads `returncode`."""
253        return subprocess.run([*self.command, *args], capture_output=True, text=True, check=False)

256def detect_host_cuda(*, run: Callable[[list[str]], str] = _run) -> HostCuda:
257    """Return the CUDA version `nvidia-smi` reports, else the one `nvcc --version` reports, else none.
258
259    `run` executes a command and returns its stdout, raising when it cannot; tests inject it.
260    """
261    for command, pattern, source in (
262        (["nvidia-smi"], _SMI_CUDA, "nvidia-smi"),
263        (["nvcc", "--version"], _NVCC_CUDA, "nvcc"),
264    ):
265        try:
266            match = pattern.search(run(command))
267        except (OSError, subprocess.SubprocessError):
268            continue
269        if match:
270            return HostCuda((int(match[1]), int(match[2])), source)  # type: ignore[arg-type]
271    return HostCuda(None, "none")

274def pick_torch_index(
275    host: HostCuda, *, override: str | None = None, max_cuda: tuple[int, int] | None = None
276) -> TorchIndex:
277    """Return `override` when set, else the newest CUDA index at or below the host's CUDA capped at `max_cuda`.
278
279    `max_cuda` defaults to `DEFAULT_MAX_CUDA`; a host without CUDA, or older than every index, gets the CPU index.
280    """
281    if override:
282        return TorchIndex(override, torch_build(override), "profile")
283    if host.version is not None:
284        effective = min(host.version, max_cuda or DEFAULT_MAX_CUDA)
285        for tag, version in CUDA_INDEXES.items():
286            if version <= effective:
287                url = f"{PYTORCH_INDEX_BASE}/{tag}"
288                return TorchIndex(url, torch_build(url), "probe")
289    return TorchIndex(CPU_INDEX, torch_build(CPU_INDEX), "probe")

292def ensure_env(  # noqa: PLR0913 - the env, its lock, root and profile settings, and two injectable boundaries
293    env: Env,
294    lock: Path | None,
295    root: Path,
296    *,
297    settings: Envs,
298    lease: JobLease,
299    requirements: Sequence[str] = (),
300    workers: Mapping[str, Path] | None = None,
301    uv: Uv | None = None,
302    host_cuda: HostCuda | None = None,
303    network: Network = "online",
304    clock: Clock = SystemClock(),  # noqa: B008 - stateless
305) -> EnvHandle:
306    """Return the env `env` built from `lock` and overlaid with `workers` under `root`, built once across processes.
307
308    A published env is never changed: an env of another lock, torch index, worker runtime or worker module source is
309    a separate build, named by its overlay digest, and the build it replaces stays for the workers running in it.
310
311    For `Env.host()`, `lock` is the senselab lock, or None when none ships with the install, in which case the installed
312    distributions are checked against `requirements`; the env is this interpreter's, which must be uv-managed.
313
314    Args:
315        env: The env a backend declares.
316        lock: The lock file the env is installed from.
317        root: The directory the profile's `env_roots` gives for `env.kind`; unused for a host env.
318        settings: The profile's `envs`: `torch_index` replaces the probed index when set.
319        lease: The profile's `lease`: the build's lease beats under it, and a waiter waits `stale_after_s` plus
320            `lease.wait_s`.
321        requirements: The backend's requirements; required for a host env, which is refused unless `lock` covers each.
322        workers: The backend's worker modules, dotted name to source file; each is copied into the env's
323            site-packages as that module, beside the worker runtime, when the env is built.
324        uv: The `uv` command; `Uv.find()` when None.
325        timeout_s: Seconds one request to a PyTorch index may take; the packaged profile's `timeouts.fetch_s` when None.
326        host_cuda: The host's CUDA; probed with `detect_host_cuda` when None and the lock pins torch.
327        network: The profile's `network`; an `offline` profile uses a built env and builds none.
328        clock: How the build lease reads time and waits.
329
330    Raises:
331        ManagedPythonMissing: no uv-managed interpreter of the needed version, or the host's is not uv-managed.
332        CudaCompatibilityError: the routed index has no wheel for the lock's torch pins.
333        EnvLockError: the lock's header lacks what a build needs, or a host env got no `requirements`.
334        HostLockLacksBackend: a host env whose `lock` does not pin every one of `requirements`.
335        RuntimeError: another `uv` step failed; the message carries uv's stderr.
336        LeaseTimeout, LeaseUnavailable: as `Lease`.
337        Offline: the env is not built and `network` is `offline`.
338    """
339    uv = uv or Uv.find()
340    if env.kind == "host":
341        if not requirements:
342            raise EnvLockError("an Env.host() backend must pass its requirements, so the senselab lock can be checked")
343        if lock is None:
344            lacking = installed_gaps(requirements)
345            if lacking:
346                raise HostLockLacksBackend(f"the installed packages do not satisfy {lacking}")
347            return _host_env(None, uv, requirements)
348        lacking = host_lock_gaps(lock, requirements)
349        if lacking:
350            raise HostLockLacksBackend(f"{lock} does not cover {lacking}; declare the backend Env.isolated instead")
351        return _host_env(lock, uv, requirements)
352    if lock is None:
353        raise EnvLockError(f"a {env.kind} env is built from its lock, none was given")
354    parsed = EnvLock.read(lock)
355    stem, index, host_cuda = _stem(env, parsed, settings, workers, host_cuda)
356    root = Path(root)
357    root.mkdir(parents=True, exist_ok=True)
358    lock_sha = file_digest("file@1", parsed.path)
359    wanted = (lock_sha, without_userinfo(index.url) if index else None)
360    current = _read_record(root, stem)
361    if current is not None and (current.lock_sha, current.torch_index) == wanted:
362        return _handle(root, current, parsed.path)
363    held = Lease(root / stem, settings=lease, timeout=lease.stale_after_s + lease.wait_s, clock=clock)
364    with held, held.keep_alive(lease.renew_fraction) as state:
365        current = _read_record(root, stem)
366        if current is None or (current.lock_sha, current.torch_index) != wanted:
367            require_online(network, f"building env {stem}")
368            current = _build(root, stem, parsed, lock_sha, index, uv, host_cuda, workers or {})
369            if state.lost or not held.prove():
370                logger.warning("env %s: its build lease was lost; the build is used here and not published", stem)
371            else:
372                write_atomically(root / f"{stem}.json", current.model_dump_json().encode())
373    return _handle(root, current, parsed.path)

376def _stem(
377    env: Env, parsed: EnvLock, settings: Envs, workers: Mapping[str, Path] | None, host_cuda: HostCuda | None
378) -> tuple[str, TorchIndex | None, HostCuda | None]:
379    """Return the build name of `env` from `parsed`, its torch index, and the host CUDA probed to pick it."""
380    index = None
381    if parsed.torch_pins:
382        host_cuda = host_cuda or detect_host_cuda()
383        index = pick_torch_index(host_cuda, override=settings.torch_index, max_cuda=parsed.max_cuda)
384    overlay = overlay_digest(workers or {})
385    stem = f"{env.name}-{index.tag}-{overlay[:16]}" if index else f"{env.name}-{overlay[:16]}"
386    return stem, index, host_cuda

389def current_env(  # noqa: PLR0913 - the env, its lock, root and profile settings, and the injectable CUDA probe
390    env: Env,
391    lock: Path,
392    root: Path,
393    *,
394    settings: Envs,
395    workers: Mapping[str, Path] | None = None,
396    host_cuda: HostCuda | None = None,
397) -> EnvHandle | None:
398    """Return the handle `ensure_env` would return for `env` built from `lock` under `root`, or None if not built.
399
400    Nothing is built, locked or created.
401
402    Raises:
403        EnvLockError: the lock's header lacks what a build needs.
404    """
405    parsed = EnvLock.read(lock)
406    stem, index, _ = _stem(env, parsed, settings, workers, host_cuda)
407    current = _read_record(Path(root), stem) if Path(root).is_dir() else None
408    wanted = (file_digest("file@1", parsed.path), without_userinfo(index.url) if index else None)
409    if current is None or (current.lock_sha, current.torch_index) != wanted:
410        return None
411    return _handle(Path(root), current, parsed.path)

442def _read_record(root: Path, stem: str) -> EnvRecord | None:
443    try:
444        record = EnvRecord.model_validate_json((root / f"{stem}.json").read_bytes())
445    except FileNotFoundError:

448        logger.warning("env record %s.json is unreadable and is rebuilt: %s", stem, err)

456def _host_env(lock: Path | None, uv: Uv, requirements: Sequence[str]) -> EnvHandle:
457    suffix = "t" if sysconfig.get_config_var("Py_GIL_DISABLED") else ""
458    managed = _check(uv("python", "dir"), "find the uv-managed interpreters").stdout.strip()

462            f"the host interpreter {sys.executable} is not uv-managed",

464    return EnvHandle(
465        python=sys.executable,
466        python_version=platform.python_version() + suffix,
467        lock_sha=file_digest("file@1", lock) if lock else _requirements_digest(requirements),
468        wheel_set=_wheel_set(Path(sysconfig.get_paths()["purelib"])),

565    """Give the group read on every file and traverse on every directory, execute where the owner has it."""
566    for here, _, files in os.walk(directory):
567        for path, is_dir in ((Path(here), True), *((Path(here) / f, False) for f in files)):
568            try:
569                mode = path.lstat().st_mode
570                if stat.S_ISLNK(mode):
571                    continue
572                exec_bit = stat.S_IXGRP if is_dir or mode & stat.S_IXUSR else 0
573                path.chmod(mode | stat.S_IRGRP | exec_bit)

575                if err.errno not in (errno.EPERM, errno.EACCES):

662def host_lock_gaps(lock: Path, requirements: Sequence[str]) -> list[str]:
663    """Return the `requirements` the `uv.lock` at `lock` does not cover.
664
665    A requirement whose marker is false on this host is skipped. Any other is covered when its distribution is a
666    locked package at a version its specifier admits (pre-releases included) and, for each extra it names, the
667    package's locked optional dependencies for that extra are locked too. A requirement naming a URL is never covered.
668
669    Raises:
670        EnvLockError: a requirement is not a valid requirement string.
671    """
672    packages: dict[str, dict[str, Any]] = {
673        _normalize(str(p["name"])): p for p in tomllib.loads(Path(lock).read_text("utf-8")).get("package", [])
674    }
675    gaps = []
676    for text in requirements:
677        try:
678            req = Requirement(text)
679        except InvalidRequirement as err:
680            raise EnvLockError(f"invalid requirement {text!r}: {err}") from err
681        if req.marker is not None and not req.marker.evaluate():
682            continue
683        package = packages.get(_normalize(req.name))
684        covered = (
685            package is not None and req.url is None and req.specifier.contains(package["version"], prereleases=True)
686        )
687        if covered and package is not None:
688            extras = package.get("optional-dependencies", {})
689            covered = all(
690                extra in extras and all(_normalize(d["name"]) in packages for d in extras[extra])
691                for extra in req.extras
692            )
693        if not covered:
694            gaps.append(text)
695    return gaps

771    """Names left out of a lock body: torch, torchaudio, what only they pull in, and triton under torch."""

774    while grew:

779                grew = True
780    triton = {
781        n
782        for n, (_, parents) in packages.items()
783        if n.startswith(("triton", "pytorch-triton")) and parents & TORCH_PACKAGES
784    }

804def compile_lock(  # noqa: PLR0913 - the inputs, then keyword-only checks, bounds and seams
805    inputs: LockInputs,
806    *,
807    check_torch_index: bool = True,
808    max_steps: int = 12,
809    uv: Uv | None = None,
810    timeout_s: float | None = None,
811) -> str:
812    """Return a lock's full text: `uv pip compile --universal --generate-hashes` under a header.
813
814    Needs network. When the resolution pins `torch`, its and `torchaudio`'s versions must be served by the top CUDA
815    index and the CPU index on Linux x86_64 and by the CPU index on macOS arm64; otherwise `torch` is bounded below
816    the rejected version and the resolution repeated, up to `max_steps` times. `torch`, `torchaudio`, the CUDA
817    packages only they pull in and `inputs.excluded` are left out of the body; the builder installs the header's pins
818    from the PyTorch index. uv runs with `--python-preference only-managed`.
819
820    Args:
821        inputs: What the lock is compiled from.
822        check_torch_index: Whether to check the PyTorch indexes (needs network).
823        max_steps: How many times `torch` may be bounded down.
824        uv: The `uv` command; `Uv.find()` when None.
825        timeout_s: Seconds one request to a PyTorch index may take; the packaged profile's `timeouts.fetch_s` when None.
826
827    Raises:
828        EnvLockError: no torch within the requirements is served on every index.
829        RuntimeError: `uv pip compile` failed; the message carries uv's stderr.
830    """
831    uv = uv or Uv.find()
832    if timeout_s is None:
833        from senselab_core.runtime.profiles import ExecutionProfile  # noqa: PLC0415 - profiles import this module
834
835        timeout_s = ExecutionProfile.packaged().timeouts.fetch_s
836    python, name = inputs.python, inputs.name
837    tag = _top_cuda_tag(inputs.max_cuda)
838    targets = [(tag, _LINUX_X86_64), ("cpu", _LINUX_X86_64), ("cpu", _MACOS_ARM64)]
839    with tempfile.TemporaryDirectory(prefix=f"env-lock-{name}-") as tmp:
840        work = Path(tmp)
841        in_file, constraints = work / "requirements.in", work / "constraints.txt"
842        in_file.write_text("\n".join([*inputs.requirements, *inputs.base]) + "\n")
843        extra: list[str] = []
844        notes: list[str] = []
845        flags = ["--constraint", str(constraints), "--exclude-newer", inputs.exclude_newer]
846        for _ in range(max_steps):
847            constraints.write_text("\n".join([*inputs.quarantined, *extra]) + "\n")
848            packages = _resolved(_pip_compile(uv, in_file, python, [*flags, "--annotation-style", "line"]))
849            pins = tuple(f"{p}=={packages[p][0]}" for p in sorted(TORCH_PACKAGES) if p in packages)
850            if not pins or not check_torch_index:
851                break
852            missing = [
853                f"{pin} on {t}/{plat}"
854                for pin in pins
855                for t, plat in targets
856                if not _wheel_on_index(pin.partition("==")[0], pin.partition("==")[2], t, python, plat, timeout_s)
857            ]
858            if not missing:
859                break
860            if "torch" not in packages:
861                raise EnvLockError(f"{name!r}: torchaudio without torch is not served: {missing}")
862            notes.append(f"torch {packages['torch'][0]} skipped: not served as " + "; ".join(missing))
863            extra.append(f"torch<{packages['torch'][0]}")
864        else:
865            raise EnvLockError(f"{name!r}: no torch within the requirements is served on every index: {notes}")
866        no_emit = [*(_torch_only(packages) if pins else ()), *sorted(_normalize(n) for n in inputs.excluded)]
867        skip = [a for n in no_emit for a in ("--no-emit-package", n)]
868        body = _pip_compile(uv, in_file, python, [*flags, "--generate-hashes", *skip])
869    header = [
870        f"# senselab env lock: {name}",
871        f"# identity-digest: {LOCK_DIGEST_TAG} {lock_digest(inputs)}",
872        f"# python: {python}",
873        f"# torch: {' '.join(pins) if pins else 'none'}",
874        f"# torch-index: {tag if pins else 'none'}",
875        f"# exclude-newer: {inputs.exclude_newer}",
876        f"# uv: {_check(uv('--version'), 'report its version').stdout.strip()}",
877        *([f"# exclude: {' '.join(sorted(_normalize(n) for n in inputs.excluded))}"] if inputs.excluded else []),
878        *(f"# note: {note}" for note in notes),
879    ]
880    return "\n".join(header) + "\n" + body

For Env.host(), lock is the senselab lock, or None when none ships with the install, in which case the installed distributions are checked against requirements; the env is this interpreter's, which must be uv-managed.

Arguments:
  • env: The env a backend declares.
  • lock: The lock file the env is installed from.
  • root: The directory the profile's env_roots gives for env.kind; unused for a host env.
  • settings: The profile's envs: torch_index replaces the probed index when set.
  • lease: The profile's lease: the build's lease beats under it, and a waiter waits stale_after_s plus lease.wait_s.
  • requirements: The backend's requirements; required for a host env, which is refused unless lock covers each.
  • workers: The backend's worker modules, dotted name to source file; each is copied into the env's site-packages as that module, beside the worker runtime, when the env is built.
  • uv: The uv command; Uv.find() when None.
  • timeout_s: Seconds one request to a PyTorch index may take; the packaged profile's timeouts.fetch_s when None.
  • host_cuda: The host's CUDA; probed with detect_host_cuda when None and the lock pins torch.
  • network: The profile's network; an offline profile uses a built env and builds none.
  • clock: How the build lease reads time and waits.
Raises:
  • ManagedPythonMissing: no uv-managed interpreter of the needed version, or the host's is not uv-managed.
  • CudaCompatibilityError: the routed index has no wheel for the lock's torch pins.
  • EnvLockError: the lock's header lacks what a build needs, or a host env got no requirements.
  • HostLockLacksBackend: a host env whose lock does not pin every one of requirements.
  • RuntimeError: another uv step failed; the message carries uv's stderr.
  • LeaseTimeout, LeaseUnavailable: as Lease.
  • Offline: the env is not built and network is offline.

665

667    locked package at a version its specifier admits (pre-releases included) and, for each extra it names, the
668    package's locked optional dependencies for that extra are locked too. A requirement naming a URL is never covered.
669
670    Raises:
671        EnvLockError: a requirement is not a valid requirement string.
672    """
673    packages: dict[str, dict[str, Any]] = {
674        _normalize(str(p["name"])): p for p in tomllib.loads(Path(lock).read_text("utf-8")).get("package", [])
675    }
676    gaps = []
677    for text in requirements:
678        try:
679            req = Requirement(text)
680        except InvalidRequirement as err:
681            raise EnvLockError(f"invalid requirement {text!r}: {err}") from err
682        if req.marker is not None and not req.marker.evaluate():
683            continue

686            package is not None and req.url is None and req.specifier.contains(package["version"], prereleases=True)
