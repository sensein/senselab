senselab_core.runtime.vllm_server

A vllm serve process in an Env.server env, owned by the calling process, and its HTTP clients.

VllmServer starts <env>/bin/vllm serve on a pinned model under a beating Lease named by the host and the devices, so servers that share a card start one at a time. It waits for /health, reads the served version and model, and holds an Attestation (engine version, model identity, config digest) that every reply it returns carries and that its context_fields put in the execution context. The server is launched through the supervisor in force (Supervisor), in its own process group, which the supervisor's keeper kills if the owner dies; leaving the context stops it (SIGTERM, then SIGKILL). A svc pool worker starts one and registers it for other processes to consume (execution.services); a reply through the registry is checked against the attestation by attest_reply, as here.

The clients chat (with content_part for text, image, video and audio parts) and complete call a running server's OpenAI-compatible endpoints over loopback and return the parsed JSON; they log nothing.

156def _status_fault(exc: urllib.error.HTTPError) -> Exception:
157    text = f"HTTP {exc.code}: {exc.reason}"
158    if exc.code < HTTPStatus.INTERNAL_SERVER_ERROR:
159        return ServiceUnavailable(text) if exc.code in _TRANSIENT_STATUSES else ServerRejected(exc.code, text)
160    if exc.code == HTTPStatus.INSUFFICIENT_STORAGE or _MEMORY_WORDS.search(str(exc.reason)):
161        return ServerOutOfMemory(text)
162    return ServiceUnavailable(text) if exc.code in _TRANSIENT_STATUSES else ServerFailed(text)

165def server_fault(exc: Exception) -> Exception:
166    """Return the typed fault for what a request to a server raised: the one mapper both server paths use.
167
168    - A timeout, a connect failure, a reset or any other transport error: `ServiceUnavailable`.
169    - An HTTP 408 or 429, which a retry may cure: `ServiceUnavailable`; any other 4xx: `ServerRejected`, a content
170      fault, whatever its message says.
171    - An HTTP 507, or a 5xx whose message names memory: `ServerOutOfMemory`, a resource fault.
172    - An HTTP 503: `ServiceUnavailable`; any other 5xx: `ServerFailed`.
173    - Anything else is returned as it is.
174    """
175    if isinstance(exc, urllib.error.HTTPError):
176        return _status_fault(exc)
177    if isinstance(exc, (OSError, http.client.HTTPException)):
178        return ServiceUnavailable(f"the server did not answer: {type(exc).__name__}")
179    return exc

202def expected_attestation(lock: Path | None, model: ResolvedModel, settings: Mapping[str, Any]) -> Attestation:
203    """Return what a server of `model` with `settings` in an env built from `lock` must attest, without starting it.
204
205    The engine version is the ``vllm==`` pin of the lock.
206
207    Raises:
208        VllmStartError: ``version``, there is no lock or the lock pins no ``vllm``.
209        ValueError: as `server_args`.
210    """
211    text = lock.read_text("utf-8") if lock is not None else ""
212    pin = re.search(r"^vllm==([^\s\\;]+)", text, re.MULTILINE)
213    if pin is None:
214        raise VllmStartError("version", "the env's lock pins no vllm")
215    return Attestation(
216        engine_version=pin.group(1),
217        served_model=_served(model)[2],
218        model_identity=model.identity,
219        config_digest=config_digest(settings),
220    )

256def server_args(settings: Mapping[str, Any]) -> list[str]:
257    """Return the ``vllm serve`` arguments `settings` names, each as one ``--flag=value`` or switch.
258
259    Args:
260        settings: Every key of `SERVER_FLAGS`, `SWITCHES` and `ENGINE_VARIABLES` and no other;
261            ``limit_mm_per_prompt`` is a mapping of per-modality counts, written as compact JSON with sorted keys.
262
263    Raises:
264        ValueError: A key is missing or unknown.
265    """
266    known = {key for key, _ in SERVER_FLAGS} | {key for key, _, _ in SWITCHES} | {key for key, _ in ENGINE_VARIABLES}
267    if set(settings) != known:
268        raise ValueError(f"vLLM settings lack {sorted(known - set(settings))} and add {sorted(set(settings) - known)}")
269    args: list[str] = []
270    for key, flag in SERVER_FLAGS:
271        value = settings[key]
272        if isinstance(value, Mapping):
273            value = json.dumps(dict(value), sort_keys=True, separators=(",", ":"))
274        args.append(f"{flag}={value}")
275    args += [on if settings[key] else off for key, on, off in SWITCHES]
276    return args

285def _probe(url: str, key: str, timeout_s: float = 60.0) -> Any:  # noqa: ANN401 - a JSON value
286    """``GET url`` and return the body's `key`; any failure is a ``version`` start error."""
287    try:
288        with _NO_PROXY.open(url, timeout=timeout_s) as response:
289            body = json.loads(response.read())
290        if not isinstance(body, dict):
291            raise ValueError(f"the body is {type(body).__name__}, not an object")
292        return body.get(key)
293    except (OSError, http.client.HTTPException, ValueError) as exc:
294        raise VllmStartError("version", f"probe {url} failed: {exc}") from exc

312def _served(model: ResolvedModel) -> tuple[str, list[str], str]:
313    """Return what ``vllm serve`` is given for `model`, the arguments that bind it to its identity, its served name.
314
315    A hub model is served under its reference; a local model under its content identity, so its path never reaches a
316    reply, an attestation or a stored context.
317    """
318    if not model.pinned:
319        raise ValueError(f"an unpinned {model.ref.source} model; a server loads only a pinned model")
320    if model.ref.source == "hf":
321        return model.ref.id, [f"--revision={model.identity}", f"--tokenizer-revision={model.identity}"], model.ref.id
322    if model.ref.source == "local":
323        return model.ref.id, [f"--served-model-name={model.identity}"], model.identity
324    raise ValueError(f"a server loads a hub or local model, not a {model.ref.source} one")

327class VllmServer:
328    """One ``vllm serve`` of a pinned model in a built `Env.server` env, listening on loopback while open.
329
330    On entry it takes a `Lease` on ``<start_lock_dir>/vllm-start-<host token>-<devices>``, starts
331    ``<env>/bin/vllm serve``
332    in its own session once each of its devices has `gpu_memory_utilization` of its memory free (a hub model with
333    ``--revision`` and ``--tokenizer-revision`` set to its commit; ``CUDA_VISIBLE_DEVICES`` set to `devices` when
334    given), waits for ``/health`` and releases the lease; the lease beats while it is held. A start whose lease is lost
335    before the launch is refused; one lost after it is logged, and the healthy server is kept, since the lease only
336    orders starts. It then reads ``/version`` and checks that ``/v1/models`` serves the model. On exit, and on any
337    start failure, the server's process group gets SIGTERM, then SIGKILL.
338
339    Attributes:
340        label: The model as a message names it: the hub reference, or a local model's identity prefix, never a path.
341        url: ``http://127.0.0.1:<port>`` while the context is open, else None.
342        attestation: The running server's `Attestation` while the context is open, else None.
343    """
344
345    def __init__(  # noqa: PLR0913 - the env, model and settings, then keyword-only placement and bounds
346        self,
347        env: EnvHandle,
348        model: ResolvedModel,
349        settings: Mapping[str, Any],
350        *,
351        start_lock_dir: Path,
352        start_margin_s: float,
353        lease: JobLease,
354        vllm: VllmSettings | None = None,
355        devices: Sequence[int | str] | None = None,
356        log_dir: Path | None = None,
357        network: Network = "online",
358        memory: MemoryProbe = device_free_memory,
359        clock: Clock = SystemClock(),  # noqa: B008 - stateless
360    ) -> None:
361        """Describe the server; nothing starts until the context is entered.
362
363        Args:
364            env: The built env (`ensure_env` of an `Env.server` env) whose ``bin/vllm`` runs the server.
365            model: The model to serve, pinned to a hub commit or a local weights digest.
366            settings: The server settings `server_args` reads.
367            start_lock_dir: Where the start lease lives.
368            start_margin_s: The profile's `envs.server_start_margin_s`; a waiter for the start lease waits
369                `vllm.startup_s` plus this plus the lease's stale bound and `lease.wait_s`.
370            lease: The profile's `lease`, under which the start lease beats.
371            vllm: The profile's `vllm` section: the start-up wait, the health poll and read, and the stop grace; the
372                packaged profile's when None.
373            devices: The server's visible devices; when None the inherited environment is left alone and the lease
374                is named by the inherited ``CUDA_VISIBLE_DEVICES`` (``all`` if unset).
375            log_dir: Where the server's output goes, one mode-600 file per server; the owner's stderr when None.
376            network: The profile's `network`; `offline` gives the server `OFFLINE_VARIABLES`.
377            memory: Reads the free and total memory of the server's devices (None: those it inherits).
378            clock: The clock the start lease and the wait for health run on.
379
380        Raises:
381            ValueError: `model` is unpinned or a url model, or `settings` is not exactly the known keys.
382        """
383        self.target, self._model_args, self.served_name = _served(model)
384        self.label = str(model.ref) if model.ref.source == "hf" else f"local model {model.identity[:12]}"
385        self.model = model
386        self.args = server_args(settings)
387        self.variables = server_variables(settings)
388        self._digest = config_digest(settings)
389        self.memory_share = float(settings["gpu_memory_utilization"])
390        self._memory, self._clock = memory, clock
391        self.executable = Path(env.python).parent / "vllm"
392        self.devices = None if devices is None else [str(device) for device in devices]
393        self.start_lock_dir = Path(start_lock_dir)
394        self.vllm = vllm if vllm is not None else ExecutionProfile.packaged().vllm
395        self.start_margin_s = float(start_margin_s)
396        self.lease = lease
397        self.log_dir = log_dir
398        self.url: str | None = None
399        self.attestation: Attestation | None = None
400        self._env = env
401        self._network = network
402        self._server: subprocess.Popen[bytes] | None = None
403        self._supervisor = current()
404        self._log: IO[bytes] | None = None
405        self.log_path: Path | None = None
406
407    @property
408    def config_digest(self) -> str:
409        """The digest of the server arguments and variables, which the attestation carries."""
410        return self._digest
411
412    @property
413    def lock_path(self) -> Path:
414        """The start lease's resource, shared by every server on this host's devices."""
415        return self.start_lock_dir / f"vllm-start-{host_token()}-{device_set(self.devices)}"
416
417    def __enter__(self) -> "VllmServer":
418        """Start the server and attest it.
419
420        Raises:
421            VllmStartError: The server exited before it was healthy, was not healthy in time, or serves another
422                model; `VllmOutOfMemory` when a device lacked the memory it asks for.
423            LeaseTimeout, LeaseUnavailable: as `Lease`; `LeaseUnavailable` also when the start lease was lost before
424                the launch.
425        """
426        self.start_lock_dir.mkdir(parents=True, exist_ok=True)
427        try:
428            stale_after = self.lease.stale_after_s
429            wait = self.vllm.startup_s + self.start_margin_s + stale_after + self.lease.wait_s
430            held = Lease(self.lock_path, settings=self.lease, timeout=wait, clock=self._clock)
431            with held, held.keep_alive(self.lease.renew_fraction) as state:
432                if state.lost or not held.prove():
433                    raise LeaseUnavailable(held.path, f"the start lease {held.path} was lost before the server started")
434                self._check_memory()
435                self._launch()
436                self._await_health()
437                if state.lost:
438                    logger.warning(
439                        "start lease %s was lost while the server started; the healthy server is kept", held.path
440                    )
441            self.attestation = self._attest()
442        except BaseException:
443            self.stop()
444            raise
445        return self

615    Args:

663def _error_message(raw: bytes) -> str:
664    """Return the message of a vLLM error body, at most `ERROR_DETAIL_CHARS` long.
665
666    Only ``error.message``, a top-level ``message`` or ``detail`` is kept (a validation ``detail`` list as
667    ``loc: msg`` items); ``input`` and every other field, which can echo the prompt, are dropped, as is a body
668    that is not JSON.
669    """
670    try:
671        body = json.loads(raw)
672    except ValueError:
673        return ""
674    if not isinstance(body, dict):
675        return ""
676    error = body.get("error")
677    message: Any = error.get("message") if isinstance(error, dict) else body.get("message", body.get("detail"))
678    if isinstance(message, list):
679        items = [item for item in message if isinstance(item, dict)]
680        message = "; ".join(f"{'.'.join(map(str, item.get('loc') or ()))}: {item.get('msg', '')}" for item in items)
681    return message[:ERROR_DETAIL_CHARS] if isinstance(message, str) else ""

684def _post_json(url: str, body: Mapping[str, Any], timeout_s: float) -> Any:  # noqa: ANN401 - a JSON value
685    """POST `body` without any proxy and parse the JSON reply; nothing is logged.

688    message after the reason.
689    """

691        url, data=json.dumps(body).encode(), headers={"Content-Type": "application/json"}, method="POST"

699        raise server_fault(urllib.error.HTTPError(exc.url, exc.code, reason, exc.headers, None)) from None
