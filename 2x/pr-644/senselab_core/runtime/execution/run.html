senselab_core.runtime.execution.run

A run's lifecycle and its results: start a plan, follow it as a Run, read its ResultSet.

A run lives in <cache root>/2x/runs/<plan hash>/: plan.json (the logical plan and its input manifest, written once), inputs.json (what each record's inputs are read from, merged across attempts), manifest.json (the plan hash and the latest attempt's result store), placements/<attempt>.json (one per attempt, never overwritten), attempts/<n> (the attempt id claimed n-th), events/<writer>.jsonl (EventLog), prov/ (the provenance documents), and dead/, queues/ and epoch/ for the pool. A Run over a directory derives its status and its results from the events and the result store, and binds its input values from inputs.json, so it works from a run another process started, unless an input was held in memory by the process that planned it.

 171class ResultSet:
 172    """The results of a run over records: `(record, output, fault or not-run)` in record order, read when asked.
 173
 174    An entry may be given as a callable, which is called once, on first access; a `Run`'s results read each record's
 175    outputs from the result store that way. `shared` holds the graph outputs of nodes that run once for every record
 176    (a gather and what follows it); `run` is the `Run` they belong to, if any.
 177    """
 178
 179    def __init__(
 180        self,
 181        entries: Sequence[Entry],
 182        shared: Mapping[str, "Given | Callable[[], Given]"] | None = None,
 183        *,
 184        run: "Run | None" = None,
 185    ) -> None:
 186        """Hold `entries` in the order given."""
 187        self._entries = list(entries)
 188        self._shared = dict(shared or {})
 189        self.run = run
 190
 191    def _given(self, index: int) -> Item:
 192        record, held = self._entries[index]
 193        if callable(held):
 194            held = held()
 195            self._entries[index] = (record, held)
 196        return record, held
 197
 198    @property
 199    def shared(self) -> Mapping[str, Given]:
 200        """The graph outputs of the nodes that run once, by name."""
 201        for name, held in self._shared.items():
 202            if callable(held):
 203                self._shared[name] = held()
 204        return dict(self._shared)  # type: ignore[arg-type]
 205
 206    def __iter__(self) -> Iterator[Item]:
 207        """Iterate `(record, what it gave)` in record order."""
 208        return (self._given(i) for i in range(len(self._entries)))
 209
 210    def __len__(self) -> int:
 211        """Return the number of records."""
 212        return len(self._entries)
 213
 214    def __getitem__(self, index: int) -> Item:
 215        """Return the `index`-th record and what it gave."""
 216        return self._given(index)
 217
 218    @property
 219    def failures(self) -> list[tuple[Record, FaultRecord]]:
 220        """The records that failed, with their faults."""
 221        return [(r, o) for r, o in self if isinstance(o, FaultRecord)]
 222
 223    def outputs(self, port: str) -> list[object]:
 224        """Return output `port` of each record in record order, a record's fault or `NotRun` standing in for it.
 225
 226        Raises:
 227            KeyError: a record that gave outputs has no output `port`.
 228        """
 229        return [o.outputs[port] if isinstance(o, Output) else o for _, o in self]
 230
 231    def to_frame(self) -> Any:  # noqa: ANN401 - a pandas DataFrame
 232        """Return a data frame of the records: an `id` column, a column per output, `fault` and `not_run` columns.
 233
 234        The index is the records' entities, one level per entity name; records with no entities are indexed by id.
 235
 236        Raises:
 237            MissingExtra: pandas is not installed.
 238        """
 239        try:
 240            import pandas  # noqa: PLC0415 - optional
 241        except ImportError as err:
 242            raise MissingExtra("pandas") from err
 243        items = list(self)
 244        ports = sorted({p for _, o in items if isinstance(o, Output) for p in o.outputs})
 245        names = sorted({n for r, _ in items for n in r.entities})
 246        if names:
 247            index = pandas.MultiIndex.from_tuples(
 248                [tuple(r.entities.get(n) for n in names) for r, _ in items], names=names
 249            )
 250        else:
 251            index = pandas.Index([r.id for r, _ in items], name="id")
 252        columns: dict[str, list[object]] = {"id": [r.id for r, _ in items]}
 253        for port in ports:
 254            columns[port] = [o.outputs.get(port) if isinstance(o, Output) else None for _, o in items]
 255        columns["fault"] = [o.fault if isinstance(o, FaultRecord) else None for _, o in items]
 256        columns["not_run"] = [o.kind if isinstance(o, NotRun) else None for _, o in items]
 257        return pandas.DataFrame(columns, index=index)

 300def latest_placement(run_dir: Path) -> PlanPlacement:

 366def bind_sources(placed: Plan) -> dict[InputIdentity, object]:
 367    """Return the value of each input identity of `placed`, bound from its sources and its live values.
 368
 369    A path or JSON source is bound again (`bind_input`) and must give the identity the manifest holds, so a file
 370    changed since planning is refused; a `{"memory": <identity>}` source takes the value `placed.live` holds. Every
 371    record is checked before anything is raised.
 372
 373    Raises:
 374        InputChanged: inputs have another identity than the manifest's, or cannot be read; it names every one, and the
 375            first `InputUnreadable` is its cause.
 376        InputsUnavailable: an input held in memory is not in `placed.live`.
 377        SourcesUnrecorded: the plan records no sources.
 378    """
 379    types = port_types(placed)
 380    rows, shared = recorded_sources(placed)
 381    values: dict[InputIdentity, object] = {}
 382    missing: list[tuple[str, str]] = []
 383    changed: list[tuple[str, str]] = []
 384    unreadable: list[InputUnreadable] = []
 385    for row, read in zip(placed.manifest.rows, rows, strict=True):
 386        for port, expected in row.inputs.items():
 387            if port in shared and expected in values:
 388                continue
 389            source = read[port] if port in read else shared[port]
 390            held = _memory(source)
 391            if held is not None and held == expected and held not in placed.live:
 392                missing.append((row.id, port))
 393                continue
 394            bound = _bound(port, types[port], source, placed.live) if held in (None, expected) else None
 395            if isinstance(bound, InputUnreadable):
 396                unreadable.append(bound)
 397            if not isinstance(bound, tuple) or bound[0] != expected:
 398                changed.append((row.id, port))
 399                continue
 400            values[expected] = bound[1]
 401    if changed:
 402        raise InputChanged(changed) from next(iter(unreadable), None)
 403    if missing:
 404        raise InputsUnavailable(missing)
 405    return values

 408def _bound(
 409    port: str, kind: str, source: object, live: Mapping[InputIdentity, object]
 410) -> tuple[str, object] | InputUnreadable:
 411    """Return the identity and value of an input read from `source`, else why its file cannot be read.
 412
 413    A `{"memory": <identity>}` source is that identity's value in `live`, which holds it.
 414    """
 415    held = _memory(source)
 416    if held is not None:
 417        return held, live[held]
 418    try:
 419        identity, value, _ = bind_input(port, kind, source)
 420    except InputUnreadable as err:
 421        return err
 422    return identity, value

 435class _Reader:
 436    """Reads the graph outputs of one record, or of the run, from the terminal event of each job."""
 437
 438    def __init__(self, graph: Graph, events: Sequence[Event], store: ResultStore, epoch: int | None = None) -> None:
 439        self.graph = graph
 440        self.mapped = mapped_nodes(graph)
 441        self.store = store
 442        self.latest = terminal_events(events, epoch=epoch)
 443        self.held: dict[tuple[str, str], Mapping[str, object] | None] = {}
 444
 445    def _because(self, index: int | None) -> FaultRecord | None:
 446        """Return the failure of a job of record `index` (any, for None) that failed, if one did."""
 447        for _, event in sorted(self.latest.items()):
 448            if event.kind == "failed" and event.failure is not None and event.job is not None:
 449                if index is None or event.job.element == (index,):
 450                    return event.failure
 451        return None
 452
 453    def _stopped(self, event: Event | None, index: int | None) -> Given | None:
 454        """Return why the job whose terminal event is `event` gave nothing, or None if it gave something."""
 455        if event is None:
 456            return NotRun("pending")
 457        if event.kind == "failed":
 458            return event.failure
 459        if event.kind == "blocked":
 460            return NotRun("blocked", self._because(index))
 461        return NotRun("cancelled") if event.kind == "cancelled" else None
 462
 463    def gave(self, name: str, index: int | None) -> tuple[str, str, object] | Given:
 464        """Return graph output `name` of record `index` as (key digest, output identity, value), or why it is not."""
 465        node, port = split(self.graph.outputs[name])
 466        event = self.latest.get((node, (index,) if index is not None and node in self.mapped else ()))
 467        stopped = self._stopped(event, index)
 468        if stopped is not None:
 469            return stopped
 470        assert event is not None and event.key is not None and event.output is not None
 471        wanted = (event.key, event.output)
 472        if wanted not in self.held:
 473            self.held[wanted] = self.store.outputs_of(*wanted)
 474        found = self.held[wanted]
 475        if found is None or port not in found:
 476            return FaultRecord.gap_of("MissingOutput", f"output {port!r} of {node!r} is not in the result store")
 477        return event.key, event.output, found[port]
 478
 479    def collect(self, names: Sequence[str], index: int | None) -> Given:
 480        """Return the graph outputs `names` of record `index` (the run's, for None), or why the record gave none.
 481
 482        A record any of whose jobs failed, was blocked or cancelled gives that even when outputs are held.
 483        """
 484        for node in order(self.graph):
 485            if (index is None) != (node in self.mapped):
 486                stopped = self._stopped(self.latest.get((node, () if index is None else (index,))), index)
 487                if stopped is not None:
 488                    return stopped
 489        values: dict[str, object] = {}
 490        keys: dict[str, str] = {}
 491        identities: dict[str, str] = {}
 492        for name in names:
 493            found = self.gave(name, index)
 494            if isinstance(found, Output | FaultRecord | NotRun):
 495                return found
 496            keys[name], identities[name], values[name] = found
 497        return Output(values, keys, identities)

 509    disk, written by any process. `store` is where the results are read from, by default the one the manifest names.
 510    `epoch` is the epoch of the attempt `start` made, None for a run opened over a directory.
 511    """
 512
 513    def __init__(
 514        self,
 515        run_dir: Path,
 516        *,
 517        store: ResultStore | None = None,
 518        profile: ExecutionProfile | None = None,
 519        clock: Clock = SystemClock(),  # noqa: B008 - stateless
 520    ) -> None:
 521        """Open the run in `run_dir`; `wait` and its liveness reads are timed on `clock`.
 522
 523        `profile` is the one the run is under: its `pool.event_poll_s` paces `wait` and its `store` section opens the
 524        results store; None reads the packaged profile's.
 525        """
 526        self.run_dir = Path(run_dir)
 527        self._store, self._clock = store, clock
 528        self._profile = profile if profile is not None else ExecutionProfile.packaged()
 529        self._pool = self._profile.pool
 530        self._thread: threading.Thread | None = None
 531        self.epoch: int | None = None
 532        self._ended = threading.Event()
 533        self._cancel = threading.Event()
 534        self._error: BaseException | None = None
 535
 536    @property
 537    def plan(self) -> Plan:
 538        """The plan written in the run directory (`load_plan`)."""
 539        return load_plan(self.run_dir)
 540
 541    def manifest(self) -> dict[str, Any]:
 542        """Return the manifest (plan hash and result store), `{}` if the run has none.
 543
 544        Raises:
 545            RunPrivate: the run is another account's, so its manifest is not readable.
 546        """
 547        try:
 548            found = json.loads((self.run_dir / MANIFEST_FILE).read_text(encoding="utf-8"))
 549        except FileNotFoundError:
 550            return {}
 551        except PermissionError as err:
 552            raise RunPrivate(self.run_dir) from err
 553        return found if isinstance(found, dict) else {}

 555    def events(self) -> list[Event]:
 556        """Return the run's events, from every writer, in order."""
 557        return read_events(self.run_dir)

 559    def status(self) -> RunStatus:
 560        """Return the status derived from the events under the run's epoch, and the lines refused.
 561
 562        The run's epoch is the one its attempt holds (`epoch`), else the current one. Once a newer epoch is current,
 563        the status is of the events of the attempt's epoch and earlier, with `superseded` set.
 564        """
 565        current = self._epoch()
 566        mine = self.epoch if self.epoch is not None else current
 567        scan = scan_events(self.run_dir)
 568        superseded = mine is not None and current is not None and current > mine
 569        events = [e for e in scan.events if mine is None or e.epoch <= mine] if superseded else scan.events
 570        status = status_from(events, epoch=mine, quarantined=len(scan.quarantined))
 571        return dataclasses.replace(status, superseded=True) if superseded else status

 573    def _epoch(self) -> int | None:
 574        """Return the run's current epoch, None where none was taken."""
 575        return Epoch(self.run_dir).current() or None
 576
 577    def wait(self, timeout: float | None = None, *, poll: float | None = None) -> RunStatus:
 578        """Wait up to `timeout` seconds for the run to be done and return its status.
 579
 580        A run started here is waited on through its thread; any run is then polled, reading the event log every `poll`
 581        seconds (`pool.event_poll_s` by default), until its events say it is done, so a run another process drives
 582        can be waited for. An attempt a newer epoch stopped raises its `StaleCoordinator`. While no thread of
 583        this `Run` is alive, a run whose latest coordinator ended or whose lease is past its stale bound
 584        (`Epoch.stopped`), with no live claim on any of its queues, is not waited for further: its status is returned
 585        with `interrupted` set.
 586
 587        Raises:
 588            BaseException: the error that stopped a run started here, other than its cancellation.
 589        """
 590        poll = self._pool.event_poll_s if poll is None else poll
 591        deadline = None if timeout is None else self._clock.now() + timeout
 592        thread = self._thread
 593        if thread is not None:
 594            if timeout is None:
 595                self._ended.wait()
 596            else:
 597                self._clock.wait(self._ended, timeout)
 598            if not thread.is_alive() and self._error is not None:
 599                raise self._error
 600        status = self.status()
 601        while not status.done and (deadline is None or self._clock.now() < deadline):
 602            if (thread is None or not thread.is_alive()) and self._stopped():
 603                return dataclasses.replace(self.status(), interrupted=True)
 604            self._clock.sleep(poll if deadline is None else max(0.0, min(poll, deadline - self._clock.now())))
 605            status = self.status()
 606        return status
 607
 608    def _stopped(self) -> bool:
 609        """Return whether the latest coordinator stopped (`Epoch.stopped`) and no worker holds a live claim."""
 610        return (
 611            Epoch(self.run_dir, self._clock).stopped()
 612            and not QueueSet.for_run(self.run_dir, clock=self._clock).claimed_any()
 613        )

 615    @property
 616    def cancelled(self) -> threading.Event:
 617        """The event `cancel` sets, which the runner reads."""
 618        return self._cancel
 619
 620    def fail(self, error: BaseException) -> None:
 621        """Hold `error`, the first one that stopped the run, for `wait` to raise."""
 622        if self._error is None:
 623            self._error = error
 624
 625    def launch(self, thread: threading.Thread) -> None:
 626        """Start `thread`, the run's body, and hold it for `wait`."""
 627        self._thread = thread
 628        thread.start()
 629        threading.Thread(target=self._signal_end, args=(thread,), name="senselab-run-ended", daemon=True).start()
 630
 631    def _signal_end(self, thread: threading.Thread) -> None:
 632        thread.join()
 633        self._ended.set()
 634
 635    def cancel(self) -> None:
 636        """Stop releasing jobs: fragments already sent finish or are left to their lease; no other is sent."""
 637        self._cancel.set()
 638
 639    def results(self) -> ResultSet:
 640        """Return what each record gave, read from the result store as each is asked for.
 641
 642        The events are read now, under the run's epoch (`status`); a record whose job failed, was blocked or cancelled
 643        gives that, and an earlier attempt's result is served only under the key the attempt gives its job.
 644
 645        Raises:
 646            FileNotFoundError: the run has no `plan.json`.
 647            StaleCoordinator: a newer epoch than this attempt's is current.
 648        """
 649        placed = self.plan
 650        graph = placed.logical.graph
 651        current = self._epoch()
 652        epoch = self.epoch if self.epoch is not None else current
 653        if epoch is not None and current is not None and current > epoch:
 654            raise StaleCoordinator(epoch, current)
 655        reader = _Reader(graph, self.events(), self._results_store(), epoch)
 656        per_record = [n for n in order(graph) if n in reader.mapped]
 657        outputs = sorted(graph.outputs)
 658        mine = [name for name in outputs if split(graph.outputs[name])[0] in per_record]
 659        entries: list[Entry] = [
 660            (Record(id=row.id, entities=row.entities), _later(reader, mine, i))
 661            for i, row in enumerate(placed.manifest.rows)
 662        ]
 663        shared = {name: _later(reader, [name], None) for name in outputs if name not in mine}
 664        return ResultSet(entries, shared, run=self)
 665
 666    def _results_store(self) -> ResultStore:
 667        if self._store is not None:
 668            return self._store
 669        return results_store(self._profile, self.manifest().get("store"))

 680def default_executor(placed: Plan, executor: Executor | None, store: ResultStore | None) -> Executor:
 681    """Return the executor to run `placed` on: the one given, else `InProcess` (local substrate) or `Pool` (slurm).
 682
 683    A plan read from disk carries no profile or run config: the packaged default profile is used if its digest is the
 684    one the plan was placed under. The `Pool` is over `Slurm` and the run's own queues, bound to the run's epoch by
 685    `start`.
 686
 687    Raises:
 688        ValueError: the default profile is not the one the plan was placed under.
 689    """
 690    if executor is not None:
 691        return executor
 692    profile = placed.execution if placed.execution is not None else ExecutionProfile.load()
 693    if profile.canonical_digest() != placed.placement.profile_digest:
 694        raise ValueError(f"plan {placed.hash} was placed under another execution profile; give the executor to run on")
 695    config = placed.config if placed.config is not None else RunConfig(reuse=ReusePolicy(placed.logical.reuse))
 696    if profile.substrate.kind == "local":
 697        from senselab_core.runtime.execution.in_process import InProcess  # noqa: PLC0415 - imports the planner
 698
 699        return InProcess(store=store, profile=profile, run=config)
 700    from senselab_core.runtime.execution.pool import Pool  # noqa: PLC0415 - imports the planner
 701    from senselab_core.runtime.execution.substrates import substrate_of  # noqa: PLC0415 - as above
 702
 703    if store is None:
 704        raise ValueError("a Pool over Slurm needs the result store it reads")
 705    run_dir = run_dir_of(placed.hash)
 706    return Pool(
 707        run_dir,
 708        queues=QueueSet.for_run(run_dir, profile, own=profile.lease),
 709        store=store,
 710        events=EventLog(run_dir, "a0", 1, epoch=0),
 711        profile=profile,
 712        substrate=substrate_of(profile, run_dir, placed.hash),
 713        epoch=0,
 714    )

 717def _record_placements(
 718    placed: Plan, executor: Executor, values: Mapping[InputIdentity, object], log: EventLog, epoch: int
 719) -> Callable[[], None]:
 720    """Append each class's placement to `log` for an executor that starts no scheduler jobs, as a `Pool` does itself.
 721
 722    The explicit values and the hook apply to every executor; one with no workers to place records them as given.
 723    Return what closes the epoch: it appends each class's record again with the run's wall time beside the request.
 724    """
 725    from senselab_core.runtime.execution.pool import Pool  # noqa: PLC0415 - imports the planner
 726
 727    execution = placed.execution
 728    if isinstance(executor, Pool) or execution is None:
 729        return lambda: None
 730    actuals = Actuals(SystemClock())
 731    ledger = Placements(
 732        log,
 733        lambda: placed.placement_request(values=values),
 734        actuals,
 735        lambda name: execution.resource_classes[name].device == "cuda",
 736    )
 737    placer = Placer(
 738        execution,
 739        placed.overrides,
 740        placed.hook,
 741        request=lambda: placed.placement_request(values=values),
 742        record=ledger.record,
 743        inert="this executor starts no scheduler jobs: the values are recorded, not applied",
 744    )
 745    names = sorted({p.resource_class for p in placed.placement.nodes.values()})
 746    for name in names:
 747        placer.for_class(name, epoch)
 748        actuals.began(name)
 749
 750    def close() -> None:
 751        for name in names:
 752            actuals.landed(name, ())
 753        ledger.close()
 754
 755    return close

 800def _kept(old: object, new: object, port: str, kind: str, expected: InputIdentity) -> object | None:
 801    """Return the source an input keeps, `old` recorded and `new` given; None when the two durable ones conflict.
 802
 803    A durable source (a path or JSON value) is never replaced by a `{"memory": ...}` marker, and replaces one. Two
 804    different durable sources keep the new one unless the old one still reads, as another identity than `expected`.
 805    """
 806    if old is None or old == new or _memory(old) is not None:
 807        return new
 808    if _memory(new) is not None:
 809        return old
 810    try:
 811        identity = bind_input(port, kind, old)[0]
 812    except InputUnreadable:
 813        return new
 814    return new if identity == expected else None

 817def merged_sources(placed: Plan, run_dir: Path) -> dict[str, Any]:
 818    """Return the input sources of `placed` merged, per record and port, over those `run_dir` records (`_kept`).
 819
 820    Raises:
 821        ConflictingSources: a recorded durable source reads another identity than the one the plan holds.
 822    """
 823    given = cast("dict[str, Any]", _durable(dict(placed.sources)))
 824    path = Path(run_dir) / INPUTS_FILE
 825    if not path.is_file():
 826        return given
 827    held = json.loads(path.read_text(encoding="utf-8"))
 828    types = port_types(placed)
 829    rows, conflicts = placed.manifest.rows, []
 830    merged_rows = []
 831    for row, old, new in zip(rows, held["rows"], given["rows"], strict=True):
 832        kept = {p: _kept(old.get(p), s, p, types[p], row.inputs[p]) for p, s in new.items()}
 833        conflicts += [(row.id, p) for p, s in kept.items() if s is None]
 834        merged_rows.append(kept)
 835    shared = {p: _kept(held["shared"].get(p), s, p, types[p], rows[0].inputs[p]) for p, s in given["shared"].items()}
 836    conflicts += [("*", p) for p, s in shared.items() if s is None]
 837    if conflicts:
 838        raise ConflictingSources(conflicts)
 839    return {"rows": merged_rows, "shared": shared}

 842def record_sources(placed: Plan, run_dir: Path) -> None:
 843    """Merge the input sources of `placed` into `run_dir`'s `inputs.json` (`merged_sources`) under its lease.
 844
 845    The lease, kept alive under the run's lease (`run_lease`, with the plan's profile's `lease` section among its
 846    epochs), is proven before the write, so two coordinators
 847    starting one plan merge one after the other and neither loses a durable source. A merge whose proof failed is
 848    made again, up to `lease.record_attempts` times.
 849
 850    Raises:
 851        ConflictingSources: as `merged_sources`; nothing is written.
 852        LeaseUnavailable: the merge's lease was lost on every attempt.
 853        LeaseTimeout: another holder kept the lease for its whole timeout.
 854    """
 855    lease = run_lease(run_dir, own=_lease_of(placed))
 856    private_dir(run_dir / PRIVATE_DIR)
 857    for _ in range(lease.record_attempts):
 858        held = Lease(run_dir / INPUTS_FILE, settings=lease)
 859        with held, held.keep_alive(lease.renew_fraction):
 860            sources = merged_sources(placed, run_dir)
 861            if held.prove():
 862                _write(run_dir / INPUTS_FILE, sources)
 863                return
 864    raise LeaseUnavailable(run_dir / INPUTS_FILE, f"the lease on {run_dir / INPUTS_FILE} was lost on every merge")

 867def prepare(placed: Plan, run_dir: Path, store: ResultStore | None) -> tuple[str, int]:
 868    """Write the run's plan, input sources, manifest and license policy once, then claim an attempt.
 869
 870    The digest of the stored policy is written beside the attempt's placement (`<attempt>.licenses`).
 871
 872    Nothing here is visible to a reader of the run's epochs or status. Returns the attempt id and its ordinal.
 873
 874    Raises:
 875        ValueError: `run_dir` holds another plan.
 876        ConflictingSources, LeaseUnavailable, LeaseTimeout: as `record_sources`; a source conflicts only with a
 877            recorded run, whose plan is already written, so nothing is written.
 878        LaxerLicensePolicy: the plan's policy allows what the run's stored policy refuses.
 879    """
 880    _write_once(run_dir / PLAN_FILE, _plan_document(placed))
 881    record_sources(placed, run_dir)
 882    for name in POOL_DIRS:
 883        (run_dir / name).mkdir(exist_ok=True)
 884    _write(run_dir / MANIFEST_FILE, {"plan_hash": placed.hash, "store": None if store is None else str(store.root)})
 885    digest = record_policy(run_dir, None if placed.config is None else placed.config.licenses)
 886    attempt, ordinal = _claim_attempt(run_dir, placed.placement)
 887    publish(run_dir / PLACEMENTS_DIR / f"{attempt}.licenses", digest.encode())
 888    return attempt, ordinal

 951def _open(
 952    run_dir: Path,
 953    given: tuple[Plan, Executor, ProvStore],
 954    attempt: tuple[str, int],
 955    leases: tuple[JobLease, JobLease],
 956) -> tuple[contextlib.ExitStack, int, EventLog]:
 957    """Take the next epoch (not yet current), hold its coordinator lease, reconcile the queues and bind the executor.
 958
 959    `given` is the plan, the executor and the provenance store the plan's entity is added to; `leases` are the plan's
 960    own lease section and the run's. Return what releases the epoch, the epoch and its event log; on a failure
 961    the epoch is released, marked ended, and the error raised.
 962    """
 963    placed, executor, written = given
 964    own, lease = leases
 965    epochs, held = Epoch(run_dir), contextlib.ExitStack()
 966    try:
 967        epoch = held.enter_context(epochs.opening(own))
 968        held.enter_context(epochs.held(epoch, lease))
 969        QueueSet.for_run(run_dir, placed.execution, own=own).reconcile()
 970        log = EventLog(run_dir, *attempt, epoch=epoch)
 971        executor.bind_epoch(epoch, log)
 972        add_plan_entity(written, placed.hash)
 973    except BaseException:
 974        held.close()
 975        raise
 976    return held, epoch, log

 994def start(
 995    placed: Plan,
 996    executor: Executor | None = None,
 997    *,
 998    store: ResultStore | None = None,
 999    values: Mapping[InputIdentity, object] | None = None,
1000) -> Run:
1001    """Start `placed` on `executor` in a background thread, as a new attempt, and return its `Run`.
1002
1003    With no executor, `InProcess` runs a plan under a local substrate, and a `Pool` over `Slurm` under a slurm one, over
1004    `store` (default: the result store under the cache root) with the plan's own profile and run config. `store` is also
1005    where the run's results are read from, so it is given with any other executor that stores results, and it is the
1006    store that executor writes: the plan's `senselab:Plan` entity is added to its provenance. The input values are bound
1007    (`bind_sources`) before anything is written: from their paths or JSON values, or from the values the plan holds in
1008    memory; `values` gives them already bound. The sources are merged into `inputs.json` under its lease
1009    (`record_sources`), also before anything else is written. Every item in the run's `queues/` with neither a slot nor
1010    an outcome is put again (`QueueSet.reconcile`). The run claims its attempt (`prepare`), then takes the next epoch,
1011    holds its coordinator lease until the run ends and hands the epoch to the executor (`bind_epoch`); the epoch is
1012    current only once its thread binds it (`Epoch.bind`), the last step before the plan runs.
1013
1014    A node whose unit does not fit its class runs on the smallest class it fits (`Plan.fit`).
