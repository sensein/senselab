senselab_core.runtime.execution.runner

The plan runner: a placed plan compiled to a pydra workflow, its jobs sent through an Executor (PydraRunner).

compile_plan turns a plan into a pydra workflow: a records node split over the manifest's rows, one node per graph node (a map edge reads the row, so every mapped node shares the one record axis, and a node with several mapped upstreams is split over them zipped), and, for each node a gather edge reads, a copy of it combined over the rows. Each node's checksum is the job's senselab key digest (job_checksum), so pydra hashes no senselab code. ExecutorWorker runs pydra's jobs: it sends a node's ready elements to the Executor as fragments of the placement's batch size (node_fragment), or as one fragment to an executor that splits fragments, and raises on a failed element, so pydra fails only that element's descendants. A fragment is submitted and returns at once; one Follower thread per run polls every open fragment and settles each element as its result lands. SenselabStore holds the run's completion records; no result is pickled. Its job leases take no lock file: the run's coordinator epoch is the one exclusion of a run (ADR-003). Each attempt has a pydra root of its own, <work dir>/<attempt>, so no two coordinators share a job directory or the lock pydra's save takes there, and nothing a killed coordinator leaves is on a later attempt's path.

118class SenselabStore:
119    """pydra's `ResultStore` over one run: completion records in memory by checksum and context slot, never pickled.
120
121    A job's results live in senselab's store, which the executor serves under the run's reuse policy; this store
122    holds what pydra reads to resolve a downstream input and to mark a job done, and `lookup` returns a record of its
123    own context's slot only. `cache_root` holds only the workflow job's own files; a lease writes nothing there.
124    """
125
126    def __init__(self, cache_root: Path) -> None:
127        """Hold the run's records under `cache_root`."""
128        self.cache_root = Path(cache_root)
129        self._records: dict[str, dict[str, Result]] = {}
130
131    def __getstate__(self) -> dict[str, Any]:
132        """Return the state a pickled job carries: the root, never the records."""
133        return {"cache_root": self.cache_root, "_records": {}}
134
135    def cache_dir(self, checksum: str, context: Mapping[str, Any]) -> Path:
136        """Return the directory of the job `checksum` in `context`."""
137        return self.cache_root / checksum / str(context_key(context))
138
139    def lookup(
140        self,
141        checksum: str,
142        context: Mapping[str, Any],
143        task: object,  # noqa: ARG002
144        stored: list[tuple[Path, Mapping[str, Any]]] | None = None,  # noqa: ARG002
145    ) -> Result | None:
146        """Return the record of `checksum` in `context`'s slot, or None."""
147        return self._records.get(checksum, {}).get(context_key(context))
148
149    def stored_contexts(self, checksum: str) -> Iterator[tuple[Path, Mapping[str, Any]]]:
150        """Yield the slot and context of each record of `checksum`."""
151        for slot, result in sorted(self._records.get(checksum, {}).items()):
152            yield self.cache_root / checksum / slot, result.context
153
154    def lockfile(self, checksum: str, context: Mapping[str, Any]) -> Path:
155        """Return the name pydra's `ResultStore` gives the lock of job `checksum` in `context`; nothing writes it."""
156        return self.cache_root / f"{checksum}-{context_key(context)}.lock"
157
158    def acquire(self, checksum: str, context: Mapping[str, Any]) -> Lease:
159        """Return the lease of the job `checksum` in `context`: one that takes no lock (`_EpochLease`)."""
160        return _EpochLease(checksum, self.cache_dir(checksum, context), self.lockfile(checksum, context))
161
162    def commit(self, lease: Lease, result: Result, job: object = None) -> None:  # noqa: ARG002
163        """Record `result` as the job's held under `lease`."""
164        self.record(lease.checksum, result)
165
166    def record(self, checksum: str, result: Result) -> None:
167        """Record `result` for `checksum` in its own context's slot."""
168        self._records.setdefault(checksum, {})[context_key(result.context)] = result
169
170    def checksums(self) -> list[str]:
171        """Return the checksum of every job recorded, sorted."""
172        return sorted(self._records)

205class _NodeKey:
206    """The checksum of a compiled node's job (`job_checksum`), from the identities its inputs resolve to."""
207
208    def __init__(self, placed: Plan, node: str) -> None:
209        """Key jobs of `node` of `placed`."""
210        self.placed = placed
211        self.node = node
212
213    def identities(self, inputs: Mapping[str, object]) -> dict[str, InputIdentity]:
214        """Return the input identity of each port, as `node_fragment` names it, from the job's resolved inputs."""
215        rows = self.placed.manifest.rows
216        found: dict[str, InputIdentity] = {}
217        for feed in _feeds(self.placed, self.node):
218            value = inputs[feed.port]
219            if feed.source == "input":
220                row = typing.cast(Mapping[str, int], value)[RECORD_FIELD] if feed.edge.kind == "map" else 0
221                found[feed.port] = rows[row].inputs[feed.source_port]
222            elif feed.edge.kind == "gather":
223                ports = [port_output_identity(r.key_digest, feed.source_port) for r in _refs(value, feed.source)]
224                found[feed.port] = gather_identity(ports)
225            else:
226                found[feed.port] = port_output_identity(_ref(value).key_digest, feed.source_port)
227        return found
228
229    def checksum(self, inputs: Mapping[str, object]) -> str:
230        """Return the job's `job_checksum`."""
231        return job_checksum(self.placed, self.node, self.identities(inputs))

267def compile_plan(placed: Plan) -> Any:  # noqa: ANN401 - a pydra workflow task
268    """Return `placed` as a pydra workflow task, every node's checksum the senselab key digest of its job.
269
270    A `map` edge from a graph input reads the `records` node's row; a broadcast graph input is the one identity every
271    row holds; a node with several mapped upstreams (the records included) is split over them zipped; a `gather` edge
272    reads a copy of its source combined over `records.row`. Each node carries its resource class as a routing tag.
273
274    Raises:
275        ValueError: the graph has a nested fan-out (`fan_out_issues`).
276        UnmappedOutput: a `map` edge reads a node's output (`check_mapped_outputs`).
277    """
278    graph = placed.logical.graph
279    refused = fan_out_issues(graph)
280    if refused:
281        raise ValueError("; ".join(i.message for i in refused))
282    check_mapped_outputs(graph)
283    mapped = mapped_nodes(placed.logical.graph)
284    classes = {n.id: _node_class(placed, n.id) for n in graph.nodes}
285    records = python.define(
286        _record, inputs={RECORD_FIELD: Any}, outputs={REF: Any}, name=RECORDS, checksum=_records_key(placed.hash)
287    )
288    records.senselab_node = None
289    gathered = {split(e.source)[0] for e in graph.edges if e.kind == "gather"}
290    ordered = order(graph)
291
292    def construct(rows: list[int]) -> Any:  # noqa: ANN401 - the lazy outputs pydra reads
293        added: dict[str, Any] = {}
294        if mapped:
295            added[RECORDS] = workflow.add(records().split(row=rows), name=RECORDS)
296        outputs = []
297        for node in ordered:
298            resources = ComputeResources(tags=(placed.placement.nodes[node].resource_class,))
299            added[f"n_{node}"] = workflow.add(
300                _wired(placed, node, classes, added, mapped), name=f"n_{node}", resources=resources
301            )
302            if node in gathered:
303                combined = _wired(placed, node, classes, added, mapped).combine(f"{RECORDS}.{RECORD_FIELD}")
304                added[f"g_{node}"] = workflow.add(combined, name=f"g_{node}", resources=resources)
305            outputs.append(getattr(added[f"n_{node}"], REF))
306        return tuple(outputs)
307
308    names = {f"n_{node}": Any for node in ordered}
309    plan_check = f"plan-{placed.hash}"
310    return workflow.define(
311        construct, inputs={"rows": list[int]}, outputs=names, name="plan", checksum=lambda t, i: plan_check
312    )(rows=list(range(len(placed.manifest.rows))))

315def _wired(placed: Plan, node: str, classes: Mapping[str, Any], added: Mapping[str, Any], mapped: set[str]) -> Any:  # noqa: ANN401 - a pydra task
316    """Return a task of `node`, each input wired to its lazy source, split over its mapped upstreams zipped."""
317    values: dict[str, Any] = {}
318    upstreams: list[str] = []
319    rows = placed.manifest.rows
320    for feed in _feeds(placed, node):
321        if feed.source == "input" and feed.edge.kind == "map":
322            values[feed.port] = getattr(added[RECORDS], REF)
323            upstreams.append(RECORDS)
324        elif feed.source == "input":
325            values[feed.port] = rows[0].inputs[feed.source_port] if rows else None
326        elif feed.edge.kind == "gather":
327            values[feed.port] = getattr(added[f"g_{feed.source}"], REF)
328        else:
329            values[feed.port] = getattr(added[f"n_{feed.source}"], REF)
330            if feed.source in mapped:
331                upstreams.append(f"n_{feed.source}")
332    task = classes[node](**values)
333    states = sorted(set(upstreams))
334    return task.split(tuple(f"_{u}" for u in states)) if len(states) > 1 else task

338class _Run:
339    """One run's state, shared by its workers: the plan, admission and its values, executor, batches and outcomes."""
340
341    placed: Plan
342    admission: Admission
343    values: Mapping[InputIdentity, object]
344    executor: Executor
345    store: SenselabStore
346    events: EventLog
347    cancel: threading.Event
348    senders: ThreadPoolExecutor
349    follower: "Follower"
350    pending: dict[str, list[tuple[Any, tuple[int, ...], asyncio.Future[ResultRef]]]] = field(default_factory=dict)
351    scheduled: set[str] = field(default_factory=set)
352    results: dict[str, dict[tuple[int, ...], ResultRef]] = field(default_factory=dict)
353    errors: list[BaseException] = field(default_factory=list)
354    sending: set[asyncio.Task[None]] = field(default_factory=set)
355    element_failures: set[str] = field(default_factory=set)
356    ended: set[tuple[str, tuple[int, ...]]] = field(default_factory=set)
357    settled: dict[str, ResultRef] = field(default_factory=dict)
358
359    def expected(self) -> list[tuple[str, tuple[int, ...]]]:
360        """Return every job of the plan as (node, element): a mapped node's per record, any other once."""
361        graph = self.placed.logical.graph
362        mapped = mapped_nodes(graph)
363        rows = len(self.placed.manifest.rows)
364        return [(n, (i,)) for n in order(graph) if n in mapped for i in range(rows)] + [
365            (n, ()) for n in order(graph) if n not in mapped
366        ]
367
368    def fragment_size(self, node: str) -> int | None:
369        """Return the most jobs of `node` one fragment holds, as the executor sizes it; None for one that splits."""
370        return None if self.executor.splits_fragments else self.executor.fragment_size(self.placed, node)
371
372    def close(self) -> None:
373        """Give every job with no terminal event its own: `cancelled` once cancelled, else `blocked`.
374
375        A run stopped by a held env or a newer coordinator (`Interrupted`) leaves them without one, for a later epoch.
376        """
377        if any(isinstance(e, Interrupted) for e in self.errors):
378            return
379        for node, element in self.expected():
380            if (node, element) not in self.ended:
381                self.log("cancelled" if self.cancel.is_set() else "blocked", node, element)
382
383    def fail_batch(self, node: str, elements: list[tuple[int, ...]], error: Exception) -> None:
384        """Log `failed` for each of `elements` with no result, with the record of `error` (`from_exception`)."""
385        failure = FaultRecord.from_exception(error)
386        for element in elements:
387            if (node, element) not in self.ended:
388                self.log("failed", node, element, failure=failure)
389
390    def settle(self, node: str, ref: ResultRef) -> None:
391        """Hold `ref` as its element's result and log it `failed`, `cache_hit` or `finished`."""
392        self.results.setdefault(node, {})[ref.job.element] = ref
393        kind: EventKind = "failed" if ref.failure is not None else "cache_hit" if ref.served else "finished"
394        self.log(
395            kind,
396            node,
397            ref.job.element,
398            key=ref.key_digest,
399            failure=ref.failure,
400            output=ref.output_identity,
401            reuse=ref.served,
402        )
403
404    def serve_shared(self) -> None:
405        """Settle each job pydra shared with a settled job of the same checksum, which it never sent.
406
407        A job whose upstream results all landed is given the failure of a failed same-key job, or is sent again, so its
408        executor serves it the stored result as a reuse, or runs it when the store's hit rules refuse the entry. Nodes
409        are taken in graph order, so a shared job's descendants read its result; each fragment is followed to its end
410        before the next is sent.
411        """
412        if self.errors or self.cancel.is_set():
413            return
414        mapped = mapped_nodes(self.placed.logical.graph)
415        for node in order(self.placed.logical.graph):
416            again: list[tuple[int, ...]] = []
417            upstream: dict[str, list[ResultRef]] = {}
418            for i in range(len(self.placed.manifest.rows)) if node in mapped else ():
419                read = None if (node, (i,)) in self.ended else self._read(node, (i,), mapped)
420                shared = None if read is None else self.settled.get(_NodeKey(self.placed, node).checksum(read[0]))
421                if read is None or shared is None:
422                    continue
423                if shared.failure is not None:
424                    self.settle(node, _at(shared, node, (i,)))
425                    continue
426                again.append((i,))
427                for source, ref in read[1].items():
428                    upstream.setdefault(source, []).append(ref)
429            size = self.fragment_size(node) or max(1, len(again))
430            for start in range(0, len(again), size):
431                handle = self.send(node, again[start : start + size], upstream)
432                for ref in follow(self.executor, handle, first_s=self.follower.first_s, max_s=self.follower.max_s):
433                    self.settle(node, ref)
434
435    def _read(
436        self, node: str, element: tuple[int, ...], mapped: set[str]
437    ) -> tuple[dict[str, object], dict[str, ResultRef]] | None:
438        """Return the inputs and upstream results of the job of `node` at `element`, None if one is missing.
439
440        An upstream result is held at `element` of a mapped source and at `()` of any other, and must have succeeded.
441        """
442        inputs: dict[str, object] = {}
443        refs: dict[str, ResultRef] = {}
444        for feed in _feeds(self.placed, node):
445            if feed.source == "input":
446                inputs[feed.port] = {RECORD_FIELD: element[0]}
447                continue
448            ref = self.results.get(feed.source, {}).get(element if feed.source in mapped else ())
449            if ref is None or ref.failure is not None:
450                return None
451            inputs[feed.port] = refs[feed.source] = ref
452        return inputs, refs
453
454    def send(self, node: str, elements: list[tuple[int, ...]], upstream: Mapping[str, Sequence[ResultRef]]) -> Handle:
455        """Submit the jobs of `node` at `elements` as one fragment, reading `upstream`, and return its handle.
456
457        Each job is logged `started` as it is sent, unless the executor logs its own claims (`logs_claims`).
458
459        Raises:
460            RunCancelled: the run was cancelled before the fragment was sent.
461        """
462        if self.cancel.is_set():
463            raise RunCancelled(f"run {self.placed.hash} was cancelled before node {node!r} was sent")
464        fragment = node_fragment(self.placed, self.admission, node, upstream=upstream, elements=elements)
465        if not self.executor.logs_claims:
466            for job, key in zip(fragment.jobs, fragment_keys(fragment, self.placed, self.values), strict=True):
467                self.log("started", node, job.element, key=key)
468        return self.executor.submit(fragment)

473        node: str,

475        *,

503class ExecutorWorker(Worker):  # type: ignore[misc]
504    """A pydra worker sending a compiled node's jobs to a senselab `Executor`, by batch size or by ready set.
505
506    Jobs of one node that become ready together are sent as fragments of at most the placement's batch size, or as one
507    fragment per ready set to an executor that splits fragments (`splits_fragments`). A job returns when its element's
508    result arrives and raises `ElementFailed` when the executor failed it.
509    """
510
511    _plugin_name = "senselab-executor"
512
513    def __init__(self, run_state: _Run | None = None, resource_class: str | None = None) -> None:
514        """Run the jobs of `run_state`, those routed to `resource_class` (None for the submitter's own worker)."""
515        super().__init__()
516        self.run_state = run_state
517        self.resource_class = resource_class
518
519    def __getstate__(self) -> dict[str, Any]:
520        """Return the state a pickled job carries: no run, whose executor lives in this process."""
521        return {"loop": None, "run_state": None, "resource_class": self.resource_class}
522
523    async def run(self, job: Any, rerun: bool = False) -> Result:  # noqa: ANN401, ARG002, FBT001, FBT002
524        """Return the job's result once its executor gave it; raise `ElementFailed` for a failed element."""
525        run = self.run_state
526        assert run is not None, "an unpickled ExecutorWorker has no run"
527        node = getattr(type(job.task), "senselab_node", None)
528        if node is None:
529            outputs: object = {RECORD_FIELD: int(job.task.row)}
530        else:
531            try:
532                outputs = (await self._submit(run, node, job)).model_dump(mode="json")
533            except ElementFailed:
534                index = job.state_index
535                run.element_failures.add(job.name if index is None else f"{job.name}({index})")
536                raise
537        result = Result(
538            outputs=job.task.Outputs(**{REF: outputs}),
539            runtime=None,
540            errored=False,
541            cache_dir=run.store.cache_dir(job.checksum, job.context),
542            task=job.task,
543            context=job.context,
544        )
545        run.store.record(job.checksum, result)

548    async def _submit(self, run: _Run, node: str, job: Any) -> ResultRef:  # noqa: ANN401 - a pydra job
549        loop = asyncio.get_running_loop()
550        future: asyncio.Future[ResultRef] = loop.create_future()
551        element = _element(job)
552        run.pending.setdefault(node, []).append((job, element, future))
553        size = run.fragment_size(node)
554        if size is not None and len(run.pending[node]) >= size:
555            self._flush(run, node)
556        elif node not in run.scheduled:
557            run.scheduled.add(node)
558            loop.call_soon(self._flush, run, node)
559        ref = await future
560        run.settled[job.checksum] = ref
561        if ref.failure is not None:
562            raise ElementFailed(ref.failure)
563        return ref
564
565    def _flush(self, run: _Run, node: str) -> None:
566        run.scheduled.discard(node)
567        waiting = run.pending.pop(node, [])
568        size = run.fragment_size(node)
569        if size is None:
570            size = max(1, len(waiting))
571        for start in range(0, len(waiting), size):
572            sending = asyncio.ensure_future(self._send(run, node, waiting[start : start + size]))
573            run.sending.add(sending)
574            sending.add_done_callback(run.sending.discard)
575
576    async def _send(self, run: _Run, node: str, batch: list[tuple[Any, tuple[int, ...], Any]]) -> None:
577        """Dispatch `batch` as one fragment; once every result it gave is settled, fail the elements it gave none."""
578        futures = {element: future for _, element, future in batch}
579
580        def settle(ref: ResultRef) -> None:
581            run.settle(node, ref)
582            futures[ref.job.element].set_result(ref)
583
584        try:
585            error = await _dispatched(run, node, batch, settle)
586        except BaseException as signal:
587            for future in futures.values():
588                if not future.done():
589                    future.set_exception(signal)
590            raise
591        if error is not None:
592            run.errors.append(error)
593            if not isinstance(error, (RunCancelled, *Interrupted)):
594                run.fail_batch(node, list(futures), error)
595        missing = [element for element, future in futures.items() if not future.done()]
596        if missing and not run.errors:
597            run.errors.append(RuntimeError(f"the executor gave node {node!r} no result for elements {missing}"))
598        for element in missing:
599            futures[element].set_exception(run.errors[-1])

602async def _dispatched(
603    run: _Run, node: str, batch: list[tuple[Any, tuple[int, ...], Any]], settle: Callable[[ResultRef], None]
604) -> Exception | None:
605    """Submit `batch` on the run's senders, then await the follower, each result passed to `settle` on this loop.
606
607    No thread is held while the fragment's results are awaited. Returns once every result given is settled, with the
608    error the submission or the following raised, or None.
609    """
610    loop = asyncio.get_running_loop()
611    ended: asyncio.Future[Exception | None] = loop.create_future()
612
613    def end(error: Exception | None) -> None:
614        if not ended.done():
615            ended.set_result(error)
616
617    def soon(callback: Callable[..., None], *args: object) -> None:
618        with contextlib.suppress(RuntimeError):
619            loop.call_soon_threadsafe(callback, *args)
620
621    try:
622        elements, upstream = [element for _, element, _ in batch], _upstream(run.placed, node, batch)
623        handle = await loop.run_in_executor(
624            run.senders, contextvars.copy_context().run, run.send, node, elements, upstream
625        )
626    except Exception as raised:  # noqa: BLE001 - the caller fails the batch with it, and the run re-raises it
627        return raised
628    run.follower.follow(handle, lambda ref: soon(settle, ref), lambda error: soon(end, error))
629    return await ended

632class Follower:
633    """One thread following every open handle of a run, `Executor.poll` per handle per round, with no thread each.
634
635    Each result is passed to its handle's `give`, and `end` is called once per handle: with None once every result
636    was given, else with the error its poll raised, or `RunCancelled` once `cancel` is set (the handle is cancelled
637    through the executor after its landed results are given). A round that gives nothing waits `first_s`, doubling up
638    to `max_s`; a new handle or `cancel` ends the wait. The thread runs in the context the follower was made in.
639    """
640
641    def __init__(  # noqa: PLR0913 - the executor and cancel, then keyword-only waits and the clock
642        self,
643        executor: Executor,
644        cancel: threading.Event,
645        *,
646        first_s: float,
647        max_s: float,
648        cancel_check_s: float,
649        clock: Clock = SystemClock(),  # noqa: B008 - stateless
650    ) -> None:
651        """Follow the handles of `executor`, stopping at `cancel`; the waits between rounds run on `clock`.
652
653        A waiting follower checks `cancel` every `cancel_check_s` seconds.
654        """
655        self.executor, self.cancel, self.first_s, self.max_s, self._clock = executor, cancel, first_s, max_s, clock
656        self.cancel_check_s = cancel_check_s
657        self._open: dict[str, tuple[Handle, Callable[[ResultRef], object], Callable[[Exception | None], object]]] = {}
658        self._lock = threading.Lock()
659        self._wake = threading.Event()
660        self._closed = False
661        self._thread: threading.Thread | None = None
662        self._context = contextvars.copy_context()
663
664    def follow(
665        self, handle: Handle, give: Callable[[ResultRef], object], end: Callable[[Exception | None], object]
666    ) -> None:
667        """Follow `handle` from the next round, which starts at once."""
668        with self._lock:
669            self._open[handle.id] = (handle, give, end)
670            if self._thread is None:
671                self._thread = threading.Thread(
672                    target=self._context.run, args=(self._loop,), name="senselab-follower", daemon=True
673                )
674                self._thread.start()
675        self._wake.set()
676
677    def open_handles(self) -> int:
678        """Return the handles followed and not yet ended."""
679        with self._lock:
680            return len(self._open)
681
682    def _round(self) -> bool:
683        """Poll every open handle once; return whether any result was given."""
684        with self._lock:
685            followed = list(self._open.values())
686        gave = False
687        for handle, give, end in followed:
688            try:
689                refs, done = self.executor.poll(handle)
690            except Exception as error:  # noqa: BLE001 - ended with it; the runner re-raises it
691                self._end(handle, end, error)
692                continue
693            for ref in refs:
694                give(ref)
695            gave = gave or bool(refs)
696            if done:
697                self._end(handle, end, None)
698            elif self.cancel.is_set():
699                self.executor.cancel(handle)
700                self._end(handle, end, RunCancelled(f"the run was cancelled while {handle.id} was followed"))
701        return gave
702
703    def _end(self, handle: Handle, end: Callable[[Exception | None], object], error: Exception | None) -> None:
704        with self._lock:
705            self._open.pop(handle.id, None)
706        end(error)
707
708    def _loop(self) -> None:
709        pause = self.first_s
710        while not self._closed:
711            self._wake.clear()
712            if self._round():
713                pause = self.first_s
714            elif self.open_handles():
715                self._nap(pause)
716                pause = min(2 * pause, self.max_s)
717            else:
718                self._wake.wait()
719
720    def _nap(self, seconds: float) -> None:
721        """Wait up to `seconds`, ending early on a new handle, on `close`, or once `cancel` is set."""
722        deadline = self._clock.now() + seconds
723        while not self._wake.is_set() and not self.cancel.is_set() and (left := deadline - self._clock.now()) > 0:
724            self._clock.wait(self._wake, min(left, self.cancel_check_s))
725
726    def close(self) -> None:
727        """Stop following; open handles are left as they are."""
728        self._closed = True
729        self._wake.set()
730        if self._thread is not None:
731            self._thread.join()

735    """Return each upstream node's results the jobs of `batch` read, from their resolved inputs.
736
737    A per-element result is held at the reading job's element: a result pydra shared is another element's.
738    """
739    found: dict[str, dict[tuple[int, ...], ResultRef]] = {}
740    for job, element, _ in batch:

771

778        self.work_dir, self._clock = work_dir, clock

780            senders if senders is not None else sum(c.max_workers for c in self.profile.resource_classes.values())

785    def run_plan(  # noqa: PLR0913 - the plan, its executor and values, then keyword-only run inputs
786        self,
787        placed: Plan,
788        executor: Executor,
789        *,
790        values: Mapping[InputIdentity, object],
791        events: EventLog,
792        store: SenselabStore | None = None,
793        cancel: threading.Event | None = None,
794    ) -> RunOutcome:

796
797        Each job is first logged `queued` with its key under this plan (`plan_keys`).
798
799        A failed element fails only its descendants, which are reported `blocked`; every other element runs. A job
800        sharing its key with another job of the run is that job's reuse, or its failure (`_Run.serve_shared`). Each
801        job's `queued`, `started` and `finished` or `failed` event is appended to `events`. Once `cancel` is set no
802        further fragment is sent: running fragments finish and the run raises `RunCancelled`. While it runs,
803        `attempt_id()` is the attempt of `events`. pydra's files are under `<work dir>/<attempt of events>`.
804
805        Raises:
806            ValueError: as `compile_plan`, or as `executor.admit`.
807            RunCancelled: `cancel` was set before every fragment was sent.
808            WorkflowError: pydra errored a job other than an element the executor failed, or named no job.
809            BaseException: an exception pydra raised its `WorkflowError` over, such as a cancellation.
810            Exception: the first error an executor call raised, or a result it did not give, after the run.
811        """
812        with attempt_scope(events.attempt):
813            return self._run_plan(placed, executor, values, events, store, cancel)

815    def _run_plan(  # noqa: PLR0913 - as `run_plan`
816        self,
817        placed: Plan,
818        executor: Executor,
819        values: Mapping[InputIdentity, object],
820        events: EventLog,
821        store: SenselabStore | None,
822        cancel: threading.Event | None,
823    ) -> RunOutcome:
824        if self.work_dir is None:
825            private_dir(run_dir_of(placed.hash) / PRIVATE_DIR)
826        base = private_dir(self.work_dir if self.work_dir is not None else run_dir_of(placed.hash) / PYDRA_DIR)
827        root = private_dir(base / events.attempt)
828        store = store if store is not None else SenselabStore(root)
829        admission = executor.admit(placed, values)
830        cancel = cancel if cancel is not None else threading.Event()
831        pool = self.profile.pool
832        senders = ThreadPoolExecutor(self.senders, thread_name_prefix="senselab-send")
833        follower = Follower(
834            executor,
835            cancel,
836            first_s=pool.first_poll_s,
837            max_s=pool.poll_s,
838            cancel_check_s=pool.cancel_check_s,
839            clock=self._clock,
840        )
841        run = _Run(placed, admission, values, executor, store, events, cancel, senders, follower)
842        keys = plan_keys(placed, values)
843        for node, element in run.expected():
844            run.log("queued", node, element, key=keys.get((node, element)))
845        classes = sorted({p.resource_class for p in placed.placement.nodes.values()})
846        try:
847            _submit_plan(run, store, classes)
848            run.serve_shared()
849        finally:
850            follower.close()
851            senders.shutdown(cancel_futures=True)
852            executor.release(admission)
853            run.close()
854        return _outcome(placed, run)

634    """One thread following every open handle of a run, `Executor.poll` per handle per round, with no thread each.
635
636    Each result is passed to its handle's `give`, and `end` is called once per handle: with None once every result
637    was given, else with the error its poll raised, or `RunCancelled` once `cancel` is set (the handle is cancelled
638    through the executor after its landed results are given). A round that gives nothing waits `first_s`, doubling up
639    to `max_s`; a new handle or `cancel` ends the wait. The thread runs in the context the follower was made in.
640    """
641
642    def __init__(  # noqa: PLR0913 - the executor and cancel, then keyword-only waits and the clock

667    ) -> None:
668        """Follow `handle` from the next round, which starts at once."""
669        with self._lock:
670            self._open[handle.id] = (handle, give, end)
671            if self._thread is None:
672                self._thread = threading.Thread(
673                    target=self._context.run, args=(self._loop,), name="senselab-follower", daemon=True
674                )
675                self._thread.start()

685        with self._lock:

687        gave = False
688        for handle, give, end in followed:
689            try:
