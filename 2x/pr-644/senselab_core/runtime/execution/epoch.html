senselab_core.runtime.execution.epoch

A run's coordinator epochs: epoch/<n>, each taken by one mkdir, held under a beating lease (ADR-003).

epoch/<n>/ holds lease.json (its coordinator's whole lease section, run-lease@2, written as it is taken), attempt.json (the attempt id and ordinal the coordinator claimed under it, written last), coordinator.lease (the Lease its coordinator keeps alive) and, once that coordinator ends, ended. The highest n with an attempt.json is the current epoch; a coordinator of an older one releases no further work.

 80def read_section(directory: Path) -> JobLease:
 81    """Return the lease section `directory/lease.json` holds.
 82
 83    Raises:
 84        RunLeaseUnreadable: it is missing, is no `run-lease@2` document, or lacks a key `JobLease` requires.
 85    """
 86    path = directory / BOUND_FILE
 87    try:
 88        found = json.loads(path.read_text(encoding="utf-8"))
 89        if not isinstance(found, dict) or found.get("format") != LEASE_FORMAT:
 90            raise ValueError(
 91                f"its format is {found.get('format') if isinstance(found, dict) else None!r}, not {LEASE_FORMAT}"
 92            )
 93        return JobLease.model_validate(found["lease"])
 94    except (OSError, ValueError, KeyError) as err:
 95        raise RunLeaseUnreadable(path, err) from err

 98def combined(sections: Sequence[JobLease]) -> JobLease:
 99    """Return the lease every judge of a run uses.
100
101    It has the longest bounds and wait, the shortest renewal and the rest of the newest's.
102
103    `sections` run oldest first; the last is the newest. With none, the packaged profile's section.
104    """
105    if not sections:
106        return ExecutionProfile.load().lease
107    stale_after_s = max(s.stale_after_s for s in sections)
108    attribute_delay_s = max(s.attribute_delay_s for s in sections)
109    return sections[-1].model_copy(
110        update={
111            "stale_after_s": stale_after_s,
112            "attribute_delay_s": attribute_delay_s,
113            "renew_fraction": min(s.renew_fraction for s in sections),
114            "wait_s": max(*(s.wait_s for s in sections), stale_after_s + attribute_delay_s),
115        }
116    )

119def run_lease(run_dir: Path, own: JobLease | None = None) -> JobLease:
120    """Return the lease bound every judge and holder of the run in `run_dir` uses (`combined`).
121
122    It is built from the lease section of every epoch with an `attempt.json`, oldest first, and `own` last when given.
123
124    Raises:
125        RunLeaseUnreadable: a bound epoch's section is unreadable.
126    """
127    epochs = Epoch(run_dir)
128    sections = [read_section(epochs.root / str(n)) for n in epochs.bound()]
129    return combined([*sections, own] if own is not None else sections)

137class Epoch:
138    """The epochs of the run in `run_dir`."""
139
140    def __init__(self, run_dir: Path, clock: Clock = SystemClock()) -> None:  # noqa: B008 - stateless
141        """Hold the epochs under `run_dir/epoch`; their leases and the storage clock are read on `clock`."""
142        self.root = Path(run_dir) / EPOCH_DIR
143        self._clock = clock
144
145    def bound(self) -> list[int]:
146        """Return the epochs with an `attempt.json`, oldest first."""
147        if not self.root.is_dir():
148            return []
149        found = (int(n) for n in os.listdir(self.root) if EPOCH_NAME.fullmatch(n))
150        return sorted(n for n in found if (self.root / str(n) / ATTEMPT_FILE).exists())
151
152    def _taken(self) -> int:
153        """Return the highest epoch directory, 0 when none was made."""
154        if not self.root.is_dir():
155            return 0
156        return max((int(n) for n in os.listdir(self.root) if EPOCH_NAME.fullmatch(n)), default=0)
157
158    def current(self) -> int:
159        """Return the highest bound epoch (one with an `attempt.json`), 0 when none is."""
160        return max(self.bound(), default=0)
161
162    def _make(self) -> int:
163        """Make the next epoch directory by `mkdir`, skipping a number another coordinator made, and return `n`."""
164        self.root.mkdir(parents=True, exist_ok=True)
165        n = self._taken() + 1
166        while True:
167            try:
168                (self.root / str(n)).mkdir()
169            except FileExistsError:
170                n += 1
171                continue
172            return n
173
174    def take(self, lease: JobLease | None = None) -> int:
175        """Take the next epoch by `mkdir epoch/<n>`, record `lease` in it, and return `n`.
176
177        `lease` is the packaged profile's `lease` section when None. The epoch is current only once `bind` is called.
178        """
179        n = self._make()
180        _create(self.root / str(n) / BOUND_FILE, _document(lease))
181        return n
182
183    @contextlib.contextmanager
184    def opening(self, lease: JobLease | None = None) -> Iterator[int]:
185        """Take the next epoch, recording `lease` in it, and yield it; on exit mark it ended unless it was bound.
186
187        Until `bind`, the epoch is current to no reader: it fences no coordinator and changes no status.
188        """
189        n = self._make()
190        try:
191            _create(self.root / str(n) / BOUND_FILE, _document(lease))
192            yield n
193        finally:
194            if not (self.root / str(n) / ATTEMPT_FILE).exists():
195                with contextlib.suppress(OSError):
196                    _create(self.root / str(n) / ENDED, b"")
197
198    def bind(self, n: int, attempt: str, ordinal: int) -> None:
199        """Make epoch `n` current: record the attempt, the `ordinal`-th, its coordinator claimed; written once.
200
201        The last step of a coordinator's setup, so a coordinator that did not become ready was never current.
202        """
203        _create(self.root / str(n) / ATTEMPT_FILE, json.dumps({"attempt": attempt, "ordinal": ordinal}).encode())
204
205    def attempt(self, n: int) -> tuple[str, int] | None:
206        """Return the attempt id and ordinal epoch `n`'s coordinator claimed, None before it claimed one."""
207        try:
208            found = json.loads((self.root / str(n) / ATTEMPT_FILE).read_text(encoding="utf-8"))
209        except (FileNotFoundError, ValueError):
210            return None
211        return str(found["attempt"]), int(found["ordinal"])
212
213    @contextlib.contextmanager
214    def held(self, n: int, lease: JobLease) -> Iterator[LeaseState]:
215        """Hold epoch `n`'s coordinator lease, kept alive under `lease`, for the block; mark the epoch ended after.
216
217        Raises:
218            LeaseTimeout: another coordinator holds epoch `n`.
219        """
220        directory = self.root / str(n)
221        held = Lease(directory / COORDINATOR, settings=lease, timeout=0, clock=self._clock)
222        try:
223            with held, held.keep_alive(lease.renew_fraction) as state:
224                yield state
225        finally:
226            _create(directory / ENDED, b"")
227
228    def stopped(self) -> bool:
229        """Return whether the current epoch's coordinator ended, or its lease is older than the run's stale bound.
230
231        A lease is aged on the storage clock, against `run_lease`. With no epoch bound, the highest epoch taken is
232        judged as it is: by its `ended` mark, else by its directory's age against the bound its own `lease.json` holds
233        (the packaged profile's when unreadable). False when no epoch was taken.
234        """
235        n = self.current() or self._taken()
236        if n == 0:
237            return False
238        directory = self.root / str(n)
239        if (directory / ENDED).exists():
240            return True
241        if not (directory / ATTEMPT_FILE).exists():
242            try:
243                bound = read_section(directory)
244            except RunLeaseUnreadable:
245                bound = combined([])
246        else:
247            bound = run_lease(self.root.parent)
248        age = Lease(directory / COORDINATOR, settings=bound, clock=self._clock).age()
249        if age is None:
250            with StorageDir.open_root(self.root) as opened:
251                age = self._clock.storage_now(opened) - directory.stat().st_mtime
252        return bool(age > bound.stale_after_s + bound.attribute_delay_s)

def combined( sections: Sequence[senselab_core.runtime.execution_settings.JobLease]) -> senselab_core.runtime.execution_settings.JobLease: 
 99def combined(sections: Sequence[JobLease]) -> JobLease:
100    """Return the lease every judge of a run uses.
101
102    It has the longest bounds and wait, the shortest renewal and the rest of the newest's.
103
104    `sections` run oldest first; the last is the newest. With none, the packaged profile's section.
105    """
106    if not sections:
107        return ExecutionProfile.load().lease
108    stale_after_s = max(s.stale_after_s for s in sections)
109    attribute_delay_s = max(s.attribute_delay_s for s in sections)
110    return sections[-1].model_copy(
111        update={
112            "stale_after_s": stale_after_s,
113            "attribute_delay_s": attribute_delay_s,
114            "renew_fraction": min(s.renew_fraction for s in sections),
115            "wait_s": max(*(s.wait_s for s in sections), stale_after_s + attribute_delay_s),
116        }
117    )
