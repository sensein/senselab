senselab_core.runtime.execution.supervisor

The supervisor of the processes a host launches: its table of them, their orphans, and a keeper outliving the host.

Every process senselab starts goes through Supervisor.popen, in a new session, so it leads its own process group. The supervisor in force (current) is the innermost one entered in this process, else a process-wide one that supervises by process group alone.

An entered supervisor on Linux sets PR_SET_CHILD_SUBREAPER, so a descendant orphaned by its parent's exit (a fork that never execs, a setsid escapee) is re-parented to this process and enters the table as an orphan. Orphans are found in the kernel's list of this process's children. An orphan in a launched group is a member of it; one in any other group is held for the groups launched when it was adopted and is SIGKILLed once each of them is released, at once when there were none. A child started in its own session by code that bypasses the supervisor is taken for such an orphan. A group is live while its leader has not exited or one of its orphans has not (group_live). Orphans are reaped one by one, by pid; a launched process is never adopted or reaped here until its owner has reaped it. Children that were already running when it was entered, their groups, and children of this process's own group are never adopted.

Elsewhere, and in the process-wide supervisor, a group is supervised by its process group: a setsid escapee is not reached, and liveness is probed by group (live_members).

Every launch, and every orphan an entered supervisor adopts, is also recorded with the keeper, one process per supervisor, in its own session, that reads the records on its stdin. Once this process has died, by any signal, the keeper SIGKILLs every group and orphan still recorded whose start time it reads unchanged. Leaving an entered supervisor SIGKILLs every group and orphan it holds, reaps the orphans and the leaders, stops the keeper and restores the subreaper flag.

 56def read_start_time(pid: int, probe_s: float) -> str | None:
 57    """Return the start time of process `pid`, or None when no process has that pid.
 58
 59    On Linux it is field 22 of `/proc/<pid>/stat`; elsewhere it is `ps -o lstart=`, which may take `probe_s` seconds.
 60    The keeper runs this function's source.
 61
 62    Raises:
 63        LookupError: the start time cannot be read (`/proc` unreadable, `ps` missing, failing or too slow).
 64    """
 65    if sys.platform == "linux":
 66        if not os.access("/proc/self/stat", os.R_OK):
 67            raise LookupError("/proc is not readable")
 68        try:
 69            with open(f"/proc/{pid}/stat", "rb") as f:
 70                data = f.read()
 71        except (FileNotFoundError, ProcessLookupError):
 72            return None
 73        except OSError as err:
 74            raise LookupError(f"/proc/{pid}/stat: {err}") from err
 75        return data[data.rfind(b")") + 2 :].split()[19].decode()
 76    try:
 77        os.kill(pid, 0)
 78    except ProcessLookupError:
 79        return None
 80    except PermissionError:
 81        pass
 82    try:
 83        argv = ["ps", "-o", "lstart=", "-p", str(pid)]
 84        out = subprocess.run(argv, capture_output=True, text=True, check=False, timeout=probe_s)
 85    except (OSError, subprocess.SubprocessError) as err:
 86        raise LookupError(f"ps: {err}") from err
 87    found = " ".join(out.stdout.split())
 88    if out.returncode != 0 or not found:
 89        raise LookupError(f"ps exited {out.returncode} with {found!r}")
 90    return found

 93_KEEPER_HEAD = r"""
 94import os, select, signal, subprocess, sys
 95
 96owner, probe_s = int(sys.argv[1]), float(sys.argv[2])
 97UNKNOWN = object()
 98try:
 99    os.setsid()
100except OSError:
101    pass
102held = {"g": {}, "p": {}}
103rest = b""
104
105
106def on_term(signum, frame):
107    pass
108
109
110"""
111
112_KEEPER_BODY = r"""
113
114def apply(data):
115    global rest
116    *lines, rest = (rest + data).split(b"\n")
117    for line in lines:
118        sign, kind, (number, _, given) = line[:1], line[1:2].decode(), line[2:].decode().partition(" ")
119        number = int(number)
120        if sign == b"+":
121            held[kind][number] = given or UNKNOWN
122        else:
123            held[kind].pop(number, None)
124
125
126def drain():
127    os.set_blocking(0, False)
128    while True:
129        try:
130            data = os.read(0, 65536)
131        except OSError:
132            return
133        if not data:
134            return
135        apply(data)
136
137
138def kill(kind, number, recorded):
139    if recorded is UNKNOWN:
140        return
141    try:
142        now = read_start_time(number, probe_s)
143    except LookupError:
144        return
145    if (now is not None and now != recorded) or (now is None and kind == "p"):
146        return
147    try:
148        (os.killpg if kind == "g" else os.kill)(number, signal.SIGKILL)
149    except OSError:
150        pass
151
152
153signal.signal(signal.SIGTERM, on_term)
154if sys.platform == "linux":
155    try:
156        import ctypes
157        ctypes.CDLL(None).prctl(1, signal.SIGTERM)
158    except Exception:
159        pass
160while os.getppid() == owner:
161    if select.select([0], [], [], 0.2)[0]:
162        data = os.read(0, 65536)
163        if not data:
164            break
165        apply(data)
166drain()
167for kind in ("g", "p"):
168    for number, recorded in held[kind].items():
169        kill(kind, number, recorded)
170"""
171
172KEEPER = _KEEPER_HEAD + inspect.getsource(read_start_time) + _KEEPER_BODY
173"""Run as ``python -I -c KEEPER <owner pid> <probe seconds>`` with records on stdin: ``+g``/``-g`` a group,
174``+p``/``-p`` a pid.
175
176A ``+`` record carries the start time its owner read at launch (`read_start_time`) after the number; one without is
177of an unknown start time. At the end the keeper reads each start time again and SIGKILLs a recorded pid only while a
178process of that pid has the recorded start time, and a recorded group while its leader has it or no process has the
179group's number. A start time unknown when recorded, or unreadable at the end, kills nothing.
180"""

194def live_members(pgid: int) -> bool:
195    """Whether group `pgid` has a member that is neither a zombie nor exiting: by `/proc`, else by `killpg(pgid, 0)`."""
196    if os.path.isdir("/proc/self/task"):
197        return any(
198            len(fields) > _STAT_PGRP and int(fields[_STAT_PGRP]) == pgid and fields[_STAT_STATE] not in (b"Z", b"X")
199            for _, fields in proc_stats()
200        )
201    try:
202        os.killpg(pgid, 0)
203    except (ProcessLookupError, PermissionError):
204        return False
205    return True

208def own_children() -> set[int]:
209    """The pids of this process's children, from the kernel's list of each of its threads' children."""
210    found: set[int] = set()
211    try:
212        tasks = os.listdir("/proc/self/task")
213    except OSError:
214        return found
215    for task in tasks:
216        try:
217            with open(f"/proc/self/task/{task}/children") as f:
218                found.update(int(pid) for pid in f.read().split())
219        except OSError:
220            continue
221    return found

224def subreaper_supported() -> bool:
225    """Whether this host can make a process a subreaper whose orphans it can list: Linux with a children list."""
226    return sys.platform == "linux" and os.path.exists(f"/proc/self/task/{threading.get_native_id()}/children")

229def _prctl(option: int, value: object) -> None:
230    libc = ctypes.CDLL(None, use_errno=True)
231    if libc.prctl(option, value, 0, 0, 0) != 0:
232        errno = ctypes.get_errno()
233        raise OSError(errno, os.strerror(errno))

236def _subreaper_flag() -> int:
237    value = ctypes.c_int(0)
238    _prctl(PR_GET_CHILD_SUBREAPER, ctypes.byref(value))
239    return value.value

242def _exited(pid: int) -> bool:
243    """Whether child `pid` has exited, without reaping it; a pid that is no longer an unreaped child has (Linux)."""
244    if sys.platform != "linux":
245        raise NotImplementedError("orphans are adopted on Linux only")
246    try:
247        return os.waitid(os.P_PID, pid, os.WEXITED | os.WNOHANG | os.WNOWAIT) is not None
248    except ChildProcessError:
249        return True
250
251
252def _reap(pid: int) -> int | None:
253    """Reap exited child `pid` and return its exit status, negative for the signal that ended it; else None (Linux)."""
254    if sys.platform != "linux":
255        raise NotImplementedError("orphans are adopted on Linux only")
256    try:
257        info = os.waitid(os.P_PID, pid, os.WEXITED | os.WNOHANG)
258    except ChildProcessError:
259        return None
260    if info is None:
261        return None
262    return info.si_status if info.si_code == os.CLD_EXITED else -info.si_status

265def _kill(pid: int, sig: int, *, group: bool = False) -> None:
266    with contextlib.suppress(ProcessLookupError, PermissionError):
267        (os.killpg if group else os.kill)(pid, sig)

282class Keeper:
283    """The keeper process of this process, started on the first record; records are written to its stdin.
284
285    In a process forked from its owner, the next record starts that process's own keeper.
286    """
287
288    def __init__(
289        self,
290        settings: SupervisorSettings | None = None,
291        clock: Clock = SystemClock(),  # noqa: B008 - stateless
292    ) -> None:
293        """Start no process yet; the keeper reads start times under `settings` (`packaged_settings()` when None).
294
295        Its reap bound in `stop` is read on `clock`.
296        """
297        self.settings = settings if settings is not None else packaged_settings()
298        self._clock = clock
299        self.pid: int | None = None
300        self._owner = os.getpid()
301        self._stdin: IO[bytes] | None = None
302        self._broken = False
303
304    def record(self, sign: str, kind: str, number: int, start: str | None = None) -> None:
305        """Record (`+`) or drop (`-`) group or pid `number` (`kind` `g` or `p`); a dead keeper is logged once.
306
307        `start` is the start time of the group leader or pid read at launch (`read_start_time`); None when unknown,
308        and then the keeper never kills it.
309        """
310        if os.getpid() != self._owner:
311            self._disown()
312        if self._broken or (self.pid is None and sign == "-"):
313            return
314        try:
315            stdin = self._stdin if self._stdin is not None else self._start()
316            stdin.write(f"{sign}{kind}{number}{'' if start is None else ' ' + start}\n".encode())
317            stdin.flush()
318        except (OSError, ValueError) as exc:
319            self._broken = True
320            logger.warning("the keeper is gone; launches are no longer kept: %s", exc)
321
322    def stop(self, timeout: float) -> None:
323        """Close the keeper's stdin, so it kills what is still recorded and exits; reap it within `timeout`."""
324        if os.getpid() != self._owner:
325            self._disown()
326        pid, stdin, self.pid, self._stdin = self.pid, self._stdin, None, None
327        if stdin is not None:
328            with contextlib.suppress(OSError):
329                stdin.close()
330        if pid is None:
331            return
332        deadline = self._clock.now() + timeout
333        with contextlib.suppress(ChildProcessError):
334            while os.waitpid(pid, os.WNOHANG) == (0, 0):
335                if self._clock.now() > deadline:
336                    _kill(pid, signal.SIGKILL)
337                    os.waitpid(pid, 0)
338                    return
339                self._clock.sleep(self.settings.reap_poll_s)
340
341    def _start(self) -> IO[bytes]:
342        read, write = os.pipe()
343        try:
344            self.pid = os.posix_spawn(
345                sys.executable,
346                [sys.executable, "-I", "-c", KEEPER, str(os.getpid()), repr(self.settings.start_time_probe_s)],
347                os.environ,
348                file_actions=[
349                    (os.POSIX_SPAWN_DUP2, read, 0),
350                    (os.POSIX_SPAWN_OPEN, 1, os.devnull, os.O_WRONLY, 0),
351                ],
352            )
353        finally:
354            os.close(read)
355        self._stdin = os.fdopen(write, "wb")
356        return self._stdin
357
358    def _disown(self) -> None:
359        """In a forked copy, close this process's copy of the owner's keeper stdin and keep nothing of it."""
360        if self._stdin is not None:
361            with contextlib.suppress(OSError):
362                self._stdin.close()
363        self.pid, self._stdin, self._broken, self._owner = None, None, False, os.getpid()

366class Supervisor:
367    """The table of the processes launched through it and of the orphans it adopted; a context manager.
368
369    Attributes:
370        reaped: The latest reaped orphans' pids and exit statuses (a negative status is the signal that ended it).
371    """
372
373    def __init__(
374        self,
375        *,
376        subreaper: bool | None = None,
377        children: Callable[[], set[int]] = own_children,
378        poll_s: float | None = None,
379        settings: SupervisorSettings | None = None,
380        clock: Clock = SystemClock(),  # noqa: B008 - stateless
381    ) -> None:
382        """Supervise; once entered, as a subreaper where `subreaper` (by default where `subreaper_supported`).
383
384        `children` lists this process's children; `poll_s` is the reaper's interval (`settings.reap_poll_s` when
385        None); `settings` bound the start-time
386        reads (`packaged_settings()` when None); the exit bound and the reaper's waits run on `clock`.
387        """
388        self._want_subreaper = subreaper_supported() if subreaper is None else subreaper
389        self._children = children
390        self._keeper = Keeper(settings, clock)
391        self._poll_s = self._keeper.settings.reap_poll_s if poll_s is None else poll_s
392        self._clock = clock
393        self._lock = threading.RLock()
394        self._groups: dict[int, subprocess.Popen[bytes]] = {}
395        self._launched: dict[int, subprocess.Popen[bytes]] = {}
396        self._orphans: dict[int, Orphan] = {}
397        self._ended: set[int] = set()
398        self._foreign: set[int] = set()
399        self._foreign_groups: set[int] = set()
400        self._own_pgid = os.getpgid(0)
401        self._entered = False
402        self._adopting = False
403        self._previous_flag = 0
404        self._stop = threading.Event()
405        self._reaper: threading.Thread | None = None
406        self.reaped: collections.deque[tuple[int, int]] = collections.deque(maxlen=REAPED_KEPT)
407
408    @property
409    def adopting(self) -> bool:
410        """Whether this supervisor is entered as a subreaper, so orphans re-parent to it."""
411        return self._adopting
412
413    def __enter__(self) -> Self:
414        """Become the supervisor in force; as a subreaper, start reaping orphans."""
415        with self._lock:
416            self._entered = True
417            if self._want_subreaper:
418                try:
419                    self._previous_flag = _subreaper_flag()
420                    _prctl(PR_SET_CHILD_SUBREAPER, 1)
421                except OSError as exc:
422                    logger.warning("cannot become a subreaper; supervising by process group only: %s", exc)
423                    self._want_subreaper = False
424            if self._want_subreaper:
425                self._foreign = self._children() - self._launched.keys()
426                self._foreign_groups = {g for g in map(_group_of, self._foreign) if g is not None}
427                self._adopting = True
428                self._stop.clear()
429                self._reaper = threading.Thread(target=self._reap_loop, name="senselab-supervisor", daemon=True)
430                self._reaper.start()
431        _push(self)
432        return self
433
434    def __exit__(
435        self, exc_type: type[BaseException] | None, exc: BaseException | None, tb: TracebackType | None
436    ) -> None:
437        """SIGKILL every group and orphan held, reap them, stop the keeper and restore the subreaper flag."""
438        _pop(self)
439        self._stop.set()
440        if self._reaper is not None:
441            self._reaper.join()
442            self._reaper = None
443        deadline = self._clock.now() + self._keeper.settings.exit_bound_s
444        with self._lock:
445            for pgid, proc in self._groups.items():
446                if proc.returncode is None:
447                    _kill(pgid, signal.SIGKILL, group=True)
448            while self._adopting:
449                self._adopt()
450                for orphan in self._orphans.values():
451                    _kill(orphan.pid, signal.SIGKILL)
452                self.reap()
453                leaders = any(p.returncode is None and not _exited(g) for g, p in self._groups.items())
454                if (not self._orphans and not leaders) or self._clock.now() > deadline:
455                    break
456                self._clock.sleep(self._poll_s)
457            for proc in self._groups.values():
458                with contextlib.suppress(subprocess.TimeoutExpired):
459                    proc.wait(max(deadline - self._clock.now(), 0.0))
460            self._prune()
461            self._keeper.stop(max(deadline - self._clock.now(), 0.0))
462            self._groups.clear()
463            self._launched.clear()
464            if self._adopting:
465                with contextlib.suppress(OSError):
466                    _prctl(PR_SET_CHILD_SUBREAPER, self._previous_flag)
467            self._adopting = self._entered = False
468
469    def popen(self, argv: Sequence[str], **popen: Any) -> "subprocess.Popen[bytes]":  # noqa: ANN401
470        """Start `argv` in a new session as `subprocess.Popen` does with `popen`; enter its group in the table.
471
472        The group is recorded with the keeper.
473        """
474        with self._lock:
475            proc: subprocess.Popen[bytes] = subprocess.Popen(argv, **{**popen, "start_new_session": True})  # noqa: S603
476            self._prune()
477            self._groups[proc.pid] = self._launched[proc.pid] = proc
478            self._ended.discard(proc.pid)
479            self._keeper.record("+", "g", proc.pid, self._start_of(proc.pid))
480            return proc
481
482    def released(self, pgid: int) -> None:
483        """Drop launched group `pgid` before its leader is reaped; SIGKILL its orphans and those held only for it."""
484        with self._lock:
485            self._prune()
486            if pgid in self._groups:
487                self._drop(pgid)
488
489    def group_live(self, pgid: int) -> bool:
490        """Whether group `pgid` can still run code: from the table when adopting, else by probing the group."""
491        if not self._adopting:
492            return live_members(pgid)
493        with self._lock:
494            self._adopt()
495            leader = self._groups.get(pgid)
496            if leader is not None and leader.returncode is None and not _exited(pgid):
497                return True
498            return any(o.pgid == pgid and not _exited(o.pid) for o in self._orphans.values())
499
500    def orphans(self) -> dict[int, int]:
501        """Each adopted orphan not yet reaped, by pid, with its process group."""
502        with self._lock:
503            self._adopt()
504            return {pid: orphan.pgid for pid, orphan in self._orphans.items()}
505
506    def reap(self) -> None:
507        """Adopt new orphans and reap each that has exited, recording its status in `reaped`."""
508        with self._lock:
509            self._prune()
510            self._adopt()
511            for pid in list(self._orphans):
512                if not _exited(pid):
513                    continue
514                self._keeper.record("-", "p", pid)
515                status = _reap(pid)
516                del self._orphans[pid]
517                if status is not None:
518                    self.reaped.append((pid, status))
519
520    def _adopt(self) -> None:
521        if not self._adopting:
522            return
523        listed = self._children()
524        self._foreign &= listed
525        self._launched = {pid: proc for pid, proc in self._launched.items() if proc.returncode is None}
526        known = self._launched.keys() | self._orphans.keys() | self._foreign
527        if self._keeper.pid is not None:
528            known.add(self._keeper.pid)
529        for pid in listed - known:
530            pgid = _group_of(pid)
531            if pgid is None or pgid == self._own_pgid or pgid in self._foreign_groups:
532                continue
533            owners = set() if pgid in self._groups else set(self._groups)
534            orphan = self._orphans[pid] = Orphan(pid, pgid, owners)
535            self._keeper.record("+", "p", pid, self._start_of(pid))
536            if pgid not in self._groups and (pgid in self._ended or not owners):
537                self._end(orphan)
538
539    def _start_of(self, pid: int) -> str | None:
540        """Return the start time of child `pid`, unreaped so its pid is not reused; None when it cannot be read."""
541        try:
542            return read_start_time(pid, self._keeper.settings.start_time_probe_s)
543        except LookupError as exc:
544            logger.warning("the start time of %d is unknown; the keeper will not kill it: %s", pid, exc)
545            return None
