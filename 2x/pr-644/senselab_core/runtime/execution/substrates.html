senselab_core.runtime.execution.substrates

Substrates: what starts the pool workers that serve a run's queues, and scale, which sizes them from queue depth.

A Substrate starts workers for a queue, lists those alive, counts them and stops given ones. Local runs each worker as a senselab execute process under the supervisor that launched it, a cuda class's on a visible device of its own; Slurm submits one sbatch job per worker, with the class's partition, gres, CPUs, memory and time, named by plan hash, run, attempt, epoch and a token of its own, and never on an excluded node. A queue is named by its resource class, with #<model identity> for a resident model; every worker serves one queue.

scale is called by the coordinator each round, with one live() listing: per queue, the workers its waiting jobs (Queue.work) want, as the coordinator sizes them, within the class's max_workers and the QOS CPU cap over the class's CPUs per worker. A surplus is stopped, pending workers first, running ones only once the queue has no unresolved item.

A worker marks itself ready (mark_ready) once it can reach the run and its store; the mark names its host by a digest, its process and its HardwareClass per resource class, and writes no hardware index. Fleet counts a worker that left live() without a mark as a failed start of its class, backs the class's next start off, and reports the class broken after faults.infrastructure_replacements consecutive failures; it then starts no more workers of the class.

 236def mark_ready(run_dir: Path, queues: Mapping[str, QueueReady] | None = None) -> None:
 237    """Record that this worker started: it reached the run and its store, and what it says of each queue it serves."""
 238    directory = Path(run_dir) / WORKERS_DIR
 239    directory.mkdir(parents=True, exist_ok=True)
 240    document = {
 241        "format": WORKER_READY_FORMAT,
 242        "host": host_token(),
 243        "pid": os.getpid(),
 244        "started": process_started(os.getpid()),
 245        "queues": {
 246            name: {
 247                "hardware": None if ready.hardware is None else ready.hardware.model_dump(mode="json"),
 248                "envs": {k: dict(v) for k, v in ready.envs.items()},
 249            }
 250            for name, ready in (queues or {}).items()
 251        },
 252    }
 253    write_atomically(directory / worker_id(), jcs_dumps(document))

 256def read_mark(path: Path) -> ReadyMark | None:
 257    """Return the `worker-ready@1.0` mark at `path`; None when it is unreadable or of another format (logged once)."""
 258    try:
 259        document = jcs_loads(Path(path).read_bytes())
 260        if not isinstance(document, dict):
 261            return None
 262        if check(WORKER_READY_FORMAT, document.get("format")) is not FormatVerdict.ACCEPTED:
 263            _say_once(
 264                ("mark", str(document.get("format"))),
 265                "a worker mark is %r; %r is read",
 266                document.get("format"),
 267                WORKER_READY_FORMAT,
 268            )
 269            return None
 270        with reading(WORKER_READY_FORMAT, document["format"]):
 271            return ReadyMark(
 272                document["host"],
 273                document["pid"],
 274                document["started"],
 275                {
 276                    name: QueueReady(
 277                        None if found["hardware"] is None else HardwareClass.model_validate(found["hardware"]),
 278                        {k: dict(v) for k, v in found["envs"].items()},
 279                    )
 280                    for name, found in document["queues"].items()
 281                },
 282            )
 283    except (OSError, ValueError, KeyError, TypeError):
 284        return None

 296class Peers:
 297    """The other pool workers of a run alive on this host, from the marks in its `workers` directory.
 298
 299    A mark never changes once written, so each is read once; a listing and a liveness check per worker remain. A
 300    worker is alive while its process is, and, where `/proc` says when it started, the same process.
 301
 302    Attributes:
 303        reads: The marks read so far.
 304    """
 305
 306    def __init__(self, run_dir: Path) -> None:
 307        """Watch the workers of the run in `run_dir`."""
 308        self._directory = Path(run_dir) / WORKERS_DIR
 309        self._marks: dict[str, ReadyMark] = {}
 310        self.reads = 0
 311
 312    def alive(self) -> dict[int, ReadyMark]:
 313        """Return each other worker of this host alive now, by pid; this process is not one."""
 314        names = [p.name for p in self._directory.iterdir()] if self._directory.is_dir() else []
 315        for name in names:
 316            if name not in self._marks and not name.startswith("."):
 317                self.reads += 1
 318                if (mark := read_mark(self._directory / name)) is not None:
 319                    self._marks[name] = mark
 320        here, mine = host_token(), os.getpid()
 321        return {
 322            m.pid: m
 323            for n, m in self._marks.items()
 324            if n in names and m.host == here and m.pid != mine and _alive(m.pid, m.started)
 325        }
 326

 328def _alive(pid: int, started: str | None) -> bool:
 329    try:
 330        os.kill(pid, 0)
 331    except ProcessLookupError:
 332        return False
 333    except PermissionError:
 334        return started is not None and process_started(pid) == started
 335    return started is None or process_started(pid) in (None, started)

 357class Local:
 358    """A `Substrate` of local processes: each worker is `command` with `--classes <queue>`, in its own group.
 359
 360    At most `max_workers` are alive at once, across queues. A worker of a `cuda` class (`cuda`, by name) is given one
 361    of `devices` no other live worker holds, as its `CUDA_VISIBLE_DEVICES`, so no more run than there are devices. A
 362    process is launched through the supervisor in force when it starts, and stopped and released through that one.
 363    """
 364
 365    def __init__(
 366        self,
 367        max_workers: int,
 368        *,
 369        command: Sequence[str],
 370        cuda: Collection[str] = (),
 371        devices: Sequence[str] = (),
 372        stop_grace_s: float | None = None,
 373    ) -> None:
 374        """Run at most `max_workers` workers, each `command` plus `--classes <queue>`; `cuda` classes on `devices`.
 375
 376        A stopped worker is given `stop_grace_s` seconds after SIGTERM before it is killed (the packaged profile's
 377        `substrate.stop_grace_s` when None).
 378
 379        Raises:
 380            ValueError: `max_workers` is below 1.
 381        """
 382        if max_workers < 1:
 383            raise ValueError(f"a local substrate runs at least one worker, got {max_workers}")
 384        self.max_workers = max_workers
 385        self.command = list(command)
 386        self.cuda = frozenset(cuda)
 387        self.devices = tuple(devices)
 388        self.stop_grace_s = ExecutionProfile.packaged().substrate.stop_grace_s if stop_grace_s is None else stop_grace_s
 389        self._procs: dict[str, _Process] = {}
 390        self._lock = threading.Lock()
 391
 392    def _alive(self) -> dict[str, "_Process"]:
 393        for key, held in list(self._procs.items()):
 394            if held.proc.poll() is not None:
 395                held.supervisor.released(held.proc.pid)
 396                del self._procs[key]
 397        return self._procs
 398
 399    def start(self, resource_class: str, count: int) -> list[WorkerHandle]:
 400        """Start workers of `resource_class`, as many as `count`, the room under `max_workers` and free devices allow.
 401
 402        Raises:
 403            LocalError: a process could not be launched, those launched before it kept and listed; or `resource_class`
 404                is a `cuda` class and no device is visible.
 405        """
 406        if class_of(resource_class) in self.cuda and not self.devices:
 407            raise LocalError(f"no device is visible to run {resource_class!r} on (CUDA_VISIBLE_DEVICES, nvidia-smi)")
 408        with self._lock:
 409            alive = self._alive()
 410            room = self.max_workers - len(alive)
 411            held = {p.device for p in alive.values()}
 412            free: list[str | None] = [d for d in self.devices if d not in held]
 413            if class_of(resource_class) not in self.cuda:
 414                free = [None] * max(0, room)
 415            handles: list[WorkerHandle] = []
 416            for device in free[: max(0, min(count, room))]:
 417                supervisor = current()
 418                argv = [*self.command, "--classes", resource_class]
 419                env = None if device is None else {**os.environ, "CUDA_VISIBLE_DEVICES": device}
 420                try:
 421                    proc = supervisor.popen(argv, env=env)
 422                except OSError as err:
 423                    raise LocalError(f"could not launch {shlex.join(argv)}: {err}", handles) from err
 424                self._procs[str(proc.pid)] = _Process(resource_class, proc, supervisor, device)
 425                handles.append(WorkerHandle(resource_class, str(proc.pid)))
 426            return handles
 427
 428    def live(self) -> list[WorkerHandle]:
 429        """Return the workers alive."""
 430        with self._lock:
 431            return [WorkerHandle(p.queue, key) for key, p in self._alive().items()]
 432
 433    def running(self, resource_class: str) -> int:
 434        """Return the workers of `resource_class` alive."""
 435        return sum(1 for h in self.live() if h.queue == resource_class)
 436
 437    def stop(self, handles: Sequence[WorkerHandle]) -> None:
 438        """SIGTERM each worker of `handles`, which commits what finished; kill one that outlasts `stop_grace_s`."""
 439        with self._lock:
 440            found = [self._procs.pop(h.id) for h in handles if h.id in self._procs]
 441        for held in found:
 442            if held.proc.poll() is None:
 443                held.proc.terminate()
 444        for held in found:
 445            try:
 446                held.proc.wait(self.stop_grace_s)
 447            except subprocess.TimeoutExpired:
 448                held.proc.kill()
 449                held.proc.wait()
 450            held.supervisor.released(held.proc.pid)

 477class Slurm:
 478    """A `Substrate` of Slurm jobs: one `sbatch` job per worker, within a QOS CPU cap, never on an excluded node.
 479
 480    A job is named `senselab-<plan hash prefix>-<run digest>-<attempt>-<token>-e<epoch>`, the token drawn per intended
 481    worker, and carries its queue as its comment. The jobs of this run, whatever their attempt or epoch, are those of
 482    the user (`squeue --me`) whose name starts `senselab-<plan hash prefix>-<run digest>-`. `live` cancels those of an
 483    earlier epoch, ignores those of a later one, and returns this epoch's. The class's `partition`, `gres`, `cpus`,
 484    `mem_bytes` and `time_s` become the job's `--partition`, `--gres`, `--cpus-per-task`, `--mem` and `--time`; `${VAR}`
 485    in a partition, a gres or an excluded node is expanded from the environment where the job is submitted. A Slurm
 486    command runs within `commands.timeout_s` and is tried `commands.attempts` times, `commands.retry_first_s` apart and
 487    doubling, before `SlurmError`. A submission that fails or times out is looked up by its name before it is tried
 488    again; a job found under that name is the one submitted.
 489
 490    Attributes:
 491        attempt: The attempt id the jobs are named by; the pool sets it to its attempt's.
 492        epoch: The coordinator epoch the jobs are named by; the pool sets it to its own.
 493    """

 495    def __init__(  # noqa: PLR0913 - the classes and cap, then keyword-only submission settings
 496        self,
 497        classes: Mapping[str, ResourceClass],
 498        qos_cpu_cap: int | None,
 499        *,
 500        command: Sequence[str],
 501        plan_hash: str,
 502        run_dir: Path,
 503        attempt: str = "a0",
 504        epoch: int = 0,

 520        self.plan_hash = plan_hash

 539    def _class(self, queue: str) -> ResourceClass:
 540        try:
 541            return self.classes[class_of(queue)]

 547    def _granted(self, queue: str) -> tuple[ResourceClass, tuple[str, ...], Mapping[str, str], Mapping[str, str]]:
 548        """Return the class of `queue` as it is placed this epoch, its excluded nodes, each value's layer, its names."""
 549        base = self._class(queue)

 551            names = site_names([base.partition], self.exclude_nodes)

 556    def sbatch_args(self, queue: str, name: str) -> list[str]:
 557        """Return the `sbatch` command line of one worker of `queue`, named `name`, under its placement.
 558
 559        Raises:
 560            ValueError: the queue's class is unknown, or names a variable that is not set.
 561            PlacementAnswerRefused: the site's hook answered with a value that could not run the work.
 562        """
 563        grant, excluded, _, _ = self._granted(queue)
 564        args = [SBATCH, "--parsable", f"--job-name={name}", f"--comment={queue}"]
 565        if grant.partition is not None:
 566            args.append(f"--partition={expand(grant.partition)}")
 567        if grant.gres is not None:
 568            args.append(f"--gres={expand(grant.gres)}")
 569        if grant.cpus is not None:
 570            args.append(f"--cpus-per-task={grant.cpus}")
 571        if grant.mem_bytes is not None:
 572            args.append(f"--mem={math.ceil(grant.mem_bytes / 2**20)}M")
 573        if grant.time_s is not None:
 574            args.append(f"--time={_minutes(grant.time_s)}")
 575        if excluded:
 576            args.append("--exclude=" + ",".join(expand(n) for n in excluded))
 577        if self.log_dir is not None:
 578            args.append(f"--output={self.log_dir}/%x-%j.out")
 579        return [*args, "--wrap", shlex.join([*self.command, "--classes", queue])]

 581    def _try(self, argv: Sequence[str]) -> tuple[str | None, str, str | None]:
 582        """Run `argv` once; return its output, or None, why it failed and what the command said when it exited.
 583
 584        Raises:
 585            UndeclaredClient: `argv[0]` is not one of `CLIENTS`.
 586        """
 587        if not argv or argv[0] not in CLIENTS:
 588            raise UndeclaredClient(
 589                f"{argv[0] if argv else '(none)'!r} is not one of the Slurm clients {sorted(CLIENTS)}"
 590            )
 591        timeout = self.commands.timeout_s
 592        try:
 593            done = subprocess.run(argv, capture_output=True, text=True, check=False, timeout=timeout)  # noqa: S603
 594        except subprocess.TimeoutExpired:
 595            return None, f"timed out after {timeout} s", None
 596        except OSError as err:
 597            return None, f"could not run: {err}", None
 598        if done.returncode == 0:
 599            return done.stdout, "", None
 600        said = (done.stderr or done.stdout).strip()
 601        return None, f"exited {done.returncode}: {said}", said

 603    def _run(self, argv: Sequence[str], recover: Callable[[], str | None] | None = None) -> str:
 604        """Run `argv` until it succeeds, at most `commands.attempts` times, `commands.retry_first_s` apart and doubling.
 605
 606        Without `recover` the command must be idempotent. With it, each failed try is followed, after the wait, by
 607        `recover()`, which returns the output the try had when it took effect after all; that ends the tries.
 608
 609        Raises:
 610            SlurmError: no try succeeded and `recover` found nothing.
 611        """
 612        wait, last, said = self.commands.retry_first_s, "", None
 613        for attempt in range(self.commands.attempts):
 614            out, last, said = self._try(argv)
 615            if out is not None:
 616                return out
 617            if recover is None and attempt + 1 == self.commands.attempts:
 618                break
 619            self._clock.sleep(wait)
 620            wait *= 2
 621            if recover is not None and (found := recover()) is not None:
 622                return found
 623        raise SlurmError(f"{shlex.join(argv)} {last}", site_message=said if argv[0] == SBATCH else None)
 624
 625    def _submit(self, queue: str) -> WorkerHandle:
 626        """Submit one worker of `queue` under a name of its own; return its pending handle.
 627
 628        A failed try is followed by a lookup of the name (`squeue --me --name`); a job found is the one submitted.
 629
 630        Raises:
 631            SlurmError: no try submitted a job found under its name, or the lookup failed, so the outcome is unknown.
 632        """
 633        name = self.job_name(uuid.uuid4().hex[:TOKEN_HEX])
 634
 635        def lookup() -> str | None:
 636            out = self._run([SQUEUE, "--me", "--noheader", "--states=all", f"--name={name}", SQUEUE_FORMAT])
 637            return next((job for job, _, _, job_name in _rows(out) if job_name == name), None)
 638
 639        args = self.sbatch_args(queue, name)
 640        try:
 641            job = self._run(args, recover=lookup).strip().split(";")[0]
 642        except SlurmError as err:
 643            if err.site_message is None:
 644                raise
 645            _, _, layers, names = self._granted(queue)
 646            raise SiteRefused(class_of(queue), shlex.join(args), err.site_message, layers, names) from err
 647        return WorkerHandle(queue, job, pending=True, name=name)
 648
 649    def live(self) -> list[WorkerHandle]:
 650        """Return this epoch's live jobs of the run, from one `squeue`; cancel those of an earlier epoch."""
 651        out = self._run([SQUEUE, "--me", "--noheader", f"--states={LIVE_STATES}", SQUEUE_FORMAT])
 652        mine, stale = [], []
 653        for job, state, comment, name in _rows(out):
 654            if not name.startswith(self.prefix):
 655                continue
 656            found = EPOCH_SUFFIX.search(name)
 657            epoch = int(found[1]) if found else 0
 658            handle = WorkerHandle(comment, job, pending=state == "PENDING", name=name)
 659            if epoch == self.epoch:
 660                mine.append(handle)
 661            elif epoch < self.epoch:
 662                stale.append(handle)
 663        if stale:
 664            self._run([SCANCEL, *(h.id for h in stale)])
 665        self._cache = mine
 666        return mine
 667
 668    def _cached(self) -> list[WorkerHandle]:
 669        return self._cache if self._cache is not None else self.live()
 670
 671    def running(self, resource_class: str) -> int:
 672        """Return the live jobs of queue `resource_class`, from the last listing."""
 673        return sum(1 for h in self._cached() if h.queue == resource_class)
 674
 675    def start(self, resource_class: str, count: int) -> list[WorkerHandle]:
 676        """Submit workers of `resource_class`, as many as `count` and the room under `qos_cpu_cap` allow.
 677
 678        Raises:
 679            ValueError: the queue's class is unknown, or names a variable that is not set.
 680            PlacementAnswerRefused: the site's hook answered with a value that could not run the work.
 681            SlurmError: `sbatch` kept failing; it carries the workers submitted before it (`started`).
 682            SiteRefused: the scheduler refused the submission; names the `sbatch` line, its reply and the layers.
 683        """
 684        self._granted(resource_class)
 685        cached = self._cached()
 686        if self.qos_cpu_cap is not None:
 687            used = sum(self._cpus(h.queue) for h in cached)
 688            count = min(count, (self.qos_cpu_cap - used) // self._cpus(resource_class))
 689        if self.log_dir is not None:
 690            private_dir(self.log_dir)
 691        handles: list[WorkerHandle] = []
 692        for _ in range(max(0, count)):
 693            try:
 694                handles.append(self._submit(resource_class))
 695            except SlurmError as err:
 696                err.started = tuple(handles)
 697                raise
 698            cached.append(handles[-1])
 699        return handles
 700
 701    def _cpus(self, queue: str) -> int:
 702        return self._class(queue).cpus or 1
 703
 704    def stop(self, handles: Sequence[WorkerHandle]) -> None:
 705        """Cancel the jobs of `handles`."""
 706        if handles:
 707            self._run([SCANCEL, *(h.id for h in handles)])
 708            gone = {h.id for h in handles}
 709            self._cache = [h for h in self._cached() if h.id not in gone]

 712class Fleet:
 713    """A `Substrate` that records the workers its substrate starts, budgets their starts and lists them once a round.
 714
 715    A worker that left `live()` without having marked itself ready (`mark_ready`, in `workers_dir`) and that this
 716    fleet did not stop is a failed start of its class: the class's next start waits `first_s`, doubling per failure up
 717    to `max_s`, and after `budget` failures in a row the class is broken (`broken`) and starts no more workers.
 718    A worker that marked itself ready ends the run of failures. `fail_substrate` counts the same way the failures of
 719    the substrate's own commands, under the class `*`; `substrate_due` says when the substrate may be used again.
 720
 721    Attributes:
 722        broken: Each broken class (and `*`), once.
 723    """
 724
 725    def __init__(  # noqa: PLR0913 - the substrate, then keyword-only budget and clock
 726        self,
 727        substrate: Substrate,
 728        *,
 729        workers_dir: Path | None,
 730        budget: int,
 731        first_s: float,
 732        max_s: float,
 733        clock: Clock = SystemClock(),  # noqa: B008 - stateless
 734    ) -> None:
 735        """Wrap `substrate`; `budget` consecutive failed starts of a class break it."""
 736        self.substrate = substrate
 737        self.workers_dir = workers_dir
 738        self.budget, self.first_s, self.max_s, self._clock = budget, first_s, max_s, clock
 739        self.broken: dict[str, BrokenClass] = {}
 740        self.starts: Counter[str] = Counter()
 741        self._tracked: dict[str, WorkerHandle] = {}
 742        self._failures: Counter[str] = Counter()
 743        self._not_before: dict[str, float] = {}
 744        self._snapshot: list[WorkerHandle] | None = None
 745        self._began: dict[str, tuple[str, float]] = {}
 746        self._lived: defaultdict[str, float] = defaultdict(float)
 747
 748    def reset(self) -> None:
 749        """Forget every failure, every broken class and the worker lifetimes, as a new epoch does."""
 750        self.broken.clear()
 751        self._failures.clear()
 752        self._not_before.clear()
 753        self._lived.clear()
 754        now = self._clock.now()
 755        self._began = {i: (c, now) for i, (c, _) in self._began.items()}
 756
 757    def worker_seconds(self, resource_class: str) -> float:
 758        """Return the seconds the workers of `resource_class` have lived since the epoch began, as `live` saw them.
 759
 760        A worker lives from the first `live` that lists it started (not pending) to the first that does not list it.
 761        """
 762        now = self._clock.now()
 763        return self._lived[resource_class] + sum(now - b for c, b in self._began.values() if c == resource_class)
 764
 765    def _ready(self, handle: WorkerHandle) -> bool:
 766        return self.workers_dir is not None and (self.workers_dir / handle.id).exists()
 767
 768    def _fail(self, resource_class: str, reason: str) -> None:
 769        self._failures[resource_class] += 1
 770        n = self._failures[resource_class]
 771        self._not_before[resource_class] = self._clock.now() + min(self.first_s * 2 ** (n - 1), self.max_s)
 772        if n > self.budget and resource_class not in self.broken:
 773            self.broken[resource_class] = BrokenClass(resource_class, n, reason)
 774            logger.error("workers of %s are broken after %d failed starts: %s", resource_class, n, reason)
 775
 776    def fail_substrate(self, err: Exception) -> None:
 777        """Count a failure of the substrate's commands: against the class a `SiteRefused` names, else the substrate.
 778
 779        A class, or the substrate (`*`), is broken after `budget` failures in a row.
 780        """
 781        self._fail(err.resource_class if isinstance(err, SiteRefused) else "*", str(err))
 782
 783    def substrate_due(self) -> bool:
 784        """Whether the substrate may be used now: it is not broken, and its latest failure's wait has passed."""
 785        return "*" not in self.broken and self._clock.now() >= self._not_before.get("*", 0.0)
 786
 787    def substrate_ok(self) -> None:
 788        """End the run of the substrate's failures."""
 789        self._failures["*"] = 0
 790
 791    def broken_for(self, queue: str) -> BrokenClass | None:
 792        """Return the break that stops starts of `queue`: its class's, else the substrate's."""
 793        return self.broken.get(class_of(queue)) or self.broken.get("*")
 794
 795    def live(self) -> list[WorkerHandle]:
 796        """List the substrate's workers once, and judge the tracked workers that left it."""
 797        found = self.substrate.live()
 798        ids = {h.id for h in found}
 799        now = self._clock.now()
 800        for handle in found:
 801            if not handle.pending:
 802                self._began.setdefault(handle.id, (class_of(handle.queue), now))
 803        for gone in [i for i in self._began if i not in ids]:
 804            cls, began = self._began.pop(gone)
 805            self._lived[cls] += now - began
 806        for handle in list(self._tracked.values()):
 807            if handle.id in ids:
 808                if self._ready(handle):
 809                    self._failures[class_of(handle.queue)] = 0
 810                continue
 811            del self._tracked[handle.id]
 812            if self._ready(handle):
 813                self._failures[class_of(handle.queue)] = 0
 814            else:
 815                self._fail(
 816                    class_of(handle.queue), f"a worker of {handle.queue} (id {handle.id}) exited before it started"
 817                )
 818        self._snapshot = list(found)
 819        return found
 820
 821    def start(self, resource_class: str, count: int) -> list[WorkerHandle]:
 822        """Start through the substrate unless the class is broken or backing off; record the workers.
 823
 824        Once the class has a failed start on record, at most `budget + 1` less its failures less its started workers
 825        not yet ready (pending jobs excluded) are started, so a command that fails at once is run at most `budget + 1`
 826        times; before that only `max_workers` and the QOS cap bound starts.
 827        """
 828        cls = class_of(resource_class)
 829        if self.broken_for(resource_class) is not None or self._clock.now() < self._not_before.get(cls, 0.0):
 830            return []
 831        if self._failures[cls] > 0:
 832            pending = {h.id for h in self._snapshot or [] if h.pending}
 833            unproven = sum(
 834                1
 835                for h in self._tracked.values()
 836                if class_of(h.queue) == cls and h.id not in pending and not self._ready(h)
 837            )
 838            count = min(count, self.budget + 1 - self._failures[cls] - unproven)
 839            if count <= 0:
 840                return []
 841        try:
 842            started = self.substrate.start(resource_class, count)
 843        except SubstrateError as err:
 844            self._track(cls, err.started)
 845            raise
 846        self._track(cls, started)
 847        return started
 848
 849    def _track(self, cls: str, started: Sequence[WorkerHandle]) -> None:
 850        for handle in started:
 851            if self.workers_dir is not None:
 852                (self.workers_dir / handle.id).unlink(missing_ok=True)
 853            self._tracked[handle.id] = handle
 854        self.starts[cls] += len(started)
 855
 856    def running(self, resource_class: str) -> int:
 857        """Return the workers of the queue in the last listing."""
 858        if self._snapshot is None:
 859            self.live()
 860        return sum(1 for h in self._snapshot or [] if h.queue == resource_class)
 861
 862    def stop(self, handles: Sequence[WorkerHandle]) -> None:
 863        """Stop `handles` through the substrate and forget them."""
 864        self.substrate.stop(handles)
 865        gone = {h.id for h in handles}
 866        for key in gone:
 867            self._tracked.pop(key, None)
 868        if self._snapshot is not None:
 869            self._snapshot = [h for h in self._snapshot if h.id not in gone]
 870
 871    def substrate_wait(self) -> float:
 872        """Return the seconds until the substrate may be used again; 0 when it may be now or is broken."""
 873        return max(0.0, self._not_before.get("*", 0.0) - self._clock.now())
 874
 875    def close(self) -> None:
 876        """Stop every worker the substrate lists, late duplicates included; when it cannot list, every one tracked."""
 877        try:
 878            alive = self.substrate.live()
 879        except SubstrateError:
 880            alive = list(self._tracked.values())
 881        self.stop(alive)

 884def scale(  # noqa: PLR0913, C901 - the queues and substrate, then keyword-only state; one pass over each queue
 885    queues: QueueSet,
 886    substrate: Substrate,
 887    profile: ExecutionProfile,
 888    *,
 889    workers: Callable[[str, int], int],
 890    surplus_since: dict[str, float] | None = None,
 891    clock: Clock = SystemClock(),  # noqa: B008 - stateless
 892    granted: Callable[[str], ResourceClass] | None = None,
 893) -> None:
 894    """Start or stop the workers each queue of `queues` needs, from one `live()` listing of `substrate`.
 895
 896    A queue wants `workers(name, work)` workers, `work` being the jobs of its unresolved items. Fewer alive
 897    start workers, within the room left in the class (`max_workers` less the workers alive on its queues) and, under a
 898    QOS cap, the CPUs left over the class's CPUs per worker. More alive stop the surplus, pending workers first; running
 899    ones are stopped only once the queue has no unresolved item, so none is working, and never on a service class,
 900    whose workers exit when their consumers stop. With `surplus_since` (the caller
 901    keeps it between rounds) a surplus is stopped only once it has lasted `2 × pool.poll_s`, so a queue that briefly
 902    empties between items keeps its workers; without it a surplus is stopped at once. The surplus is timed on `clock`.
 903    `granted` gives a class as it is placed (`max_workers` is read from it, only for a queue that wants to start
 904    workers); by default the profile's.
 905    """
 906    names = queues.names()
 907    live = substrate.live()
 908    alive = {name: [h for h in live if h.queue == name] for name in names}
 909    cap = profile.substrate.qos_cpu_cap
 910
 911    def cpus(name: str) -> int:
 912        grant = profile.resource_classes.get(class_of(name))
 913        return grant.cpus or 1 if grant is not None else 1
 914
 915    used = sum(len(hs) * cpus(name) for name, hs in alive.items())
 916    for name in sorted(names):
 917        grant = profile.resource_classes.get(class_of(name))
 918        if grant is None:
 919            continue
 920        work = queues.queue(name).work()
 921        wanted = workers(name, work)
 922        have = len(alive[name])
 923        if have > wanted:
 924            if surplus_since is not None:
 925                at = clock.now()
 926                if at - surplus_since.setdefault(name, at) < 2 * profile.pool.poll_s:
 927                    continue
 928                del surplus_since[name]
 929            victims = sorted(alive[name], key=lambda h: not h.pending)
 930            victims = victims if work == 0 and not is_service_class(name) else [h for h in victims if h.pending]
 931            substrate.stop(victims[: have - wanted])
 932            continue
 933        if surplus_since is not None:
 934            surplus_since.pop(name, None)
 935        if wanted == have:
 936            continue
 937        limit = (granted(name) if granted is not None else grant).max_workers
 938        room = limit - sum(len(hs) for other, hs in alive.items() if class_of(other) == class_of(name))
 939        if cap is not None:
 940            room = min(room, (cap - used) // cpus(name))
 941        count = min(wanted - have, room)
 942        if count > 0:
 943            started = substrate.start(name, count)
 944            used += len(started) * cpus(name)
 945            alive[name].extend(started)
 946

 948def profile_argument(profile: ExecutionProfile, run_dir: Path) -> str:
 949    """Return what `senselab execute --profile` takes to load `profile`: its packaged name, else a file written.

 958    except (FileNotFoundError, ValueError):
 959        pass

 962    return str(path)

 965def substrate_of(
 966    profile: ExecutionProfile,
 967    run_dir: Path,
 968    plan_hash: str,
 969    *,
 970    hardware: HardwareInfo | None = None,
 971) -> Substrate:
 972    """Return the substrate `profile` names for the run in `run_dir` of plan `plan_hash`.
 973
 974    `local` is `Local` over the sum of the classes' `max_workers`, its `cuda` classes on the GPUs this process sees
 975    (`hardware`, probed when None); `slurm` is `Slurm` over the profile's classes, QOS cap and excluded nodes, logging
 976    under `<run dir>/private/slurm`.
 977    """
 978    command = worker_command(run_dir, profile_argument(profile, run_dir))
 979    spec = profile.substrate
 980    if spec.kind == "local":
 981        classes = profile.resource_classes
 982        return Local(
 983            sum(c.max_workers for c in classes.values()),
 984            command=command,
 985            cuda=[name for name, c in classes.items() if c.device == "cuda"],
 986            devices=visible_devices(hardware or probe_hardware(timeout_s=profile.sizing.device_query_s)),
 987            stop_grace_s=spec.stop_grace_s,
 988        )
 989    if spec.commands is None:
 990        raise ValueError("a slurm substrate needs its commands settings")
 991    return Slurm(
 992        profile.resource_classes,
