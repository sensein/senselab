 69class WorkerLoadFailed(Exception):  # noqa: N818 - a fault named for what failed, as its callers know it
 70    """A worker did not start or load its backend; `__cause__` is what it raised.
 71
 72    Its fault class and gap are its cause's (`carried_class`), so a load that ran out of memory is a resource fault,
 73    re-placed on a larger class; a content fault stated at load is the environment's (`faults.at_load`), and with no
 74    cause it is an infrastructure fault of its env. It is raised by a checkout, never by a request to a loaded worker,
 75    so it is never split.
 76    """
 77
 78    @property
 79    def carried_class(self) -> tuple[FaultClass, bool]:
 80        """The fault class of the cause and whether the table decided it."""
 81        cause = self.__cause__
 82        return at_load(*classification(cause)) if cause is not None else (FaultClass.INFRASTRUCTURE, False)

177class WorkerSet:
178    """The pooled workers of a process, one per backend, env, init and budget or network, and the servers.
179
180    At most `bound` pooled workers are kept: once more exist, the least recently used idle ones are closed. A
181    worker is held by one job at a time. Host workers keep their state in this process's model cache of their budget
182    (`process_cache`), shared by every set. A server is started once per (env, backend, model); its start holds a
183    lease of that triple alone, so starts of other servers proceed. Servers are not counted in the bound.
184
185    With a `registry`, a server is consumed from the registry while a live service attests what it must, and started
186    otherwise; a set that `provide`s registers each server it starts there, so every later set consumes that one.
187    """
188
189    def __init__(
190        self,
191        bound: int,
192        *,
193        start_server: ServerStart = start_vllm,
194        registry: ServiceRegistry | None = None,
195        provide: bool = False,
196    ) -> None:
197        """Keep at most `bound` pooled workers; servers start through `start_server`.
198
199        Raises:
200            ValueError: `bound` is below 1, or `provide` without a registry.
201        """
202        if bound < 1:
203            raise ValueError(f"a worker set holds at least one worker, got bound {bound}")
204        if provide and registry is None:
205            raise ValueError("a worker set that provides services needs their registry")
206        self.bound = bound
207        self._start = start_server
208        self._registry, self._provide = registry, provide
209        self._provided: dict[tuple[str, str, str], ServiceLease] = {}
210        self._lock = threading.Lock()
211        self._pool: collections.OrderedDict[tuple[str, ...], Worker] = collections.OrderedDict()
212        self._held: dict[tuple[str, ...], threading.Lock] = {}
213        self._busy: collections.Counter[tuple[str, ...]] = collections.Counter()
214        self._holders: dict[tuple[str, ...], int] = {}
215        self._servers: dict[tuple[str, str, str], Server] = {}
216        self._leases: dict[tuple[str, str, str], threading.Lock] = {}
217
218    def tighten(self, bound: int) -> None:
219        """Lower the bound to `bound` when that is lower, closing the idle workers beyond it; a higher one is ignored.
220
221        Raises:
222            ValueError: `bound` is below 1.
223        """
224        if bound < 1:
225            raise ValueError(f"a worker set holds at least one worker, got bound {bound}")
226        with self._lock:
227            self.bound = min(self.bound, bound)
228        self._evict()
229
230    def pooled(self) -> list[Worker]:
231        """Return the pooled workers, least recently used first."""
232        with self._lock:
233            return list(self._pool.values())

235    @contextlib.contextmanager
236    def checkout(  # noqa: PLR0913 - the backend, env and model, then keyword-only placement
237        self,
238        backend: BackendSpec,
239        env: EnvHandle,
240        model: ResolvedModel | None,
241        *,
242        device: str,
243        profile: ExecutionProfile,
244        expected: Attestation | None = None,
245    ) -> Iterator[Worker]:
246        """Hold the worker of this backend, env, model and device until the block ends; other holders wait.
247
248        A server backend's worker runs in this process over its running server, which must attest `expected`. A dead
249        pooled worker is closed and replaced.
250
251        Raises:
252            ValueError: a server backend without `expected`; a host backend declaring `models` without `state_bytes`.
253            LeaseTimeout, LeaseUnavailable: a server's start lease is busy or cannot be read.
254            WorkerLoadFailed: a worker or a server failed to start or load, whatever it raised, as its cause
255                (`VllmOutOfMemory`, a resource fault, `VllmStartError` and `AttestationMismatch` among them).
256        """
257        init = worker_init(device, model)
258        if backend.env.kind == "server":
259            if expected is None or model is None:
260                raise ValueError(f"backend {backend.name!r} is a server backend; its expected attestation is needed")
261            yield _loaded(
262                lambda: self.server_worker(backend, init, self.server(backend, env, model, profile, expected))
263            )
264            return
265        host = backend.env.kind == "host"
266        variant = str(profile.model_cache_budget_bytes) if host else profile.network
267        named = (backend.task, backend.name, backend.version, backend.worker)
268        key = (backend.env.kind, env.python, *named, jcs_dumps(init).decode(), variant)
269        with self._lock:
270            held = self._held.setdefault(key, threading.Lock())
271            self._busy[key] += 1
272        try:
273            with held:
274                with self._lock:
275                    self._holders[key] = threading.get_ident()
276                try:
277                    yield self._worker(key, backend, env, init, model, profile)
278                finally:
279                    with self._lock:
280                        self._holders.pop(key, None)
281        finally:
282            with self._lock:
283                self._busy[key] -= 1
284                if self._busy[key] == 0:
285                    del self._busy[key]
286            self._evict()
287
288    def _worker(  # noqa: PLR0913 - the pool key and what a new worker is built from
289        self,
290        key: tuple[str, ...],
291        backend: BackendSpec,
292        env: EnvHandle,
293        init: dict[str, Any],
294        model: ResolvedModel | None,
295        profile: ExecutionProfile,
296    ) -> Worker:
297        with self._lock:
298            worker = self._pool.get(key)
299        if isinstance(worker, HostWorker) or (worker is not None and worker.alive):
300            with self._lock:
301                self._pool.move_to_end(key)
302            return worker
303        if worker is not None:
304            worker.close()
305        worker = self._new_worker(backend, env, init, model, profile)
306        with self._lock:
307            self._pool[key] = worker
308        self._evict()
309        return worker
310
311    def _evict(self) -> None:
312        """Close the least recently used idle workers while more than `bound` are pooled."""
313        closing: list[Worker] = []
314        with self._lock:
315            idle = [k for k in self._pool if self._busy[k] == 0]
316            while len(self._pool) > self.bound and idle:
317                closing.append(self._pool.pop(idle.pop(0)))
318        for worker in closing:
319            if isinstance(worker, WorkerClient):
320                worker.close()
321
322    def _new_worker(
323        self,
324        backend: BackendSpec,
325        env: EnvHandle,
326        init: dict[str, Any],
327        model: ResolvedModel | None,
328        profile: ExecutionProfile,
329    ) -> Worker:
330        if backend.env.kind != "host":
331            return _loaded(
332                lambda: WorkerClient(
333                    env.python,
334                    backend.worker,
335                    init=init,
336                    variables=network_variables(profile.network),
337                    load_timeout=profile.timeouts.load_s,
338                    request_timeout=profile.timeouts.request_s,
339                    settings=profile.worker,
340                )
341            )
342        budget = profile.model_cache_budget_bytes
343        key = host_model_key(backend, env, model, init["device"])
344        size_of = _size_of(backend)
345        return _loaded(
346            lambda: HostWorker(
347                backend.worker,
348                init=backend_init(init),
349                key=key,
350                cache=process_cache(budget, profile.sizing.device_query_s),
351                size_of=size_of,
352            )
353        )

355    def server(
356        self,
357        backend: BackendSpec,
358        env: EnvHandle,
359        model: ResolvedModel,
360        profile: ExecutionProfile,
361        expected: Attestation,
362    ) -> Server:
363        """Return the running server of this env, backend and model, starting it (or replacing a dead one) if needed.
364
365        The start holds the lease of this env, backend and model only. A live registered service attesting `expected`
366        is consumed (`ConsumerLease`) instead; a providing set registers what it starts.
367
368        Raises:
369            VllmStartError, LeaseUnavailable: as the server start or its registration.
370            LeaseTimeout: a start lease is busy, or another providing set is starting this server.
371            AttestationMismatch: the server attests something other than `expected`.
372        """
373        key = (env.python, backend.name, model.identity)
374        with self._lock:
375            lease = self._leases.setdefault(key, threading.Lock())
376        with lease:
377            with self._lock:
378                server = self._servers.get(key)
379            if server is not None:
380                try:
381                    server.context_fields()
382                    return server
383                except StaleAttestation:
384                    with self._lock:
385                        self._servers.pop(key, None)
386                    self._retire(key, server)
387            server = self._consumed(expected)
388            if server is None:
389                server = self._provided_server(key, backend, env, model, profile, expected)
390            with self._lock:
391                self._servers[key] = server
392            return server
393
394    def _consumed(self, expected: Attestation) -> Server | None:
395        """Return a consumer lease on a live registered service attesting `expected`; None without one."""
396        if self._registry is None:
397            return None
398        consumer = self._registry.consume(expected)
399        if self._registry.lookup(expected) is None:
400            consumer.stop()
401            return None
402        return consumer
403
404    def _provided_server(  # noqa: PLR0913 - the server's key and what starts it
405        self,
406        key: tuple[str, str, str],
407        backend: BackendSpec,
408        env: EnvHandle,
409        model: ResolvedModel,
410        profile: ExecutionProfile,
411        expected: Attestation,
412    ) -> Server:
413        """Start the server; a providing set reserves its registration first and publishes it once started.
414
415        Raises:
416            LeaseTimeout: another live registration of `expected` exists; nothing was started.
417        """
418        if not self._provide:
419            return self._started(backend, env, model, profile, expected)
420        assert self._registry is not None
421        service = self._registry.reserve(expected)
422        try:
423            server = self._started(backend, env, model, profile, expected)
424            try:
425                assert server.url is not None
426                service.publish(server.url)
427            except BaseException:
428                server.stop()
429                raise
430        except BaseException:
431            service.close()
432            raise
433        with self._lock:
434            self._provided[key] = service
435        return server
436
437    def _started(
438        self,
439        backend: BackendSpec,
440        env: EnvHandle,
441        model: ResolvedModel,
442        profile: ExecutionProfile,
443        expected: Attestation,
444    ) -> Server:
445        server = self._start(backend, env, model, profile)
446        if server.attestation != expected:
447            attested = server.attestation
448            server.stop()
449            raise AttestationMismatch(f"the server of {server.label} attests {attested}, the lock declares {expected}")
450        return server
451
452    def _retire(self, key: tuple[str, str, str], server: Server) -> None:
453        """Stop `server`, then withdraw its registration if this set provides it, even when the stop raised."""
454        with self._lock:
455            provided = self._provided.pop(key, None)
456        run_all(server.stop, provided.close if provided is not None else _nothing)
457
458    def provided(self) -> list[ServiceLease]:
459        """Return the registrations of the servers this set started and provides."""
460        with self._lock:
461            return list(self._provided.values())

464    def server_worker(backend: BackendSpec, init: dict[str, Any], server: Server) -> HostWorker:
465        """Return the in-process worker of a server backend, given `server` as `init["server"]`.
466
467        Its requests are counted per attested server instance, in this process's server cache: a second consumer of a
468        server another has used reports no load.
469        """
470        instance = value_digest("server", {"url": server.url, "attestation": _attested(server)})
471        return HostWorker(
472            backend.worker,
473            init={**backend_init(init), "server": server},
474            key=(backend.name, instance, init["device"], PRECISION),
475            cache=_SERVERS,
476            size_of=lambda _: 0,

479    def abandon(self, thread: int) -> None:
480        """Close every pooled worker process the thread `thread` holds now, so its `run` raises `WorkerLost`.
481
482        A later checkout starts another. A host worker runs in this process and cannot be stopped; it is left to finish.
483        """
484        with self._lock:
485            held = [
486                (k, w) for k, w in self._pool.items() if self._holders.get(k) == thread and isinstance(w, WorkerClient)
487            ]
488            for key, _ in held:
489                del self._pool[key]
490        for _, worker in held:
491            worker.close()
492
493    def release(self) -> None:
494        """Free what this set and this process hold; a later checkout loads again.
495
496        The set is closed (`close`), every process model cache dropped, and the device memory torch caches, when
497        torch is loaded, freed.
498        """
499        self.close()
500        with _CACHES_LOCK:
501            caches = list(_CACHES.values())
502        for cache in caches:
503            cache.clear()
504        gc.collect()
505        torch = sys.modules.get("torch")
506        if torch is not None and torch.cuda.is_initialized():
507            torch.cuda.empty_cache()
508
509    def close(self) -> None:
510        """Stop every pooled worker process and every server, withdrawing what it provides; a checkout starts again."""
511        with self._lock:
512            workers, servers = list(self._pool.values()), list(self._servers.items())
513            self._pool.clear()
514            self._servers.clear()
515        run_all(
516            *(w.close for w in workers if isinstance(w, WorkerClient)),
517            *(functools.partial(self._retire, key, server) for key, server in servers),
518        )

525def host_model_key(backend: BackendSpec, env: EnvHandle, model: ResolvedModel | None, device: str) -> ModelKey:
526    """Return the process `ModelCache` key of a host backend, as (backend code, identity, device, dtype).
527
528    The backend code is `<name>:<worker module>@<version>`, so backends of different code never share a loaded state.
529    The identity is the model's, so two tasks on one backend code and one model share it; without a model it is the
530    digest of the backend's task, name, version and env.
531    """
532    identity = (
533        model.identity
534        if model is not None
535        else value_digest(
536            "backend", {"task": backend.task, "name": backend.name, "version": backend.version, "env": env.python}
537        )
538    )
539    return (f"{backend.name}:{backend.worker}@{backend.version}", identity, device, PRECISION)

542def _loaded(start: Callable[[], Started]) -> Started:
543    """Return the worker `start` starts and loads.
544
545    Raises:
546        LeaseTimeout, LeaseUnavailable: a start lease is busy or cannot be read, as raised.
547        WorkerLoadFailed: whatever else the start or load raised, as its cause; it has the cause's fault class.
548    """
549    try:
550        return start()
551    except (LeaseTimeout, LeaseUnavailable):
552        raise
553    except Exception as exc:
554        failed = WorkerLoadFailed(f"{type(exc).__name__}: {exc}")
555        for note in getattr(exc, "__notes__", ()):
556            failed.add_note(note)
557        raise failed from exc

560def _size_of(backend: BackendSpec) -> Callable[[object], int]:
561    """Return how a backend's state counts against the model-cache budget: by its module's `state_bytes`.
562
563    A backend that declares no `models` and no `state_bytes` holds none, and counts 0.
564
565    Raises:
566        ValueError: the backend declares `models` and its module no `state_bytes`.
567    """
568    size_of: Callable[[object], int] | None = getattr(importlib.import_module(backend.worker), "state_bytes", None)
569    if size_of is not None:
570        return size_of
571    if backend.models:
572        raise ValueError(
573            f"backend {backend.name!r} serves models and its module {backend.worker!r} defines no state_bytes, "
574            "so the model-cache budget cannot hold it"
575        )
576    return lambda _: 0
