 69    def part(self, indices: Sequence[int], **update: object) -> PlanFragment:
 70        """Return the fragment of jobs `indices`, in that order, with the inputs they read and `update`."""
 71        named = {i for index in indices for i in self.inputs[index].values()}
 72        gathered = {g: self.gathered[g] for g in named if g in self.gathered}
 73        read = {i for i in named if i.startswith("output:")} | {e for es in gathered.values() for e in es}
 74        return PlanFragment.model_validate(
 75            {
 76                **self.shared,
 77                "jobs": [self.jobs[i] for i in indices],
 78                "inputs": [self.inputs[i] for i in indices],
 79                "upstream": {i: self.upstream[i] for i in sorted(read) if i in self.upstream},
 80                "gathered": gathered,
 81                **update,
 82            }
 83        )

 91def split(fragment: PlanFragment, quantum: int, request: int = 1) -> list[PlanFragment]:
 92    """Return `fragment` as items of at most `quantum` jobs each, in job order; `[fragment]` when it fits in one.
 93
 94    The fragment is dumped once and each item is built from its own slice, so the work is linear in the jobs.
 95
 96    `request` is the jobs per request of the fragment's backend (`node_request_size`). When it is at most `quantum`,
 97    every item but the last holds the most whole requests that fit, so no request is cut; a larger request is cut at
 98    `quantum` jobs.
 99
100    Raises:
101        ValueError: `quantum` or `request` is below 1.
102    """
103    if quantum < 1 or request < 1:
104        raise ValueError(f"an item and a request hold at least one job, got {quantum} and {request}")
105    count = len(fragment.jobs)
106    if count <= quantum:
107        return [fragment]
108    size = quantum - quantum % request if request <= quantum else quantum
109    dumped = Dumped.of(fragment)
110    return [dumped.part(range(start, min(start + size, count))) for start in range(0, count, size)]

222def worth(plan: ProbePlan, estimate: Estimate | None) -> bool:
223    """Return whether `plan` is worth running: it has a `SINGLE` item, or warm-ups that a batched phase can use.
224
225    A batched phase needs a measured single fit and a device memory fit (`Probe.plan_batched`); warm-ups alone, with
226    neither, measure nothing.
227    """
228    if any(i.phase == "SINGLE" for i in plan.items):
229        return True
230    return bool(plan.items and plan.batched and estimate is not None and estimate.device is not None)

233def covers(estimate: Estimate | None, jobs: Sequence[Job]) -> bool:
234    """Return whether `estimate` measures every length of `jobs`: it has no unit of length, or a slope and the range.
235
236    An `oom` record a probe can lift (`probe_regime`) leaves its stratum uncovered.
237    """
238    single = None if estimate is None else estimate.single
239    if estimate is None or single is None or any(probe_regime(r) for r in estimate.oom):
240        return False
241    if estimate.unit_kind == "none":
242        return True
243    lengths = [job_units(j, estimate.unit_kind) for j in jobs]
244    return single.u is not None and all(n is None or single.lo <= n <= single.hi for n in lengths)

247def placed_backends(placed: Plan) -> dict[str, set[tuple[str, str]]]:
248    """Return the backends, as `(task, backend name)`, the plan places on each queue."""
249    found: dict[str, set[tuple[str, str]]] = {}
250    tasks = {n.id: n.task for n in placed.logical.graph.nodes}
251    for node, record in placed.placement.nodes.items():
252        model = placed.logical.models.get(node)
253        name = record.resource_class if model is None else f"{record.resource_class}#{model.identity}"
254        found.setdefault(name, set()).add((tasks[node], record.backend))
255    return found

def worth( plan: senselab_core.runtime.calibration.ProbePlan, estimate: senselab_core.runtime.calibration.Estimate | None) -> bool: 
223def worth(plan: ProbePlan, estimate: Estimate | None) -> bool:
224    """Return whether `plan` is worth running: it has a `SINGLE` item, or warm-ups that a batched phase can use.
225
226    A batched phase needs a measured single fit and a device memory fit (`Probe.plan_batched`); warm-ups alone, with
227    neither, measure nothing.
228    """
229    if any(i.phase == "SINGLE" for i in plan.items):
230        return True
231    return bool(plan.items and plan.batched and estimate is not None and estimate.device is not None)

def covers( estimate: senselab_core.runtime.calibration.Estimate | None, jobs: Sequence[senselab_core.runtime.execution.job.Job]) -> bool: 
234def covers(estimate: Estimate | None, jobs: Sequence[Job]) -> bool:
235    """Return whether `estimate` measures every length of `jobs`: it has no unit of length, or a slope and the range.
236
237    An `oom` record a probe can lift (`probe_regime`) leaves its stratum uncovered.
238    """
239    single = None if estimate is None else estimate.single
240    if estimate is None or single is None or any(probe_regime(r) for r in estimate.oom):
241        return False
242    if estimate.unit_kind == "none":
243        return True
244    lengths = [job_units(j, estimate.unit_kind) for j in jobs]
245    return single.u is not None and all(n is None or single.lo <= n <= single.hi for n in lengths)

def placed_backends( placed: senselab_core.runtime.planner.Plan) -> dict[str, set[tuple[str, str]]]: 
248def placed_backends(placed: Plan) -> dict[str, set[tuple[str, str]]]:
249    """Return the backends, as `(task, backend name)`, the plan places on each queue."""
250    found: dict[str, set[tuple[str, str]]] = {}
251    tasks = {n.id: n.task for n in placed.logical.graph.nodes}
252    for node, record in placed.placement.nodes.items():
253        model = placed.logical.models.get(node)
254        name = record.resource_class if model is None else f"{record.resource_class}#{model.identity}"
255        found.setdefault(name, set()).add((tasks[node], record.backend))
256    return found
