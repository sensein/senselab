senselab_core.runtime.execution.probe_machine

The per-unit probe state machine of the Pool: resolve a unit, look up its estimate, wait on a peer, probe, end.

ProbeMachine holds the state of each unit this run (State), the parts waiting on a hello, the worker marks read, and the stragglers of closed probe queues. The Pool supplies what it acts through: the queues and roots, the plan and keys of a fragment, the grant, the scaler, and _produce, which puts a part sized by an estimate. A unit that drifts at co-tenancy 1 is confirming: its next part is held and probed afresh, and a probe that disagrees with the estimate supersedes it.

227    def _place(self, held: Held) -> None:
228        """Move `held`'s computed jobs through their unit's `ProbeState`: resolve, look up, and probe or put them."""
229        placed, values = self.host.plan_of(held.part)
230        jobs, placement = fragment_jobs(held.part, placed, values)
231        decoder = decoder_of(jobs)
232        ready, hello = self._marks.get(held.queue), self._hello.get(class_of(held.queue))
233        unit: MeasureUnit | Unresolved = (
234            decoder if isinstance(decoder, Unresolved) else self.resolver.unit(jobs[0].call, placement, decoder, hello)
235        )
236        if ready is not None and not isinstance(decoder, Unresolved):
237            entry = ready.envs.get(backend_key(placement.backend))
238            if ready.hardware is None:
239                unit = Unresolved(f"a worker of {held.queue} names no hardware class")
240            elif entry is None and isinstance(unit, Pending):
241                unit = Pending(f"the ready mark of a worker of {held.queue} holds no env of {placement.backend.name!r}")
242            elif entry is not None and "unbuilt" in entry:
243                why = entry["unbuilt"]
244                unit = Unresolved(f"the env of backend {placement.backend.name!r} could not be built: {why}")
245            elif entry is not None:
246                unit = self.resolver.unit(jobs[0].call, placement, decoder, hello or ready.hardware, entry)
247        if isinstance(unit, Pending):
248            held.followed.waiting.append(held.part)
249            with self.lock:
250                self._resolving.append(held)
251            self.host.wake()
252        elif isinstance(unit, Unresolved):
253            self._release(Release(held, None, None), jobs, placement)
254        else:
255            self._lookup(unit, held, jobs, placement)

257    def _lookup(self, unit: MeasureUnit, held: Held, jobs: Sequence[Job], placement: Placement) -> None:
258        """Take `held` to its unit's state: an ended unit sizes it, a waiting or probing one holds it, else look up."""
259        digest = unit.digest()
260        self.resolved[digest] = unit
261        predicted = self._hello.get(class_of(held.queue)) is None
262        state = self._states.get(digest)
263        if state is not None:
264            if state.state == "confirming" and state.run is None and not self._inflight(state.unit):
265                self._confirm(state, held, jobs, placement)
266            elif state.state in ("peer_wait", "probing", "confirming"):
267                held.followed.waiting.append(held.part)
268                state.held.append(held)
269            else:
270                self._release(Release(held, state.estimate, unit), jobs, placement)
271            return
272        estimate = self.measures.get(unit)
273        if self._gate(unit, held, estimate):
274            return
275        if held.following:
276            self._release(Release(held, estimate, unit), jobs, placement)
277            return
278        plan = None if covers(estimate, jobs) else self._plan_probe(unit, held, jobs, placement)
279        if plan is None or not worth(plan, estimate):
280            ended = State(
281                unit,
282                held.queue,
283                "measured" if estimate else "unmeasured",
284                self._clock.now(),
285                estimate,
286                predicted=predicted,
287            )
288            with self.lock:
289                self._states[digest] = ended
290            self._release(Release(held, estimate, unit), jobs, placement)
291            return
292        taken = self._take_lease(unit, digest)
293        if taken == "peer":
294            held.followed.waiting.append(held.part)
295            waiting = State(unit, held.queue, "peer_wait", self._clock.now(), estimate, [held], predicted=predicted)
296            with self.lock:
297                self._states[digest] = waiting
298            return
299        self._start_probe(unit, held, plan, taken, predicted, placement)

301    def _plan_probe(
302        self, unit: MeasureUnit, held: Held, jobs: Sequence[Job], placement: Placement, *, fresh: bool = False
303    ) -> ProbePlan | None:
304        """Return `Probe.plan` over `held`'s jobs against what the roots hold of `unit`; None when unplannable.
305
306        `fresh` leaves the stored observations out of the prior, as a confirming probe does.
307        """
308        found, kind = [detected_units(j) for j in jobs], detected_kind(jobs)
309        keys = self.host.keys(held.part)
310        candidates = [
311            Candidate(fragment="part", index=i, key=keys[i], units=u if k == kind != "none" else None)
312            for i, (k, u) in enumerate(found)
313        ]
314        edges = bucket_edges(placement.backend, kind)
315        if unit.formation.startswith("bucket:") and not edges:
316            return None
317        stored = () if fresh else tuple(self.roots.observations(unit))
318        prior = Prior(observations=stored, records=tuple(self.roots.regime_records(unit)))
319        return Probe.plan(
320            unit,
321            candidates,
322            prior,
323            self.profile.calibration,
324            unit.formation,
325            self.probe_s,
326            quantum_items=self.profile.pool.quantum_items,
327            bucket_edges=edges,
328        )

330    def _take_lease(
331        self, unit: MeasureUnit, digest: str
332    ) -> Literal["peer"] | tuple[Lease | None, contextlib.ExitStack | None]:
333        """Take the unit's probe lease; "peer" while an admitted peer holds it and this unit has not yet waited."""
334        found = self.roots.probe_lease(unit)
335        if isinstance(found, PeerProbing):
336            return (None, None) if digest in self._waited else "peer"
337        if isinstance(found, Unleased):
338            return None, None
339        stack = contextlib.ExitStack()
340        try:
341            stack.enter_context(found)
342            stack.enter_context(found.keep_alive(self.profile.lease.renew_fraction))
343        except LeaseTimeout:
344            stack.close()
345            return (None, None) if digest in self._waited else "peer"
346        except LeaseUnavailable:
347            stack.close()
348            return None, None
349        return found, stack

351    def _start_probe(  # noqa: PLR0913 - the unit and its part, the plan, the lease, and what the queue records
352        self,
353        unit: MeasureUnit,
354        held: Held,
355        plan: ProbePlan,
356        taken: tuple[Lease | None, contextlib.ExitStack | None],
357        predicted: bool,
358        placement: Placement,
359        confirming: State | None = None,
360    ) -> None:
361        """Put `plan`'s items on a new probe queue pinned to the unit's hardware class; its other jobs wait.
362
363        A `confirming` unit's probe leaves the stored observations out of its prior and keeps its state.
364        """
365        probe_id = uuid.uuid4().hex
366        name = probe_queue(held.queue, probe_id)
367        env = {k: v for k, v in unit.env.model_dump(mode="json").items() if k != "decoder"}
368        self.host.queue(name).set_probe_unit(unit.hardware, backend_key(placement.backend), env)
369        prior = () if confirming is not None else tuple(self.roots.observations(unit))
370        run = Run(
371            probe_id, name, plan, held, self._clock.now(), prior, last=self._clock.now(), lease=taken[0], stack=taken[1]
372        )
373        state = confirming or State(unit, held.queue, "probing", self._clock.now(), predicted=predicted)
374        state.run = run
375        with self.lock:
376            self._states[unit.digest()] = state
377        held.followed.waiting.append(held.part)
378        self._put_items(state, plan.items)

380    def _put_items(self, state: State, items: Sequence[ProbeItem]) -> None:
381        """Put probe `items` on the run's queue, one fragment of one request shape each."""
382        run, dumped = state.run, Dumped.of(state.run.held.part)  # type: ignore[union-attr]
383        assert run is not None
384        queue = self.host.queue(run.queue)
385        for item in items:
386            indices = [c.index for c in item.jobs]
387            placement = run.held.part.placement.model_copy(update={"batch_size": item.request_jobs})
388            fragment = dumped.part(indices, placement=placement.model_dump(mode="json"))
389            queue.put_probe(fragment, item.phase, item.stratum)
390            flight = Flight(run.queue, item_digest(fragment), fragment)
391            self._origins[flight.digest] = Origin(state.unit, None, state.queue, run.id)
392            run.held.followed.flights.append(flight)
393            run.awaiting[flight.digest] = (item, flight)
394            run.used.update(indices)
395            run.put[flight.digest] = len(indices)
396        self.probes[state.unit.digest()] = ProbedUnit(run.id, frozenset(run.put), sum(run.put.values()))
397        self.host.wake()

399    def _advance(self) -> None:
400        """Advance every unit's state: greet new workers, resolve what waited on a hello, tick waits and probes."""
401        fresh = self._greet()
402        if fresh and self._resolving:
403            with self.lock:
404                waiting, self._resolving = self._resolving, []
405            for held in waiting:
406                held.followed.waiting = [p for p in held.followed.waiting if p is not held.part]
407                self._place(held)
408        if self._resolving:
409            self._expire_resolving()
410        for state in list(self._states.values()):
411            if state.state == "peer_wait":
412                self._peer_tick(state)
413            elif state.run is not None and state.state in ("probing", "confirming"):
414                self._probe_tick(state)
415            elif state.state == "confirming" and state.held and not self._inflight(state.unit):
416                self._start_confirming(state)
417        self._release_stragglers()
418        while self._releasing:
419            release = self._releasing[0]
420            placed, values = self.host.plan_of(release.held.part)
421            jobs, placement = fragment_jobs(release.held.part, placed, values)
422            self._release(release, jobs, placement)
423            self._releasing.pop(0)

425    def _release_stragglers(self) -> None:
426        """Put as production the jobs of an item of a closed probe queue whose live claim has gone stale."""
427        for held, flight, indices in list(self._stragglers):
428            queue = self.host.queue(flight.queue)
429            if queue.outcome(flight.digest) is None and not queue.claimed(flight.digest):
430                self._stragglers.remove((held, flight, indices))
431                held.followed.flights.remove(flight)
432                part = Held(held.followed, held.queue, Dumped.of(held.part).part(indices))
433                held.followed.waiting.append(part.part)
434                self._releasing.append(Release(part, None, None))
435            elif queue.outcome(flight.digest) is not None:
436                self._stragglers.remove((held, flight, indices))

438    def _greet(self) -> bool:
439        """Read each new ready mark: write its class into the hardware index; a class's first hello decides it.
440
441        Return whether a mark was read that had not been.
442        """
443        fresh, disagreed = False, []
444        for name, mark in self.host.marks():
445            if name in self._greeted:
446                continue
447            if mark is None:
448                self._unreadable.add(name)
449                continue
450            self._unreadable.discard(name)
451            self._greeted.add(name)
452            fresh = True
453            for queue, ready in mark.queues.items():
454                self._marks.setdefault(queue, ready)
455                resource = self.profile.resource_classes.get(class_of(queue))
456                if resource is not None and ready.hardware is not None:
457                    self.roots.note_seen(resource, ready.hardware)
458                    if class_of(queue) not in self._hello:
459                        self._hello[class_of(queue)] = ready.hardware
460                        disagreed.append((class_of(queue), ready.hardware))
461        for resource_class, seen in disagreed:
462            self._reresolve(resource_class, seen)
463        return fresh

465    def _expire_resolving(self) -> None:
466        """Run as production the jobs of a unit still `Resolving` `resolve_s` after its queue's first live worker."""
467        now = self._clock.now()
468        for handle in self.host.live_workers():
469            if not handle.pending:
470                self._live_since.setdefault(handle.queue, now)
471        late = [
472            h
473            for h in self._resolving
474            if (since := self._live_since.get(h.queue)) is not None and now - since >= self.profile.resolve_s
475        ]
476        with self.lock:
477            self._resolving = [h for h in self._resolving if h not in late]
478        for held in late:
479            if held.queue not in self._said:
480                self._said.add(held.queue)
481                logger.warning(
482                    "no readable ready mark from a worker of %s within resolve_s of starting (%d unreadable); "
483                    "its jobs run on the safe values",
484                    held.queue,
485                    len(self._unreadable),
486                )
487            placed, values = self.host.plan_of(held.part)
488            jobs, placement = fragment_jobs(held.part, placed, values)
489            self._release(Release(held, None, None), jobs, placement)

491    def _peer_tick(self, state: State) -> None:
492        """Return a waiting unit to `New` once its peer's lease is gone or `peer_wait_s` passed; else keep waiting."""
493        now = self._clock.now()
494        if state.checked is not None and now - state.checked < self.profile.pool.poll_s:
495            return
496        state.checked = now
497        expired = now - state.since >= self.profile.peer_wait_s
498        if not expired and self.roots.peer_holds(state.unit):
499            return
500        if expired:
501            self._waited.add(state.unit.digest())
502        with self.lock:
503            del self._states[state.unit.digest()]
504        for held in state.held:
505            held.followed.waiting = [p for p in held.followed.waiting if p is not held.part]
506            self._place(held)

512    def _read_outcomes(self, state: State, *, replace: bool) -> None:
513        """Take the observations of every awaited item that has an outcome; `replace` swaps a served item."""
514        run = state.run
515        assert run is not None
516        queue = self.host.queue(run.queue)
517        for digest, (item, _) in list(run.awaiting.items()):
518            record = queue.outcome(digest)
519            if record is None:
520                if queue.claimed(digest):
521                    run.last = self._clock.now()
522                continue
523            run.last = self._clock.now()
524            del run.awaiting[digest]
525            reported = record.get("observations") if record.get("outcome") == "done" else None
526            found = [Observation.model_validate(o) for o in reported] if isinstance(reported, list) else []
527            run.observations.extend(found)
528            served = any(o.outcome == "served" for o in found)
529            if replace and served and self._warm_s(run.observations) < self.probe_s:
530                old = run.plan
531                run.plan = Probe.replace_served(old, item)
532                self._put_items(state, [i for i in run.plan.items if i not in old.items])

534    def _probe_tick(self, state: State) -> None:
535        """Read the probe's outcomes; replace a served item, plan the batched phase, or end it on its budgets."""
536        run, cfg = state.run, self.profile.calibration
537        assert run is not None

802    def _gate(self, unit: MeasureUnit, held: Held, estimate: Estimate | None) -> bool:

876    def oom(self, digest: str, observations: Sequence[Observation], worker: str | None) -> None:
877        """Write the `oom` record of item `digest` from its requests that failed with a resource fault (§7.4).
878
879        The record is of the largest faulting request (the shortest of equals): its batch, its longest job's length
880        and the co-tenancy of its grant (the device's, the host's on a shared host, else 1; an unread count is 1). The
881        unit's estimate is read again at once.
882        """
883        origin = self._origins.get(digest)
884        faulted = [o for o in observations if o.fault == "resource"]
885        if origin is None or not faulted or digest in self._oomed:
886            return
887        self._oomed.add(digest)
888        grant = self.host.grant(origin.queue)
889        if grant.memory_bytes is None:
890            logger.warning("no oom record of %s: its grant states no memory", origin.unit.label())
891            return
892        worst = max(faulted, key=lambda o: (o.batch, -o.padded_units / o.batch))
893        shared = worst.cotenancy.device if grant.on_device else worst.cotenancy.host if grant.shared_host else 1
894        basis = origin.estimate.fault_basis(origin.probe) if origin.estimate else tuple(filter(None, [origin.probe]))
895        record = OomRecord(
896            resource="device" if grant.on_device else "host",
897            batch=worst.batch,
898            length=worst.padded_units / worst.batch,
899            cotenancy=shared or 1,
900            grant_bytes=grant.memory_bytes,
901            basis=basis,
902            worker=worker,
903        )
904        shape = (origin.unit.digest(), record.resource, record.batch, record.cotenancy, record.grant_bytes, basis)
905        with self._machine:
906            if self._shortest.get(shape, math.inf) <= record.length:
907                return
908            self._shortest[shape] = record.length
909            self.roots.put_record(origin.unit, "oom", record, self._provenance(origin.unit.hardware))
910            if (state := self._states.get(origin.unit.digest())) is not None:
