senselab_core.runtime.execution.telemetry

What the Pool reads of production requests: drift from the estimate that sized them, and the epoch's actuals.

Drift keeps the last drift_window requests of each (unit, co-tenancy) as ratios wall_s / predicted and says when their median lies outside the band the estimate's spread allows. peak_beyond and disagree are the two other tests of production against an estimate. Actuals collects what a class's work took, to close its placement event.

 22def predicted(estimate: Estimate, request: Observation) -> tuple[float, float | None] | None:
 23    """Return the seconds `estimate` predicts for `request` at its batch, and the fit's spread; None without a fit.
 24
 25    The prediction is in the probe regime: the single fit at batch 1, else the batched fit of that batch.
 26    """
 27    if request.batch == 1:
 28        fit, units = estimate.single, request.units
 29    else:
 30        fit, units = next((f for f in estimate.batched if f.batch == request.batch), None), request.padded_units
 31    if fit is None:
 32        return None
 33    return fit.r + (fit.u or 0.0) * units, fit.spread

 36def is_evidence(request: Observation, hardware: HardwareClass, warmup_requests: int) -> bool:
 37    """Return whether `request` is drift evidence: a warm, `ok` production request on `hardware`, not served."""
 38    return (
 39        request.probe is None
 40        and request.outcome == "ok"
 41        and request.load_s is None
 42        and request.since_load is not None
 43        and request.since_load >= warmup_requests
 44        and request.hardware == hardware
 45        and request.wall_s > 0
 46    )

 50    """Return the half-width of the band a median of `window` ratios may lie in around 1."""

 91class Drift:
 92    """The last `window` readings of each (unit, co-tenancy), and the test of their median against its band."""
 93
 94    def __init__(self, window: int, confidence: float) -> None:
 95        """Judge windows of `window` readings by the two-sided z `confidence`."""
 96        self.window, self.confidence = window, confidence
 97        self._seen: dict[tuple[str, int], deque[Reading]] = {}
 98
 99    def add(self, unit: str, k: int, reading: Reading) -> Drifted | None:
100        """Take `reading` of unit `unit` under co-tenancy `k`; return the window if it is full and has drifted.
101
102        A window that drifted starts over, so the next verdict is of `window` new requests.
103        """
104        seen = self._seen.setdefault((unit, k), deque(maxlen=self.window))
105        seen.append(reading)
106        if len(seen) < self.window:
107            return None
108        rho = statistics.median(r.rho for r in seen)
109        expected = statistics.median(r.expected for r in seen)
110        half = band(self.confidence, statistics.median(r.spread for r in seen), self.window)
111        if abs(statistics.median(r.rho / r.expected for r in seen) - 1.0) <= half:
112            return None
113        found = Drifted(
114            k,
115            rho,
116            len(seen),
117            round(statistics.median(r.batch for r in seen)),
118            tuple(r.rho for r in seen),
119            expected * (1.0 - half),
120            expected * (1.0 + half),
121        )
122        seen.clear()
123        return found

126def peak_beyond(estimate: Estimate, request: Observation, confidence: float, window: int) -> float | None:
127    """Return the `m1` a device peak beyond its band implies, None when the request is within it.
128
129    The prediction is `m0 + m1·padded_units` of the device fit; the band is the drift test's, at the spread of the
130    request's batch. None as well where the estimate has no device fit, the request no device reading, or the batch no
131    measured spread.
132    """
133    fit, found = estimate.device, predicted(estimate, request)
134    if fit is None or found is None or found[1] is None or request.padded_units <= 0:
135        return None
136    if request.device_peak is None or request.device_after is None:
137        return None
138    actual = request.device_peak - request.device_after
139    if actual <= (fit.m0 + fit.m1 * request.padded_units) * (1.0 + band(confidence, found[1], window)):
140        return None
141    return (actual - fit.m0) / request.padded_units

144def disagree(suspect: Estimate, probed: Estimate, confidence: float) -> bool:
145    """Return whether two single fits differ beyond their bands at the middle of `probed`'s measured range.
146
147    The bands are the drift test's, each of its fit's spread and requests. A fit with no measured spread cannot
148    disagree.
149    """
150    a, b = suspect.single, probed.single
151    if a is None or b is None or a.spread is None or b.spread is None:
152        return False
153    x = (b.lo + b.hi) / 2
154    ta, tb = a.r + (a.u or 0.0) * x, b.r + (b.u or 0.0) * x
155    if ta <= 0 or tb <= 0:
156        return False
157    error = MEDIAN_SE * math.hypot(ta * a.spread / math.sqrt(a.n), tb * b.spread / math.sqrt(b.n))
158    return abs(tb - ta) > confidence * error

161class Actuals:
162    """What each class's work took this epoch: from its first put to its last outcome, and its largest device peak."""
163
164    def __init__(self, clock: Clock) -> None:
165        """Time every class on `clock`."""
166        self._clock = clock
167        self._first: dict[str, float] = {}
168        self._last: dict[str, float] = {}
169        self._peak: dict[str, int] = {}
170
171    def began(self, resource_class: str) -> None:
172        """Note that work of `resource_class` was put, the first time this epoch."""
173        self._first.setdefault(resource_class, self._clock.now())
174
175    def landed(self, resource_class: str, observations: Sequence[Observation]) -> None:
176        """Note an outcome of `resource_class` and the device peaks of its `observations`."""
177        self._last[resource_class] = self._clock.now()
178        peaks = [o.device_peak for o in observations if o.device_peak is not None]
179        if peaks:
180            self._peak[resource_class] = max(self._peak.get(resource_class, 0), *peaks)
181
182    def closing(self, resource_class: str, worker_s: float | None) -> ClassActuals:
183        """Return what the class's work took; `worker_s` is its workers' lifetimes, given for a `cuda` class."""
184        first, last = self._first.get(resource_class), self._last.get(resource_class)
185        return ClassActuals(
186            wall_s=None if first is None or last is None else max(0.0, last - first),
187            gpu_hours=None if worker_s is None else worker_s / HOUR_S,
188            device_peak_bytes=self._peak.get(resource_class),
189        )

192class Placements:
193    """One epoch's placement events for one log: each class's record as placed, and again, closed, at the epoch's end.
194
195    `request` builds the run's request, from which a class placed with no hook takes its needs; `cuda` says whether
196    a class is a GPU class, whose GPU hours are its workers' lifetimes.
197    """
198
199    def __init__(
200        self,
201        log: EventLog,
202        request: Callable[[], PlacementRequest],
203        actuals: Actuals,
204        cuda: Callable[[str], bool],
205    ) -> None:
206        """Log to `log`, closing each class with what `actuals` read of it."""
207        self._log, self._request, self._actuals, self._cuda = log, request, actuals, cuda
208        self._placed: dict[str, PlacementRecord] = {}
209
210    def record(  # noqa: PLR0913 - what a `Placer` hands its recorder
211        self, _: int, name: str, needs: ClassNeeds | None, hook: HookRef | None, placed: ClassPlacement
212    ) -> None:
213        """Append class `name`'s placement to the log and keep it with the request it was placed against."""
214        record = placement_record(name, needs, hook, placed)
215        self._log.record("placement", None, placement=record)
216        asked = needs if needs is not None else self._request().held(name)
217        self._placed[name] = record.model_copy(update={"request": asked})
218
219    def close(self, worker_seconds: Callable[[str], float] | None = None) -> None:
220        """Append each placed class's record again with its actuals; `worker_seconds` gives a class's lifetimes."""
221        for name, record in self._placed.items():
222            worker_s = worker_seconds(name) if worker_seconds is not None and self._cuda(name) else None
223            closing = self._actuals.closing(name, worker_s)
224            self._log.record("placement", None, placement=record.model_copy(update={"closing": closing}))
225        self._placed.clear()
