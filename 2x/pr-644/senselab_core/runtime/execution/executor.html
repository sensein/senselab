 54def stored_message(message: str) -> str:
 55    """Return `message` for a record other users read: paths, account and host names rendered to labels.
 56
 57    The full text goes to the operator log. An account is the running user's, the owner `owner_remedy` names, or the
 58    lease holder `Lease.holder` names; a host is this one's name or its first label; a path is any absolute one. A
 59    name is matched as a whole token.
 60    """
 61    kept = _OWNER.sub(r"owned by uid \1; as its owner,", message)
 62    kept = _HOLDER.sub(r"\1'<account>'", kept)
 63    shown = _token(pwd.getpwuid(os.getuid()).pw_name).sub("<account>", _PATH.sub("<path>", kept))
 64    host = socket.gethostname()
 65    for name in sorted({host, host.split(".")[0]} - {""}, key=len, reverse=True):
 66        shown = _token(name).sub("<host>", shown)
 67    if shown != message:
 68        logger.info("a fault message was rendered for the record; the operator's text was: %s", message)
 69    return shown

115class FaultRecord(Frozen):
116    """A job's failure: its fault class, the exception's type and message, whether it is a gap, whether it is charged.
117
118    Only a content fault that is not a gap is charged against the item's retry budget; a resource or infrastructure
119    fault re-places the job, and a gap fault, one the fallback table classified, is never charged. `load` is true for a
120    fault stated while a worker started, loaded or reloaded its model: it is the environment's, not an item's.
121    """
122
123    fault: FaultName
124    type: NonEmptyStr
125    message: str
126    gap: bool
127    charged: bool
128    load: bool = False
129
130    @model_serializer(mode="wrap")
131    def _omit_a_false_load(self, handler: SerializerFunctionWrapHandler) -> dict[str, Any]:
132        """Serialise `load` only when it is true, so a record that is no load's reads as it did before the field."""
133        data: dict[str, Any] = handler(self)
134        if not data.get("load"):
135            data.pop("load", None)
136        return data

138    @field_validator("message")
139    @classmethod
140    def _render_message(cls, v: str) -> str:
141        return stored_message(v)

143    @model_validator(mode="after")
144    def _check_charged(self) -> "FaultRecord":
145        if self.charged and (self.gap or self.fault != FaultClass.CONTENT.value):
146            raise ValueError("only a content fault that is not a gap is charged")
147        return self

149    @property
150    def counts_against_env(self) -> bool:
151        """Whether the failure is its environment's: infrastructure, a gap that is not resource, or a load's fault."""
152        return self.fault == "infrastructure" or (self.gap and self.fault != "resource") or self.load

154    @property
155    def fault_class(self) -> FaultClass:
156        """The fault class, as a `FaultClass`."""
157        return FaultClass(self.fault)

159    @classmethod
160    def from_exception(cls, exc: BaseException) -> "FaultRecord":
161        """Return the record of `exc` as a worker classifies it (`error_record`), its notes after its message."""
162        error = error_record(exc)
163        notes = [str(n) for n in getattr(exc, "__notes__", ())]
164        message = wire_text("\n".join([error["message"], *notes]))
165        return cls.of({**error, "message": message, "load": bool(getattr(exc, "from_load", False))})
166
167    @classmethod
168    def gap_of(cls, kind: str, message: str) -> "FaultRecord":
169        """Return the infrastructure gap `kind` with `message`: a failure no typed fault names, never charged."""
170        return cls.of({"type": kind, "message": message, "fault": FaultClass.INFRASTRUCTURE.value, "gap": True})
171
172    @classmethod
173    def of(cls, error: Mapping[str, Any]) -> "FaultRecord":
174        """Return the record of a worker's error object (`error_record`): charged for a content fault, not a gap.
175
176        The one place a `FaultRecord` is built from its parts; every other site calls it, `from_exception` or `gap_of`.
177        """
178        gap = bool(error.get("gap", True))
179        fault = FaultClass(error["fault"])
180        return cls(
181            fault=fault.value,
182            type=str(error["type"]),
183            message=str(error.get("message", "")),
184            gap=gap,
185            charged=fault is FaultClass.CONTENT and not gap,
186            load=bool(error.get("load", False)),
187        )

239class PlanFragment(Formatted):
240    """A `plan-fragment@3.0` document: jobs of one node of one admitted plan, their input identities, the placement.
241
242    `inputs` holds one mapping per job, in the order of `jobs`; every job names `plan_hash` and `node`. `admission`
243    names the admission whose graph-input values the jobs read; `upstream` gives, for each `output:` input
244    identity (one port of one upstream job, `port_output_identity`), where that port's value is stored; `gathered`
245    gives, for each `gather:` input identity, its elements' port identities in element order, each in `upstream`.
246    `attempt` counts the submissions of these jobs.
247    """
248
249    format: FormatTag = PLAN_FRAGMENT_FORMAT
250    port_value: ClassVar[bool] = False
251    plan_hash: Hex64
252    node: NodeId
253    admission: NonEmptyStr
254    jobs: Annotated[tuple[JobRef, ...], Field(min_length=1)]
255    inputs: tuple[JobInputs, ...]
256    upstream: Upstream = FrozenMapping({})
257    gathered: Gathered = FrozenMapping({})
258    placement: NodePlacement
259    attempt: Annotated[int, Field(ge=1)] = 1
260
261    @model_validator(mode="after")
262    def _check_jobs(self) -> "PlanFragment":
263        if len(self.inputs) != len(self.jobs):
264            raise ValueError(f"a fragment of {len(self.jobs)} jobs holds {len(self.inputs)} input mappings")
265        if len(set(self.jobs)) != len(self.jobs):
266            raise ValueError("a fragment names each job once")
267        if any((j.plan_hash, j.node) != (self.plan_hash, self.node) for j in self.jobs):
268            raise ValueError(f"every job of the fragment is of plan {self.plan_hash} node {self.node!r}")
269        named = {i for inputs in self.inputs for i in inputs.values()}
270        if {i for i in named if i.startswith("gather:")} != set(self.gathered):
271            raise ValueError(f"gathered names {sorted(self.gathered)}; the jobs read other gathers")
272        if any(gather_identity(elements) != g for g, elements in self.gathered.items()):
273            raise ValueError("a gather identity is the identity of its elements in order")
274        read = {i for i in named if i.startswith("output:")} | {e for es in self.gathered.values() for e in es}
275        if read != set(self.upstream):
276            raise ValueError(f"upstream names {sorted(self.upstream)}; the jobs read {sorted(read)}")
277        if any(identity != ref.identity() for identity, ref in self.upstream.items()):
278            raise ValueError("an upstream identity is the identity of its key digest and port")
279        return self

355class Executor(Protocol):
356    """Runs plan fragments of admitted plans: `admit` a plan, `submit` its fragments, follow, stop and read them.
357
358    An admission holds the values of the plan's graph inputs by content identity, until it is released; an
359    `output:` input is read from the result store. Two admissions of one plan are independent. `logs_claims` says
360    whether the executor's workers log each job `claimed`, in which case the runner logs no `started` as it sends one.
361    `splits_fragments` says whether the executor splits a fragment into its own units of work, in which case the runner
362    sends the jobs of a node that are ready together as one fragment, not in fragments of the batch size.
363    """
364
365    logs_claims: ClassVar[bool]
366    splits_fragments: ClassVar[bool]
367
368    def bind_epoch(self, epoch: int, events: "EventLog") -> None:
369        """Run what follows as the coordinator of `epoch`, logging to `events` (that epoch's attempt)."""
370
371    def admit(self, plan: Plan, values: Mapping[InputIdentity, object]) -> Admission:
372        """Hold `plan` and the value of each graph-input identity its fragments name, and return the admission."""
373
374    def release(self, admission: Admission) -> None:
375        """Let go of `admission` and its values."""
376
377    def fragment_size(self, plan: Plan, node: str) -> int | None:
378        """Return the most jobs of `node` one fragment holds; None for an executor that splits fragments."""
379
380    def submit(self, fragment: PlanFragment) -> Handle:
381        """Take `fragment` and return its handle."""
382
383    def status(self, handle: Handle) -> Status:
384        """Return where the fragment of `handle` is."""
385
386    def cancel(self, handle: Handle) -> None:
387        """Stop the jobs of `handle` not yet run."""
388
389    def poll(self, handle: Handle) -> tuple[list[ResultRef], bool]:
390        """Return the results of `handle` landed and not yet given, and whether every one is given; never wait."""
391

393def follow(
394    executor: Executor,
395    handle: Handle,
396    *,
397    first_s: float = 0.01,
398    max_s: float = 1.0,
399    clock: Clock = SystemClock(),  # noqa: B008 - stateless
400) -> Iterator[ResultRef]:
401    """Yield each result of `handle` as `executor.poll` gives it, until every one is given.
402
403    A poll that gives nothing is followed by a wait of `first_s`, doubling up to `max_s`; one that gives a result
404    resets it; the waits run on `clock`. Errors `poll` raises are raised here, after the results given before them.
405    """
406    pause = first_s
407    while True:
408        given, done = executor.poll(handle)
409        yield from given
410        if done:
411            return
412        if given:
413            pause = first_s
414        else:
415            clock.sleep(pause)
416            pause = min(2 * pause, max_s)
