senselab_core.runtime.execution.events

A run's event log and the status derived from it: Event, EventLog, read_events and status_from.

Every state change of a job, and each class's placement in an epoch, is one run-event@2 line in <run>/events/<writer>.jsonl; a writer appends to its own file only, so any number of processes log one run without sharing a file. A run's status, its counts and its faults are computed from the job events alone.

129class Event(Formatted):
130    """A `run-event@2` document: one state change of one job of one attempt, or a run-scoped `placement` of a class.
131
132    `job` names the job (None for an event of the node as a whole), `key` its key digest, `attempt` the id of the
133    attempt (run or resume) that wrote it, `ordinal` that attempt's place among the run's attempts (taken when it was
134    claimed), `epoch` the coordinator epoch the attempt belongs to (0 for none), `writer` the writer whose file holds
135    it, `seq` its position in that file, `placement_digest` the digest of the node's placement, `fault` the class of a
136    failure, `failure` the record of it, `reuse` the policy that served a `cache_hit` (its value), `env` the env the job
137    ran in, `replaced` whether a failed job was queued again on a larger resource class (its fault is then not counted
138    against the env), `output` the output identity of a `finished` or `cache_hit` job, `lease` the stale lease a
139    `reclaimed` claim broke and `t_ns` the wall-clock nanoseconds of the event.
140    """
141
142    format: FormatTag = RUN_EVENT_FORMAT
143    port_value: ClassVar[bool] = False
144    kind: EventKind
145    node: NodeId | None = None
146    key: Hex64 | None = None
147    job: JobRef | None = None
148    attempt: NonEmptyStr
149    ordinal: Annotated[int, Field(ge=1)]
150    epoch: Annotated[int, Field(ge=0)] = 0
151    writer: NonEmptyStr
152    seq: Annotated[int, Field(ge=0)]
153    placement_digest: Annotated[str, StringConstraints(pattern=r"^[0-9a-f]{64}$")] | None = None
154    fault: FaultName | None = None
155    failure: FaultRecord | None = None
156    reuse: Literal["exact", "equivalent", "tolerant"] | None = None
157    env: NonEmptyStr | None = None
158    output: OutputIdentity | None = None
159    placement: PlacementRecord | None = None
160    replaced: bool = False
161    lease: LeaseBreak | None = None
162    t_ns: Annotated[int, Field(ge=0)]
163
164    @model_validator(mode="after")
165    def _check(self) -> "Event":
166        if self.kind == "placement":
167            if (
168                self.placement is None
169                or any(
170                    v is not None
171                    for v in (
172                        self.node,
173                        self.job,
174                        self.key,
175                        self.fault,
176                        self.failure,
177                        self.output,
178                        self.reuse,
179                        self.env,
180                        self.lease,
181                    )
182                )
183                or self.placement_digest is not None
184            ):
185                raise ValueError("a placement event carries its record alone: no node, job, key, fault, output or env")
186            return self
187        if self.node is None or self.placement is not None:
188            raise ValueError("a job event names its node and carries no placement record")
189        if self.job is not None and self.job.node != self.node:
190            raise ValueError(f"event of node {self.node!r} names job {self.job.node!r}")
191        if self.failure is not None and self.fault != self.failure.fault:
192            raise ValueError("an event's fault is the class of its failure")
193        if self.kind == "failed" and self.failure is None:
194            raise ValueError("a failed event carries its failure")
195        if self.kind != "failed" and (self.fault is not None or self.failure is not None or self.replaced):
196            raise ValueError("only a failed event carries a fault, and whether the job was placed on a larger class")
197        if self.reuse is not None and self.kind != "cache_hit":
198            raise ValueError("only a cache_hit event carries a reuse policy")
199        if (self.kind == "reclaimed") != (self.lease is not None):
200            raise ValueError("a reclaimed event carries the lease it broke, and no other event does")
201        return self

216@dataclass(frozen=True)
217class RunStatus:
218    """Where a run is: its jobs by state, its failures by fault class, its broken envs and whether it is over.
219
220    Derived from events alone (`status_from`). `done` is True once every job seen has a terminal event, and
221    `cancelled` once any job was cancelled, so a cancelled run is never plain done. `last_event_ns` is the time of the
222    newest event. `interrupted` is set by `Run.wait` alone, when the run is not done and its coordinator stopped.
223    `reclaimed` counts the claims that broke a stale lease first (`reclaimed` events); `quarantined` the event lines a
224    reader could not decode (`EventScan`), so a job whose terminal event was one reads as not done for that reason.
225    `unplaceable` holds each distinct message of a plan that fits no class (`Unplaceable`): an outcome of the plan,
226    never counted against an env. `superseded` is set by `Run.status` alone, for an attempt whose epoch a newer one
227    replaced: its counts are of its own epoch and earlier, never the newer attempt's.
228    """
229
230    counts: Mapping[JobState, int]
231    faults: Mapping[str, int]
232    broken: tuple[BrokenEnv, ...] = ()
233    done: bool = False
234    cancelled: bool = False
235    last_event_ns: int | None = None
236    interrupted: bool = False
237    nodes: Mapping[str, Mapping[JobState, int]] = field(default_factory=dict)
238    reclaimed: int = 0
239    quarantined: int = 0
240    unplaceable: tuple[str, ...] = ()
241    superseded: bool = False

244def _latest(events: Sequence[Event]) -> list[Event]:
245    """Return the events of a job's latest attempt: the highest `epoch`, then the highest `ordinal` within it."""
246    latest = max((e.epoch, e.ordinal, e.attempt) for e in events)
247    return [e for e in events if (e.epoch, e.ordinal, e.attempt) == latest]

250def _superseded(event: Event, events: Sequence[Event]) -> bool:
251    """Return whether `event` is a failure its writer then queued again: a retry within one attempt."""
252    return event.kind == "failed" and any(
253        e.kind == "queued" and e.writer == event.writer and e.seq > event.seq for e in events
254    )

257def _current(events: Sequence[Event], epoch: int | None) -> list[Event]:
258    """Return the events of a job's latest attempt (`_latest`); none when it has no event of epoch `epoch` or later."""
259    mine = _latest(events)
260    return [] if epoch is not None and mine[0].epoch < epoch else mine

263def _latest_terminal(events: Sequence[Event], epoch: int | None = None) -> Event | None:
264    """Return the terminal event of a job's latest attempt that has precedence, else an earlier result under its key.
265
266    A result (`finished`, `cache_hit`) outranks `failed`, which outranks `blocked`, then `cancelled`; among events of
267    one kind the one later in its writer's file, then the greater writer, is taken. A `failed` its writer queued
268    again is not terminal. While the latest attempt has no terminal event, a result an earlier attempt gave stands
269    only when its key is the key the latest attempt's events give the job. With `epoch`, a job with no event of that
270    epoch has no latest attempt (`_current`).
271    """
272    mine = _current(events, epoch)
273    terminal = [e for e in mine if e.kind in TERMINAL and not _superseded(e, mine)]
274    if not terminal:
275        keys = {e.key for e in mine if e.key is not None}
276        terminal = [e for e in events if e.kind in RESULTS and e.key in keys]
277    return max(terminal, key=lambda e: (TERMINAL[e.kind], e.epoch, e.ordinal, e.writer, e.seq)) if terminal else None

280def _job_state(events: Sequence[Event], epoch: int | None = None) -> JobState:
281    """Return a job's state: its terminal state (`_latest_terminal`), else `running` or `queued` by counts.
282
283    In the latest attempt a job is running while its claims (`claimed`, `started`) outnumber its yields and at least
284    match its submissions (`queued`), so a retry queued after a failure is queued until it is claimed again. Counts
285    read no clock and no order across writers.
286    """
287    found = _latest_terminal(events, epoch)
288    if found is not None:
289        return _STATE[found.kind]
290    kinds = Counter(e.kind for e in _current(events, epoch))
291    claims = kinds["claimed"] + kinds["started"]
292    return "running" if claims > kinds["yielded"] and claims >= kinds["queued"] else "queued"

295def _by_job(events: Iterable[Event]) -> dict[tuple[str, tuple[int, ...]], list[Event]]:
296    """Return each job's state-changing events, in the order given."""
297    jobs: dict[tuple[str, tuple[int, ...]], list[Event]] = {}
298    for event in events:
299        if event.kind in _STATE and event.node is not None:
300            jobs.setdefault((event.node, event.job.element if event.job is not None else ()), []).append(event)
301    return jobs

304def terminal_events(events: Iterable[Event], *, epoch: int | None = None) -> dict[tuple[str, tuple[int, ...]], Event]:
305    """Return, for each job `(node, element)` with one, the terminal event that states how it ended (under `epoch`)."""
306    found = {job: _latest_terminal(each, epoch) for job, each in _by_job(events).items()}
307    return {job: event for job, event in found.items() if event is not None}

310def status_from(
311    events: Iterable[Event], *, broken_after: int | None = None, epoch: int | None = None, quarantined: int = 0
312) -> RunStatus:
313    """Return the status of a run from its `events`, each writer's in file order (`read_events`).
314
315    A job's state is the terminal event of its latest attempt (highest `epoch`, then `ordinal`) with precedence
316    (`TERMINAL`), a failure queued again excepted, else a result of an earlier attempt under the same key, else
317    running or queued as the latest attempt's claims and submissions count (`_job_state`); `verify` changes none.
318    `seq` is compared within one writer only and no clock order is assumed for terminal states. `faults` counts
319    every `failed` event by its class; an env with `broken_after` infrastructure failures (the packaged profile's
320    `infrastructure_replacements` when None) is `broken`. Cancelled and blocked jobs are not faults. With `epoch`
321    (the run's current one), a job with no event of
322    that epoch is queued. `quarantined` is the number of lines the reader refused, carried into the status.
323    """
324    if broken_after is None:
325        from senselab_core.runtime.profiles import ExecutionProfile  # noqa: PLC0415 - read only when needed
326
327        broken_after = ExecutionProfile.load().faults.infrastructure_replacements
328    seen = [e for e in events if e.kind != "placement"]
329    jobs = _by_job(seen)
330    faults: Counter[str] = Counter()
331    failures: Counter[str] = Counter()
332    latest: dict[str, FaultRecord] = {}
333    for event in seen:
334        if event.kind == "failed" and event.fault is not None:
335            faults[event.fault] += 1
336            if (
337                event.env is not None
338                and event.failure is not None
339                and event.failure.counts_against_env
340                and not event.replaced
341            ):
342                failures[event.env] += 1
343                latest[event.env] = event.failure
344    tally: Counter[str] = Counter()
345    per_node: dict[str, Counter[str]] = {}
346    for (node, _), found in jobs.items():
347        state = _job_state(found, epoch)
348        tally[state] += 1
349        per_node.setdefault(node, Counter())[state] += 1
350    broken = tuple(
351        BrokenEnv(env=env, fault=latest[env], failures=n) for env, n in sorted(failures.items()) if n >= broken_after
352    )
353    return RunStatus(
354        counts={s: tally[s] for s in STATES},
355        faults=dict(faults),
356        broken=broken,
357        done=bool(jobs) and not tally["queued"] and not tally["running"],
358        cancelled=tally["cancelled"] > 0,
359        last_event_ns=max((e.t_ns for e in seen), default=None),
360        nodes={n: {s: c[s] for s in STATES} for n, c in sorted(per_node.items())},
361        reclaimed=sum(1 for e in seen if e.kind == "reclaimed"),
362        quarantined=quarantined,
363        unplaceable=tuple(
364            sorted({e.failure.message for e in seen if e.failure is not None and e.failure.type == UNPLACEABLE})
365        ),
366    )

409class EventLog:
410    """One writer's append-only log of a run: `<run dir>/events/<writer uuid>.jsonl`, written by this writer only.
411
412    `attempt` is the id of the attempt every event of this log belongs to, `ordinal` its claim order and `epoch` its
413    coordinator's epoch (0 for none), `clock` gives
414    the `t_ns` of an event `record` makes, and each event gets the next `seq` of this writer. A writer used after a
415    `fork` is a new writer in the child.
416    """
417
418    def __init__(
419        self, run_dir: Path, attempt: str, ordinal: int, *, epoch: int = 0, clock: Callable[[], int] = time.time_ns
420    ) -> None:
421        """Log to `run_dir` as attempt `attempt`, the `ordinal`-th, of `epoch`; nothing is created until an append."""
422        self.run_dir = Path(run_dir)
423        self.attempt = attempt
424        self.ordinal = ordinal
425        self.epoch = epoch
426        self.clock = clock
427        self._writer = (os.getpid(), uuid.uuid4().hex)
428        self._seq = 0
429        self._lock = threading.Lock()
430
431    @property
432    def writer(self) -> str:
433        """This process's writer uuid4 hex: the name of the file `append` writes."""
434        if self._writer[0] != os.getpid():
435            self._writer = (os.getpid(), uuid.uuid4().hex)
436            self._seq = 0
437        return self._writer[1]
438
439    @property
440    def path(self) -> Path:
441        """The file this writer appends to."""
442        return self.run_dir / EVENTS_DIR / f"{self.writer}.jsonl"
443
444    def append(self, event: Event) -> None:
445        """Append `event` as one line of this writer's file: the whole line, or nothing.
446
447        Raises:
448            EventWriteFailed: the line could not be written whole; the file holds only the lines before it.
449        """
450        data = (json.dumps(event.model_dump(mode="json"), sort_keys=True, separators=(",", ":")) + "\n").encode()
451        with self._lock:
452            path = self.path
453            path.parent.mkdir(parents=True, exist_ok=True)
454            try:
455                with StorageDir.open_root(path.parent) as directory:
456                    directory.append(path.name, data, grammar=EVENT_FILE)
457            except OSError as err:
458                raise EventWriteFailed(path, event.seq, err) from err

460    def record(  # noqa: PLR0913 - the fields of an event
461        self,
462        kind: EventKind,
463        node: str | None,
464        *,
465        job: JobRef | None = None,
466        key: str | None = None,
467        placement_digest: str | None = None,
468        failure: FaultRecord | None = None,
469        reuse: str | None = None,
470        env: str | None = None,
471        output: str | None = None,
472        placement: PlacementRecord | None = None,
473        replaced: bool = False,
474        lease: LeaseBreak | None = None,
475    ) -> Event:
476        """Append the event of `kind` for `node` (None for a `placement`), stamped now with the next `seq`."""
477        with self._lock:
478            writer = self.writer
479            seq, self._seq = self._seq, self._seq + 1
480        event = Event(
481            kind=kind,
482            node=node,
483            job=job,
484            key=key,
485            attempt=self.attempt,
486            ordinal=self.ordinal,
487            epoch=self.epoch,
488            writer=writer,
489            seq=seq,
490            placement_digest=placement_digest,
491            fault=None if failure is None else failure.fault,
492            failure=failure,
493            reuse=reuse,
494            env=env,
495            output=output,
496            placement=placement,
497            replaced=replaced,
498            lease=lease,
499            t_ns=self.clock(),
500        )
501        self.append(event)
502        return event
