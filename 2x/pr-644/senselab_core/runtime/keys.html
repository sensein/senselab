 39def profile_slices(
 40    task: TaskSpec, settings: Mapping[str, Any], dataset_profile: Mapping[str, Any] | None = None
 41) -> dict[str, ProfileSliceIdentity]:
 42    """Return each profile slice `task` declares, by name, keyed by the identity of its value (`slice_identity`).
 43
 44    A `run.<path>` slice is cut from `settings`, a `dataset.<path>` one from `dataset_profile`, following the dotted
 45    path one mapping key per part. Nothing else of either mapping is read.
 46
 47    Raises:
 48        MissingProfileSlice: a declared slice's path is absent.
 49    """
 50    sources: dict[str, Mapping[str, Any] | None] = {"run": settings, "dataset": dataset_profile}
 51    cut: dict[str, ProfileSliceIdentity] = {}
 52    for name in task.profile_slices:
 53        source, *path = name.split(".")
 54        value: Any = sources[source]
 55        for part in path:
 56            if not isinstance(value, Mapping) or part not in value:
 57                given = "no dataset profile" if sources[source] is None else f"no {part!r} under {source}"
 58                raise MissingProfileSlice(f"{task.name!r} reads profile slice {name!r}; there is {given}")
 59            value = value[part]
 60        cut[name] = slice_identity(value)
 61    return cut

 85def keys_bucket(backend: BackendSpec) -> bool:
 86    """Whether `backend` is not batch-invariant and declares its batch formation output-affecting."""
 87    return not backend.batch_invariant and "batch_formation" in backend.output_affecting

 90def fixed_formation(backend: BackendSpec) -> str | None:
 91    """Return the formation of every request to `backend`: `mixed`, `single`, or None when each job's length picks."""
 92    if backend.batch_invariant:
 93        return MIXED
 94    return None if keys_bucket(backend) else SINGLE

 97def request_size(backend: BackendSpec, batch_size: int) -> int:
 98    """Return the most items per request to `backend` under a placement of `batch_size`: 1 for formation `single`.
 99
100    Raises:
101        ValueError: `batch_size` is below 1.
102    """
103    if batch_size < 1:
104        raise ValueError(f"a request holds at least one item, got batch size {batch_size}")
105    return 1 if fixed_formation(backend) == SINGLE else batch_size

108def request_components(backend: BackendSpec, formation: str | None = None) -> RequestComponents:
109    """Return the batch formation and chunk grid of a request to `backend` of a batch of `formation` (`form_batches`).
110
111    `formation` omitted is the backend's `fixed_formation`. The chunk grid is the backend's.
112
113    Raises:
114        ValueError: `formation` is not one `backend` is sent in: `mixed` exactly when it is batch-invariant, a bucket
115            only when it keys one.
116    """
117    if formation is None:
118        formation = fixed_formation(backend)
119    elif backend.batch_invariant != (formation == MIXED) or (
120        formation.startswith(f"{BUCKET}:") and not keys_bucket(backend)
121    ):
122        raise ValueError(f"backend {backend.name!r} is not sent batches of formation {formation!r}")
123    return {"batch_formation": formation, "chunk_grid": backend.chunk_grid}

127class Placement:
128    """Where a call runs: the backend and its params, the device, the resource class, and how batches are formed.
129
130    `batch_formation` and `chunk_grid` describe the placement whether or not the backend declares them
131    output-affecting; `batch_formation` is None while each job's length picks it. `resource_class` is the class of
132    worker it runs on, `batch_size` the most items per request, `partition` the scheduler partition as the profile
133    names it, and `why` the reason `explain` gives; none of these reaches a key.
134    """
135
136    backend: BackendSpec
137    backend_params: BackendParams
138    device: str
139    batch_formation: str | None = None
140    chunk_grid: str | None = None
141    why: str = ""
142    resource_class: str = "cpu"
143    batch_size: int = 1
144    partition: str | None = None
145
146    def env(self) -> str:
147        """Return the backend's env as `host` or `<kind>:<name>`."""
148        env = self.backend.env
149        return env.kind if env.name is None else f"{env.kind}:{env.name}"
150
151    def record(self) -> NodePlacement:
152        """Return this placement as the `NodePlacement` a plan's placement document holds."""
153        return NodePlacement(
154            backend=self.backend.name,
155            backend_version=self.backend.version,
156            backend_params=self.backend_params.model_dump(mode="json"),
157            stack=self.backend.stack,
158            env=self.env(),
159            device=self.device,
160            batch_size=self.batch_size,
161            resource_class=self.resource_class,
162            partition=self.partition,
163            why=self.why,
164        )
165
166    def explain(self) -> str:
167        """Return the backend, stack, env, device, batch, resource class, partition and the reason, as one line."""
168        return (
169            f"{self.backend.task}: backend {self.backend.name!r} (stack {self.backend.stack!r}, env {self.env()}, "
170            f"device {self.device}, batch {self.batch_size}, resource class {self.resource_class}, "
171            f"partition {self.partition or 'none'}); {self.why}"
172        )

175def build_key(call: Call, placement: Placement, context: ExecutionContext) -> LogicalKey:
176    """Return the logical key of running `call` under `placement` (`call_key`); `context` never reaches it.
177
178    Raises:
179        TypeError: if `context` is not an `ExecutionContext`.
180        ValueError, UnpinnedModel, pydantic.ValidationError: as `key_for`.

182    if not isinstance(context, ExecutionContext):

184    return call_key(call, placement)

187def call_key(call: Call, placement: Placement) -> LogicalKey:
188    """Return the logical key of running `call` under `placement`.
189
190    The key holds the backend's name and version, its params, and the batch formation and chunk grid
191    the backend declares output-affecting; no other placement field: never the device, env root, reason, resource
192    class, batch size or partition.
193
194    Raises:
195        ValueError, UnpinnedModel, pydantic.ValidationError: as `key_for`.
196    """
197    declared = placement.backend.output_affecting
198    return key_for(
199        call.task,
200        placement.backend,
201        params=call.params,
202        backend_params=placement.backend_params,
203        inputs=call.inputs,
204        model=call.model,
205        profile_slices=call.profile_slices,
206        batch_formation=placement.batch_formation if "batch_formation" in declared else None,
207        chunk_grid=placement.chunk_grid if "chunk_grid" in declared else None,
208        service=call.service,
209    )
