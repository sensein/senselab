 61        + config.min_requests * config.single_item_jobs

 82def _inversions(z: np.ndarray, work: Work | None = None) -> int:
 83    """Count the pairs `i < j` with `z[i] > z[j]`, by merging runs of doubling length level by level.
 84
 85    Each merge level charges its `len(z)` values to `work`.
 86    """
 87    ranks = np.unique(z, return_inverse=True)[1].ravel().astype(np.int64)
 88    n, span, total = len(ranks), int(ranks.max(initial=0)) + 1, 0
 89    index, run = np.arange(n), 1
 90    while run < n:
 91        _charge(work, n)
 92        block, right = index // (2 * run), (index // run) % 2 == 1
 93        keys = block * span + ranks
 94        left = np.sort(keys[~right])
 95        at_most = np.searchsorted(left, keys[right], side="right") - np.searchsorted(left, block[right] * span)
 96        total += int((run - at_most).sum())
 97        run *= 2
 98    return total

101def _inverted_pairs(z: Sequence[float], work: Work | None = None) -> list[tuple[int, int]]:
102    """List the pairs `i < j` with `z[i] > z[j]`, in time linear in their number beyond n log n.
103
104    Each merge charges its run's length to `work`, and each pair listed charges one.
105    """
106
107    def merge(run: list[int]) -> list[int]:
108        if len(run) <= 1:
109            return run
110        _charge(work, len(run))
111        half = len(run) // 2
112        left, right = merge(run[:half]), merge(run[half:])
113        keys = [z[i] for i in left]
114        before = len(pairs)
115        pairs.extend((i, j) for j in right for i in left[bisect_right(keys, z[j]) :])
116        _charge(work, len(pairs) - before)
117        return sorted(left + right, key=z.__getitem__)
118
119    pairs: list[tuple[int, int]] = []
120    merge(list(range(len(z))))
121    return pairs

135def _rank_bound(points: Sequence[tuple[float, float]], slope: float) -> float:
136    """Return `2ε·(max|y − t·x̃| + |t|·max|x̃|) / min Δx` at `t = slope`: `x̃` is x centred at its median, `ε` epsilon.
137
138    `_kth_slope` compares points by `y − t·x̃`, which rounds once in the product and once in the subtraction; a pair
139    compares two of them, so it is ranked wrongly only if its slope is within this distance of `t`.
140    """
141    xs, ys = np.array([x for x, _ in points]), np.array([y for _, y in points])
142    centred = xs - float(np.median(xs))
143    spread = float(np.abs(ys - slope * centred).max()) + abs(slope) * float(np.abs(centred).max())
144    return 2 * sys.float_info.epsilon * spread / float(np.diff(np.unique(xs)).min())

147def _kth_slope(xs: np.ndarray, ys: np.ndarray, k: int, total: int, work: Work | None = None) -> float:
148    """Return the `k`-th smallest (from 0) of the `total` slopes between points of distinct `x`, sorted by `(x, y)`.
149
150    A pair's slope is below `t` exactly when the pair is inverted in `y − t·x`, so the slopes below `t` are counted by
151    inversion counting and bisected over the doubles until at most n lie between the bounds, which are then listed.
152    Slopes tied beyond that are the one value of the pairs adjacent in `y − t·x` at the bound. The comparisons use `x`
153    centred at its median, which leaves every slope unchanged and makes differences of nearby values exact; a pair is
154    then ranked wrongly only if its slope is within `_rank_bound` of `t`, and the slopes returned are computed from the
155    original `x`.
156    """
157    centred = xs - float(np.median(xs))
158    bound = max(float(np.ptp(ys)) / float(np.diff(np.unique(xs)).min()), 1.0)
159    low, high, below, above = _ordered(-2 * bound), _ordered(2 * bound), 0, total
160    while above - below > len(xs) and high - low > 1:
161        middle = (low + high) // 2
162        count = _inversions(ys - _unordered(middle) * centred, work)
163        low, below = (middle, count) if count <= k else (low, below)
164        high, above = (high, above) if count <= k else (middle, count)
165    if above - below > len(xs):
166        pivot = _unordered(low)
167        near = np.argsort(ys - pivot * centred, kind="stable")
168        _charge(work, len(near))
169        tied = [(ys[b] - ys[a]) / (xs[b] - xs[a]) for a, b in itertools.pairwise(near.tolist()) if xs[a] != xs[b]]
170        return float(min(tied, key=lambda slope: abs(slope - pivot), default=pivot))
171    lower, upper = ys - _unordered(low) * centred, ys - _unordered(high) * centred
172    order = sorted(range(len(xs)), key=lambda i: (lower[i], i))
173    slopes = sorted(
174        (ys[order[j]] - ys[order[i]]) / (xs[order[j]] - xs[order[i]])
175        for i, j in _inverted_pairs([upper[o] for o in order], work)
176    )
177    return float(slopes[min(max(k - below, 0), len(slopes) - 1)])

180def _theil_sen(points: Sequence[tuple[float, float]], work: Work | None = None) -> tuple[float, float] | None:
181    """Return `(intercept, slope)`: the median of the pairwise slopes, and the median intercept at that slope.
182
183    The slope is the median of the n(n−1)/2 pairwise slopes, found in O(n log² n) time and O(n) memory; it equals the
184    reference's except for slopes closer than `_rank_bound` to it. `work` counts the steps taken.
185    """
186    ordered = sorted(points)
187    xs, ys = np.array([x for x, _ in ordered]), np.array([y for _, y in ordered])
188    ties = sum(c * (c - 1) // 2 for c in Counter(xs.tolist()).values())
189    total = len(ordered) * (len(ordered) - 1) // 2 - ties
190    if total == 0:
191        return None
192    middle = (total - 1) // 2
193    slope = _kth_slope(xs, ys, middle, total, work)
194    if total % 2 == 0:
195        slope = statistics.median([slope, _kth_slope(xs, ys, middle + 1, total, work)])
196    return _median([y - slope * x for x, y in ordered]), slope

203def _cuts(units: Sequence[float], count: int) -> list[tuple[int, int]]:
204    """Return `[start, stop)` runs cutting ascending `units` into at most `count` of equal count, no value in two."""
205    n, parts = len(units), min(count, len(units))
206    bounds = set()
207    for i in range(1, parts):
208        b = i * n // parts
209        moves = [b] if units[b - 1] != units[b] else [bisect_left(units, units[b]), bisect_right(units, units[b])]
210        if inside := [p for p in moves if 0 < p < n]:
211            bounds.add(min(inside, key=lambda p: (abs(p - b), p)))
212    edges = [0, *sorted(bounds), n]
213    return [(a, z) for a, z in itertools.pairwise(edges) if a < z]

216def _by_stratum(items: Sequence[Observation], ranges: Sequence[tuple[float, float]]) -> dict[int, list[Observation]]:
217    """Group `items` by the length range holding each one's per-job length."""
218    grouped: dict[int, list[Observation]] = {}
219    los = [lo for lo, _ in ranges]
220    for o in items:
221        i = bisect_right(los, _length(o)) - 1
222        if i >= 0 and _length(o) <= ranges[i][1]:
223            grouped.setdefault(i, []).append(o)
224    return grouped

252def _select(observations: Sequence[Observation], hardware: HardwareClass, config: CalibrationSettings) -> _Kept:
253    """Keep `ok` probe requests of the unit's hardware in the regime each probe ran under (§5.4 steps 1-2).
254
255    The strata are the warm requests' per-job lengths cut into `length_strata` ranges of equal count.
256    """
257    bisected = {(o.probe.id, o.item) for o in observations if o.probe is not None and o.outcome == "bisected"}
258    ok = [
259        o
260        for o in observations
261        if o.outcome == "ok"
262        and o.probe is not None
263        and o.hardware == hardware
264        and (o.probe.id, o.item) not in bisected
265        and o.cotenancy.device == 1
266    ]
267    quietest: dict[str, int] = {}
268    for o in ok:
269        if o.probe is not None:
270            quietest[o.probe.id] = min(o.cotenancy.host, quietest.get(o.probe.id, o.cotenancy.host))
271    kept = [o for o in ok if o.probe is not None and o.cotenancy.host == quietest[o.probe.id]]
272    warm = [o for o in kept if o.load_s is None and o.since_load is not None and o.since_load >= config.warmup_requests]
273    admitted = {id(o) for o in warm}
274    lengths = sorted(_length(o) for o in warm)
275    whole: dict[tuple[str, str], list[Observation]] = {}
276    for o in observations:
277        if o.probe is not None and o.hardware == hardware:
278            whole.setdefault((o.probe.id, o.item), []).append(o)
279    extras: list[float] = []
280    for requests in whole.values():
281        last = [o.item_wall_s for o in requests if o.item_wall_s is not None]
282        if last and all(id(o) in admitted for o in requests):
283            extras.append(max(0.0, last[0] - sum(o.wall_s for o in requests)))
284    return _Kept(
285        single=[o for o in warm if o.batch == 1],
286        batched=[o for o in warm if o.batch > 1],
287        loads=[o for o in kept if o.load_s is not None],
288        items=extras,
289        warm=warm,
290        ranges=[(lengths[a], lengths[z - 1]) for a, z in _cuts(lengths, config.length_strata)],
291    )

294def _request_fit(
295    observations: Sequence[Observation],
296    batch: int,
297    units: Callable[[Observation], float],
298    config: CalibrationSettings,
299    ranges: Sequence[tuple[float, float]],
300) -> RequestFit | None:
301    """Fit `wall_s = r + u·units` over the strata with `min_requests` observations; None when none has them."""
302    strata = {s: o for s, o in _by_stratum(observations, ranges).items() if len(o) >= config.min_requests}
303    points = [(units(o), o.wall_s) for group in strata.values() for o in group]
304    if not points:
305        return None
306    xs = [x for x, _ in points]
307    fit = _theil_sen(points) if len(strata) >= 2 else None  # noqa: PLR2004 - a slope needs two points in length
308    if fit is not None and fit[1] >= 0:
309        r, u = max(fit[0], 0.0), fit[1]
310        predicted = [r + u * x for x in xs]
311        spread = _relative_spread([y - p for (_, y), p in zip(points, predicted, strict=True)], predicted)
312        return RequestFit(batch=batch, r=r, u=u, n=len(points), spread=spread, lo=min(xs), hi=max(xs))
313    walls = [y for _, y in points]
314    centre = _median(walls)
315    spread = _relative_spread([y - centre for y in walls], [centre] * len(walls))
316    return RequestFit(batch=batch, r=centre, u=None, n=len(points), spread=spread, lo=min(xs), hi=max(xs))

319def _warmup_unsettled(single: RequestFit | None, kept: _Kept, config: CalibrationSettings) -> bool | None:
320    """Whether the first three warm requests are slower than the later ones by more than the single fit's band.
321
322    The comparison is within one probe's planned length stratum (`ProbeRef.stratum`). The early requests are those
323    with `since_load` from `warmup_requests` to `warmup_requests + 2`; the later ones follow them and must number
324    `min_requests`. The baseline is `median(later walls) + u·(units − median(later units))`, with `u` the single
325    fit's slope (0 when unmeasured). True when a stratum holding both has the early requests' median ratio to it
326    above `1 + confidence × spread`; False when every such stratum is within it; None (unknown) when no stratum
327    holds both, or the fit has no spread or a zero spread.
328    """
329    if single is None or single.spread is None or single.spread <= 0:
330        return None
331    first = config.warmup_requests
332    slope = single.u or 0.0
333    verdicts = []
334    groups: dict[tuple[str, int | None], list[Observation]] = {}
335    for o in kept.single:
336        if o.probe is not None:
337            groups.setdefault((o.probe.id, o.probe.stratum), []).append(o)
338    for group in groups.values():
339        warm = [o for o in group if o.since_load is not None and o.since_load >= first]
340        early = [o for o in warm if o.since_load is not None and o.since_load < first + EARLY_REQUESTS]
341        later = [o for o in warm if o not in early]
342        if not early or len(later) < config.min_requests:
343            continue
344        level = _median([o.wall_s for o in later])
345        centre = _median([o.units for o in later])
346        ratios = [o.wall_s / base for o in early if (base := level + slope * (o.units - centre)) > 0]
347        if ratios:
348            verdicts.append(_median(ratios) - 1.0 > config.confidence * single.spread)
349    return any(verdicts) if verdicts else None

352def _relative_spread(residuals: Sequence[float], predicted: Sequence[float]) -> float | None:
353    centre = _median(predicted)
354    if centre <= 0:
355        return None
356    middle = _median(residuals)
357    return MAD_TO_SIGMA * _median([abs(e - middle) for e in residuals]) / centre

360def _memory_fit(
361    kept: _Kept, kind: Literal["device", "host"], config: CalibrationSettings
362) -> tuple[MemoryFit | None, EstimateReason | None]:
363    """Return the memory fit of `kind` over warm requests, and the reason it is None."""
364    device = kind == "device"
365    rows = {
366        id(o): (peak, after)
367        for o in kept.warm
368        if (peak := o.device_peak if device else o.host_peak) is not None
369        and (after := o.device_after if device else o.host_after) is not None
370    }
371    if not rows:
372        return None, "device_unmeasured" if device else "host_unmeasured"
373    groups = _by_stratum([o for o in kept.warm if id(o) in rows], kept.ranges).values()
374    points = [
375        (o.padded_units, float(rows[id(o)][0] - rows[id(o)][1]))
376        for group in groups
377        if len(group) >= config.min_requests
378        for o in group
379    ]
380    fit = _theil_sen(points)
381    if fit is None:
382        return None, "device_below_min_requests" if device else "host_below_min_requests"
383    if fit[1] < 0:
384        return None, "device_fit_nonphysical" if device else "host_fit_nonphysical"
385    resident = min(after for _, after in rows.values())
386    return MemoryFit(resident=resident, m0=fit[0], m1=fit[1], n=len(points)), None

395    """Return the most jobs of `length` units whose batch peak fits `budget_bytes`, within `[1, ceiling]`."""

418def _in_force(records: Sequence[RegimeRecord], kept: _Kept, minimum: int) -> tuple[OomRecord, ...]:
419    """Return the distinct `oom` records in force, oldest first (§7.4).
420
421    A probe measures and lifts in the probe regime only: a record there is lifted once a probe outside its `basis`
422    has `minimum` warm batched observations in the stratum holding its length; any other record stays in force.
423    """
424    live = []
425    for r in dict.fromkeys(x for x in records if isinstance(x, OomRecord)):
426        span = _holding(kept.ranges, r.length)
427        later = Counter(
428            o.probe.id
429            for o in kept.batched
430            if span is not None
431            and o.probe is not None
432            and o.probe.id not in r.basis
433            and span[0] <= _length(o) <= span[1]

435        if not (probe_regime(r) and any(n >= minimum for n in later.values())):
436            live.append(r)
437    return tuple(live)

440def raised(fit: MemoryFit | None, records: Sequence[OomRecord], memory_headroom: float) -> MemoryFit | None:
441    """Return `fit` with `m1` raised so that every faulting pair of `records` (batch of two or more) is infeasible.
442
443    A record's room is its grant divided among the processes that shared the device when it faulted (§7.4).
444    """
445    if fit is None:
446        return None
447    m1 = fit.m1
448    for r in records:
449        if r.length > 0 and r.batch >= 2:  # noqa: PLR2004 - a batch of one has none to shrink
450            share = r.grant_bytes / r.cotenancy
451            m1 = max(m1, (memory_headroom * share - fit.resident - fit.m0) / ((r.batch - 1) * r.length))
452    return fit.model_copy(update={"m1": m1})

455def _batch_choice(
456    single: RequestFit, batched: Sequence[RequestFit], lengths: Mapping[tuple[float, float], float], z: float
457) -> tuple[StratumBatch, ...]:
458    """Return per stratum the best measured batch that beats one beyond the margin, else 1 (§5.4 step 6)."""
459    chosen = []
460    for (lo, hi), length in sorted(lengths.items()):
461        t1 = single.r + (single.u or 0.0) * length
462        best, best_t = 1, t1
463        for fit in sorted(batched, key=lambda f: f.batch):
464            if single.spread is None or fit.spread is None or t1 <= 0:
465                continue
466            t_b = (fit.r + (fit.u or 0.0) * fit.batch * length) / fit.batch
467            sigma = MEDIAN_SE * math.sqrt(single.spread**2 / single.n + fit.spread**2 / fit.n)
468            if (t1 - t_b) / t1 > z * sigma and t_b < best_t:
469                best, best_t = fit.batch, t_b
470        chosen.append(StratumBatch(lo=lo, hi=hi, batch=best))
471    return tuple(chosen)

483def _strata(
484    candidates: Sequence[Candidate], count: int
485) -> tuple[tuple[Stratum, ...], tuple[tuple[Candidate, ...], ...]]:
486    """Return one job of each key cut into `count` strata of equal count by units; lengthless ones form one more."""
487    unique = sorted({c.key: c for c in sorted(candidates, key=_order, reverse=True)}.values(), key=_order)

528def _single_items(  # noqa: PLR0913 - the free pools, the strata wanted, the jobs used, the settings and the budget
529    free: list[list[Candidate]],
530    wanted: Sequence[int],
531    used: int,
532    config: CalibrationSettings,
533    unplanned: list[Unplanned],
534    budget: int,
535) -> tuple[list[ProbeItem], int]:
536    """Take the `SINGLE` items of the `wanted` strata from `free` (shortest first); return them and the jobs used.
537
538    The items follow the order of `wanted`, so the middle stratum's, planned first, hold the first warm requests and
539    its multi-job requests give that stratum later ones (the warm-up diagnostic compares them).
540
541    The items stay within `budget` jobs.
542    """
543    singles: list[ProbeItem] = []
544    for index in wanted:
545        counts = [(config.min_requests, 1)]
546        if index == len(free) // 2:
547            counts.append((config.min_requests * config.single_item_jobs, config.single_item_jobs))
548        for count, per_item in counts:
549            if len(free[index]) < count:
550                unplanned.append(Unplanned(phase="SINGLE", stratum=index, reason="too_few_candidates"))
551            elif used + count > budget:
552                unplanned.append(Unplanned(phase="SINGLE", stratum=index, reason="over_budget"))
553            else:
554                used += count
555                taken = [free[index].pop(0) for _ in range(count)]
556                singles += [
557                    ProbeItem(phase="SINGLE", stratum=index, request_jobs=1, jobs=tuple(taken[i : i + per_item]))
558                    for i in range(0, count, per_item)
559                ]
560    return singles, used

581def _buckets(edges: Sequence[float], pool: Sequence[Candidate]) -> list[list[int]]:
582    """Return the positions of `pool` grouped by bucket of `edges`, ascending (one group when there are no edges)."""
583    groups: dict[int, list[int]] = {}

589def _pack(sizes: Sequence[int], widths: Sequence[int]) -> list[list[int]] | None:
590    """Return per bucket the widths of the items it supplies, each item from one bucket; None when they do not fit.
591
592    Each item, widest first, goes to the bucket with the most jobs left, so one width is supplied while
593    `Σ_b ⌊n_b / w⌋` reaches the items asked.
594    """
595    left = list(sizes)
596    packed: list[list[int]] = [[] for _ in sizes]
597    for w in widths:
598        b = max(range(len(left)), key=lambda i: (left[i], -i), default=None)
599        if b is None or left[b] < w:
600            return None
601        left[b] -= w
602        packed[b].append(w)
603    return packed

606def _supply(first: int, pool: Sequence[Candidate], edges: Sequence[float], config: CalibrationSettings) -> int:
607    """Return the largest `B₁` within `first` whose items the `pool`'s buckets supply, one bucket per item."""
608    sizes = [len(g) for g in _buckets(edges, pool)]
609    while first >= 2 and _pack(sizes, _widths(first, config)) is None:  # noqa: PLR2004 - a batch of one is single
610        first -= 1
611    return first

625def _deal(jobs: Sequence[Candidate], widths: Sequence[int]) -> list[list[Candidate]]:
626    """Deal `jobs` in turn to items of `widths`, skipping full ones, so each item spans the jobs' range."""
627    items: list[list[Candidate]] = [[] for _ in widths]
628    slot = 0
629    for job in jobs:
630        while len(items[slot % len(widths)]) >= widths[slot % len(widths)]:
631            slot += 1
632        items[slot % len(widths)].append(job)
633        slot += 1
634    return items

637def _batched_items(
638    plan: ProbePlan, sizes: Mapping[int, int], none: ProbePlan, config: CalibrationSettings, jobs: int
639) -> ProbePlan:
640    """Return `none` with the batched items of `sizes` (`B₁` per stratum), their jobs spread by rank."""
641    items: list[ProbeItem] = []
642    remaining = [list(r) for r in plan.remaining]
643    for index, first in sizes.items():
644        groups = _buckets(plan.edges, remaining[index])
645        packed = _pack([len(g) for g in groups], _widths(first, config))
646        assert packed is not None, "_supply sized this batch to fit its buckets"
647        gone: set[int] = set()
648        for group, widths in zip(groups, packed, strict=True):
649            if not widths:
650                continue
651            picks = [group[i] for i in _spread_idx(len(group), sum(widths))]
652            gone |= set(picks)
653            dealt = _deal([remaining[index][i] for i in picks], widths)
654            items += [ProbeItem(phase="BATCHED", stratum=index, request_jobs=len(b), jobs=tuple(b)) for b in dealt]
655        remaining[index] = [c for i, c in enumerate(remaining[index]) if i not in gone]
656    return none.model_copy(
657        update={
658            "items": tuple(items),
659            "batched": tuple(sizes),
660            "remaining": tuple(tuple(r) for r in remaining),
661            "used_jobs": plan.used_jobs + jobs,
662        }
663    )

765    def plan(  # noqa: PLR0913 - the protocol's inputs (§4.1)
766        unit: "MeasureUnit",
767        candidates: Sequence[Candidate],
768        prior: Prior,
769        config: CalibrationSettings,
770        formation: str,
771        seconds: float,
772        *,
773        quantum_items: int,
774        bucket_edges: Sequence[float],
775    ) -> ProbePlan:
776        """Return the `WARMUP` and `SINGLE` items of a probe over `candidates`, and what a batched phase may use.
777
778        `candidates` are the held jobs the store would not serve. One job of each key is kept, the rest cut into
779        `length_strata` strata of equal count, and each (phase, stratum) `prior` already covers with `min_requests`
780        steady observations is left out, a batched stratum an `oom` record of `prior` re-opens excepted. `seconds`
781        is the warm-work budget the plan carries for the Pool. The plan's job budget is `probe_items(config,
782        quantum_items)`. `bucket_edges` are the input-unit edges of a `bucket:`
783        formation's grid (empty for any other): a batched item draws its jobs from one bucket.
784
785        Raises:
786            ValueError: the formation is bucketed and `bucket_edges` is empty.
787        """
788        if formation.startswith("bucket:") and not bucket_edges:
789            raise ValueError("a bucketed formation plans over its bucket edges; none were given")
790        budget = probe_items(config, quantum_items)
791        kept = _select(prior.observations, unit.hardware, config)
792        strata, pools = _strata(candidates, config.length_strata)
793        plan = ProbePlan(
794            items=(),
795            strata=strata,
796            remaining=pools,
797            batched=(),
798            unplanned=(),
799            formation=formation,
800            seconds=seconds,
801            probe_items=budget,
802            used_jobs=0,
803            edges=tuple(sorted(bucket_edges)),
804        )
805        if not strata:
806            return plan
807        middle = len(strata) // 2
808        unplanned: list[Unplanned] = []
809        wanted = []
810        for st in strata:
811            covered = _covered(kept.single, st, config.min_requests, lambda o: o.units)
812            if st.index == middle:
813                covered = covered and len(kept.items) >= config.min_requests
814            if covered:
815                unplanned.append(Unplanned(phase="SINGLE", stratum=st.index, reason="covered"))
816            else:
817                wanted.append(st.index)
818        reopened = _reopened(strata, _in_force(prior.records, kept, config.min_requests))
819        batched = tuple(
820            st.index
821            for st in strata
822            if formation != "single"
823            and (
824                st.index in reopened
825                or not _covered(kept.batched, st, config.min_requests, lambda o: o.padded_units / o.batch)
826            )
827        )
828        plan = plan.model_copy(update={"batched": batched})
829        free = [list(p) for p in pools]
830        warmup = config.warmup_requests
831        if warmup > len(free[0]) or warmup > budget:
832            wanted = []
833            unplanned.append(Unplanned(phase="WARMUP", stratum=None, reason="too_few_candidates"))
834            warmups: list[ProbeItem] = []
835        else:
836            warmups = [
837                ProbeItem(phase="WARMUP", stratum=0, request_jobs=1, jobs=(free[0].pop(0),)) for _ in range(warmup)
838            ]
839        middle_first = sorted(wanted, key=lambda i: i != len(free) // 2)
840        singles, used = _single_items(free, middle_first, warmup, config, unplanned, budget)
841        if not singles and not (batched and warmups):
842            return plan.model_copy(update={"unplanned": tuple(unplanned)})
843        return plan.model_copy(
844            update={
845                "items": (*warmups, *singles),
846                "remaining": tuple(tuple(f) for f in free),
847                "unplanned": tuple(unplanned),
848                "used_jobs": used,
849            }
850        )

852    @staticmethod
853    def plan_batched(  # noqa: PLR0913 - the batched phase's inputs (§4.3)
854        plan: ProbePlan,
855        single_estimate: Estimate,
856        grant: "Grant",
857        config: CalibrationSettings,
858        seconds_left: float,
859        *,
860        memory_headroom: float,
861    ) -> ProbePlan:
862        """Return `plan` with the `BATCHED` items its `SINGLE` estimate allows (§4.3).
863
864        `B₁` per stratum is the least of the memory batch at its longest candidate under `memory_headroom` of the
865        grant, the batch the item budget leaves, the batch whose items the single fit's upper prediction fits in
866        `seconds_left`, `grant.quantum_items`, and what the stratum's remaining candidates supply. `B₂` is the integer
867        nearest √B₁ when `batched_sizes` is 2. The result's `items` are the batched ones only.
868        """
869        none = plan.model_copy(update={"items": (), "batched": ()})
870        single, device = single_estimate.single, single_estimate.device
871        if plan.formation != "single" and plan.strata and not plan.batched:
872            return _unplanned(none, "covered")
873        blocked = _blocked(plan, single_estimate, grant)
874        if blocked is not None or single is None or device is None or grant.memory_bytes is None:
875            return _unplanned(none, blocked or "single_unmeasured")
876        spans = {st.index: st for st in plan.strata}
877        caps: dict[int, int] = {}
878        times: dict[int, float] = {}
879        for index in plan.batched:
880            hi = spans[index].hi
881            if hi is None:
882                none = _unplanned(none, "no_lengths", index)
883                continue
884            budget = memory_headroom * grant.memory_bytes
885            caps[index] = memory_batch(device, budget, hi, grant.quantum_items)
886            times[index] = (single.r + (single.u or 0.0) * hi) * (1 + config.confidence * (single.spread or 0.0))
887        left = plan.probe_items - plan.used_jobs
888        item_s = single_estimate.item_s or 0.0
889        supplied = fits_jobs = False
890        refused: PlanReason | None = None
891        for top in range(max(caps.values(), default=0), 1, -1):
892            sizes = {i: _supply(min(c, top), plan.remaining[i], plan.edges, config) for i, c in caps.items()}
893            sizes = {i: b for i, b in sizes.items() if b >= 2}  # noqa: PLR2004 - a batch of one is the single phase
894            if not sizes:
895                continue
896            supplied = True
897            jobs = sum(config.min_requests * _per_set(b, config)[0] for b in sizes.values())
898            wall = sum(
899                config.min_requests * (_per_set(b, config)[0] * times[i] + item_s * _per_set(b, config)[1])
900                for i, b in sizes.items()
901            )
902            fits_jobs = fits_jobs or jobs <= left
903            if jobs <= left and wall <= seconds_left:
904                planned = _batched_items(plan, sizes, none, config, jobs)
905                return planned if refused is None else _unplanned(planned, refused)
906            refused = refused or ("over_budget" if jobs > left else "over_seconds")
907        if not caps:
908            return none
909        if max(caps.values()) < 2:  # noqa: PLR2004 - a batch of one is the single phase
910            return _unplanned(none, "memory_below_two")
911        return _unplanned(
912            none, "too_few_candidates" if not supplied else "over_seconds" if fits_jobs else "over_budget"
913        )

854    def plan_batched(  # noqa: PLR0913 - the batched phase's inputs (§4.3)
855        plan: ProbePlan,
856        single_estimate: Estimate,
857        grant: "Grant",
858        config: CalibrationSettings,
859        seconds_left: float,
860        *,
861        memory_headroom: float,
862    ) -> ProbePlan:
863        """Return `plan` with the `BATCHED` items its `SINGLE` estimate allows (§4.3).
864
865        `B₁` per stratum is the least of the memory batch at its longest candidate under `memory_headroom` of the
866        grant, the batch the item budget leaves, the batch whose items the single fit's upper prediction fits in
867        `seconds_left`, `grant.quantum_items`, and what the stratum's remaining candidates supply. `B₂` is the integer
868        nearest √B₁ when `batched_sizes` is 2. The result's `items` are the batched ones only.
869        """
870        none = plan.model_copy(update={"items": (), "batched": ()})
871        single, device = single_estimate.single, single_estimate.device
872        if plan.formation != "single" and plan.strata and not plan.batched:
873            return _unplanned(none, "covered")
874        blocked = _blocked(plan, single_estimate, grant)
875        if blocked is not None or single is None or device is None or grant.memory_bytes is None:
876            return _unplanned(none, blocked or "single_unmeasured")
877        spans = {st.index: st for st in plan.strata}
878        caps: dict[int, int] = {}
879        times: dict[int, float] = {}
880        for index in plan.batched:
881            hi = spans[index].hi
882            if hi is None:
883                none = _unplanned(none, "no_lengths", index)
884                continue
885            budget = memory_headroom * grant.memory_bytes
886            caps[index] = memory_batch(device, budget, hi, grant.quantum_items)
887            times[index] = (single.r + (single.u or 0.0) * hi) * (1 + config.confidence * (single.spread or 0.0))
888        left = plan.probe_items - plan.used_jobs
889        item_s = single_estimate.item_s or 0.0
890        supplied = fits_jobs = False
891        refused: PlanReason | None = None
892        for top in range(max(caps.values(), default=0), 1, -1):
893            sizes = {i: _supply(min(c, top), plan.remaining[i], plan.edges, config) for i, c in caps.items()}
894            sizes = {i: b for i, b in sizes.items() if b >= 2}  # noqa: PLR2004 - a batch of one is the single phase
895            if not sizes:
896                continue
897            supplied = True
898            jobs = sum(config.min_requests * _per_set(b, config)[0] for b in sizes.values())
899            wall = sum(
900                config.min_requests * (_per_set(b, config)[0] * times[i] + item_s * _per_set(b, config)[1])
901                for i, b in sizes.items()
902            )
903            fits_jobs = fits_jobs or jobs <= left
904            if jobs <= left and wall <= seconds_left:
905                planned = _batched_items(plan, sizes, none, config, jobs)
906                return planned if refused is None else _unplanned(planned, refused)
907            refused = refused or ("over_budget" if jobs > left else "over_seconds")
908        if not caps:
909            return none
910        if max(caps.values()) < 2:  # noqa: PLR2004 - a batch of one is the single phase
911            return _unplanned(none, "memory_below_two")
912        return _unplanned(
913            none, "too_few_candidates" if not supplied else "over_seconds" if fits_jobs else "over_budget"
914        )
