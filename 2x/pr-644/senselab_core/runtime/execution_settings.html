 27class ResourceClass(Frozen):
 28    """A class of worker grants: the device and, for a GPU, its type; what each worker is granted; the caps.
 29
 30    Attributes:
 31        device: `cpu`, `cuda` or `mps`.
 32        gpu_type: The GPU type a `cuda` class is granted, such as `h100`; None for any GPU and for other devices.
 33        cpus: CPUs per worker; None for the host's.
 34        mem_bytes: Memory per worker; None for the host's.
 35        time_s: Wall-time limit per worker; None for none.
 36        partition: The scheduler partition, `${VAR}` expanded where a worker is submitted; None off a scheduler.
 37        gres: The scheduler's generic-resource request, such as `gpu:h100:1`; None for none.
 38        max_workers: Workers of the class alive at once.
 39        memory_cap_bytes: The static memory the models resident at once on one worker may hold; None for no cap.
 40    """
 41
 42    device: Literal["cpu", "cuda", "mps"]
 43    gpu_type: NonEmptyStr | None = None
 44    cpus: PositiveInt | None = None
 45    mem_bytes: PositiveInt | None = None
 46    time_s: PositiveSeconds | None = None
 47    partition: NonEmptyStr | None = None
 48    gres: NonEmptyStr | None = None
 49    max_workers: PositiveInt
 50    memory_cap_bytes: PositiveInt | None = None

 52    @model_validator(mode="after")
 53    def _check_gpu(self) -> "ResourceClass":
 54        if self.gpu_type is not None and self.device != "cuda":
 55            raise ValueError(f"a {self.device} class has no gpu_type")

 58    def matches(self, name: str) -> bool:
 59        """Whether `name` fits this device: `cpu[:v]`, `mps[:v]`, `gpu` or `gpu:<gpu_type>[:v]`, or `svc:<id>`."""
 60        kind, _, rest = name.partition(":")
 61        if kind == SERVICE_CLASS:
 62            return True
 63        if kind == "gpu":

158class CalibrationSettings(Frozen):
159    """How a unit without a measure is probed, where measures are kept, and how a measure is judged stale.
160
161    Attributes:
162        measures_roots: Roots read in order and pooled; writes go to the first writable one.
163        trust: `group` reads every root's files, `own` only the reader's own.
164        warmup_requests: Requests a model state answers before one counts as warm (the floor is 1: the first
165            request after a load is cold).
166        length_strata: Strata of equal count a probe cuts its candidate jobs into.
167        min_requests: Warm requests a stratum needs for a quantity to be measured.
168        single_item_jobs: Jobs of the one item per probe that separates the per-item cost from the per-job cost.
169        batched_sizes: Batch sizes the batched phase measures.
170        confidence: The one-sided z of the batch choice and of the item cut.
171        drift_confidence: The two-sided z of the drift test.
172        observations_kept: Superseded probes' observation files kept for audit.
173        unmeasured_workers: Workers a unit without a measure runs on.
174    """
175
176    measures_roots: Annotated[tuple[NonEmptyStr, ...], Field(min_length=1)]
177    trust: Trust
178    warmup_requests: Annotated[int, Field(ge=0)]
179    length_strata: PositiveInt
180    min_requests: PositiveInt
181    single_item_jobs: Annotated[int, Field(ge=2)]
182    batched_sizes: Annotated[int, Field(ge=1, le=2)]
183    confidence: PositiveFloat
184    drift_confidence: PositiveFloat
185    observations_kept: PositiveInt
186    unmeasured_workers: PositiveInt
187
188    @model_validator(mode="after")
189    def _check_roots(self) -> "CalibrationSettings":
190        if blank := [r for r in self.measures_roots if not r.strip()]:
191            raise ValueError(f"measures_roots holds a blank root {blank[0]!r}")
192        if relative := [r for r in self.measures_roots if r.strip() and r.strip()[0] not in "/~$"]:
193            raise ValueError(f"measures_roots holds {relative[0]!r}, which is not an absolute path")
194        if repeated := sorted({r for r in self.measures_roots if self.measures_roots.count(r) > 1}):
195            raise ValueError(f"measures_roots holds {repeated[0]!r} twice")
196        return self
197
198    @property
199    def drift_window(self) -> int:
200        """Production requests of a regime judged for drift together: `length_strata × min_requests`."""
201        return self.length_strata * self.min_requests

204class LeaseBoundError(ValueError):
205    """`1 + missed_beats` renewal intervals plus the storage's attribute delay do not stay below a stale bound."""

208class JobLease(Frozen):
209    """How every lease is kept and taken (`Lease`, `Lease.keep_alive`).
210
211    Attributes:
212        stale_after_s: Seconds on the storage clock after which another process breaks the lease.
213        renew_fraction: The share of `stale_after_s` between two beats.
214        attribute_delay_s: Seconds a client of the storage may show a directory's stale mtime.
215        missed_beats: Beats a live holder may miss without its lease being broken, at least 1; the bound allows
216            `1 + missed_beats` renewal intervals between two beats.
217        wait_s: Seconds a wait for a lease may take by default; at least `stale_after_s + attribute_delay_s`, the time a
218            taker needs to break a dead holder's lease.
219        poll_s: Seconds between two attempts to take a lease.
220        remove_attempts: Times a lease moved aside is removed while a taker that reached it before the move races
221            the removal.
222        record_attempts: Times a holder displaced before its write takes the lease again and merges, for a run's
223            `inputs.json` and a run manifest.
224        dir_mode: The octal mode of a lease directory and its token directory, such as `2770`.
225    """
226
227    stale_after_s: PositiveSeconds
228    renew_fraction: Annotated[float, Field(gt=0, lt=1)]
229    attribute_delay_s: Annotated[float, Field(ge=0, allow_inf_nan=False)]
230    missed_beats: PositiveInt
231    wait_s: PositiveSeconds
232    poll_s: PositiveSeconds
233    remove_attempts: PositiveInt
234    record_attempts: PositiveInt
235    dir_mode: Annotated[str, StringConstraints(pattern=r"^[0-7]{3,4}$")]
236
237    @model_validator(mode="after")
238    def _check_wait(self) -> "JobLease":
239        if self.wait_s < self.stale_after_s + self.attribute_delay_s:
240            raise ValueError(
241                f"lease.wait_s {self.wait_s} is below lease.stale_after_s {self.stale_after_s} + "
242                f"lease.attribute_delay_s {self.attribute_delay_s}: a taker would give up before it could break a "
243                "dead holder's lease"
244            )
245        return self
246
247    @property
248    def mode(self) -> int:
249        """`dir_mode` as the integer mode bits."""
250        return int(self.dir_mode, 8)
251
252    def check_bound(self) -> None:
253        """Raise `LeaseBoundError` unless the bound holds.
254
255        The bound is `(1 + missed_beats) × renew_fraction × stale_after_s + attribute_delay_s < stale_after_s`.
256        """
257        if (
258            1 + self.missed_beats
259        ) * self.renew_fraction * self.stale_after_s + self.attribute_delay_s >= self.stale_after_s:
260            raise LeaseBoundError(
261                f"(1 + {self.missed_beats}) × lease.renew_fraction {self.renew_fraction} × lease.stale_after_s "
262                f"{self.stale_after_s} + lease.attribute_delay_s {self.attribute_delay_s} is not below "
263                f"lease.stale_after_s {self.stale_after_s}: a beating lease that missed "
264                f"{self.missed_beats} beat(s) could be seen as stale and broken"
265            )
