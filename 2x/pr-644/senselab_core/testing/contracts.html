senselab_core.testing.contracts

The contract suite every declared task and backend gets by being declared (testing.md §3).

contract_cases(registry) returns (id, case) pairs; a case is a no-argument callable that raises AssertionError (or the validation error it provokes) when the declaration breaks the contract. Ids are <task>::<case> and <backend>::<case>; a backend name declared under several tasks is written <task>/<backend>.

Task cases: params_round_trip and params_reject_unknown. Backend cases, each building keys with key_builder (default build_key): key_deterministic; key_changes_with_<field> for every LogicalKey field, except that a batch formation or chunk grid the backend does not declare output-affecting gets key_ignores_<field>; key_ignores_<field> for every other Placement field and every ExecutionContext field; key_independent_of_hash_seed; and content_fault_isolated, which runs the backend's worker on its load stub.

backend_license derives the backend's code license from lock_records(backend) and checks every package record and the result against the license table; it is skipped when no lock_records is given.

key_independent_of_hash_seed builds the base call's key in two interpreters started with PYTHONHASHSEED=1 and PYTHONHASHSEED=2. Each imports the params models and the key builder by __module__ and __qualname__, so they must be defined at module level.

101def _regular_series(item: str, port: str, frames: int = 800) -> dict[str, Any]:
102    """Return the wire form of a one-channel regular group of `frames` samples, its phase set by the item and port."""
103    phase = int(identity_digest("contract-probe@1", [item, port])[:8], 16) % 1000
104    data = numpy.sin(2.0 * numpy.pi * (numpy.arange(frames) + phase) / 40.0).astype(numpy.float32)[None, :]
105    return {
106        "format": "regular-group@3.0",
107        "channels": [{"name": "ch0", "type": "audio", "units": "FS", "placement": None, "reference": None}],
108        "sampling_rate": float(AUDIO_RATE),
109        "data": data,
110        "source": None,
111        "source_stamp": None,
112        "decoded": None,
113    }

116def _audio(item: str, port: str, frames: int = AUDIO_FRAMES) -> dict[str, Any]:
117    """Return the wire form of the audio whose one mono group is `_regular_series`'s, anchored on its identity."""
118    group = {k: v for k, v in _regular_series(item, port, frames).items() if k != "format"}
119    data = group["data"]
120    fields = {"npy": npy_digest("npy@1", data), "sampling_rate": float(AUDIO_RATE), "channels": group["channels"]}
121    identity = "content:" + identity_digest("audio@4", fields)
122    return {
123        "format": "audio@3.0",
124        "groups": {"audio": group},
125        "anchor": {"clock_id": identity, "absolute_start_ns": None},
126        "identity": identity,
127    }

140def _transcript(item: str, port: str) -> dict[str, Any]:
141    """Return the wire form of a one-segment transcript, its last word set by the item and port."""
142    span = Span(start_ns=0, end_ns=1_000_000_000, clock_id="contract-clock")
143    node = TranscriptNode(text=f"{item} spoke twice then {_digits(item, port) % 7}", span=span)
144    return wire_form(Transcript(segments=(node,), extent=span))

147def _intervals(item: str, port: str) -> dict[str, Any]:
148    """Return the wire form of two speaker turns, their boundary set by the item and port."""
149    cut = 400_000_000 + _digits(item, port) % 200_000_000
150    return wire_form(
151        Intervals(
152            start_ns=[0, cut],
153            end_ns=[cut, 1_000_000_000],
154            label=numpy.array(["spk0", "spk1"]),
155            clock_id="contract-clock",
156        )
157    )

160def _trials(item: str, port: str) -> dict[str, Any]:
161    """Return the wire form of four trials, two of each kind, over three-dimensional embeddings."""
162    rng = numpy.random.default_rng(_digits(item, port))
163    first = rng.standard_normal((4, 3)) + 1.0
164    return wire_form(
165        Trials(
166            first=first,
167            second=first + 0.1 * rng.standard_normal((4, 3)) * numpy.array([[1], [5], [1], [5]]),
168            same=numpy.array([True, False, True, False]),
169            space="contract-space",
170        )
171    )

220def contract_cases(
221    registry: Registry,
222    *,
223    key_builder: KeyBuilder = build_key,
224    hash_seed_timeout_s: float = HASH_SEED_TIMEOUT_S,
225    lock_records: LockRecords | None = None,
226) -> list[tuple[str, Case]]:
227    """Return the contract cases of every task and backend in `registry`, with their ids.
228
229    Each `key_independent_of_hash_seed` interpreter may take `hash_seed_timeout_s` seconds.
230    """
231    names = Counter(b.name for t in registry.tasks() for b in registry.backends(t.name))
232    cases: list[tuple[str, Case]] = []
233    for task in registry.tasks():
234        cases += [(f"{task.name}::{case}", functools.partial(fn, task)) for case, fn in _task_cases(task)]
235        for backend in registry.backends(task.name):
236            label = backend.name if names[backend.name] == 1 else f"{task.name}/{backend.name}"
237            cases += [
238                (f"{label}::{case}", functools.partial(fn, task, backend))
239                for case, fn in _backend_cases(backend, key_builder, hash_seed_timeout_s, lock_records)
240            ]
241    return cases

248def probe_params(task: TaskSpec) -> TaskParams:
249    """Return the task's params built from defaults, each required field taking its declared `PROBE_KEY` sample.
250
251    Raises:
252        AssertionError: if a required field declares no sample.
253    """
254    probes: dict[str, Any] = {}
255    for name, field in task.params.model_fields.items():
256        if field.is_required():
257            extra = field.json_schema_extra if isinstance(field.json_schema_extra, dict) else {}
258            assert PROBE_KEY in extra, f"required param {name!r} of {task.name!r} declares no {PROBE_KEY!r} sample"
259            probes[name] = extra[PROBE_KEY]
260    return task.params.model_validate(probes)

267def _backend_cases(
268    backend: BackendSpec, build: KeyBuilder, hash_seed_timeout_s: float, lock_records: LockRecords | None
269) -> list[tuple[str, Callable[[TaskSpec, BackendSpec], None]]]:
270    cases: list[tuple[str, Callable[[TaskSpec, BackendSpec], None]]] = [
271        ("key_deterministic", functools.partial(_key_deterministic, build)),
272        ("output_affecting_produced", _output_affecting_produced),
273    ]
274    for f in LogicalKey.model_fields:
275        if f in OUTPUT_AFFECTING and f not in backend.output_affecting:
276            cases.append((f"key_ignores_{f}", functools.partial(_key_ignores_placement, build, f)))
277        else:
278            cases.append((f"key_changes_with_{f}", functools.partial(_key_changes_with, build, f)))
279    cases += [(f"key_ignores_{f}", functools.partial(_key_ignores_placement, build, f)) for f in PLACEMENT_ONLY]
280    cases += [
281        (f"key_ignores_{f}", functools.partial(_key_ignores_context, build, f)) for f in ExecutionContext.model_fields
282    ]
283    cases.append(
284        ("key_independent_of_hash_seed", functools.partial(_key_independent_of_hash_seed, build, hash_seed_timeout_s))
285    )
286    cases.append(("content_fault_isolated", _content_fault_isolated))
287    if backend.min_input_ns is not None:
288        cases.append(("input_below_declared_minimum_is_content_fault", _input_below_minimum))
289    cases.append(("backend_license", functools.partial(_backend_license, lock_records)))
290    return cases

293def _output_affecting_produced(task: TaskSpec, backend: BackendSpec) -> None:
294    produced: dict[str, object] = dict(request_components(backend, fixed_formation(backend) or SINGLE))
295    missing = [c for c in backend.output_affecting if produced[c] is None]
296    assert not missing, f"{backend.name!r} declares {missing} output-affecting, and no request produces a value for it"
297    assert backend.batch_invariant or "batch_formation" in backend.output_affecting, (
298        f"{backend.name!r} is not batch-invariant and does not declare batch_formation output-affecting"
299    )

302def _backend_license(lock_records: LockRecords | None, task: TaskSpec, backend: BackendSpec) -> None:
303    if lock_records is None:
304        raise unittest.SkipTest("no lock records supplied; nothing to derive a license from")
305    records = lock_records(backend)

313def _params_round_trip(task: TaskSpec) -> None:
314    params = probe_params(task)
315    assert task.params.model_validate_json(params.model_dump_json()) == params, "params do not survive JSON"

320def _params_reject_unknown(task: TaskSpec) -> None:
321    known = list(task.params.model_fields)
322    probe = f"{known[0]}_" if known else "unknown_param"
323    try:
324        task.params.model_validate({**probe_params(task).model_dump(), probe: 0})
325    except ValidationError as e:
326        message = str(e)
327    else:
328        raise AssertionError(f"{task.params.__name__} accepted the unknown key {probe!r}")
329    assert probe in message, f"the error does not name the unknown key {probe!r}: {message}"
330    if known:
331        assert "did you mean" in message and repr(known[0]) in message, f"no suggestion of {known[0]!r}: {message}"

334def base_call(task: TaskSpec, backend: BackendSpec) -> tuple[Call, Placement]:
335    """Return the call and placement every key case starts from: default params, `BASE_MODEL`, every component set.
336
337    A server backend's call carries `BASE_SERVICE`.
338    """
339    call = Call(
340        task=task,
341        params=probe_params(task),
342        inputs={port: "content:" + identity_digest("contract-probe@1", ["input", port]) for port in task.inputs},
343        model=BASE_MODEL,
344        service=None if backend.server is None else BASE_SERVICE,
345    )
346    placement = Placement(
347        backend=backend,
348        backend_params=backend.params(),
349        device=backend.devices[0],
350        batch_formation="contract-batches",
351        chunk_grid="contract-grid",
352    )
353    return call, placement

360def _key_deterministic(build: KeyBuilder, task: TaskSpec, backend: BackendSpec) -> None:
361    call, placement = base_call(task, backend)
362    key = build(call, placement, BASE)
363    assert build(*base_call(task, backend), BASE).digest() == key.digest(), "the same call gave two keys"
364    again = build(
365        replace(
366            call,
367            params=task.params.model_validate_json(call.params.model_dump_json()),
368            inputs=dict(reversed(list(call.inputs.items()))),
369        ),
370        replace(
371            placement, backend_params=backend.params.model_validate_json(placement.backend_params.model_dump_json())
372        ),
373        BASE,
374    )
375    assert again.digest() == key.digest(), "input order or a params JSON round trip changed the key"

378def _literal_choices(annotation: object) -> list[object]:
379    """Return the values the `Literal`s in `annotation` admit, through unions; empty for any other type."""
380    if typing.get_origin(annotation) is typing.Literal:
381        return list(typing.get_args(annotation))
382    return [c for arg in typing.get_args(annotation) for c in _literal_choices(arg)]

385def _perturbed(field: str, value: object) -> object:
386    if isinstance(value, bool):
387        return not value
388    if isinstance(value, int):
389        return value + 1
390    if isinstance(value, str):
391        return value + "~"
392    if value is None:
393        return "perturbed"
394    if isinstance(value, Mapping):
395        return {**value, "~": "perturbed"}
396    raise TypeError(f"no perturbation for field {field!r} holding {type(value).__name__}; extend _perturbed")

399def _varied_params(model: type[P], params: P) -> P:
400    dumped = params.model_dump(mode="json")
401    for name, value in dumped.items():
402        try:
403            varied = model.model_validate({**dumped, name: _perturbed(name, value)})
404        except (ValidationError, TypeError):
405            continue
406        if varied != params:
407            return varied
408    raise unittest.SkipTest(f"no field of {model.__name__} can be varied generically; nothing to vary")

411def _varied_call(field: str, call: Call, placement: Placement) -> tuple[Call, Placement]:  # noqa: PLR0911 - one return per key field
412    """Return the call and placement with the one declaration or argument behind key field `field` changed."""
413    task, backend = call.task, placement.backend
414    if field in ("task", "task_version"):
415        attr = "name" if field == "task" else "version"
416        changed = task.model_copy(update={attr: getattr(task, attr) + "~"})
417        return replace(call, task=changed), replace(
418            placement, backend=backend.model_copy(update={"task": changed.name})
419        )
420    if field in ("backend", "backend_version"):
421        attr = "name" if field == "backend" else "version"
422        return call, replace(placement, backend=backend.model_copy(update={attr: getattr(backend, attr) + "~"}))
423    if field == "params":
424        return replace(call, params=_varied_params(task.params, call.params)), placement
425    if field == "backend_params":
426        return call, replace(placement, backend_params=_varied_params(backend.params, placement.backend_params))
427    if field == "inputs":
428        if not call.inputs:
429            raise unittest.SkipTest(f"{task.name!r} declares no inputs; nothing to vary")
430        port = sorted(call.inputs)[0]
431        varied = "content:" + identity_digest("contract-probe@1", ["varied"])
432        return replace(call, inputs={**call.inputs, port: varied}), placement
433    varied_call: dict[str, Any] = {
434        "model": VARIED_MODEL,
435        "profile_slices": {"profile": "profile:" + identity_digest("contract-probe@1", ["slice"])},
436        "service": VARIED_SERVICE,
437    }
438    if field in varied_call:
439        if field == "service" and call.service is None:
440            raise unittest.SkipTest(f"{backend.name!r} is not a server backend; no service to vary")
441        return replace(call, **{field: varied_call[field]}), placement
442    if field in OUTPUT_AFFECTING:
443        return call, _placed(placement, field, f"{getattr(placement, field)}~")
444    raise AssertionError(f"LogicalKey.{field} has no contract variation: add it to _varied_call or KEY_FORMAT_FIELDS")

452def _key_changes_with(build: KeyBuilder, field: str, task: TaskSpec, backend: BackendSpec) -> None:
453    call, placement = base_call(task, backend)
454    key = build(call, placement, BASE)
455    if field in KEY_FORMAT_FIELDS:
456        dumped = key.model_dump()
457        changed = LogicalKey.model_validate({**dumped, field: _perturbed(field, dumped[field])})
458    else:
459        changed = build(*_varied_call(field, call, placement), BASE)
460    assert changed.digest() != key.digest(), f"varying the {field} of a call left its key unchanged"

463def _key_ignores_placement(build: KeyBuilder, field: str, task: TaskSpec, backend: BackendSpec) -> None:
464    call, placement = base_call(task, backend)
465    varied = _placed(placement, field, _perturbed(field, getattr(placement, field)))
466    assert build(call, varied, BASE).digest() == build(call, placement, BASE).digest(), (
467        f"the placement's {field} reached the key"

471def _key_ignores_context(build: KeyBuilder, field: str, task: TaskSpec, backend: BackendSpec) -> None:
472    assert field not in LogicalKey.model_fields, f"context field {field!r} is also a key field"
473    value = BASE.model_dump()[field]
474    choices = _literal_choices(ExecutionContext.model_fields[field].annotation)
475    optional = type(None) in typing.get_args(ExecutionContext.model_fields[field].annotation)
476    if choices:
477        varied = next(c for c in choices if c != value)
478    else:
479        varied = None if optional and value is not None else _perturbed(field, value)

560    assert all("outputs" in r for r in clean), f"a fault-free run failed: {clean}"
561    assert faulted[1].get("error", {}).get("fault") is FaultClass.CONTENT, f"item b did not fail as content: {faulted}"

590def _comparable(value: object) -> object:
591    """Return `value` with each array replaced by its `npy_digest`, so results holding arrays compare with `==`."""
592    if isinstance(value, numpy.ndarray):
593        return npy_digest("npy@1", value)
594    if isinstance(value, Mapping):
595        return {k: _comparable(v) for k, v in value.items()}
596    if isinstance(value, (list, tuple)):
597        return [_comparable(v) for v in value]
598    return value
