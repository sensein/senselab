senselab_core.runtime.execution.portable

senselab's results on a pydra PortableStore: the job a store entry is of, its identity, reuse rule and records.

open_store opens a fresh store at a root every node sees: entries keyed by the senselab logical key's digest (logical-key@7), slotted by the execution context's fingerprint (context@5), the pool and senselab sections of a context recorded and never fingerprinted. SenselabReuse admits stored entries by the run's reuse policy and measured equivalence classes, refuses a result whose env license the run's policy refuses, and samples cross-context hits for verification. Each executed entry's activity is written to its run's provenance before the entry is renamed into place (record_execution, the store's on_commit), and each reuse when it is served.

197@dataclass(frozen=True)
198class SenselabIdentity:
199    """`senselab-identity@2`: the key is the job's `LogicalKey`, the slot its context's `context@5` fingerprint.
200
201    A job whose pydra checksum carries an absence suffix (`<digest>-absent-<hex>`) keys its entry by the logical key's
202    fields plus `absent`, its absence triples in pydra's form, and its checksum is the logical key's digest plus the
203    suffix of those triples. `name` is `UPGRADED` only to read a store this account cannot upgrade.
204    """
205
206    recorded_only: frozenset[str] = RECORDED_ONLY
207    name: str = IDENTITY
208    fingerprint_name: ClassVar[str] = "context@5"
209    scheme: ClassVar[int | None] = None
210
211    def key(self, job: "PydraJob") -> dict[str, Any]:
212        """Return the record the job's checksum digests: `entry_key` of its logical key and its absence triples."""
213        return entry_key(doc_of(job.task).key, job.absent)
214
215    def supplied_key(self, checksum: str) -> None:
216        """Return None: senselab holds no supplied checksum, and `check_key` refuses a key-less publish."""
217        return None
218
219    def check_key(self, checksum: str, key: Mapping[str, Any]) -> bool:
220        """Whether `key` is a logical key whose digest, plus the absence suffix of its `absent`, is `checksum`."""
221        if not isinstance(key, Mapping):
222            return False
223        fields = {k: v for k, v in key.items() if k != ABSENT_KEY}
224        try:
225            suffix = ""
226            if ABSENT_KEY in key:
227                triples = absent_triples(key[ABSENT_KEY])
228                if not triples or absent_form(triples) != key[ABSENT_KEY]:
229                    return False
230                suffix = absence_suffix(triples)
231            return LogicalKey.model_validate(fields).digest() + suffix == checksum
232        except (TypeError, ValueError):
233            return False
234
235    def context(self, job: "PydraJob", phase: str) -> dict[str, Any]:
236        """Return the lookup context the job carries, or the context of running it here, with its `senselab` section.
237
238        The execute context also records when the call ran and its output identity (`ran`).
239        """
240        doc = doc_of(job.task)
241        section = doc.section()
242        if phase == "lookup":
243            return {**doc.context.model_dump(mode="json"), JOB: section}
244        section.update(_ran(job.checksum), profile=doc.profile.identity_digest())
245        return {**doc.execute_context().model_dump(mode="json"), JOB: section}
246
247    def fingerprint(self, context: Mapping[str, Any]) -> str:
248        """Return the context's `context@5` fingerprint, its recorded-only sections left out."""
249        return stored_context(context).fingerprint()

261def stored_evidence(store: PortableStore) -> list[EquivalenceEvidence]:
262    """Return every verify-sampling comparison the store holds, by digest; an unreadable one is logged and skipped."""
263    found = []
264    with store.sidecar(EVIDENCE) as side:
265        for name in sorted(side.names()):
266            try:
267                with side.open_read(name) as member:
268                    found.append(read_evidence(name, member.read()))
269            except (OSError, ValueError) as err:
270                logger.warning("an equivalence record was skipped: %s", err)
271    return found

274def ranked(
275    stored: Sequence[tuple[Any, Mapping[str, Any], str]],
276    here: ExecutionContext,
277    policy: ReusePolicy,
278    backend: str,
279    equivalence: Equivalence,
280) -> list[Candidate]:
281    """Return the stored entries admitted for `here`: the equal fingerprint first, then those `policy` accepts."""
282    mine = here.fingerprint()
283    found: list[Candidate] = []
284    for entry, context, digest in stored:
285        try:
286            theirs = stored_context(context)
287        except ValueError as err:
288            logger.warning("a stored context of backend %s was skipped: %s", backend, err)
289            continue
290        if entry.fingerprint == mine:
291            found.insert(0, Candidate(entry, context, digest, 0, ReusePolicy.EXACT.value))
292        elif acceptable(theirs, here, policy, backend=backend, equivalence=equivalence):
293            found.append(Candidate(entry, context, digest, 1, policy.value))
294    return found

297def serving(  # noqa: PLR0913 - the store and key, then the job's context, run, backend and classes
298    store: PortableStore,
299    key: str,
300    here: ExecutionContext,
301    run: RunConfig,
302    backend: str,
303    equivalence: Equivalence = NO_EQUIVALENCE,
304) -> Candidate | None:
305    """Return the entry `store` would serve the job of key digest `key` in `here` under `run`, or None.
306
307    The first admitted entry is the one a lookup takes; it is served unless the run refuses its license or samples it
308    for verification. Only each entry's `context.json` is read.
309    """
310    found = ranked(store.stored(key), here, ReusePolicy(run.reuse), backend, store.reuse.equivalence(equivalence))
311    if not found:
312        return None
313    first = found[0]
314    entry = StoredContext(first.entry.fingerprint, stored_context(first.context), ReusePolicy(first.admitted_by))
315    return first if servable(run, key, here, entry) is Servable.SERVE else None

333class SenselabReuse:
334    """The reuse rule of a senselab store: the run's policy and classes admit, its license and verify rules decide.
335
336    Entries are ranked in the process that looks up, against the context of running the job there: the equal
337    fingerprint first, then each entry the job's reuse policy accepts under the measured classes (split by the store's
338    disagreeing evidence). A result whose env license the run refuses is refused; a sampled cross-context hit is
339    verified. A served entry's reuse activity is written to its run's provenance.
340    """
341
342    def __init__(self) -> None:
343        """Bind to no store yet."""
344        self._store: PortableStore | None = None
345
346    def __getstate__(self) -> dict[str, Any]:
347        """Return the state a pickled store carries: no bound store."""
348        return {"_store": None}
349
350    def __setstate__(self, state: dict[str, Any]) -> None:
351        """Restore an unbound hook."""
352        self.__dict__.update(state)
353
354    def bind(self, store: PortableStore) -> None:
355        """Bind to `store`, whose sidecar holds the evidence."""
356        self._store = store
357
358    def equivalence(self, given: Equivalence) -> Equivalence:
359        """Return `given` split by every disagreeing comparison the store holds."""
360        if self._store is None or not any(given.values()):
361            return given
362        table = given
363        for evidence in self._evidence():
364            if not evidence.agree:
365                table = split_class(table, evidence.backend, evidence.stored.member(), evidence.current.member())
366        return table
367
368    def _evidence(self) -> list[EquivalenceEvidence]:
369        assert self._store is not None
370        return stored_evidence(self._store)
371
372    def rank(
373        self, current: Mapping[str, Any], stored: Sequence[tuple[Any, Mapping[str, Any], str]], task: object
374    ) -> list[Candidate]:
375        """Return the admitted entries, the equal fingerprint first, then those the job's policy accepts."""
376        doc = doc_of(task)
377        table = self.equivalence(doc.equivalence)
378        return ranked(stored, doc.execute_context(), ReusePolicy(doc.run.reuse), doc.key.backend, table)
379
380    def verdict(self, candidate: Candidate, current: Mapping[str, Any], task: object) -> Verdict:  # noqa: ARG002
381        """Return what the run does with `candidate`: refuse a refused license, verify a sampled hit, else serve."""
382        doc = doc_of(task)
383        entry = StoredContext(
384            candidate.entry.fingerprint, stored_context(candidate.context), ReusePolicy(candidate.admitted_by)
385        )
386        said = servable(doc.run, doc.key.digest(), doc.execute_context(), entry)
387        return {Servable.SERVE: Verdict.SERVE, Servable.VERIFY: Verdict.VERIFY, Servable.REFUSE: Verdict.REFUSE}[said]
388
389    def served(
390        self,
391        candidate: Candidate,
392        current: Mapping[str, Any],
393        task: object,
394        entry: "StoredEntry",  # noqa: ARG002
395    ) -> None:
396        """Write the reuse activity of serving `entry` for the job to its run's provenance, before it is served."""
397        doc = doc_of(task)
398        record_reuse(doc, stored_context(candidate.context), entry.outputs[OUTPUTS], ReusePolicy(candidate.admitted_by))
399
400    def compare(
401        self,
402        candidate: Candidate,
403        stored: "StoredEntry",
404        fresh: Mapping[str, Any],
405        fresh_digest: str,
406        task: object,  # noqa: ARG002
407    ) -> None:
408        """Compare a verified job's fresh outputs with the stored ones under the run's thresholds; record the evidence.
409
410        A disagreement splits the class of the two contexts for every later lookup.
411        """
412        doc = doc_of(task)
413        comparison = compare(stored.outputs[OUTPUTS], fresh[OUTPUTS], doc.run.thresholds)
414        evidence = EquivalenceEvidence(
415            task=doc.key.task,
416            backend=doc.key.backend,
417            key=doc.key.digest(),
418            stored=EvidenceSide.of(stored_context(candidate.context)),
419            current=EvidenceSide.of(doc.execute_context()),
420            mismatched=comparison.differences,
421            thresholds=comparison.thresholds,
422            run=doc.attempt,
423        )
424        if self._store is not None:
425            record_evidence(self._store, evidence)
426        if not comparison.agree:
427            logger.warning(
428                "verify: %s disagrees with its stored result in %s", doc.key.backend, ", ".join(comparison.paths)
429            )
430        _verified(doc.key.digest(), evidence.canonical_digest())

439def record_execution(
440    staged: object, checksum: str, key: Mapping[str, Any], context: Mapping[str, Any], digest: str
441) -> None:  # noqa: ARG001
442    """The store's `on_commit`: write the executed activity of an entry to its run's provenance, before its rename.
443
444    The activity names the job's node and plan, the inputs it used, the output identity of its outputs and, for a
445    verified job, the evidence its comparison recorded (noted by `SenselabReuse.compare`, which runs before this).
446    """
447    section = context[JOB]
448    logical = LogicalKey.model_validate(dict(key))
449    here = stored_context(context)
450    ref = JobRef(plan_hash=section["plan"], node=section["node"], element=tuple(section["element"]))
451    used = [(port, identity) for port, identity in section["used"]]
452    attributes: dict[str, object] = {}
453    verifies = _verifies(checksum)
454    if verifies is not None:
455        attributes["senselab:verifies"] = verifies
456    started = datetime.fromisoformat(section["started"])
457    output = section["output"]
458    if output is None:
459        raise RuntimeError(f"job {ref} published outputs its call never noted (ran)")
460    provenance = ProvStore()
461    agent, activity = execution_records(logical, here, section["profile"], used, output, started, attributes, ref)
462    entities = (kprov.entity(i, core_vocabulary()) for i in dict.fromkeys([*(i for _, i in used), output]))
463    provenance.add(kprov.plan(ref.plan_hash, core_vocabulary()), *entities, agent, activity)
464    provenance.write(_run_dir(ref.plan_hash))

467def record_reuse(
468    doc: JobDoc,
469    stored: ExecutionContext,
470    outputs: Mapping[str, object],
471    policy: ReusePolicy,
472    current: ExecutionContext | None = None,
473) -> None:
474    """Write the `senselab:Reuse` of serving a stored entry for `doc`'s job to its run's provenance.
475
476    `current` is the context the entry was admitted for: where the job would run (`execute_context`) when None.
477    """
478    key = doc.key
479    current = current if current is not None else doc.execute_context()
480    identity = output_identity(dict(outputs))
481    agent, stored_agent = agent_record(key, current), agent_record(key, stored)
482    own: dict[str, object] = {
483        "senselab:storedAgent": stored_agent.id,
484        **verify_attributes(doc.run),
485        "senselab:logicalKey": key_record(key),
486        "senselab:context": context_record(current),
487        "senselab:storedContext": context_record(stored),
488        "senselab:reusePolicy": policy.value,
489    }
490    if stored.env_license is None:
491        own["senselab:unlicensedReuse"] = True
492    if policy is ReusePolicy.EQUIVALENT:
493        own["senselab:equivalence"] = equivalence_digest(doc.equivalence)
494    now = datetime.now(timezone.utc)
495    reuse = kprov.activity(
496        core_vocabulary(),
497        types=["senselab:Reuse"],
498        attributes=own,
499        used=[kprov.Usage(identity)],
500        associations=[plan_link(agent, doc.ref)],
501        started=now,
502        ended=now,
503    )
504    provenance = ProvStore()
505    provenance.add(
506        kprov.plan(doc.ref.plan_hash, core_vocabulary()),
507        kprov.entity(identity, core_vocabulary()),
508        agent,
509        stored_agent,
510        reuse,
511    )
512    provenance.write(_run_dir(doc.ref.plan_hash))

515def open_store(root: Path, work: Path, lease: LeaseSettings | None = None) -> PortableStore:
516    """Open the senselab store at `root`, every job running in a directory of its own under `work`.
517
518    Both are on storage every node sees; `work` is the run's private directory. `lease` bounds the store's job leases
519    (`warm.lease_settings` of the run's profile); None keeps pydra's defaults, which a reader that takes no lease uses.
520    A store whose header names `UPGRADED`, and otherwise matches, has its header rewritten to `IDENTITY` when this
521    account can write the root (`upgrade_header`), and is read under `UPGRADED` when it cannot.
522
523    Raises:
524        StoreLayoutMismatch: the root's header names another identity, or differs in any other key.
525    """
526    work = Path(work).absolute()
527
528    def work_dir(checksum: str, fingerprint: str, attempt: str) -> Path:
529        return work / checksum / fingerprint / attempt
